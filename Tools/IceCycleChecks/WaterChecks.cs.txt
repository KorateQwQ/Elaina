    static float WaterProgress(NPC npc) => npc.GetGlobalNPC<ElementalGlobalNPC>().GetCurrentAccumulation(ElementType.Water) ?? 0;
    static bool WaterHit(NPC npc, uint tick, int amount = 12)
    {
        Clock(tick);
        return WaterExposureNPC.ApplyHit(FireBurnNPC.CaptureHit(npc)!.Value, 0, amount);
    }

    static void CheckWaterExposure(Player player, Skill skill)
    {
        CheckTenFrameAttachments();
        int wet = ModContent.BuffType<ElementalAffliction_Wet>(), fire = ModContent.BuffType<ElementalAffliction_Fire>();
        var npc = NpcFixture(110); var body = NpcFixture(111, 110); var other = NpcFixture(112);
        var snapshots = new List<object>();
        void Capture(string name, NPC target) {
            var p = target.GetGlobalNPC<ElementalGlobalNPC>().ElementAccumulation[ElementType.Water];
            snapshots.Add(new { Name=name, p.Current, p.Max, p.VisualTime, p.InAffliction, p.Duration, p.MaxDuration, p.UpdateCounter });
        }
        for (int i=1;i<=14;i++) {
            WaterHit(i%2 == 0 ? npc : body, (uint)(i*12));
            if(i==1 || i==7 || i==14) Capture("water-"+i,npc);
        }
        Near(WaterProgress(npc), 168, "fourteen stream pulses accumulate on shared root");
        Check(!WaterExposureNPC.IsWet(npc), "stream does not trigger Wet before three seconds of exposure");
        WaterHit(body, 180);
        Check(WaterExposureNPC.IsWet(npc) && WaterExposureNPC.IsWet(body) && !WaterExposureNPC.IsWet(other), "fifteenth stream pulse wets root and parts only");
        Near(npc.buffTime[npc.FindBuffIndex(wet)], 600, "Wet begins at ten seconds");
        Capture("wet-10s",npc);
        var shared = typeof(ElementalGlobalNPC).GetMethod("GetWaterHealthBarProgress",All)!
            .Invoke(body.GetGlobalNPC<ElementalGlobalNPC>(),[body]);
        Check(ReferenceEquals(shared,npc.GetGlobalNPC<ElementalGlobalNPC>().ElementAccumulation[ElementType.Water]),"body health bar reads shared Wet progress");
        npc.buffTime[npc.FindBuffIndex(wet)] = 590;
        WaterHit(npc, 192);
        Near(npc.buffTime[npc.FindBuffIndex(wet)], 600, "continued hits extend Wet up to ten seconds");
        WaterHit(npc, 204);
        Near(npc.buffTime[npc.FindBuffIndex(wet)], 600, "continued hits cannot exceed cap");
        npc.buffTime[npc.FindBuffIndex(wet)] = 300; npc.GetGlobalNPC<WaterExposureNPC>().PostAI(npc); Capture("wet-5s",npc);
        WaterHit(npc, 216);
        Near(npc.buffTime[npc.FindBuffIndex(wet)],600,"fresh attachment replenishes a partially expired Wet status like burning");

        var progress = npc.GetGlobalNPC<ElementalGlobalNPC>();
        using (var bytes = new MemoryStream()) {
            var writer = new BinaryWriter(bytes); progress.SendExtraAI(npc, new BitWriter(), writer);
            bytes.Position=0;
            var received = new ElementalGlobalNPC();
            received.ReceiveExtraAI(npc, new BitReader(new BinaryReader(new MemoryStream(new byte[1]))), new BinaryReader(bytes));
            Check(received.ElementAccumulation[ElementType.Water].InAffliction && received.ElementAccumulation[ElementType.Water].MaxDuration == 600,
                "Wet progress uses actual NPC synchronization");
        }

        var gated = NpcFixture(113); var part = NpcFixture(114, 113);
        Check(WaterHit(gated, 300), "first water attachment accepted");
        Check(!WaterHit(part, 300) && !WaterHit(part, 301), "different projectiles share two-frame attachment cooldown");
        Near(WaterProgress(gated), 12, "body spam cannot add buildup during cooldown");
        Check(WaterHit(part, 302), "attachment allowed at two-frame boundary");
        // 同一脉冲/水球的去重在发起端完成；服务器只保留主体级两帧冷却，不按弹幕标识或序号拒绝。
        Check(WaterHit(part, 304), "server accepts any attachment once the shared cooldown has passed");
        Near(WaterProgress(gated), 36, "each accepted attachment adds buildup");
        Check(WaterHit(other, 300), "independent target has independent attachment cooldown");
        var peerHit = FireBurnNPC.CaptureHit(part)!.Value;
        Clock(305);
        Check(!WaterExposureNPC.ApplyHit(peerHit, 1, 12), "different players also share the target attachment gate");
        Clock(306);
        Check(WaterExposureNPC.ApplyHit(peerHit, 1, 12), "second player accepted after cooldown");
        Near(WaterProgress(gated), 36, "players' incomplete buildup is not summed");

        var orb = NpcFixture(115);
        FreezeCD(orb, 1200); float lockCd = skill.CurrentCD;
        WaterHit(orb, 400, 180);
        Check(WaterExposureNPC.IsWet(orb), "one orb directly triggers Wet on a neutral target");
        Near(orb.GetGlobalNPC<ElementalGlobalNPC>().GetFreezeCooldown(orb), 1200, "orb never refunds target freeze cooldown");
        Near(skill.CurrentCD, lockCd, "orb never refunds lock cooldown");
        WaterHit(orb, 412);
        Near(orb.GetGlobalNPC<ElementalGlobalNPC>().GetFreezeCooldown(orb), 1200, "stream never refunds target freeze cooldown");
        Near(skill.CurrentCD, lockCd, "stream never refunds lock cooldown");

        var burning = NpcFixture(116); var burningPart = NpcFixture(117, 116);
        burning.AddBuff(fire, 600); burningPart.AddBuff(fire, 600);
        var batch = new WaterExposureBatch();
        Check(batch.Capture(burning), "orb captures burning multiplier before reaction");
        WaterHit(burning, 500, 180);
        Check(!FireBurnNPC.IsBurning(burning) && !burningPart.HasBuff(fire), "orb extinguishes root and every body part");
        Check(!WaterExposureNPC.IsWet(burning) && WaterProgress(burning)==0, "extinguishing hit cannot trigger Wet or gain progress");
        Check(batch.Capture(burningPart), "same orb keeps burning damage bonus on later body hits");
        // 同一水球的后续体节命中由发起端批次拦截，见 CheckActualWaterCallbacks 的真实回调检查。
        WaterHit(burningPart, 504, 12);
        Near(WaterProgress(burning), 12, "fresh water pulse may accumulate after burning is gone");
        var burningStream = NpcFixture(118); burningStream.AddBuff(fire, 600);
        for(int i=1;i<=5;i++) WaterHit(burningStream, (uint)(600+i*12));
        Check(!FireBurnNPC.IsBurning(burningStream) && WaterProgress(burningStream)==0, "one second of stream consumes a full burn without Wet buildup");

        var fireTarget = NpcFixture(119); var firePart = NpcFixture(120,119);
        WaterHit(fireTarget, 700, 180);
        for(int i=1;i<=4;i++) {
            Clock((uint)(700+i*18));
            var burn=fireTarget.GetGlobalNPC<FireBurnNPC>();
            burn.ApplyExposure(fireTarget,0,18,1);
            firePart.GetGlobalNPC<FireBurnNPC>().ApplyExposure(firePart,0,18,1);
            Near(fireTarget.GetGlobalNPC<ElementalGlobalNPC>().GetCurrentAccumulation(ElementType.Fire) ?? 0,0,"fire consumes Wet without fire buildup");
        }
        Check(!WaterExposureNPC.IsWet(fireTarget) && !FireBurnNPC.IsBurning(fireTarget), "removing Wet does not ignite in the same pulse");
        Capture("wet-consumed",fireTarget);
        for(int i=5;i<=13;i++) {Clock((uint)(700+i*18)); fireTarget.GetGlobalNPC<FireBurnNPC>().ApplyExposure(fireTarget,0,18,1);}
        Check(!FireBurnNPC.IsBurning(fireTarget), "nine fire pulses do not ignite");
        Clock(952); fireTarget.GetGlobalNPC<FireBurnNPC>().ApplyExposure(fireTarget,0,18,1);
        Check(FireBurnNPC.IsBurning(fireTarget), "ten fire pulses ignite after Wet is removed");

        var damageRoot=NpcFixture(121); var damagePart=NpcFixture(122,121);
        Clock(1000);
        FireBurnNPC.ApplyBeamHit(FireBurnNPC.CaptureHit(damageRoot)!.Value,0,18,1,1);
        FireBurnNPC.ApplyBeamHit(FireBurnNPC.CaptureHit(damagePart)!.Value,0,18,1,10000);
        Near(damageRoot.GetGlobalNPC<ElementalGlobalNPC>().GetCurrentAccumulation(ElementType.Fire) ?? 0,18,"body damage cannot stack compensation into an instant burn");
        Clock(1001); FireBurnNPC.ApplyBeamHit(FireBurnNPC.CaptureHit(damagePart)!.Value,0,18,1,10000);
        Check(!FireBurnNPC.IsBurning(damageRoot),"new projectile cannot bypass two-frame Fire cooldown");

        var stale=NpcFixture(123); var staleHit=FireBurnNPC.CaptureHit(stale)!.Value;
        NpcFixture(123); Clock(1100);
        Check(!WaterExposureNPC.ApplyHit(staleHit,0,180),"reused NPC slot rejects stale water packet");
        GameMain.netMode=NetmodeID.MultiplayerClient;
        Check(!WaterExposureNPC.ApplyHit(FireBurnNPC.CaptureHit(other)!.Value,0,180),"client cannot apply authoritative water state");
        GameMain.netMode=NetmodeID.SinglePlayer;
        var idle=NpcFixture(124); WaterHit(idle,1200);
        Clock(2099); idle.GetGlobalNPC<WaterExposureNPC>().PostAI(idle); Near(WaterProgress(idle),12,"partial water survives 899 idle ticks");
        Clock(2100); idle.GetGlobalNPC<WaterExposureNPC>().PostAI(idle); Near(WaterProgress(idle),0,"partial water clears after 900 idle ticks");
        string output=Path.Combine(rootDirectory,".vissandbox/ice-cycle-checks"); Directory.CreateDirectory(output);
        File.WriteAllText(Path.Combine(output,"progress-snapshots.json"),System.Text.Json.JsonSerializer.Serialize(snapshots));
    }

    static void CheckTenFrameAttachments()
    {
        Check(SkillBalanceConfig.TryParse(File.ReadAllText(Path.Combine(rootDirectory, SkillBalanceConfig.ResourcePath)),
            out var config, out var error), "ray configuration parses: " + error);
        int waterTicks = (int)Math.Round(config.Skills[nameof(WaterLaserSKill)].BaseCooldown * 60);
        int fireTicks = (int)Math.Round(config.Skills["FireLaserSkill"].BaseCooldown * 60);
        Check(waterTicks == 10 && fireTicks == 10, "both rays use ten-frame attachment budgets");
        var water = NpcFixture(150); var fire = NpcFixture(151); var strongFire = NpcFixture(152);
        for (int i = 1; i <= 17; i++) {
            WaterHit(water, (uint)(20000 + i * waterTicks), waterTicks);
            Clock((uint)(20000 + i * fireTicks));
            FireBurnNPC.ApplyBeamHit(FireBurnNPC.CaptureHit(fire)!.Value, 0, fireTicks, 1, 1);
            if (i <= 9)
                FireBurnNPC.ApplyBeamHit(FireBurnNPC.CaptureHit(strongFire)!.Value, 0, fireTicks, 1, 500);
            if (i == 6) {
                Near(WaterProgress(water), 60, "water still gains sixty attachment ticks per second");
                Near(fire.GetGlobalNPC<ElementalGlobalNPC>().GetCurrentAccumulation(ElementType.Fire) ?? 0, 60,
                    "fire still gains sixty base attachment ticks per second");
            }
        }
        Near(WaterProgress(water), 170, "seventeen water hits remain below Wet threshold");
        Near(fire.GetGlobalNPC<ElementalGlobalNPC>().GetCurrentAccumulation(ElementType.Fire) ?? 0, 170,
            "seventeen fire hits remain below ignition threshold");
        Near(strongFire.GetGlobalNPC<ElementalGlobalNPC>().GetCurrentAccumulation(ElementType.Fire) ?? 0, 162,
            "nine scaled fire hits retain the old five-hit damage attachment budget over ninety frames");
        WaterHit(water, 20180, waterTicks);
        FireBurnNPC.ApplyBeamHit(FireBurnNPC.CaptureHit(fire)!.Value, 0, fireTicks, 1, 1);
        Check(WaterExposureNPC.IsWet(water) && FireBurnNPC.IsBurning(fire), "both statuses trigger at 180 attachment ticks");

        var burning = NpcFixture(153); var wet = NpcFixture(154);
        int burnBuff = ModContent.BuffType<ElementalAffliction_Fire>(), wetBuff = ModContent.BuffType<ElementalAffliction_Wet>();
        burning.AddBuff(burnBuff, 600); wet.AddBuff(wetBuff, 600);
        for (int i = 1; i <= 6; i++) {
            WaterHit(burning, (uint)(21000 + i * waterTicks), waterTicks);
            Clock((uint)(21000 + i * fireTicks));
            FireBurnNPC.ApplyBeamHit(FireBurnNPC.CaptureHit(wet)!.Value, 0, fireTicks, 1, 1);
            if (i < 6) {
                Near(burning.buffTime[burning.FindBuffIndex(burnBuff)], 600 - i * 100, "water removes the same burn duration per second");
                Near(wet.buffTime[wet.FindBuffIndex(wetBuff)], 600 - i * 100, "fire removes the same Wet duration per second");
            }
        }
        Check(!FireBurnNPC.IsBurning(burning) && !WaterExposureNPC.IsWet(wet), "six pulses consume ten seconds of opposing status");
        Near(WaterProgress(burning), 0, "extinguishing does not also accumulate Wet");
        Near(wet.GetGlobalNPC<ElementalGlobalNPC>().GetCurrentAccumulation(ElementType.Fire) ?? 0, 0,
            "drying does not also accumulate fire");
    }

    static void CheckNewFreezeRules()
    {
        int wet=ModContent.BuffType<ElementalAffliction_Wet>();
        var dry=NpcFixture(130); var moist=NpcFixture(131); var body=NpcFixture(132,131);
        moist.AddBuff(wet,600); body.AddBuff(wet,600);
        for(int i=0;i<17;i++) {
            Clock((uint)(10000+2*i));
            dry.GetGlobalNPC<ElementalGlobalNPC>().OnHitByElement(ElementType.Ice,dry,new NPC.HitInfo { Damage=1 },0,5.6f);
            moist.GetGlobalNPC<ElementalGlobalNPC>().OnHitByElement(ElementType.Ice,moist,new NPC.HitInfo { Damage=1 },0,5.6f);
            body.GetGlobalNPC<ElementalGlobalNPC>().OnHitByElement(ElementType.Ice,body,new NPC.HitInfo { Damage=10000 },0,5.6f);
        }
        Near(dry.GetGlobalNPC<ElementalGlobalNPC>().GetCurrentAccumulation(ElementType.Ice)??0,47.6,"dry buildup is half speed");
        Check(!moist.HasBuff<ElementalAffliction_Freeze>(),"body hits do not double freeze buildup or damage compensation");
        Clock(10034); moist.GetGlobalNPC<ElementalGlobalNPC>().OnHitByElement(ElementType.Ice,moist,new NPC.HitInfo { Damage=1 },0,5.6f);
        Check(moist.HasBuff<ElementalAffliction_Freeze>(),"wet target freezes on eighteenth accepted hit");
        Near(moist.buffTime[moist.FindBuffIndex(wet)],600,"ice does not consume Wet duration");
        var low=NpcFixture(133); low.lifeMax=100;
        Clock(11000); low.GetGlobalNPC<ElementalGlobalNPC>().OnHitByElement(ElementType.Ice,low,new NPC.HitInfo { Damage=25 },0,5.6f);
        Near(low.GetGlobalNPC<ElementalGlobalNPC>().GetCurrentAccumulation(ElementType.Ice)??0,50,"damage compensation accelerates freezing a weak dry target");
        Clock(11002); low.GetGlobalNPC<ElementalGlobalNPC>().OnHitByElement(ElementType.Ice,low,new NPC.HitInfo { Damage=25 },0,5.6f);
        Check(low.HasBuff<ElementalAffliction_Freeze>(),"low-health target freezes quickly");
        var cone=Attach(new IceConeProj()); var shard=Attach(new ShardbindProj());
        foreach(var spell in new ModProjectile[]{cone,shard}) {
            var m=new NPC.HitModifiers(); spell.ModifyHitNPC(moist,ref m);
            Near(m.SourceDamage.ApplyTo(1),spell==cone?1.125:2.5,"ice spell gets correct Wet damage multiplier");
            m=new NPC.HitModifiers(); spell.ModifyHitNPC(dry,ref m); Near(m.SourceDamage.ApplyTo(1),1,"dry damage uses normal budget");
        }
        Near(ElementalGlobalNPC.FreezeCooldownDuration,1800,"independent-target freeze cooldown is thirty seconds");
        var state=moist.GetGlobalNPC<ElementalGlobalNPC>(); state.SetAffliction(ElementType.Ice,true); state.PostAI(moist);
        state.SetAffliction(ElementType.Ice,false); state.PostAI(moist);
        Near(state.GetFreezeCooldown(moist),1800,"thaw starts thirty-second cooldown");
        Near(dry.GetGlobalNPC<ElementalGlobalNPC>().GetFreezeCooldown(dry),0,"other target has no cooldown");
        CheckActualWaterCallbacks();
    }

    static void CheckActualWaterCallbacks()
    {
        var ballTemplate=new WaterBall(); var beamTemplate=new WaterLaser();
        typeof(ModProjectile).GetProperty("Entity",All)!.SetValue(ballTemplate,new Projectile { type=700 });
        typeof(ModProjectile).GetProperty("Entity",All)!.SetValue(beamTemplate,new Projectile { type=701 });
        ContentInstance.Register(ballTemplate); ContentInstance.Register(beamTemplate);
        var root=NpcFixture(140); var body=NpcFixture(141,140);
        int fire=ModContent.BuffType<ElementalAffliction_Fire>();
        root.AddBuff(fire,600); body.AddBuff(fire,600);
        var ball=Attach(new WaterBall()); ball.Projectile.type=700; ball.Projectile.identity=500;
        Clock(12000);
        var m=new NPC.HitModifiers(); ball.ModifyHitNPC(root,ref m);
        Near(m.SourceDamage.ApplyTo(100),200,"real orb hook doubles burning damage before reaction");
        ball.OnHitNPC(root,default,0); Check(FireBurnNPC.IsBurning(root),"zero-damage orb cannot consume a status");
        ball.OnHitNPC(root,default,100);
        Check(!FireBurnNPC.IsBurning(root) && !WaterExposureNPC.IsWet(root),"real orb callback and receiver extinguish without wetting");
        m=new NPC.HitModifiers(); ball.ModifyHitNPC(body,ref m);
        Near(m.SourceDamage.ApplyTo(100),200,"real orb keeps multiplier for later body hits");
        Clock(12004); ball.OnHitNPC(body,default,100);
        Near(WaterProgress(root),0,"same orb late body callback cannot apply another water attachment");

        var beam=Attach(new WaterLaser()); beam.Projectile.type=701; beam.Projectile.identity=501;
        typeof(WaterLaser).GetField("primaryTarget",All)!.SetValue(beam,root.whoAmI);
        Clock(12010); m=new NPC.HitModifiers(); beam.ModifyHitNPC(root,ref m); beam.OnHitNPC(root,default,100);
        Near(WaterProgress(root),10,"real stream callback forwards its ten-tick exposure budget");
        var hit=FireBurnNPC.CaptureHit(root)!.Value;
        GameMain.netMode=NetmodeID.MultiplayerClient;
        GameMain.player[0].GetModPlayer<WaterExposurePlayer>().Apply(hit.TargetIndex,hit.TargetGeneration,hit.AnchorIndex,hit.AnchorGeneration,false,700,180);
        Near(WaterProgress(root),10,"client receiver cannot settle wet state");
        GameMain.netMode=NetmodeID.Server;
        Clock(12020);
        using var send=new MonoMod.RuntimeDetour.Hook(typeof(NetMessage).GetMethod("SendData",All)!,
            (Action<int,int,int,Terraria.Localization.NetworkText,int,float,float,float,int,int,int>)((_,_,_,_,_,_,_,_,_,_,_)=>{}));
        GameMain.player[0].GetModPlayer<WaterExposurePlayer>().Apply(hit.TargetIndex,hit.TargetGeneration,hit.AnchorIndex,hit.AnchorGeneration,false,700,180);
        Check(WaterExposureNPC.IsWet(root) && WaterExposureNPC.IsWet(body),"server receiver authoritatively wets root and parts");
        GameMain.netMode=NetmodeID.SinglePlayer;
        CheckStreamPulseDeduplication();
    }

    // 回归：附着去重只针对“同一脉冲 × 同一元素主体”，与弹幕槽位/identity 无关。
    // 旧实现按 (玩家, 弹幕类型, identity) 记录脉冲序号，新施法复用旧槽位时序号从头计数，整段积蓄被静默丢弃。
    static void CheckStreamPulseDeduplication()
    {
        var batchField=typeof(WaterLaser).GetField("waterBatch",All)!;
        void Pulse(WaterLaser beam) => ((WaterExposureBatch)batchField.GetValue(beam)!).Reset();
        void Hit(WaterLaser beam,NPC target) {
            var m=new NPC.HitModifiers(); beam.ModifyHitNPC(target,ref m); beam.OnHitNPC(target,default,100);
        }
        var worm=NpcFixture(143); var segA=NpcFixture(144,143); var segB=NpcFixture(145,143); var other=NpcFixture(146);
        var first=Attach(new WaterLaser()); first.Projectile.type=701; first.Projectile.identity=510;
        typeof(WaterLaser).GetField("primaryTarget",All)!.SetValue(first,segA.whoAmI);

        Clock(13000); Pulse(first);
        Hit(first,segA); Hit(first,other);
        Clock(13003); Hit(first,segB); Hit(first,worm);
        Near(WaterProgress(worm),10,"one pulse attaches once per worm even when later segment hits pass the server cooldown");
        Near(WaterProgress(other),10,"separate enemies hit by the same pulse each receive their own attachment");
        for(int i=1;i<=3;i++) { Clock((uint)(13000+i*10)); Pulse(first); Hit(first,segB); }
        Near(WaterProgress(worm),40,"every paid pulse attaches again");

        // 新一次施法分到相同的弹幕 identity，脉冲重新开始。
        var recast=Attach(new WaterLaser()); recast.Projectile.type=701; recast.Projectile.identity=510;
        typeof(WaterLaser).GetField("primaryTarget",All)!.SetValue(recast,worm.whoAmI);
        Clock(13100); Pulse(recast); Hit(recast,worm);
        Near(WaterProgress(worm),50,"a recast reusing an old projectile identity still accumulates from its first pulse");
        Clock(13110); Pulse(recast); Hit(recast,segA);
        Near(WaterProgress(worm),60,"recast keeps accumulating on later pulses");
    }

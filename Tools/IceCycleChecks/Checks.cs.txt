    static void CheckShardbindDrawLayer()
    {
        var controller = Attach(new ShardbindProj());
        var behindTiles = new List<int>(); var behindNpcs = new List<int>();
        var behindProjectiles = new List<int>(); var overPlayers = new List<int>(); var overUi = new List<int>();
        controller.DrawBehind(17, behindTiles, behindNpcs, behindProjectiles, overPlayers, overUi);
        Check(behindProjectiles.SequenceEqual(new[] { 17 }), "Shardbind submits before the foreground queue");
        typeof(ShardbindProj).GetField("time", All)!.SetValue(controller, IceCycleBalance.LockChargeStartFrame + 1);
        KL.Drawing.LayerDrawRequestSystem.ClearFrame();
        // 控制器在 BehindProjectiles 的 PreDraw 提前登记蓄光请求。
        Color light = Color.White;
        Check(!controller.PreDraw(ref light), "Shardbind suppresses default projectile texture");
        Check(!KL.Drawing.LayerDrawRequestSystem.HasBloomRequests(
            KL.Drawing.LayerDrawRequestSystem.DrawTargetLayer.Projectiles,
            KL.Drawing.LayerDrawRequestSystem.DrawTiming.After), "Shardbind must not enqueue into the completed Projectiles layer");
        Check(KL.Drawing.LayerDrawRequestSystem.HasBloomRequests(
            KL.Drawing.LayerDrawRequestSystem.DrawTargetLayer.OverPlayers,
            KL.Drawing.LayerDrawRequestSystem.DrawTiming.Before), "Shardbind charge shares the foreground Bloom");
        KL.Drawing.LayerDrawRequestSystem.ClearFrame();
    }
    static void CheckCombatBloomLayers()
    {
        GameMain.dedServ = false;
        Queue.ClearFrame();
        var ball = Attach(new WaterBall());
        var burst = Attach(new FireBurstProj());
        var tornado = Attach(new FireTornado());
        ball.DrawBehind(0, new(), new(), new(), new(), new());
        burst.DrawBehind(1, new(), new(), new(), new(), new());
        var dust = new Dust { customData = new KL.Dusts.KLBasicDust.DustInfo(10, Vector2.One) {
            Bloom = KL.Dusts.KLBasicDust.BloomPlacement.BeforeOverPlayers } };

        static void CheckLayers(bool tornadoActive)
        {
            foreach (var layer in Enum.GetValues<Queue.DrawTargetLayer>())
            foreach (var timing in Enum.GetValues<Queue.DrawTiming>())
            {
                bool expected = layer == Queue.DrawTargetLayer.OverPlayers && timing == Queue.DrawTiming.Before
                    || tornadoActive && layer == Queue.DrawTargetLayer.BehindNPCsAndTiles && timing == Queue.DrawTiming.After;
                Check(Queue.HasBloomRequests(layer, timing) == expected, $"Combined skill Bloom at {layer}.{timing}, tornado={tornadoActive}");
            }
        }
    }

    }

    static void CheckWaterDamage()
    {
        // 槽位复用检查会替换先前的 NPC；这里使用独立实体，确保共享锚点仍指向当前目标。
        var target = NpcFixture(170); var part = NpcFixture(171, 170); var other = NpcFixture(172);
        var beam = Attach(new WaterLaser());
        typeof(WaterLaser).GetField("primaryTarget", All)!.SetValue(beam, target.whoAmI);
        foreach (int cooldown in new[] { 0, 1, 600 }) {
            FreezeCD(target, cooldown); FreezeCD(other, cooldown);
            float expected = 1f;
            foreach (var npc in new[] { target, part, other }) {
                var modifiers = new NPC.HitModifiers();
                beam.ModifyHitNPC(npc, ref modifiers);
                Near(modifiers.SourceDamage.ApplyTo(1), expected * (npc == target ? 1 : .5f),
                    "Water Stream damage is independent of freeze cooldown and retains half splash");
            }
        }

    }

    static void CheckLockSequence(NPC target, NPC part, NPC splash, Func<int> freezeEvents, Skill skill)
    {
        CheckShardbindTargetPriority();
        for (int damage = 1; damage < 1000; damage++) {
            int shards = Enumerable.Range(0, 15).Sum(i => IceCycleBalance.LockShardDamage(damage, i, 15));
            if (shards + IceCycleBalance.LockFinalDamage(damage) != damage) throw new Exception("Split loses integer damage");
        }
        Check(true, "Exact integer budget conserved for 1..999 damage");
        Near(IceCycleBalance.LockFinalDamage(12688), 6344, "Final burst gets half the reference damage");

        Clock(2000);
        int previousEvents = freezeEvents(); float oldLockCD = skill.CurrentCD;
        FreezeCD(target, 0); target.GetGlobalNPC<ElementalGlobalNPC>().SetAffliction(ElementType.Ice, false);
        target.dontTakeDamage = true; target.immortal = true;
        Array.Fill(target.buffImmune, true);
        target.position = new Vector2(500, 300); target.velocity = new Vector2(12, -5);
        part.position = new Vector2(550, 300); part.velocity = new Vector2(-3, 8);
        var source = new Projectile { owner = 0, identity = 100, active = true, Center = target.Center };
        IceboundNPC.HoldGroup(target, source, 80);
        var state = target.GetGlobalNPC<IceboundNPC>(); var partState = part.GetGlobalNPC<IceboundNPC>();
        Check(state.IsHeld(target) && partState.IsHeld(part), "Invulnerable and buff-immune main target and related parts are sealed");
        Check(!splash.GetGlobalNPC<IceboundNPC>().IsHeld(splash), "Nearby splash enemy is not sealed");
        Check(!state.PreAI(target), "Ice seal prevents AI");
        target.position += Vector2.One; target.velocity = Vector2.One * 50; state.Pin(target);
        Check(target.position == new Vector2(500, 300) && target.velocity == Vector2.Zero, "Seal restores position after local physical movement");
        Check(target.GetFreezeCooldown() == 0 && !target.HasBuff<ElementalAffliction_Freeze>() && freezeEvents() == previousEvents,
            "Seal has no freeze buff, cooldown or freeze event");
        Near(skill.CurrentCD, oldLockCD, "Seal never refunds lock cooldown");
        var second = new Projectile { owner = 1, identity = 100, active = true, Center = target.Center };
        IceboundNPC.HoldGroup(target, second, 90);
        IceboundNPC.ReleaseGroup(source); Check(state.IsHeld(target), "First caster cannot release second caster's seal");
        IceboundNPC.ReleaseGroup(second);
        Check(!state.IsHeld(target) && !partState.IsHeld(part) && target.velocity == new Vector2(12, -5), "Final release restores original motion for whole group");
        IceboundNPC.HoldGroup(target, source, 2); Clock(2002); Check(!state.IsHeld(target), "Unexpected missing source has bounded control duration");

        var lockProj = Attach(new ShardbindProj());
        uint serial = state.SpawnSerial; lockProj.BindTarget(target, serial);
        Check(lockProj.LockedTarget == target, "Projectile binds to selected enemy even during invulnerability");
        Check(!IceConeTargetSystem.IsValidTarget(target), "Shared targeting excludes invulnerable enemies");
        target.dontTakeDamage = false; target.immortal = false;
        target.chaseable = false;
        Check(!IceConeTargetSystem.IsValidTarget(target), "Shared targeting excludes unchaseable enemies");
        target.chaseable = true;
        Check(IceConeTargetSystem.IsValidTarget(target), "Shared targeting accepts chaseable damageable enemies");
        Check(IceConeTargetSystem.FindTarget(new Vector2(1000000, 1000000), -1) == null, "No target within cursor range resolves to null");
        target.friendly = true; Check(!IceConeTargetSystem.IsValidTarget(target), "Friendly targets remain excluded"); target.friendly = false;
        state.SetDefaults(target); Check(lockProj.LockedTarget == null, "Reused NPC slot cannot inherit old lock");

        var cone = new IceCone3DUnit(new Vector3(200, 0, 0), Vector3.Zero, new Vector3(-58, 0, 0), 1, 135);
        cone.OnSpawn(); int arrivals = 0, arrivalFrame = 0;
        var modelMethod = typeof(IceCone3DUnit).GetMethod("GetModelMatrix", All)!;
        for (int frame = 1; frame < 110; frame++) {
            cone.MoveWithTarget(new Vector2(3, -1)); cone.Update();
            if (cone.ArrivedThisUpdate) { arrivals++; arrivalFrame = frame; }
            Matrix model = (Matrix)modelMethod.Invoke(cone, [frame - 1])!;
            Near(Vector3.TransformNormal(Vector3.UnitX, model).Length(), 20, "Lock growth preserves width");
            Near(Vector3.TransformNormal(Vector3.UnitZ, model).Length(), 20, "Lock growth preserves thickness");
            float age = arrivalFrame == 0 ? 0 : Math.Clamp((frame - arrivalFrame) / 5f, 0, 1);
            float expectedLength = 20 * (1 + 1.2f * (1 - MathF.Pow(1 - age, 3)));
            Near(Vector3.TransformNormal(Vector3.UnitY, model).Length(), expectedLength,
                "Lock length stays continuous on arrival and reaches 2.2 in five ticks");
            var late = new IceCone3DUnit(new Vector3(200, 0, 0), Vector3.Zero, new Vector3(-58, 0, 0), 1, 135);
            late.OnSpawn(); late.MoveWithTarget(new Vector2(3 * frame, -frame)); late.CatchUp(frame);
            Matrix lateModel = (Matrix)modelMethod.Invoke(late, [frame - 1])!;
            Check(!late.ArrivedThisUpdate, "Catch-up does not replay impact events during growth");
            foreach (Vector3 point in new[] { Vector3.Zero, Vector3.UnitX, Vector3.UnitY, Vector3.UnitZ })
                Check(Vector3.Distance(Vector3.Transform(point, model), Vector3.Transform(point, lateModel)) < .001f,
                    "Late network model matches continuous growth and translated lock anchor");
        }
        Check(arrivals == 1 && arrivalFrame == 29, "Moving coordinate system keeps one actual crossing event at frame 29");
        var replay = new IceCone3DUnit(new Vector3(200, 0, 0), Vector3.Zero, new Vector3(-58, 0, 0), 1, 135);
        replay.OnSpawn(); replay.MoveWithTarget(new Vector2(3 * 109, -109)); replay.CatchUp(109);
        Check(Vector3.Distance(cone.Position3D, replay.Position3D) < .001f && replay.HasArrived && !replay.ArrivedThisUpdate,
            "Network catch-up reconstructs translated visuals without replaying impact events");
        CheckPreparedShardbindGeometry();

        // Execute the real controller AI and visual-unit updates; intercept only the engine's world damage scan.
        // This checks scheduling/pulse budgets without requiring a loaded Terraria world or graphics assets.
        var sequence = Attach(new ShardbindProj());
        sequence.Projectile.damage = 12688; sequence.Projectile.timeLeft = ShardbindVisualSettings.ControllerLifetime;
        sequence.Projectile.identity = 101;
        sequence.BindTarget(target, state.SpawnSerial);
        var pulses = new List<(int Frame, int Damage, float Radius)>();
        Vector2 strikePosition = target.Center;
        const int teleportTick = ShardbindSequence.FirstImpactTick + 1;
        void CaptureDamage(On_Projectile.orig_Damage orig, Projectile projectile)
        {
            if (projectile != sequence.Projectile) { orig(projectile); return; }
            Check(sequence.CanDamage() == true && projectile.friendly, "Real damage scan is confined to explicit pulse");
            Check(projectile.Center == strikePosition && sequence.Colliding(default, new Rectangle((int)strikePosition.X, (int)strikePosition.Y, 8, 8)) == true,
                "Every pulse covers the original strike location");
            if ((int)projectile.ai[1] >= teleportTick)
                Check(sequence.Colliding(default, target.Hitbox) == false, "Teleported target is outside the remaining pulses");
            pulses.Add(((int)projectile.ai[1], projectile.damage,
                (float)typeof(ShardbindProj).GetField("damageRadius", All)!.GetValue(sequence)!));
        }
        On_Projectile.Damage += CaptureDamage;
        try {
            for (int frame = 1; frame <= IceCycleBalance.LockBurstFrame + 1; frame++) {
                Clock((uint)(3000 + frame));
                if (frame <= ShardbindSequence.SealTick) target.position += new Vector2(5, -2);
                if (frame <= ShardbindSequence.SealTick) strikePosition = target.Center;
                if (frame == teleportTick) {
                    Vector2 destination = strikePosition + new Vector2(1000, 1000);
                    target.Center = destination;
                    state.Pin(target);
                    Check(target.Center == destination, "Seal respects a forced teleport out of the strike area");
                }
                sequence.AI();
                Check(sequence.Projectile.Center == strikePosition, "Controller stays at the position captured on seal");
                if (frame >= ShardbindSequence.SealTick && frame < teleportTick)
                    Check(target.Center == strikePosition && target.velocity == Vector2.Zero, "Target and strike center coincide before and during first piercing");
                Check(sequence.Projectile.timeLeft > 0, "Controller survives the complete charge and burst timeline");
                sequence.Projectile.timeLeft--;
                if (frame == IceCycleBalance.LockChargeStartFrame) {
                    var liveUnits = (IEnumerable<KL.Drawing.VisualUnit>)typeof(ShardbindProj).GetField("units", All)!.GetValue(sequence)!;
                    Check(liveUnits.OfType<IceCone3DUnit>().Count(c => c.Active) == 15
                        && liveUnits.OfType<IceCone3DUnit>().Count(c => c.HasArrived) is > 0 and < 15,
                        "Charge begins during the volley while all fifteen cones survive");
                }
                Check(state.IsHeld(target) == (frame >= ShardbindSequence.SealTick && frame < teleportTick), "Control starts before launch and ends when target escapes");
            }
        }
        finally { On_Projectile.Damage -= CaptureDamage; }
        Check(pulses.Count == 16 && pulses[0].Frame == ShardbindSequence.FirstImpactTick && pulses[^1].Frame == IceCycleBalance.LockBurstFrame, "Real AI schedules 15 impacts and exactly one final blast");
        Check(pulses.Take(15).All(p => p.Radius == 80) && pulses[^1].Radius == 200, "Real AI uses separate impact and final AOE sizes");
        Check(pulses.Sum(p => p.Damage) == 12688 && pulses[^1].Damage == 6344, "Real AI applies exact 50/50 total budget");
        Check(!splash.GetGlobalNPC<IceboundNPC>().IsHeld(splash) && freezeEvents() == previousEvents && target.GetFreezeCooldown() == 0,
            "Full sequence never seals splash targets or triggers freeze mechanics");
        target.Center = strikePosition;
        IceboundNPC.HoldGroup(target, sequence.Projectile, 20); sequence.OnKill(1);
        Check(!state.IsHeld(target) && !partState.IsHeld(part), "Premature projectile removal releases all related parts");
    }

            }
            On_Projectile.Damage += Capture;
            try {
                for (int frame = 1; frame <= IceCycleBalance.LockBurstFrame + 1; frame++) {
                    Clock((uint)(5000 + deathFrame * 200 + frame));
                    if (frame == deathFrame) {
                        deathPosition = sequence.Projectile.Center;
                        target.life = 0; target.active = false;
                        // 立即复用同类型NPC槽位，旧冰凌锁不能追踪或控制新敌人。
                        target = NpcFixture(20); target.Center = new Vector2(9000, 9000);
                    }

                        Check(!target.GetGlobalNPC<IceboundNPC>().IsHeld(target) && !part.GetGlobalNPC<IceboundNPC>().IsHeld(part), "Losing target releases old group and never seals reused slot");
                    }
                }
            }
            finally { On_Projectile.Damage -= Capture; }
            Check(pulses.Count == 16 && pulses.Sum() == 12688 && pulses[^1] == 6344, "Target death preserves all remaining AOE pulses and final explosion");
            using var bytes = new MemoryStream();
            using (var writer = new BinaryWriter(bytes, global::System.Text.Encoding.UTF8, true)) sequence.SendExtraAI(writer);
            bytes.Position = 0;
            var received = Attach(new ShardbindProj()); received.BindTarget(target, target.GetGlobalNPC<IceboundNPC>().SpawnSerial);
            using (var reader = new BinaryReader(bytes, global::System.Text.Encoding.UTF8, true)) received.ReceiveExtraAI(reader);
            Check(received.LockedTarget == null, "Detached state survives extra-AI serialization even with a live slot");
            received.Projectile.ai[1] = IceCycleBalance.LockBurstFrame + 1;
            received.AI();
            Check(received.Projectile.Center == deathPosition, "Late receiver reconstructs the strike at the synchronized location");
        }

    static void CheckNetworkSeal()
    {
        var target = NpcFixture(22);
        Vector2 frozen = new(2400, 1200);
        target.Center = frozen;

        Check(packets.SequenceEqual(new int[] { MessageID.SyncNPC, MessageID.SyncProjectile }) && target.GetGlobalNPC<IceboundNPC>().IsHeld(target),
            "Server freezes the target and sends the NPC position before the strike position");

        using (var writer = new BinaryWriter(bytes, global::System.Text.Encoding.UTF8, true)) server.SendExtraAI(writer);
        IceboundNPC.ReleaseGroup(server.Projectile);
        target.Center += new Vector2(60, 0);
        var client = Attach(new ShardbindProj());

        GameMain.netMode = NetmodeID.MultiplayerClient;
        try {
            for (int i = 0; i <= ShardbindSequence.FirstImpactTick; i++) client.AI();

                "Delayed client waits before piercing instead of sealing at a predicted position");
            bytes.Position = 0;
            using (var reader = new BinaryReader(bytes, global::System.Text.Encoding.UTF8, true)) client.ReceiveExtraAI(reader);
            client.Projectile.ai[1] = ShardbindSequence.SealTick;
            client.AI();
            Check(client.Projectile.Center == frozen && target.Center == frozen && target.GetGlobalNPC<IceboundNPC>().IsHeld(target),
                "Receiving the server seal aligns the moving NPC and the controller on the same tick");
            var pulses = new List<int>();
            void Pulse(On_Projectile.orig_Damage orig, Projectile projectile) => pulses.Add((int)projectile.ai[1]);
            On_Projectile.Damage += Pulse;
            try {
                client.Projectile.ai[1] = ShardbindSequence.LastImpactTick;
                for (int i = ShardbindSequence.SealTick + 1; i <= IceCycleBalance.LockBurstFrame; i++) client.AI();
            }
            finally { On_Projectile.Damage -= Pulse; }
            Check(pulses.Count == 16 && pulses[0] == ShardbindSequence.FirstImpactTick && pulses[^1] == IceCycleBalance.LockBurstFrame,
                "Owner processes all local damage ticks after a delayed server seal and a subsequent timeline correction");
        }
        finally { GameMain.netMode = NetmodeID.SinglePlayer; }
    }

    static readonly Type StrikeSystem = typeof(ElementStrikeSystem);
    static int Hit(NPC target, ElementInfusion infusion, int damage, Action<NPC> during = null, int owner = 0, Projectile projectile = null)
    {
        var hit = new NPC.HitInfo { Damage = Math.Max(1, damage), SourceDamage = 99999, Knockback = 12.34567f, HideCombatText = true };
        using var engine = new Hook(typeof(NPC).GetMethod("StrikeNPC", [typeof(NPC.HitInfo), typeof(bool), typeof(bool)])!,
            (Func<NPC,NPC.HitInfo,bool,bool,int>)((npc, actual, _, _) => {
                Near(actual.Knockback, hit.Knockback, "full knockback reaches the engine");
                during?.Invoke(npc); return damage;
            }));
        if (projectile != null) {
            projectile.GetGlobalProjectile<ElementalGlobalProjectile>().InfusionElement = infusion;
            return (int)StrikeSystem.GetMethod("StrikeProjectile", All)!.Invoke(null,[target,hit,false,false,projectile,null])!;
        }
        var data = StrikeSystem.GetMethod("Create", All)!.Invoke(null, [target, infusion, owner]);
        return (int)StrikeSystem.GetMethod("Strike", All)!.Invoke(null, [target, hit, GameMain.netMode == NetmodeID.Server, false, data])!;
    }

    static void Check(bool value, string name) { if (!value) throw new Exception(name); passed++; }
    static void Near(double value, double expected, string name) => Check(Math.Abs(value - expected) < .0001, $"{name}: {value} != {expected}");

    static void Register(Mod mod, ModType content) { ((ILoadable)content).Load(mod); ContentInstance.Register(content); }

    static Player PlayerFixture(int index)
    {
        var player = new Player { whoAmI = index, active = true };
        var parts = new ModPlayer[] { new IceCyclePlayer(), new ElainaSkillModPlayer(), new FreezeBuildupPlayer(),
            new ElainaAttributeModPlayer(), new ActionModPlayer(), new global::伊蕾娜.ElainaModplayer { Elaina = true },
            new ElainaManaElixirPlayer() };
        for (ushort i = 0; i < parts.Length; i++) {
            typeof(ModPlayer).GetProperty("Entity", All)!.SetValue(parts[i], player);
            typeof(ModPlayer).GetProperty("Index", All)!.SetValue(parts[i], i);
            if (index == 0) ContentInstance.Register(parts[i]);
        }
        typeof(Player).GetField("modPlayers", All)!.SetValue(player, parts);
        var modSkill = new ShardbindSkill { CurrentCD = 60, MaxCD = 60, CooldownDuration = 60, MaxStack = 1, Stack = 0, BasicStatus = Skill.SKillBasicStatus.UnLock };
        var skill = new Skill(); typeof(Skill).GetProperty("ModSkill", All)!.SetValue(skill, modSkill); modSkill.Skill = skill;
        player.GetModPlayer<ElainaSkillModPlayer>().UnlockedSkill[nameof(ShardbindSkill)] = skill;
        GameMain.player[index] = player;
        return player;
    }

    static NPC NpcFixture(int index, int realLife = -1)
    {
        var npc = new NPC { active = true, whoAmI = index, type = NPCID.BlueSlime, realLife = realLife, life = 10000, lifeMax = 10000 };
        npc.buffImmune = new bool[BuffLoader.BuffCount];
        var icebound = new IceboundNPC(); icebound.SetDefaults(npc);
        typeof(NPC).GetField("_globals", All)!.SetValue(npc, new GlobalNPC[] { new ElementalGlobalNPC(), icebound });
        GameMain.npc[index] = npc;
        return npc;
    }
    static void FreezeCD(NPC npc, int ticks) => typeof(ElementalGlobalNPC).GetField("freezeCooldown", All)!.SetValue(npc.GetGlobalNPC<ElementalGlobalNPC>(), ticks);
    static void TriggerFreeze(NPC npc, int sourcePlayer)
    {
        var global = npc.GetGlobalNPC<ElementalGlobalNPC>();
        global.SetCurrentAccumulation(ElementType.Ice, 98);
        global.OnHitByElement(ElementType.Ice, npc, new NPC.HitInfo { Damage = 10, SourceDamage = 10 }, sourcePlayer, 2.8f);
    }
    static T Attach<T>(T modProjectile, int owner = 0) where T : ModProjectile
    {
        var projectile = new Projectile { owner = owner, active = true, damage = 100 };
        typeof(ModProjectile).GetProperty("Entity", All)!.SetValue(modProjectile, projectile);
        return modProjectile;
    }

    static Skill Owned(ElainaSkill modSkill)
    {
        var skill = new Skill();
        typeof(Skill).GetProperty("ModSkill", All)!.SetValue(skill, modSkill);
        modSkill.Skill = skill;
        skill.BasicStatus = Skill.SKillBasicStatus.UnLock;
        return skill;
    }

    static void CheckLockSequence(NPC target, NPC part, NPC splash, Func<int> freezeEvents, Skill skill)
    {
        for (int damage = 1; damage < 1000; damage++) {
            int shards = Enumerable.Range(0, 15).Sum(i => IceCycleBalance.LockShardDamage(damage, i, 15));
            if (shards + IceCycleBalance.LockFinalDamage(damage) != damage) throw new Exception("Split loses integer damage");
        }
        Check(true, "Exact integer budget conserved for 1..999 damage");
        Near(IceCycleBalance.LockFinalDamage(12688), 6344, "Final burst gets half the reference damage");

        Clock(2000);
        int previousEvents = freezeEvents(); float oldLockCD = skill.CurrentCD;
        FreezeCD(target, 0); target.GetGlobalNPC<ElementalGlobalNPC>().SetAffliction(ElementType.Ice, false);
        target.dontTakeDamage = true; target.immortal = true;
        Array.Fill(target.buffImmune, true);
        target.position = new Vector2(500, 300); target.velocity = new Vector2(12, -5);
        part.position = new Vector2(550, 300); part.velocity = new Vector2(-3, 8);
        var source = new Projectile { owner = 0, identity = 100, active = true };
        IceboundNPC.HoldGroup(target, source, 80);
        var state = target.GetGlobalNPC<IceboundNPC>(); var partState = part.GetGlobalNPC<IceboundNPC>();
        Check(state.IsHeld(target) && partState.IsHeld(part), "Invulnerable and buff-immune main target and related parts are sealed");
        Check(!splash.GetGlobalNPC<IceboundNPC>().IsHeld(splash), "Nearby splash enemy is not sealed");
        Check(!state.PreAI(target), "Ice seal prevents AI");
        target.position += Vector2.One * 999; target.velocity = Vector2.One * 50; state.Pin(target);
        Check(target.position == new Vector2(500, 300) && target.velocity == Vector2.Zero, "Seal restores position after physical or external movement");
        Check(target.GetFreezeCooldown() == 0 && !target.HasBuff<ElementalAffliction_Freeze>() && freezeEvents() == previousEvents,

        {

        Check(pulses.Take(15).All(p => p.Radius == 80) && pulses[^1].Radius == 200, "Real AI uses separate impact and final AOE sizes");

    static void CheckTargetDeathContinuation()
    {
        foreach (int deathFrame in new[] { 1, 45, 100 }) {
            var target = NpcFixture(20); var part = NpcFixture(21, 20);
            target.Center = new Vector2(600, 400);
            var sequence = Attach(new ShardbindProj());
            sequence.Projectile.damage = 12688; sequence.Projectile.timeLeft = 120; sequence.Projectile.identity = 200 + deathFrame;
            sequence.BindTarget(target, target.GetGlobalNPC<IceboundNPC>().SpawnSerial);
            var pulses = new List<int>(); Vector2 deathPosition = target.Center;
            void Capture(On_Projectile.orig_Damage orig, Projectile projectile) {
                if (projectile == sequence.Projectile) pulses.Add(projectile.damage); else orig(projectile);
            }
            On_Projectile.Damage += Capture;
            try {
                for (int frame = 1; frame <= 111; frame++) {
                    Clock((uint)(5000 + deathFrame * 200 + frame));
                    if (frame == deathFrame) {
                        deathPosition = sequence.Projectile.Center;
                        target.life = 0; target.active = false;
                        // 立即复用同类型NPC槽位，旧冰凌锁不能追踪或控制新敌人。
                        target = NpcFixture(20); target.Center = new Vector2(9000, 9000);
                    }
                    sequence.AI();
                    if (frame >= deathFrame) {
                        Check(sequence.Projectile.active && sequence.Projectile.Center == deathPosition, "Dead target leaves animation alive at its last position");
                        Check(!target.GetGlobalNPC<IceboundNPC>().IsHeld(target) && !part.GetGlobalNPC<IceboundNPC>().IsHeld(part), "Losing target releases old group and never seals reused slot");
                    }
                }
            }
            finally { On_Projectile.Damage -= Capture; }
            Check(pulses.Count == 16 && pulses.Sum() == 12688 && pulses[^1] == 6344, "Target death preserves all remaining AOE pulses and final explosion");
            using var bytes = new MemoryStream();
            using (var writer = new BinaryWriter(bytes, global::System.Text.Encoding.UTF8, true)) sequence.SendExtraAI(writer);
            bytes.Position = 0;
            var received = Attach(new ShardbindProj()); received.BindTarget(target, target.GetGlobalNPC<IceboundNPC>().SpawnSerial);
            using (var reader = new BinaryReader(bytes, global::System.Text.Encoding.UTF8, true)) received.ReceiveExtraAI(reader);
            Check(received.LockedTarget == null, "Detached state survives extra-AI serialization even with a live slot");
        }
    }

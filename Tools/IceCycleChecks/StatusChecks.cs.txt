    static void CheckFixedAndDamageBuildup()
    {
        // 固定积蓄统一为 10 点；伤害补偿按最大生命的 50% 对应整条计算。
        foreach (var sample in new[] { (Life: 10000, Damage: 1, Expected: 10f),
            (Life: 10000, Damage: 500, Expected: 10f), (Life: 1000, Damage: 125, Expected: 25f),
            (Life: 1000, Damage: 250, Expected: 50f), (Life: 1000, Damage: 500, Expected: 100f) }) {
            Clock(24000);
            var ice = NpcFixture(183); var fire = NpcFixture(184); var water = NpcFixture(185);
            foreach (var npc in new[] { ice, fire, water }) npc.life = npc.lifeMax = sample.Life;
            ice.GetGlobalNPC<ElementalGlobalNPC>().OnHitByElement(ElementType.Ice, ice,
                new NPC.HitInfo { Damage = sample.Damage, SourceDamage = 99999 }, 0, 10);
            FireBurnNPC.ApplyBeamHit(FireBurnNPC.CaptureHit(fire)!.Value, 0, 18, 1, sample.Damage);
            WaterExposureNPC.ApplyHit(FireBurnNPC.CaptureHit(water)!.Value, 0, 18, sample.Damage);
            foreach (var pair in new[] { (Npc: ice, Element: ElementType.Ice), (Npc: fire, Element: ElementType.Fire),
                (Npc: water, Element: ElementType.Water) }) {
                bool afflicted = pair.Element == ElementType.Ice ? pair.Npc.HasBuff<ElementalAffliction_Freeze>()
                    : pair.Element == ElementType.Fire ? FireBurnNPC.IsBurning(pair.Npc) : WaterExposureNPC.IsWet(pair.Npc);
                Check(afflicted == (sample.Expected >= 100), "fixed or damage compensation triggers the same threshold across ice/fire/water");
                if (!afflicted) Near(pair.Npc.GetGlobalNPC<ElementalGlobalNPC>().GetCurrentAccumulation(pair.Element) ?? 0,
                    sample.Expected, "fixed gain and damage compensation use max, never add or halve");
            }
        }
        var root = NpcFixture(183); root.life = root.lifeMax = 1000;
        var part = NpcFixture(184,183); var beam = Attach(new WaterLaser());
        beam.Projectile.GetGlobalProjectile<ElementalGlobalProjectile>().BeginBuildupBatch();
        ElementInfusion waterHit = new(ElementType.Water, 5.55f);
        Clock(25000); Hit(part,waterHit,0,projectile:beam.Projectile);
        Near(WaterProgress(root),0,"zero damage does not claim this anchor");
        Hit(part,waterHit,125,projectile:beam.Projectile);
        Near(WaterProgress(root),25,"water uses actual damage to shared maximum life");
        Clock(25003); Hit(root,waterHit,500,projectile:beam.Projectile);
        Near(WaterProgress(root),25,"one pulse compensates once per anchor");
    }

    static void CheckElementalDamageOrder()
    {
        using var send = new MonoMod.RuntimeDetour.Hook(typeof(NetMessage).GetMethod("SendData", All)!,
            (Action<int,int,int,Terraria.Localization.NetworkText,int,float,float,float,int,int,int>)((_,_,_,_,_,_,_,_,_,_,_) => {}));
        foreach (int mode in new[] { NetmodeID.SinglePlayer, NetmodeID.Server, NetmodeID.MultiplayerClient })
        foreach (var type in new[] { ElementType.Fire, ElementType.Ice, ElementType.Water, ElementType.Lightning })
        foreach (float points in new[] {0f,7.625f}) {
            GameMain.netMode=mode; Clock(20000); var npc=NpcFixture(180);
            float Progress() => npc.GetGlobalNPC<ElementalGlobalNPC>().GetCurrentAccumulation(type) ?? 0;
            int calls=0;
            Hit(npc,new(type,points),1,_=> { calls++; Near(Progress(),0,"damage precedes attachment"); });
            Check(calls==1,"exactly one strike");
            Near(Progress(), mode == NetmodeID.MultiplayerClient ? 0 : points,"only authority applies explicit points");
        }
        GameMain.netMode=NetmodeID.SinglePlayer;
        CheckIceDamageWithoutBuildup();
        CheckStrikeTransport();
    }

    static void CheckStrikeTransport()
    {
        // 安装实际 tML 二进制上的两个 IL 补丁，校验发送位置和栈签名。
        var system = new ElementStrikeSystem(); system.Load(); system.Unload();
        var target = NpcFixture(181); var element = new ElementInfusion(ElementType.Fire,7.625f);
        var payload = StrikeSystem.GetMethod("Create",All)!.Invoke(null,[target,element,1]);
        var original = new NPC.HitInfo { Damage=1, SourceDamage=999, Crit=true, HitDirection=-1,
            DamageType=DamageClass.Magic, Knockback=12.34567f, HideCombatText=true };
        using var bytes=new MemoryStream();
        using var writer=new BinaryWriter(bytes, System.Text.Encoding.UTF8,true);
        StrikeSystem.GetMethod("Write",All)!.Invoke(null,[writer,target.whoAmI,original,payload]);
        using var reader=new BinaryReader(bytes,System.Text.Encoding.UTF8,true);
        int strikes=0, relays=0;
        using var engine = new MonoMod.RuntimeDetour.Hook(typeof(NPC).GetMethod("StrikeNPC",[typeof(NPC.HitInfo),typeof(bool),typeof(bool)])!,
            (Func<NPC,NPC.HitInfo,bool,bool,int>)((_,h,net,_)=>{
                Check(net && h.DamageType==original.DamageType && h.Crit && h.HitDirection==-1 && h.SourceDamage==999,
                    "atomic strike round-trips all damage fields"); Near(h.Knockback,original.Knockback,"wire preserves knockback"); strikes++; return 1;
            }));
        using var relay = new MonoMod.RuntimeDetour.Hook(typeof(ModPacket).GetMethod("Send",All)!,
            (Action<ModPacket,int,int>)((_,_,ignore)=>{Check(ignore==0,"server excludes predicting client"); relays++;}));
        typeof(Mod).Assembly.GetType("Terraria.ModLoader.ModNet")!.GetField("netMods",All)!.SetValue(null, Array.Empty<Mod>()); global::KL.KL.KLInstance = new CheckMod(); typeof(Mod).GetField("netID",All)!.SetValue(global::KL.KL.KLInstance,(short)0);
        void Receive(int mode) { GameMain.netMode=mode; bytes.Position=0; StrikeSystem.GetMethod("Receive",All)!.Invoke(null,[reader,0]); }
        Clock(40000); Receive(NetmodeID.Server);
        Near(target.GetGlobalNPC<ElementalGlobalNPC>().GetCurrentAccumulation(ElementType.Fire)??0,7.625,"server uses exact transmitted points");
        Clock(40005); Receive(NetmodeID.Server);
        Check(strikes == 1 && relays == 1, "replaying a strike packet cannot duplicate damage or buildup");
        Receive(NetmodeID.MultiplayerClient);
        Near(target.GetGlobalNPC<ElementalGlobalNPC>().GetCurrentAccumulation(ElementType.Fire)??0,7.625,"relay cannot settle a second attachment");
        bytes.Position=0;
        StrikeSystem.GetMethod("Write",All)!.Invoke(null,[writer,target.whoAmI,original,payload]);
        Receive(NetmodeID.Server);
        Near(target.GetGlobalNPC<ElementalGlobalNPC>().GetCurrentAccumulation(ElementType.Fire)??0,15.25,"distinct same-frame packet contributes independently");
        GameMain.netMode=NetmodeID.SinglePlayer; NpcFixture(181); Receive(NetmodeID.Server);
        Check(strikes==3 && relays==2,"slot reuse rejects both damage and points");
        GameMain.netMode=NetmodeID.SinglePlayer;
    }

            var target = NpcFixture(180);
            target.AddBuff(ModContent.BuffType<ElementalAffliction_Wet>(), 1800);
            Clock(20000);
            Hit(target, new(ElementType.Ice, 1.6f), 1, projectile: Attach(new IceConeProj()).Projectile);
            Clock(20000 + gap);
            Near(Hit(target, new(ElementType.Ice, 50), 50, projectile: Attach(new ShardbindProj()).Projectile), 50,
                "independent ice blast deals damage");
            Check(target.HasBuff<ElementalAffliction_Freeze>(), $"independent blast freezes Wet target after {gap} ticks");
        }
        var shared = NpcFixture(180); var part = NpcFixture(181,180); var other = NpcFixture(182);
        var attack = Attach(new IceConeProj()).Projectile;
        foreach (var type in new[] { ElementType.Water, ElementType.Ice, ElementType.Lightning, ElementType.Fire }) {
            var infusion = new ElementInfusion(type, 5);
            Hit(shared, infusion, 0, projectile: attack);
            Hit(part, infusion, 1, projectile: attack);
            Clock(GameMain.GameUpdateCount + 6);
            Near(Hit(shared, infusion, 1, projectile: attack), 1, "later body hits still deal damage");
            Hit(other, infusion, 1, projectile: attack);
            foreach (var npc in new[] { shared, other })
                Near(npc.GetGlobalNPC<ElementalGlobalNPC>().GetCurrentAccumulation(type) ?? 0, 5, $"one {type} attachment per attack and anchor");
        }
        attack.GetGlobalProjectile<ElementalGlobalProjectile>().BeginBuildupBatch();
        Hit(part, new(ElementType.Lightning,5), 1, projectile: attack);
        Near(shared.GetGlobalNPC<ElementalGlobalNPC>().GetCurrentAccumulation(ElementType.Lightning) ?? 0, 10, "new pulse can attach in same tick");
        shared = NpcFixture(180);
        Hit(shared, new(ElementType.Lightning,5), 1, projectile: attack);
        Near(shared.GetGlobalNPC<ElementalGlobalNPC>().GetCurrentAccumulation(ElementType.Lightning) ?? 0, 5, "slot reused by a new NPC can attach");
        foreach (var scenario in new[] { (Amount: 50f, Initial: 0f), (Amount: 30f, Initial: 0f), (Amount: 30f, Initial: 99f),
            (Amount: 7.5f, Initial: 0f), (Amount: 0f, Initial: 99f) }) {
            var wet = NpcFixture(180); var body = NpcFixture(181, 180); var dry = NpcFixture(182);
            wet.AddBuff(ModContent.BuffType<ElementalAffliction_Wet>(), 1800);
            body.AddBuff(ModContent.BuffType<ElementalAffliction_Wet>(), 1800);
            var state = wet.GetGlobalNPC<ElementalGlobalNPC>();
            state.SetCurrentAccumulation(ElementType.Ice, scenario.Initial);
            var sequence = Attach(new ShardbindProj());
            sequence.Projectile.damage = 100; sequence.Projectile.timeLeft = ShardbindVisualSettings.ControllerLifetime;
            sequence.BindTarget(wet, wet.GetGlobalNPC<IceboundNPC>().SpawnSerial);
            var element = sequence.Projectile.GetGlobalProjectile<ElementalGlobalProjectile>();
            element.InfusionElement = ElementType.Ice;
            element.InfusionElement = new(ElementType.Ice, scenario.Amount);
            int pulses = 0;
            void Damage(On_Projectile.orig_Damage orig, Projectile projectile) {
                if (projectile != sequence.Projectile) { orig(projectile); return; }
                pulses++;
                bool final = pulses == 16;
                Near(element.InfusionElement.Buildup, final ? scenario.Amount : 0, "only final blast exposes configured buildup to global hit hook");
                foreach (var target in new[] { wet, body, dry }) {
                    if (sequence.CanHitNPC(target) == false) continue;
                    var modifiers = new NPC.HitModifiers { SourceDamage = StatModifier.Default };
                    sequence.ModifyHitNPC(target, ref modifiers);
                    element.ModifyHitNPC(projectile, target, ref modifiers);
                    Near(modifiers.SourceDamage.ApplyTo(100), target == wet ? 150 : 100,
                        "all shard and final damage use the status before this hit, including freeze-triggering blast");
                    sequence.OnHitNPC(target, default, 1);
                    Hit(target,element.InfusionElement,1,projectile:projectile);
                }
                Check(sequence.CanHitNPC(body) == false, "shared-life body cannot duplicate the pulse");
                if (!final) {
                    Near(state.GetCurrentAccumulation(ElementType.Ice) ?? 0, scenario.Initial, "15 early impacts leave ice progress unchanged");
                    Check(WaterExposureNPC.IsWet(wet) && !wet.HasBuff<ElementalAffliction_Freeze>(), "early impacts retain Wet");
                }
            }
            On_Projectile.Damage += Damage;
            try {
                for (int frame = 1; frame <= IceCycleBalance.LockBurstFrame + 1; frame++) {
                    Clock((uint)(21000 + frame));
                    sequence.AI();
                    Near(element.InfusionElement.Buildup, scenario.Amount, "pulse restores cast-time buildup configuration");
                }
            }
            finally { On_Projectile.Damage -= Damage; }
            Check(pulses == 16, "one cast has 15 shards and one final blast");
            bool freezes = scenario.Initial + scenario.Amount * 2 >= 100;
            Check(wet.HasBuff<ElementalAffliction_Freeze>() == freezes && WaterExposureNPC.IsWet(wet) != freezes,
                "only final hit may freeze and consume Wet");
            if (!freezes) Near(state.GetCurrentAccumulation(ElementType.Ice) ?? 0, scenario.Initial + scenario.Amount * 2,
                "wet target receives configured buildup exactly once");
            Near(dry.GetGlobalNPC<ElementalGlobalNPC>().GetCurrentAccumulation(ElementType.Ice) ?? 0, scenario.Amount,
                "independent dry splash target receives full fixed buildup once");
        }
    }

    static void CheckMultipartHealthBars()
    {
        ElementalGlobalNPC State(NPC npc) => npc.GetGlobalNPC<ElementalGlobalNPC>();
        ElementalGlobalNPC.BuildUpProgressContext Display(NPC npc) =>
            (ElementalGlobalNPC.BuildUpProgressContext)typeof(ElementalGlobalNPC)
                .GetMethod("GetIceHealthBarProgress", All)!.Invoke(State(npc), [npc])!;
        var snapshots = new List<object>();
        void Capture(string name, NPC npc) {
            var p = Display(npc);
            snapshots.Add(new { Name=name, p.Current, p.Max, p.VisualTime, p.InAffliction,
                p.Duration, p.MaxDuration, p.UpdateCounter, Cooldown=npc.GetFreezeCooldown() });
        }

        var core = NpcFixture(150); core.type = NPCID.MoonLordCore; core.dontTakeDamage = true;
        var left = NpcFixture(151); left.type = NPCID.MoonLordHand; left.ai[3] = 150;
        var right = NpcFixture(152); right.type = NPCID.MoonLordHand; right.ai[3] = 150;
        var head = NpcFixture(153); head.type = NPCID.MoonLordHead; head.ai[3] = 150;
        var otherCore = NpcFixture(154); otherCore.type = NPCID.MoonLordCore;
        var otherHead = NpcFixture(155); otherHead.type = NPCID.MoonLordHead; otherHead.ai[3] = 154;
        foreach (NPC target in new[] { left, right, head }) {
            Clock((uint)(20000 + target.whoAmI * 2));
            State(target).OnHitByElement(ElementType.Ice, target,
                new NPC.HitInfo { Damage=1, SourceDamage=1 }, 0, 20);
        }
        Near(Display(core).Current, 60, "head and both hands accumulate on hidden Moon Lord core");
        foreach (NPC part in new[] { left, right, head }) {
            Check(ReferenceEquals(Display(part), Display(core)), "each Moon Lord part reads shared ice progress");
            Check(Display(part).VisualTime > 0, "unfrozen parts display buildup even while core is invulnerable");
            Check(!State(part).ElementAccumulation.ContainsKey(ElementType.Ice), "display does not copy state into parts");
        }
        Near(Display(otherHead).Current, 0, "second Moon Lord has independent buildup");
        Capture("ice-buildup", head);

        var shared = Display(core);
        shared.InAffliction = true; shared.Current = 0; shared.Duration = 180; shared.MaxDuration = 240;
        shared.UpdateCounter = 11;
        foreach (NPC part in new[] { left, right, head }) {
            Check(Display(part).InAffliction, "every part displays frozen state");
            Near(Display(part).Duration, 180, "every part displays same remaining freeze duration");
        }
        Capture("ice-frozen", head);
        shared.InAffliction = false; shared.Current = 0; shared.VisualTime = 120;
        FreezeCD(core, 900);
        foreach (NPC part in new[] { left, right, head }) Near(part.GetFreezeCooldown(), 900, "every part shares frost cooldown");
        Capture("ice-cooldown", head);

        // Exercise actual extra-AI deserialization: display must follow the replacement context, not cache old state.
        shared.Current = 45; shared.VisualTime = 0;
        using (var bytes = new MemoryStream()) {
            State(core).SendExtraAI(core, new Terraria.ModLoader.IO.BitWriter(), new BinaryWriter(bytes));
            bytes.Position = 0;
            GameMain.netMode = NetmodeID.MultiplayerClient;
            State(core).ReceiveExtraAI(core, new Terraria.ModLoader.IO.BitReader(new BinaryReader(new MemoryStream(new byte[1]))), new BinaryReader(bytes));
            Check(!ReferenceEquals(shared, Display(head)), "client display follows newly received core context");
            Near(Display(head).Current, 45, "client head reads synchronized accumulation");
            Check(Display(head).VisualTime > 0, "first received buildup is immediately visible on client head");
            Near(head.GetFreezeCooldown(), 900, "client head reads synchronized cooldown");
            GameMain.netMode = NetmodeID.SinglePlayer;
        }

        foreach (var group in new[] {
            (NPCID.SkeletronHead, NPCID.SkeletronHand, 1),
            (NPCID.SkeletronPrime, NPCID.PrimeLaser, 1),
            (NPCID.Golem, NPCID.GolemFistLeft, -1),
            (NPCID.MartianSaucerCore, NPCID.MartianSaucerCannon, 0),
            (NPCID.PirateShip, NPCID.PirateShipCannon, 0) }) {
            var parent = NpcFixture(156); parent.type = group.Item1;
            var part = NpcFixture(157); part.type = group.Item2;
            int previousGolem = NPC.golemBoss;
            if (group.Item3 >= 0) part.ai[group.Item3] = parent.whoAmI;
            else NPC.golemBoss = parent.whoAmI;
            State(parent).SetCurrentAccumulation(ElementType.Ice, 37);
            Near(Display(part).Current, 37, "other multipart boss reads shared ice buildup");
            NPC.golemBoss = previousGolem;
        }
        var ordinary = NpcFixture(158); State(ordinary).SetCurrentAccumulation(ElementType.Ice, 12);
        Near(Display(ordinary).Current, 12, "ordinary NPC keeps local progress");
        var worm = NpcFixture(159); var segment = NpcFixture(160,159);
        State(worm).SetCurrentAccumulation(ElementType.Ice, 27);
        Near(Display(segment).Current, 27, "worm keeps existing shared-root state");
        shared = Display(core); shared.Current = 0; shared.VisualTime = 0; FreezeCD(core, 0);
        foreach (NPC part in new[] { left, right, head }) Near(Display(part).VisualTime, 0, "clearing root hides every part");
        var output = Path.Combine(rootDirectory, ".vissandbox/ice-cycle-checks");
        Directory.CreateDirectory(output);
        File.WriteAllText(Path.Combine(output, "ice-progress-snapshots.json"), System.Text.Json.JsonSerializer.Serialize(snapshots));
    }

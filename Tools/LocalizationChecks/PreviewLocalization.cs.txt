using System.Text.Json;
using Hjson;
using System.Reflection;
using System.Text.RegularExpressions;

namespace Terraria.Localization;

// Load the real HJSON resources in isolated FNA hosts, without booting Terraria.
public static class Language
{
    public static readonly Dictionary<string, string> Text = [];
    public static string Culture { get; private set; } = "zh-Hans";
    public static bool Exists(string key) => Text.ContainsKey(key);
    public static string GetTextValue(string key, params object[] args)
    {
        string value = Text.GetValueOrDefault(key, key);
        // Mirror ordinary {$Key} string references used by game localization (without format-index offsets).
        for (int depth = 0; depth < 16; depth++)
        {
            string resolved = Regex.Replace(value, @"\{\$([\w.]+)\}", match =>
                Text.GetValueOrDefault(match.Groups[1].Value, match.Value));
            if (resolved == value) break;
            value = resolved;
        }
        return args.Length == 0 ? value : string.Format(value, args);
    }
    public static LocalizedText GetText(string key) => new(GetTextValue(key));
    public static void Load(string repo, string culture, bool uiOnly = false, string tml = null)
    {
        Culture = culture;
        if (tml != null)
        {
            // Read the installed game's strings, not a checked-in translation from a particular version.
            Assembly game = Assembly.LoadFrom(Path.Combine(tml, "tModLoader.dll"));
            string prefix = "Terraria.Localization.Content." + culture.Replace('-', '_') + ".";
            string[] resources = game.GetManifestResourceNames().Where(name => name.StartsWith(prefix, StringComparison.Ordinal)
                && name.EndsWith(".json", StringComparison.Ordinal)).ToArray();
            if (resources.Length == 0) throw new InvalidOperationException("No installed game localization for " + culture);
            foreach (string resource in resources)
            {
                using var stream = game.GetManifestResourceStream(resource);
                using var vanilla = JsonDocument.Parse(stream, new JsonDocumentOptions {
                    CommentHandling = JsonCommentHandling.Skip, AllowTrailingCommas = true });
                Flatten(vanilla.RootElement, "");
            }
            Console.WriteLine($"Installed game localization: {culture}, ItemName.Waterleaf = {GetTextValue("ItemName.Waterleaf")}");
        }
        foreach (string file in Directory.GetFiles(Path.Combine(repo, "Localization"), culture + "_*.hjson"))
        {
            if (uiOnly && !file.EndsWith(".UI.hjson")) continue;
            string prefix = Path.GetFileNameWithoutExtension(file)[(culture.Length + 1)..];
            using var doc = JsonDocument.Parse(HjsonValue.Parse(File.ReadAllText(file)).ToString(Stringify.Plain));
            Flatten(doc.RootElement, prefix);
        }
    }
    private static void Flatten(JsonElement node, string prefix)
    {
        if (node.ValueKind == JsonValueKind.Object)
            foreach (var child in node.EnumerateObject()) Flatten(child.Value, prefix.Length == 0 ? child.Name : prefix + "." + child.Name);
        else if (node.ValueKind == JsonValueKind.String) Text[prefix] = node.GetString();
    }
}

public sealed record LocalizedText(string Value)
{
    public LocalizedText WithFormatArgs(params object[] args) => new(string.Format(Value, args));
}

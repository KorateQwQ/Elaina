using KL.Drawing.Snippets;
using 伊蕾娜.ElainaModSkills.ElainaSkillUI.ConstellationSkillPanel;
using static 伊蕾娜.ElainaModSkills.ElainaSkillUI.ConstellationSkillPanel.ConstellationDrawing;
internal sealed class PanelScene
{
    private ConstellationState _state;
    private ConstellationDetail _detail;
    private readonly ConstellationDrawing _draw = new(new Terraria.ModLoader.Mod());
    private readonly ConstellationButterflyMotion _butterfly = new();
    private readonly ConstellationNodeEffects _nodeFx = new();
    private readonly ConstellationSky _sky = new();
    private readonly ConstellationBookMotion _book = new();
    private const float DesignWidth = 1100, DesignHeight = 800;
    private const int MapX = 26, MapY = 208, MapWidth = 727, MapHeight = 473, WorldWidth = 727, WorldHeight = 550;
    private static readonly Rectangle Sky = new(26, 150, 727, 558);
    // Offline frames advance the sky 7.3s: stars mid-breath, a meteor may be crossing.
    internal static float SkySeconds = 7.3f;
    private const int SlotLeft = 99, SlotTop = 741, SlotSize = 42, SlotStride = 48;
    private static readonly string[] SlotKeys = ConstellationState.SlotLabels;
    private float _zoom = .84f, _flash;
    private float _clock => SkySeconds;
    private Vector2 _pan;
    private bool _focusing;
    private float _focusElapsed, _focusStartZoom, _focusTargetZoom;
    private Vector2 _focusStartPan, _focusTargetPan;
    private string _hoverNode, _tooltip;
    private string _tooltipKey;
    private Rectangle? _tooltipAnchor;
    private KLTerm _tooltipTerm;
    private int _slotPress = -1, _slotDropTarget = -1;
    private bool _slotDragging;
    private HashSet<string> _ancestors;
    private ConstellationLayoutEditor _layoutEditor;
    private bool Editing => _layoutEditor?.IsEditing == true;
    // The fixture has no pointer: Main.MouseScreen is off-screen, so nothing is hovered unless a scene sets one.
    private Vector2? _pointer;
    private bool IsMouseHovering => _pointer != null;
    private Vector2 Design(Vector2 mouse) => _pointer ?? mouse;
    // Stands in for the production 伊蕾娜 WaterSpellbook, linked to its term in Program.
    internal const int SpellbookItem = 5000;
    internal void RenderPrimaryButton(SpriteBatch batch, float scale, string mode, bool hover, bool pressed, float seconds)
    {
        if (_state == null)
        {
            _state = Fixtures.State(mode is "equip" or "equipped" or "maxed" or "maxed-equipped");
            _state.Select(nameof(MagicMissileSkill));
            _state.Current.ModSkill.UnlockCondition = mode is "paid" or "missing"
                ? KL.SkillSystem.SkillUnlockCondition.ByItemsAndSkillPoint(20, new KL.SkillSystem.SkillUnlockItem(9, 10))
                : KL.SkillSystem.SkillUnlockCondition.None;
            var player = 伊蕾娜.ElainaModSkills.ElainaSkillModPlayer.SkillModPlayer;
            player.SkillPoint = 100;
            player.Player.inventory[0].type = 9; player.Player.inventory[0].stack = mode == "missing" ? 0 : 10;
            _state.TargetSlot = 3;
            if (mode is "equipped" or "maxed-equipped") _state.Equip();
            if (mode is "maxed" or "maxed-equipped")
                while (_state.Upgrade(_state.Selected)) { }
        }
        _draw.Batch = batch; _draw.Scale = scale; _draw.Origin = new Vector2(-756, -650) * scale;
        _draw.AdvanceToggleAnimations(seconds);
        Terraria.Main.mouseLeft = pressed;
        batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.LinearClamp, DepthStencilState.None, RasterizerState.CullNone);
        _draw.Image("NotebookSurface", 0, 0, 1100, 800);
        DrawActionButton(hover);
        batch.End();
        Terraria.Main.mouseLeft = false;
    }
    internal void RenderDebugTools(SpriteBatch batch, float scale, bool editing, string hovered)
    {
        _state = Fixtures.State();
        _layoutEditor = new(_state, new(), new());
        if (editing) _layoutEditor.Begin();
        _draw.Batch = batch; _draw.Scale = scale; _draw.Origin = new Vector2(-510, -724) * scale;
        batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.LinearClamp, DepthStencilState.None, RasterizerState.CullNone);
        _draw.Image("NotebookSurface", 0, 0, 1100, 800);
        foreach (string kind in new[] { "edit", "save", "cancel", "reset" }) DrawLayoutButton(kind, kind == hovered);
        foreach (string kind in new[] { "points", "level", "max-level", "cap", "master-skills", "reset-skills" }) DrawDebugButton(kind, kind == hovered);
        batch.End();
    }

    private int _playerLevel = 120, _playerCap = 120, _playerExperience;
    internal void RenderHeader(SpriteBatch batch, float scale, int points, int level = 120, int cap = 120, int experience = 0)
    {
        _state = Fixtures.State(true);
        _playerLevel = level; _playerCap = cap; _playerExperience = experience;
        伊蕾娜.ElainaModSkills.ElainaSkillModPlayer.SkillModPlayer.SkillPoint = points;
        _draw.Batch = batch; _draw.Scale = scale; _draw.Origin = Vector2.Zero;
        batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.LinearClamp, DepthStencilState.None, RasterizerState.CullNone);
        DrawChrome(); DrawCloseButton(false);
        string[] filters = ["learned", "ready", "locked", "hidden"];
        for (int i = 0; i < filters.Length; i++) DrawFilter(i, filters[i], false);
        DrawRuleOrnaments();
        batch.End();
    }

    // The panel delegates the header layout to ConstellationDrawing; this fixture supplies
    // the reported high-level state so the preview covers the former overlap case.
    private void DrawPlayerLevel() => _draw.PlayerLevel(_playerLevel, _playerCap, _playerExperience, 100);

    internal void Render(SpriteBatch batch, int width, int height, float uiScale, string scenario,
        Func<ConstellationDetail, float> scrollPosition = null)
    {
        _book.Request(true); _book.Finish();
        _state = Fixtures.State(scenario is not ("new" or "bottom" or "requirements-learn" or "requirements-long" or "requirements-term"));
        if (scenario is "active" or "equipped" or "maxed")
        {
            _state.Select(nameof(MagicMissileSkill));
            if (scenario == "equipped") { _state.TargetSlot = 7; _state.Equip(); }
            if (scenario == "maxed") for (int i = 0; i < 10; i++) _state.Current.ModSkill.TryLevelUp();
        }
        if (scenario == "missing") _state.Select(nameof(WaterLaserSKill));
        if (scenario == "off")
        {
            _state.Select(nameof(MagicBarrierSkill)); _state.Unlock(); _state.Toggle(nameof(AshenWitchSkill));
        }
        if (scenario == "empty") _state.SelectFilter("hidden");
        AshenWitchSkill.StatsPreview = scenario == "passive-stats";
        if (scenario == "passive-stats") _state.Select(nameof(AshenWitchSkill));
        if (scenario.StartsWith("recall-"))
        {
            var player = Fixtures.Player(typeof(伊蕾娜.ElainaModSkills.Skills.Recall.RecallSkill));
            _state = new(player);
            player.SkillPoint = 50;
            if (scenario != "recall-sealed")
            {
                player.Player.inventory[0] = new Item { type = Terraria.ID.ItemID.RecallPotion, stack = 1 };
                player.Player.GetModPlayer<伊蕾娜.ElainaModSkills.Skills.Recall.RecallDiscoveryPlayer>().PostUpdate();
                if (scenario == "recall-revealed-missing") player.Player.inventory[0].TurnToAir();
            }
            _state.Select(typeof(伊蕾娜.ElainaModSkills.Skills.Recall.RecallSkill).FullName);
            if (scenario == "recall-learned") _state.Unlock();
            _state.Current.ModSkill.MagicPointCost = 200;
            _state.Current.Skill.MaxCD = 10;
        }
        if (scenario.StartsWith("requirements"))
        {
            _state.Select(nameof(MagicMissileSkill));
            var player = 伊蕾娜.ElainaModSkills.ElainaSkillModPlayer.SkillModPlayer;
            _state.Current.ModSkill.UnlockCondition = KL.SkillSystem.SkillUnlockCondition.ByItemsAndSkillPoint(20,
                scenario == "requirements-term" ? new KL.SkillSystem.SkillUnlockItem(SpellbookItem, 1)
                    : new KL.SkillSystem.SkillUnlockItem(scenario == "requirements-long" ? 2 : 9, 10));
            player.SkillPoint = scenario == "requirements-ready" ? 20 : 4;
            player.Player.inventory[0].type = 9;
            player.Player.inventory[0].stack = scenario == "requirements-ready" ? 10 : 3;
        }
        if (scenario.StartsWith("growth"))
        {
            MagicMissileSkill.GrowthPreview = true;
            MagicMissileSkill.LongDescription = scenario == "growth-long";
            var player = 伊蕾娜.ElainaModSkills.ElainaSkillModPlayer.SkillModPlayer;
            player.UnlockSkill(KL.SkillSystem.Skill.NewSkill(typeof(MultiMissileSkill)));
            player.SkillPoint = scenario == "growth-poor" ? 0 : 10;
            _state.Refresh(); _state.Select(nameof(MagicMissileSkill));
            _state.Current.ModSkill.MagicPointCost = 18;
            _state.Current.Skill.MaxCD = 8;
            if (scenario == "growth-maxed") for (int i = 0; i < 10; i++) _state.Current.ModSkill.TryLevelUp();
        }
        if (scenario.StartsWith("forget"))
        {
            var player = 伊蕾娜.ElainaModSkills.ElainaSkillModPlayer.SkillModPlayer;
            player.Player.inventory[49] = new Item { type = 伊蕾娜.ElainaModSkills.SkillForgetting.BookType, stack = 2 };
            _state.Select(nameof(MagicMissileSkill));
            player.SkillPoint = scenario == "forget-lv1" ? 0 : 40;
            if (scenario != "forget-lv1" && !(_state.Upgrade(_state.Selected) && _state.Upgrade(_state.Selected)))
                throw new Exception("Forget preview could not reach Lv.3.");
            if (scenario == "forget-armed") _state.ForgetArmed = _state.Selected;
            if (_state.CanForget(_state.Selected) != (scenario != "forget-lv1")) throw new Exception("Forget preview availability mismatch.");
        }
        if (scenario.StartsWith("terms-"))
        {
            var type = scenario == "terms-freeze" ? typeof(IceConeSkill)
                : scenario == "terms-cauldron" ? typeof(伊蕾娜.ElainaModSkills.Skills.Alchemy.AlchemySkill) : typeof(AshenWitchSkill);
            var player = Fixtures.Types.Contains(type) ? Fixtures.Player() : Fixtures.Player(type);
            player.UnlockSkill(KL.SkillSystem.Skill.NewSkill(type));
            _state = new(player); _state.Select(type.FullName);
        }
        if (scenario.StartsWith("alchemy-"))
        {
            var type = typeof(伊蕾娜.ElainaModSkills.Skills.Alchemy.AlchemySkill);
            var player = Fixtures.Player(type);
            player.SkillPoint = 100;
            var skill = KL.SkillSystem.Skill.NewSkill(type);
            player.UnlockSkill(skill);
            skill.Level = scenario == "alchemy-max" ? 8 : 5;
            var progress = player.Player.GetModPlayer<伊蕾娜.ElainaModAlchemy.Gameplay.AlchemyProgressPlayer>();
            progress.Level = scenario == "alchemy-max" ? 8 : 3;
            progress.MaximumLevel = skill.Level;
            _state = new(player); _state.Select(type.FullName);
        }
        if (scenario.StartsWith("healing-"))
        {
            var player = Fixtures.Player();
            var character = player.Player.GetModPlayer<伊蕾娜.System.ElainaStatePlayer>();
            character.CurrentLevel = character.Cap = 95;
            player.SkillPoint = 100;
            if (scenario != "healing-new")
            {
                var skill = KL.SkillSystem.Skill.NewSkill(typeof(HealSkill));
                player.UnlockSkill(skill);
                skill.Level = scenario == "healing-level8" ? 8 : 1;
            }
            _state = new(player);
            _state.Select(nameof(HealSkill));
            var entry = 伊蕾娜.ElainaModSkills.SkillBalanceSystem.Entries[nameof(HealSkill)];
            _state.Current.ModSkill.MagicPointCost = (int)entry.ManaCost;
            _state.Current.Skill.MaxCD = (float)entry.BaseCooldown;
            string description = _state.Current.ModSkill.GetConstellationDescription();
            if (description.Contains("{0}") || description.Contains("{1}") || description.Contains("{2}")
                || !description.Contains(scenario == "healing-level8" ? "240" : "100"))
                throw new Exception("Healing description does not show the actual healing tier.");
        }
        _layoutEditor = new(_state, new(), new());
        if (scenario.StartsWith("editor"))
        {
            _layoutEditor.Begin();
            if (scenario == "editor-moved")
                _layoutEditor.Move(nameof(MagicMissileSkill), new Vector2(285, 80), new Vector2(120, 70), _zoom, true);
            _state.Select(nameof(MagicMissileSkill));
        }
        _ancestors = _state.Ancestors();
        _pan = new Vector2(MapWidth / 2f, MapHeight / 2f) - new Vector2(_state.Current.x, _state.Current.y) * _zoom;
        if (scenario is "new" or "passive" or "bottom") _pan = new Vector2((MapWidth - _state.WorldSize.X * _zoom) / 2, 30);
        if (Editing) _pan = new Vector2((MapWidth - _state.WorldSize.X * _zoom) / 2, 30);
        if (scenario.StartsWith("camera-"))
        {
            _pan = new Vector2((MapWidth - _state.WorldSize.X * _zoom) / 2, 30);
            _state.Select(nameof(MagicMissileSkill));
            CenterOn(_state.Current);
            bool retarget = scenario == "camera-retarget";
            float seconds = retarget ? .1f : int.Parse(scenario[7..]) / 1000f;
            AdvanceCamera(seconds);
            if (retarget)
            {
                Vector2 previousPan = _pan;
                float previousZoom = _zoom;
                _state.Select(nameof(WaterLaserSKill));
                CenterOn(_state.Current);
                AdvanceCamera(0);
                if (_pan != previousPan || _zoom != previousZoom) throw new Exception("Camera jumped when retargeting.");
                for (int i = 0; i < 60; i++) AdvanceCamera(1f / 144);
            }
            if (retarget || seconds >= .35f)
            {
                Vector2 center = new(MapX + MapWidth / 2f, MapY + MapHeight / 2f);
                if (Vector2.Distance(MapPoint(_state.Current.x, _state.Current.y), center) > .01f || Math.Abs(_zoom - 1.2f) > .001f)
                    throw new Exception("Selected skill did not finish centered at the focus zoom.");
            }
            _ancestors = _state.Ancestors();
        }
        UpdateButterfly(0);
        float fit = Math.Min(1, Math.Min((width / uiScale - 32) / DesignWidth, (height / uiScale - 32) / DesignHeight));
        float scale = fit * uiScale;
        Vector2 origin = (new Vector2(width, height) - new Vector2(DesignWidth, DesignHeight) * scale) / 2;
        _draw.Batch = batch; _draw.Scale = scale; _draw.Origin = origin;
        var detail = new ConstellationDetail(_state, _draw);
        _detail = detail;
        // Hover the first requirement row (the term item), as the pointer would in the panel.
        _pointer = scenario == "requirements-term" ? new Vector2(ConstellationDetail.Left + 30, detail.FooterTop + 13 + 9 + 9) : null;
        batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.LinearClamp, DepthStencilState.None, RasterizerState.CullNone);
        DrawChrome(); DrawDetails(); DrawActionButton(false); DrawCloseButton(false);
        string[] filters = ["learned", "ready", "locked", "hidden"];
        for (int i = 0; i < 4; i++) DrawFilter(i, filters[i], false);
        for (int i = 0; i < 8; i++) DrawSlot(i, false);
        foreach (string kind in new[] { "edit", "save", "cancel", "reset" }) DrawLayoutButton(kind, false);
        foreach (string kind in new[] { "points", "level", "max-level", "cap", "master-skills", "reset-skills" }) DrawDebugButton(kind, false);
        batch.End();
        using var clip = new RasterizerState { ScissorTestEnable = true, CullMode = CullMode.None };
        Rectangle Clip(float x, float y, float w, float h) => new((int)(origin.X + x * scale), (int)(origin.Y + y * scale), (int)(w * scale), (int)(h * scale));
        batch.GraphicsDevice.ScissorRectangle = Clip(Sky.X, Sky.Y, Sky.Width, Sky.Height);
        batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.LinearClamp, DepthStencilState.None, clip);
        _sky.Reset();
        for (float t = 0; t < SkySeconds; t += 1 / 60f) _sky.Advance(1 / 60f);
        DrawMap(); batch.End();
        // The camera buttons are SUI children above the map viewport, as in the panel.
        batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.LinearClamp, DepthStencilState.None, RasterizerState.CullNone);
        CameraControl("minus", false, true); CameraControl("plus", false, true); CameraControl("center", false, true);
        DrawRuleOrnaments();
        batch.End();
        batch.GraphicsDevice.ScissorRectangle = Clip(ConstellationDetail.Left, detail.Top, ConstellationDetail.Width, detail.Height);
        batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.LinearClamp, DepthStencilState.None, clip);
        float offset = scrollPosition?.Invoke(detail) ??
            (scenario is "bottom" or "growth-long" or "healing-growth" or "alchemy-growth" or "alchemy-max" or "forget" or "forget-armed" or "forget-lv1" ? Math.Max(0, detail.ContentHeight - detail.Height)
                : scenario == "healing-description" ? 120 : 0);
        float y = detail.Top - offset;
        // Term scenes hover the first registered term, as the panel does under the pointer.
        (ConstellationDetailAction Action, float Top)? term = null;
        foreach (var row in detail.Rows)
        {
            var action = scenario.StartsWith("terms-") && term == null ? row.Actions.FirstOrDefault(a => a.Kind == "term") : null;
            if (action != null) term = (action, y);
            _draw.Origin = origin + new Vector2(ConstellationDetail.Left, y) * scale;
            row.Paint(_draw, action == null ? null : "term:" + action.Id);
            y += row.Height;
        }
        batch.End(); _draw.Origin = origin;
        if (scenario.StartsWith("terms-") && term == null) throw new Exception($"{scenario}: no registered term in the description.");
        batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.LinearClamp, DepthStencilState.None, RasterizerState.CullNone);
        detail.DrawScrollHints(_draw, offset, 0);
        if (term is var (termAction, termTop))
        {
            var pointer = new Vector2(ConstellationDetail.Left + termAction.Area.Center.X, termTop + termAction.Area.Center.Y);
            var termAnchor = ConstellationDetail.TooltipAnchor(termAction, new Vector2(ConstellationDetail.Left, termTop));
            _draw.TermTooltip(KLTerms.Find(termAction.Id), pointer, new Vector2(DesignWidth, DesignHeight), termAnchor);
        }
        if (scenario == "requirements-term")
        {
            if (_tooltipTerm == null) throw new Exception("requirements-term: the requirement term is not hovered.");
            _draw.TermTooltip(_tooltipTerm, _pointer.Value, new Vector2(DesignWidth, DesignHeight), _tooltipAnchor);
        }
        if (scenario == "requirements-tooltip")
        {
            var area = ConstellationDetail.PillArea(ConstellationDetail.Pills(_state)[0]);
            _draw.Tooltip(_state.Requirements(_state.Selected, true).ActionTooltip("进修至 2 级"), new Vector2(area.Center.X, area.Y + 15),
                new Vector2(DesignWidth, DesignHeight), area);
        }
        batch.End();
        Console.WriteLine($"{scenario}: detail {detail.ContentHeight:0}/{detail.Height:0}, scale {scale:0.00}");
    }

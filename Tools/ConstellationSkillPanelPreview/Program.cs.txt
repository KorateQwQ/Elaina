using SDL2;
using ReLogic.Content;
internal sealed class PanelGame : Game
{
    internal static string Repo;
    private static string VanillaContent;
    private static GraphicsDevice Device;
    private static Microsoft.Xna.Framework.Content.ContentManager Vanilla;
    private static readonly Dictionary<string, Texture2D> Textures = [];
    private bool _done;
    private static bool IconsOnly;
    // Renders without the logic / icon checks (for layout work while an unrelated check is stale).
    private static bool SkipChecks;
    private static bool HeadersOnly;
    private static bool LocalizationOnly;
    private static bool RequirementsOnly;
    private static bool ToggleOnly;
    private static bool PolishOnly;
    private static bool PrimaryOnly;
    private static bool BookOnly;
    private static bool CameraOnly;
    private static bool OrnamentsOnly;
    private static bool ScrollOnly;
    private static bool DebugToolsOnly;
    private static bool RecallResearchOnly;
    private static bool HealingOnly;
    private static bool AlchemyOnly;
    private static bool DiscoveryOnly;
    private static bool TermsOnly;
    private static bool ForgetOnly;
    private PanelGame()
    {
        new GraphicsDeviceManager(this) { PreferredBackBufferWidth = 320, PreferredBackBufferHeight = 180, GraphicsProfile = GraphicsProfile.HiDef, SynchronizeWithVerticalRetrace = false };
        SDL.SDL_HideWindow(Window.Handle);
        IsFixedTimeStep = false;
    }
    protected override void LoadContent()
    {
        Device = GraphicsDevice;
        Content.RootDirectory = Path.GetFullPath(Path.Combine(Repo, "../KL/Fonts"));
        KL.Drawing.FontManager.HarmonyOS_Sans_SC = new(Content.Load<DynamicSpriteFont>("HarmonyOS_Sans_SC"));
        KL.Drawing.FontManager.NotoSerifSC = new(Content.Load<DynamicSpriteFont>("NotoSerifSC"));
        KL.Drawing.FontManager.Gelasio = new(Content.Load<DynamicSpriteFont>("Gelasio"));
        Content.RootDirectory = VanillaContent;
        Vanilla = Content;
        Terraria.GameContent.FontAssets.MouseText = new(Content.Load<DynamicSpriteFont>("Fonts/Mouse_Text"));
        var pixel = new Texture2D(Device, 1, 1); pixel.SetData(new[] { Color.White });
        Terraria.GameContent.TextureAssets.MagicPixel = new(pixel);
    }
    protected override void Update(GameTime time)
    {
        if (_done) return; _done = true;
        string output = Path.Combine(Repo, ".vissandbox/constellation-skill-panel/output", Terraria.Localization.Language.Culture); Directory.CreateDirectory(output);
        using var batch = new SpriteBatch(Device);
        if (DebugToolsOnly)
        {
#if DEBUG
            foreach (float scale in new[] { .86f, 1f, 1.25f, 2f })
            foreach (string scenario in new[] { "normal", "max-level", "master-skills", "editing" })
            {
                using var target = new RenderTarget2D(Device, (int)Math.Ceiling(450 * scale), (int)Math.Ceiling(76 * scale));
                Device.SetRenderTarget(target); Device.Clear(new Color(20, 21, 28));
                new PanelScene().RenderDebugTools(batch, scale, scenario == "editing", scenario);
                Device.SetRenderTarget(null);
                using var stream = File.Create(Path.Combine(output, $"debug-tools-{scenario}-{scale:0.00}.png"));
                target.SaveAsPng(stream, target.Width, target.Height);
            }
#endif
            Console.WriteLine(output); Exit(); return;
        }
        if (DiscoveryOnly) { DiscoveryPreview.Run(Device, batch, output); Exit(); return; }
        if (ScrollOnly) { ScrollPreview.Run(Device, batch, output); Exit(); return; }
        if (OrnamentsOnly) OrnamentPreview.Run(Device, batch, output);
        if (BookOnly) { BookPreview.Run(Device, batch, output); Exit(); return; }
        if (ToggleOnly) { TogglePreview.Run(Device, batch, output); Exit(); return; }
        if (HeadersOnly)
        {
            foreach (float scale in new[] { .86f, 1f, 1.25f })
            foreach (int points in new[] { 0, 9999 })
            foreach (var (level, cap, experience) in new[] { (1, 10, 0), (24, 30, 65), (99, 120, 99), (120, 120, 0), (120, 120, 100) })
            {
                using var target = new RenderTarget2D(Device, (int)Math.Ceiling(1100 * scale), (int)Math.Ceiling(146 * scale));
                Device.SetRenderTarget(target); Device.Clear(new Color(20, 21, 28));
                new PanelScene().RenderHeader(batch, scale, points, level, cap, experience);
                Device.SetRenderTarget(null);
                using var stream = File.Create(Path.Combine(output, $"study-header-{points}-lv{level}-xp{experience}-{scale:0.00}.png"));
                target.SaveAsPng(stream, target.Width, target.Height);
            }
            Console.WriteLine(output); Exit(); return;
        }
        if (IconsOnly || (!LocalizationOnly && !SkipChecks && !RecallResearchOnly)) IconChecks.Run(batch);
        if (IconsOnly) { Exit(); return; }
        if (PrimaryOnly) { PrimaryButtonPreview.Run(Device, batch, output); Exit(); return; }
        if (PolishOnly) PolishPreview.Run(Device, batch, output);
        string[] scenes = AlchemyOnly ? ["alchemy-stats", "alchemy-growth", "alchemy-max"]
            : ForgetOnly ? ["forget", "forget-armed", "forget-lv1"]
            : TermsOnly ? ["terms-freeze", "terms-magic", "terms-cauldron", "requirements-term"]
            : HealingOnly ? ["healing-new", "healing-level1", "healing-growth", "healing-level8", "healing-description"]
            : RecallResearchOnly ? ["recall-sealed", "recall-revealed-missing", "recall-ready", "recall-learned"]
            : OrnamentsOnly ? ["camera-350", "empty", "editor"]
            : CameraOnly ? ["camera-0", "camera-100", "camera-200", "camera-350", "camera-retarget"]
            : RequirementsOnly || PolishOnly ? ["requirements-learn", "requirements-upgrade", "requirements-ready", "requirements-tooltip", "requirements-long", "requirements-term"]
            : ["new", "passive", "passive-stats", "active", "equipped", "missing", "off", "empty", "bottom", "editor", "editor-moved", "maxed", "growth", "growth-poor", "growth-long", "growth-maxed"];
        foreach (var (w, h, scale) in new[] { (1600, 1000, 1f), (1280, 720, 1f), (1600, 1000, 1.25f) })
        foreach (string scene in scenes)
        {
            using var target = new RenderTarget2D(Device, w, h);
            Device.SetRenderTarget(target); Device.Clear(new Color(20, 21, 28));
            new PanelScene().Render(batch, w, h, scale, scene);
            Device.SetRenderTarget(null);
            using var stream = File.Create(Path.Combine(output, $"{scene}-{w}x{h}-{scale:0.00}.png")); target.SaveAsPng(stream, w, h);
        }
        Console.WriteLine(output); Exit();
    }
    internal static Texture2D Texture(string asset)
    {
        if (Textures.TryGetValue(asset, out var result)) return result;
        // Vanilla item icons (klicon "Item:<name>") come from the game's XNB content.
        if (asset.StartsWith("Terraria/Images/", StringComparison.Ordinal))
            return Vanilla.Load<Texture2D>(asset["Terraria/".Length..]);
        // KL assets (term icons) live in the sibling KL mod; the rest are this repository's.
        string root = asset.StartsWith("KL/", StringComparison.Ordinal) ? Path.Combine(Repo, "../KL") : Repo;
        using var stream = File.OpenRead(Path.Combine(root, asset[(asset.IndexOf('/') + 1)..] + ".png"));
        result = Texture2D.FromStream(Device, stream);
        var pixels = new Color[result.Width * result.Height]; result.GetData(pixels);
        for (int i = 0; i < pixels.Length; i++) if (pixels[i].A == 0) pixels[i] = Color.Transparent;
        result.SetData(pixels); Textures.Add(asset, result); return result;
    }
    protected override void UnloadContent()
    {
        foreach (var texture in Textures.Values) texture.Dispose();
        Terraria.GameContent.TextureAssets.MagicPixel.Value.Dispose();
        base.UnloadContent();
    }
    [STAThread] private static void Main(string[] args)
    {
        Repo = args[0]; VanillaContent = args[1]; IconsOnly = args.Contains("--icons-only");
        HeadersOnly = args.Contains("--headers-only");
        SkipChecks = args.Contains("--skip-checks");
        LocalizationOnly = args.Contains("--localization-only");
        RequirementsOnly = args.Contains("--requirements-only");
        ToggleOnly = args.Contains("--toggle-only");
        PolishOnly = args.Contains("--polish-only");
        PrimaryOnly = args.Contains("--primary-only");
        BookOnly = args.Contains("--book-only");
        CameraOnly = args.Contains("--camera-only");
        OrnamentsOnly = args.Contains("--ornaments-only");
        ScrollOnly = args.Contains("--scroll-only");
        DebugToolsOnly = args.Contains("--debug-tools-only");
        RecallResearchOnly = args.Contains("--recall-research-only");
        HealingOnly = args.Contains("--healing-only");
        AlchemyOnly = args.Contains("--alchemy-only");
        DiscoveryOnly = args.Contains("--discovery-only");
        TermsOnly = args.Contains("--terms-only");
        ForgetOnly = args.Contains("--forget-only");
        if (args.Contains("--book-checks-only")) { BookPreview.CheckMotion(); return; }
        string tml = args.FirstOrDefault(a => a.StartsWith("--tml=", StringComparison.Ordinal))?[6..];
        Terraria.Localization.Language.Load(Repo, "zh-Hans", uiOnly: true, tml: tml);
        Fixtures.Text();
        if (DiscoveryOnly) DiscoveryChecks.Run();
        else if (RecallResearchOnly) RecallResearchChecks.Run();
        else if (DebugToolsOnly) PanelChecks.RunDebugTools();
        else if (!HeadersOnly && !args.Contains("--localization-only") && !SkipChecks) PanelChecks.Run();
        Terraria.Localization.Language.Text.Clear();
        Terraria.Localization.Language.Load(Repo, args[2], tml: tml);
        Terraria.Localization.Language.Load(Path.Combine(Repo, "../KL"), args[2]);
        // The mods register these in KL ElementBuildUpSystem and 伊蕾娜 ElainaTerms; same paths, keys and freeze constants.
        KL.Drawing.Snippets.KLTerms.Register(new("KL/Drawing/Snippets/Icons/IceIcon", new Color(143, 227, 255),
            () => Terraria.Localization.Language.GetTextValue("Mods.KL.Terms.Freeze.Name"),
            () => Terraria.Localization.Language.GetTextValue("Mods.KL.Terms.Freeze.Description", 100f),
            () => [(Terraria.Localization.Language.GetTextValue("Mods.KL.Terms.Freeze.Duration"), "4", Terraria.Localization.Language.GetTextValue("Mods.KL.Terms.Seconds")),
                (Terraria.Localization.Language.GetTextValue("Mods.KL.Terms.Freeze.Cooldown"), "60", Terraria.Localization.Language.GetTextValue("Mods.KL.Terms.Seconds"))],
            () => Terraria.Localization.Language.GetTextValue("Mods.KL.Terms.Categories.Mechanic")));
        KL.Drawing.Snippets.KLTerms.Register(new("伊蕾娜/Snippets/MagicStarIcon", new Color(76, 145, 255),
            () => Terraria.Localization.Language.GetTextValue("Mods.伊蕾娜.Terms.MagicPower.Name"),
            () => Terraria.Localization.Language.GetTextValue("Mods.伊蕾娜.Terms.MagicPower.Description", 10f, 2f, 10f, 2f),
            Category: () => Terraria.Localization.Language.GetTextValue("Mods.KL.Terms.Categories.Mechanic")));
        KL.Drawing.Snippets.KLTerms.Register(new("Item:Cauldron", new Color(232, 194, 122),
            () => Terraria.Localization.Language.GetTextValue("Mods.伊蕾娜.Terms.AlchemyCauldron.Name"),
            () => Terraria.Localization.Language.GetTextValue("Mods.伊蕾娜.Terms.AlchemyCauldron.Description"),
            Category: () => Terraria.Localization.Language.GetTextValue("Mods.KL.Terms.Categories.CraftingStation")));
        KL.Drawing.Snippets.KLTerms.Register(new("伊蕾娜/Snippets/WaterSpellbookIcon", new Color(102, 195, 197),
            () => Terraria.Lang.GetItemNameValue(PanelScene.SpellbookItem),
            () => Terraria.Localization.Language.GetTextValue("Mods.伊蕾娜.Terms.WaterSpellbook.Description"),
            Category: () => Terraria.Localization.Language.GetTextValue("Mods.伊蕾娜.Terms.Categories.Spellbook")));
        KL.Drawing.Snippets.KLTerms.RegisterItem(PanelScene.SpellbookItem, "伊蕾娜/Snippets/WaterSpellbookIcon");
        // The old fixture class uses WindBladeSkill for the production WindSlashSkill.
        Terraria.Localization.Language.Text["Mods.伊蕾娜.SkillInfo.SkillName.WindBladeSkill"] = Terraria.Localization.Language.GetTextValue("Mods.伊蕾娜.SkillInfo.SkillName.WindSlashSkill");
        using var game = new PanelGame(); game.Run();
    }
}

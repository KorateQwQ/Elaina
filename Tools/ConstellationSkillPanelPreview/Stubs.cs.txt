namespace KL.SkillSystem
{
    public abstract class ModSkill
    {
        public Terraria.ModLoader.Mod Mod = new();
        public Skill Skill;
        public virtual bool IsPassiveSkill => false;
        public virtual bool IsToggleable => false;
        public bool IsEnabled = true;
        public float CooldownDuration;
        public int LockCalls;
        public virtual void OnLockSkill() => LockCalls++;
        public virtual Type[] PrerequisiteSkills => [];
        public virtual SkillUnlockCondition UnlockCondition { get; set; } = SkillUnlockCondition.None;
        public virtual bool TryUnlockSkill(KLSkillModPlayer p) => UnlockCondition.TryUnlock(p, this);
        public virtual string GetUnlockConditionDescription(KLSkillModPlayer p) => UnlockCondition.GetDescription(p, this);
        public virtual bool CanDragInSkillPanel() => Skill.BasicStatus == Skill.SKillBasicStatus.UnLock && !IsPassiveSkill;
        public int Level { get => Skill.Level; set => Skill.Level = value; }
        // LIVE_MAX_LEVEL
        // LIVE_TRY_LEVEL_UP
    }
    public sealed class Skill
    {
        public static Dictionary<string, Skill> RegisterSkill = [];
        public enum SKillBasicStatus { Lock, UnLock, Hide }
        public SKillBasicStatus BasicStatus;
        public int Level, SkillSlot = -1;
        public float CurrentCD;
        public int Stack = 1, MaxStack = 1;
        public float MaxCD = .3f;
        public ModSkill ModSkill;
        // LIVE_SKILL_MAX_LEVEL
        public Terraria.ModLoader.Mod Mod => ModSkill.Mod;
        public Asset<Texture2D> SkillIcon => new(PanelGame.Texture(Fixtures.Icon(ModSkill.GetType().Name)));
        public static Skill NewSkill(Type type, Terraria.ModLoader.Mod mod = null)
        {
            var result = new Skill { ModSkill = (ModSkill)Activator.CreateInstance(type) };
            result.ModSkill.Skill = result;
            return result;
        }
    }
    public class KLSkillModPlayer
    {
        public Player Player = new();
        public int SkillPoint;
        public Dictionary<string, Skill> UnlockedSkill = [];
        public List<Skill> ActiveSkill = Enumerable.Repeat<Skill>(null, 8).ToList();
        public bool CanCostSkillPoint(int amount) => amount <= 0 || SkillPoint >= amount;
        public bool TryCostSkillPoint(int amount) { if (!CanCostSkillPoint(amount)) return false; SkillPoint -= Math.Max(0, amount); return true; }
        public void UnlockSkill(Skill skill) { skill.Level = Math.Max(1, skill.Level); skill.BasicStatus = Skill.SKillBasicStatus.UnLock; UnlockedSkill.TryAdd(skill.ModSkill.GetType().Name, skill); }
        public void LockSkill(Skill skill)
        {
            if (!UnlockedSkill.TryGetValue(skill.ModSkill.GetType().Name, out var owned) || owned.BasicStatus != Skill.SKillBasicStatus.UnLock) return;
            UnlockedSkill.Remove(skill.ModSkill.GetType().Name);
            skill.BasicStatus = Skill.SKillBasicStatus.Lock; skill.ModSkill.OnLockSkill();
        }
        public void OnSkillsUpdated() { }
    }
    public static class KLSkillManager
    {
        public static void EquipSkill(List<Skill> skills, Skill skill, int index)
        {
            for (int i = 0; i < skills.Count; i++) if (skills[i]?.ModSkill.GetType() == skill.ModSkill.GetType()) skills[i] = null;
            skills[index] = skill; skill.SkillSlot = index;
        }
        public static void UnEquipSkill(List<Skill> skills, int index) => skills[index] = null;
    }
}

    public static class ModContent {
        public static string Root;
        public static bool UseOriginalParticles;
        public static Microsoft.Xna.Framework.Content.ContentManager Content;
        public static Microsoft.Xna.Framework.Content.ContentManager ModEffects;
        public static readonly Dictionary<string, Texture2D> Textures = new();
        public static Asset<T> Request<T>(string path, AssetRequestMode mode = AssetRequestMode.ImmediateLoad) {
            if (typeof(T) == typeof(Effect)) return new Asset<T>((T)(object)
                (path.StartsWith("KL/") ? Content : ModEffects).Load<Effect>(System.IO.Path.GetFileName(path)));
            if (!Textures.TryGetValue(path, out var texture)) {
                string file = path.StartsWith("KL/") ? System.IO.Path.Combine(Root, "..", path + ".png")
                    : System.IO.Path.Combine(Root, path[(path.IndexOf('/') + 1)..] + ".png");
                // Default follows production paths; only the old-texture comparison overrides them.
                if (UseOriginalParticles && (path == "伊蕾娜/ElainaModSkills/Skills/Water/Eff_Splash_4001_Clear"
                    || path == "伊蕾娜/ElainaModSkills/Skills/Water/Eff_Splash_4002_Clear"))
                    file = file[..^10] + ".png";
                using var stream = System.IO.File.OpenRead(file);
                texture = Texture2D.FromStream(Main.graphics.GraphicsDevice, stream);
                var pixels = new Color[texture.Width * texture.Height]; texture.GetData(pixels);
                for (int i = 0; i < pixels.Length; i++) if (pixels[i].A == 0) pixels[i] = Color.Transparent;
                texture.SetData(pixels); Textures.Add(path, texture);
            }
            return new Asset<T>((T)(object)texture);
        }
        public static int DustType<T>() => 1;
    }
    public class PreviewRandom(int seed) : Random(seed) {
        public float NextFloat(float min, float max) => min + (max - min) * (float)NextDouble();
        public Vector2 NextVector2Circular(float x, float y) {
            float angle = NextFloat(0, MathHelper.TwoPi), radius = MathF.Sqrt((float)NextDouble());
            return new Vector2(MathF.Cos(angle) * x, MathF.Sin(angle) * y) * radius;
        }
    }

    public static class Scene {
        static LaserReplay laser;
        public static int DustCount => Dust.All.Count;
        public static void CheckAttachment() {
            Dust.All.Clear();
            var owner = new Entity { position = new Vector2(300, 200), oldPosition = new Vector2(276, 200) };
            Vector2 tip = owner.position + new Vector2(20, -45);
            WaterLaserDust.Emit(tip, Vector2.Zero, new Vector2(25), 60, attachedEntity: owner);
            Dust particle = Dust.All.Single();
            void Check(Vector2 expected, string stage) {
                if (Vector2.Distance(particle.position, expected) > .001f)
                    throw new Exception($"Attachment {stage}: expected {expected}, got {particle.position}");
            }
            Check(tip, "spawn after player moved");
            DustLoader.Instance.Update(particle);
            Check(tip, "same-frame regular dust update");
            for (int step = 0; step < 8; step++) {
                Vector2 move = new(step < 4 ? 24 : -24, step % 2 == 0 ? 13 : -13);
                owner.oldPosition = owner.position; owner.position += move; tip += move;
                DustLoader.Instance.Update(particle); Check(tip, "translation " + step);
                // A second update must not apply the same owner movement again.
                DustLoader.Instance.Update(particle); Check(tip, "repeated translation " + step);
            }
            owner.position += new Vector2(400, -150); tip += new Vector2(400, -150);
            owner.oldPosition = owner.position; // Engine can reset oldPosition during teleport.
            DustLoader.Instance.Update(particle); Check(tip, "owner oldPosition reset");
            owner.active = false; DustLoader.Instance.Update(particle);
            owner.active = true; owner.position += new Vector2(1000);
            DustLoader.Instance.Update(particle); Check(tip, "detached owner reactivated");
            if (particle.DustInfo().AttachedEntity != null) throw new Exception("Inactive owner did not detach");
            Vector2 finalOrigin = new(2000, 1200);
            Prepare(40, finalOrigin, -1.4f, 600, 0);
            Vector2[] stationary = Dust.All.Where(d => d.DustInfo().AttachedEntity != null)
                .Select(d => d.position - finalOrigin).ToArray();
            foreach (Vector2 movement in new[] { new Vector2(24, 0), new Vector2(-24, 0), new Vector2(0, -18), new Vector2(22, 14) }) {
                Prepare(40, finalOrigin, -1.4f, 600, 0, movement: movement);
                Vector2[] moving = Dust.All.Where(d => d.DustInfo().AttachedEntity != null)
                    .Select(d => d.position - finalOrigin).ToArray();
                if (stationary.Length != moving.Length || stationary.Zip(moving).Any(p => Vector2.Distance(p.First, p.Second) > .01f))
                    throw new Exception("Moving muzzle particle cloud differs from stationary cloud: " + movement);
            }
            Console.WriteLine("PASS attachment: no spawn/update double movement; horizontal/vertical/reversal, oldPosition reset and owner detach.");
            Console.WriteLine("PASS actual emitter: muzzle offsets match at rest and at 24px/tick left/right, 18px/tick vertical, diagonal flight.");
        }
        public static void Prepare(int ticks, Vector2 origin, float angle, float length, float bend, int release = 0, Vector2 movement = default) {
            Dust.All.Clear(); Main.rand = new PreviewRandom(47); laser = new LaserReplay();
            for (int tick = 0; tick <= ticks + release; tick++) {
                Vector2 currentOrigin = origin - movement * (ticks + release - tick);
                laser.MoveOwner(currentOrigin + new Vector2(-20, 45), movement);
                laser.Set(tick, currentOrigin, angle, length, bend);
                if (tick <= ticks) laser.SpawnWaterDust();
                // Match Main.Update: player -> projectile/particle spawn -> Dust.UpdateDust.
                foreach (Dust dust in Dust.All) DustLoader.Instance.Update(dust);
                Dust.All.RemoveAll(d => !d.active);
            }
        }
        public static void Queue(bool beam, bool particles) {
            LayerDrawRequestSystem.ClearFrame();
            if (beam) { Color c = Color.White; laser.PreDraw(ref c); }
            // Same collection point as DrawSystem: after projectile PreDraw, before Projectiles.After flush.
            if (particles) KLBasicDust.SubmitBloomDrawRequests();
            if ((beam || particles && Dust.All.Any(d => d.active)) &&
                !LayerDrawRequestSystem.HasBloomRequests(LayerDrawRequestSystem.DrawTargetLayer.Projectiles, LayerDrawRequestSystem.DrawTiming.After))
                throw new Exception("Beam/particles missed Projectiles.After");
            foreach (var layer in Enum.GetValues<LayerDrawRequestSystem.DrawTargetLayer>())
            foreach (var timing in Enum.GetValues<LayerDrawRequestSystem.DrawTiming>())
                if ((layer != LayerDrawRequestSystem.DrawTargetLayer.Projectiles || timing != LayerDrawRequestSystem.DrawTiming.After) &&
                    LayerDrawRequestSystem.HasBloomRequests(layer, timing))
                    throw new Exception("Beam/particles created an extra bloom stage");
        }
        public static void CheckAfterDraw() {
            // Vanilla DrawDust happens later: it must neither redraw nor queue another frame's bloom.
            foreach (Dust d in Dust.All)
                if (DustLoader.Instance.PreDraw(d)) throw new Exception("Bloom dust requested a second vanilla draw");
            if (LayerDrawRequestSystem.HasBloomRequests(LayerDrawRequestSystem.DrawTargetLayer.Projectiles, LayerDrawRequestSystem.DrawTiming.After))
                throw new Exception("Dust.PreDraw requeued particles after the target had already flushed");
        }

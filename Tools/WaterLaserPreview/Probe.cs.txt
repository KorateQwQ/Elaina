using System;
using System.IO;
using System.Linq;
using System.Reflection;
using Microsoft.Xna.Framework;
using Microsoft.Xna.Framework.Content;
using Microsoft.Xna.Framework.Graphics;
using SDL2;

sealed class LaserProbe : Game
{
    const int Width = 1200, Height = 700;
    readonly string root, output, particleVariant;
    Assembly pipeline;
    SpriteBatch batch;
    ContentManager effects, modEffects;
    Texture2D pixel;
    Type main, scene, draw;
    object renderer;
    MethodInfo render;
    bool done;
    public LaserProbe(string root, string output, string particleVariant) {
        this.root = root; this.output = output; this.particleVariant = particleVariant;
        _ = new GraphicsDeviceManager(this) { PreferredBackBufferWidth = Width, PreferredBackBufferHeight = Height,
            GraphicsProfile = GraphicsProfile.HiDef, SynchronizeWithVerticalRetrace = false };
        IsFixedTimeStep = false; SDL.SDL_HideWindow(Window.Handle);
    }
    void Set(Type type, string name, object value) => type.GetField(name, BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Static).SetValue(null, value);
    protected override void LoadContent() {
        SDL.SDL_HideWindow(Window.Handle); Directory.CreateDirectory(output);
        pipeline = Pipeline.CompilePipeline(Path.GetFullPath(Path.Combine(root, "../KL")), root, output);
        batch = new SpriteBatch(GraphicsDevice);
        pixel = new Texture2D(GraphicsDevice, 1, 1); pixel.SetData(new[] { Color.White });
        main = pipeline.GetType("KL.Drawing.Main"); draw = pipeline.GetType("KL.Drawing.DrawSystem");
        scene = pipeline.GetType("伊蕾娜.ElainaModSkills.Skills.Water.Scene");
        var context = Activator.CreateInstance(pipeline.GetType("KL.Drawing.DeviceContext"));
        context.GetType().GetField("GraphicsDevice").SetValue(context, GraphicsDevice);
        Set(main, "instance", context); Set(main, "graphics", context); Set(main, "spriteBatch", batch);
        Set(main, "screenWidth", Width); Set(main, "screenHeight", Height);
        effects = new ContentManager(Services, Path.GetFullPath(Path.Combine(root, "../KL/Effects/Content")));
        var helper = pipeline.GetType("KL.Drawing.DrawHelper");
        Set(helper, "bloom", effects.Load<Effect>("Bloom")); Set(helper, "reColor", effects.Load<Effect>("ReColor"));
        Set(helper, "vertexDraw", effects.Load<Effect>("VertexDraw"));
        var assets = pipeline.GetType("KL.Drawing.ModContent"); Set(assets, "Root", root); Set(assets, "Content", effects);
        Set(assets, "UseOriginalParticles", particleVariant == "Original");
        modEffects = new ContentManager(Services, Path.Combine(root, "Effects/Content"));
        Set(assets, "ModEffects", modEffects);
        renderer = Activator.CreateInstance(draw); render = draw.GetMethod("Run");
    }
    protected override void Draw(GameTime time) {
        if (done) { Exit(); return; } done = true;
        scene.GetMethod("CheckAttachment").Invoke(null, null);
        pipeline.GetType("伊蕾娜.ElainaActions.WandScene").GetMethod("Check").Invoke(null, null);
        foreach (int tick in new[] { 6, 18, 36, 60 }) Render($"day-{tick}", tick);
        Render("night", 60, night: true);
        Render("night-90", 90, night: true);
        Render("night-120", 120, night: true);
        // Consecutive 60 Hz frames expose the fast material flow without sparse-frame aliasing.
        for (int tick = 60; tick < 66; tick++) Render($"flow-{tick}", tick, night: true, angle: 0);
        Render("zoom", 60, zoom: 1.35f);
        Render("left", 60, angle: 2.85f);
        Render("up", 60, angle: -1.4f, length: 510);
        Render("moving-right-up", 60, angle: -1.4f, length: 510, movement: new Vector2(24, 0));
        Render("moving-left-up", 60, angle: -1.4f, length: 510, movement: new Vector2(-24, 0));
        Render("moving-diagonal", 60, angle: -1.4f, length: 510, movement: new Vector2(22, 14));
        Render("short", 24, length: 28);
        Render("bent", 60, bend: -95);
        Render("backbuffer", 60, backbuffer: true);
        var off = Render("bloom-off", 60, bloom: false);
        var on = Render("bloom-on", 60);
        int lit = off.Zip(on).Count(p => p.Second.R > p.First.R + 2 || p.Second.G > p.First.G + 2 || p.Second.B > p.First.B + 2);
        // The absorption material uses covered blue water and white foam, not HDR light.
        if (lit > 2000) throw new Exception("Water material unexpectedly blooms into a luminous beam: " + lit);
        var particlesOff = Render("particles-off", 60, bloom: false, beam: false);
        var particlesOn = Render("particles-bloom", 60, beam: false);
        int particleGlow = particlesOff.Zip(particlesOn).Count(p => p.Second.G > p.First.G + 2);
        if (particleGlow > 50) throw new Exception("Water splashes unexpectedly produce glowing nodes: " + particleGlow);
        Render("release-8", 60, beam: false, release: 8);
        Render("release-30", 60, beam: false, release: 30);
        int count = (int)scene.GetProperty("DustCount").GetValue(null);
        if (count != 0) throw new Exception("Particles survived their lifetime: " + count);
        Render("wand-hold", 60, night: true, wand: true);
        Render("wand-turn", 75, night: true, wand: true, wandMoving: true, angle: -.35f);
        Render("wand-left", 75, night: true, wand: true, wandMoving: true, angle: 2.85f);
        Render("wand-zoom", 75, night: true, wand: true, wandMoving: true, zoom: 1.35f);
        Render("wand-release-1", 60, night: true, beam: false, release: 1, wand: true, wandMoving: true);
        Render("wand-release-10", 60, night: true, beam: false, release: 10, wand: true, wandMoving: true);
        Render("wand-release-20", 60, night: true, beam: false, release: 20, wand: true, wandMoving: true);
        Render("wand-default-purple", 60, night: true, beam: false, release: 30, wand: true, wandMoving: true, waterWand: false);
        if (Environment.GetEnvironmentVariable("WATER_BALL_FRAMES") == "1") {
            // Consecutive 60 Hz ticks for GIF assembly (make_ball_gif.py).
            Directory.CreateDirectory(Path.Combine(output, "frames"));
            foreach (string kind in new[] { "circle", "cone", "launch" })
            for (int tick = 0; tick <= 32; tick++)
                RenderBurst($"frames/{kind}-{tick:D2}", tick, FullRadius, kind != "circle", launch: kind == "launch");
        }
        // Launch requires time > 60 while charge fills by tick 50, so only the full radius (50 + 150) exists.
        const float radius = FullRadius;
        foreach (bool cone in new[] { false, true }) {
            string kind = cone ? "cone" : "circle";
            foreach (int tick in new[] { 1, 6, 12, 20 }) RenderBurst($"ball-{kind}-day-{tick}", tick, radius, cone);
            RenderBurst($"ball-{kind}-night-6", 6, radius, cone, night: true);
            RenderBurst($"ball-{kind}-zoom-6", 6, radius, cone, zoom: 2);
            var burstOff = RenderBurst($"ball-{kind}-bloom-off", 6, radius, cone, bloom: false);
            var burstOn = RenderBurst($"ball-{kind}-bloom-on", 6, radius, cone);
            int burstGlow = burstOff.Zip(burstOn).Count(p => p.Second.G > p.First.G + 2);
            if (burstGlow > 50) throw new Exception("Water ball burst unexpectedly glows: " + kind + " " + burstGlow);
            RenderBurst($"ball-{kind}-end", 30, radius, cone);
            int left = (int)scene.GetProperty("DustCount").GetValue(null);
            if (left != 0) throw new Exception("Burst particles survived their lifetime: " + kind + " " + left);
        }
        {
            string kind = "launch";
            foreach (int tick in new[] { 1, 6, 12 }) RenderBurst($"ball-{kind}-day-{tick}", tick, radius, true, launch: true);
            RenderBurst($"ball-{kind}-night-6", 6, radius, true, night: true, launch: true);
            RenderBurst($"ball-{kind}-zoom-6", 6, radius, true, zoom: 2, launch: true);
            RenderBurst($"ball-{kind}-end", 30, radius, true, launch: true);
            if ((int)scene.GetProperty("DustCount").GetValue(null) != 0) throw new Exception("Launch particles survived: " + kind);
        }
        Console.WriteLine("PASS water ball burst: circle/cone/launch at full charge, no glowing nodes, cleared by tick 30.");
        Console.WriteLine($"PASS: restrained Bloom changed {lit} pixels; isolated particle bloom changed {particleGlow}; particles cleared after release.");
        Console.WriteLine("Rendered actual beam/muzzle/particle drawing and KL Bloom on day/night, left/up, bend, zoom, short beam and backbuffer.");
        Exit();
    }
    Color[] Render(string name, int ticks, bool night = false, float zoom = 1, float angle = .28f,
        float length = 880, float bend = 0, bool bloom = true, bool beam = true, int release = 0, bool backbuffer = false,
        Vector2 movement = default, bool wand = false, bool wandMoving = false, bool waterWand = true) {
        using var target = new RenderTarget2D(GraphicsDevice, Width, Height, false, SurfaceFormat.Color, DepthFormat.None);
        GraphicsDevice.SetRenderTarget(backbuffer ? null : target);
        GraphicsDevice.Clear(night ? new Color(12, 23, 34) : new Color(67, 134, 141));
        // Nonzero world camera origin catches missed world-to-screen conversions.
        Vector2 camera = new(2000, 1200); Set(main, "screenPosition", camera);
        Vector2 origin = angle > 2 ? new Vector2(1040, 220) : angle < -1 ? new Vector2(490, 580) : new Vector2(160, 210);
        Vector2 zoomCenter = wand ? origin : new Vector2(600, 350);
        Matrix transform = Matrix.CreateTranslation(-zoomCenter.X, -zoomCenter.Y, 0) * Matrix.CreateScale(zoom, zoom, 1)
            * Matrix.CreateTranslation(zoomCenter.X, zoomCenter.Y, 0);
        Set(main, "Transform", transform);
        object view = main.GetField("GameViewMatrix").GetValue(null);
        view.GetType().GetField("TransformationMatrix").SetValue(view, transform);
        batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.PointClamp, null, null, null, transform);
        for (int x = 0; x < Width; x += 100) batch.Draw(pixel, new Rectangle(x, 0, 1, Height), new Color(48, 89, 96));
        for (int y = 0; y < Height; y += 100) batch.Draw(pixel, new Rectangle(0, y, Width, 1), new Color(48, 89, 96));
        // A simple wand marker locates the exact supplied action origin; this is not the player's live pose.
        batch.Draw(pixel, origin, null, new Color(194, 165, 105), angle, new Vector2(1, .5f), new Vector2(66, 5), SpriteEffects.None, 0);
        batch.End();
        scene.GetMethod("Prepare").Invoke(null, [ticks, origin + camera, angle, length, bend, release, movement]);
        scene.GetMethod("Queue").Invoke(null, [beam, true]);
        render.Invoke(renderer, [1f, 5, bloom]);
        scene.GetMethod("CheckAfterDraw").Invoke(null, null);
        // Main draws ordinary projectiles (including KL Projectiles.After), then DrawPlayers_AfterProjectiles.
        if (wand) pipeline.GetType("伊蕾娜.ElainaActions.WandScene").GetMethod("Draw").Invoke(null,
            [origin + camera, angle, waterWand ? Math.Clamp(release / 20f, 0, 1) : .5f, ticks + release, wandMoving, waterWand]);
        var pixels = new Color[Width * Height];
        if (backbuffer) GraphicsDevice.GetBackBufferData(pixels); else target.GetData(pixels);
        GraphicsDevice.SetRenderTarget(null);
        using var image = new Texture2D(GraphicsDevice, Width, Height); image.SetData(pixels);
        using var stream = File.Create(Path.Combine(output, name + ".png")); image.SaveAsPng(stream, Width, Height);
        return pixels;
    }
    const float FullRadius = 200;
    Color[] RenderBurst(string name, int ticks, float radius, bool cone, bool night = false, float zoom = 1, bool bloom = true, bool launch = false) {
        using var target = new RenderTarget2D(GraphicsDevice, Width, Height, false, SurfaceFormat.Color, DepthFormat.None);
        GraphicsDevice.SetRenderTarget(target);
        GraphicsDevice.Clear(night ? new Color(12, 23, 34) : new Color(67, 134, 141));
        Vector2 camera = new(2000, 1200); Set(main, "screenPosition", camera);
        Vector2 center = cone ? new Vector2(300, 350) : new Vector2(600, 350);
        Matrix transform = Matrix.CreateTranslation(-center.X, -center.Y, 0) * Matrix.CreateScale(zoom, zoom, 1)
            * Matrix.CreateTranslation(center.X, center.Y, 0);
        Set(main, "Transform", transform);
        object view = main.GetField("GameViewMatrix").GetValue(null);
        view.GetType().GetField("TransformationMatrix").SetValue(view, transform);
        batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.PointClamp, null, null, null, transform);
        for (int x = 0; x < Width; x += 100) batch.Draw(pixel, new Rectangle(x, 0, 1, Height), new Color(48, 89, 96));
        for (int y = 0; y < Height; y += 100) batch.Draw(pixel, new Rectangle(0, y, Width, 1), new Color(48, 89, 96));
        // Ball outline only marks the explosion radius; the ball itself is already gone on OnKill.
        for (int i = 0; i < 64; i++) {
            float a = MathHelper.TwoPi * i / 64;
            batch.Draw(pixel, center + new Vector2(MathF.Cos(a), MathF.Sin(a)) * radius * .5f, null, new Color(194, 165, 105), 0, Vector2.Zero, 3, SpriteEffects.None, 0);
        }
        if (cone && !launch) {
            // WaterBall.Colliding T shape: flight hitbox width = radius * 0.8 (resized during Hold).
            float w = (int)(radius * .8f), rotation = .2f;
            DrawBox(center, rotation, -w / 2, w * 3.5f, w);
            DrawBox(center, rotation, -w / 2, w * 1.5f, w * 3);
        }
        batch.End();
        scene.GetMethod("PrepareBurst").Invoke(null, [ticks, center + camera, radius, cone ? .2f : 0f, cone, launch]);
        scene.GetMethod("Queue").Invoke(null, [false, true]);
        render.Invoke(renderer, [1f, 5, bloom]);
        scene.GetMethod("CheckAfterDraw").Invoke(null, null);
        var pixels = new Color[Width * Height];
        target.GetData(pixels);
        GraphicsDevice.SetRenderTarget(null);
        using var image = new Texture2D(GraphicsDevice, Width, Height); image.SetData(pixels);
        using var stream = File.Create(Path.Combine(output, name + ".png")); image.SaveAsPng(stream, Width, Height);
        return pixels;
    }
    void DrawBox(Vector2 center, float rotation, float start, float end, float thickness) {
        Vector2 dir = new(MathF.Cos(rotation), MathF.Sin(rotation)), normal = new(-dir.Y, dir.X);
        Vector2[] corners = [center + dir * start - normal * thickness / 2, center + dir * end - normal * thickness / 2,
            center + dir * end + normal * thickness / 2, center + dir * start + normal * thickness / 2];
        for (int i = 0; i < 4; i++) {
            Vector2 a = corners[i], b = corners[(i + 1) % 4];
            batch.Draw(pixel, a, null, new Color(230, 120, 90), MathF.Atan2(b.Y - a.Y, b.X - a.X), new Vector2(0, .5f),
                new Vector2(Vector2.Distance(a, b), 2), SpriteEffects.None, 0);
        }
    }
    protected override void UnloadContent() {
        if (pipeline != null) {
            var visuals = pipeline.GetType("伊蕾娜.ElainaModSkills.Skills.Water.WaterLaserVisuals");
            visuals.GetMethod("Unload").Invoke(Activator.CreateInstance(visuals), null);
        }
        batch?.Dispose(); pixel?.Dispose(); effects?.Dispose(); modEffects?.Dispose();
    }
}

// Only PlayerDrawSet/DrawData storage is adapted; the action overrides, base dispatch,
// color/fade calculations, trail shader and star textures are extracted from game sources.
namespace Terraria.DataStructures {
    public struct PlayerDrawSet { public List<DrawData> DrawDataCache; }
    public struct DrawData(Texture2D texture, Vector2 position, Rectangle? sourceRect, Color color,
        float rotation, Vector2 origin, Vector2 scale, SpriteEffects effects, float layerDepth) {
        public bool ignorePlayerRotation;
        public Color Color = color;
        public void Draw(SpriteBatch batch) => batch.Draw(texture, position, sourceRect, Color,
            rotation, origin, scale, effects, layerDepth);
    }
}
namespace 伊蕾娜.ElainaActions {
    public partial class WandReplayBase {
        protected Vector2 StarCenter;
        private bool DrawWandTrail => true;
        private Vector2[] wandTrailPositions;
        private int wandTrailPointCount;

        public Color ReadStar(float progress) {
            var info = new PlayerDrawSet { DrawDataCache = new() };
            DrawStar(ref info, progress);
            return info.DrawDataCache[0].Color;
        }

        public void Replay(Vector2 origin, float angle, float progress, bool moving) {
            StarCenter = origin;
            wandTrailPointCount = moving ? 30 : 0;
            wandTrailPositions = new Vector2[wandTrailPointCount];
            for (int i = 0; i < wandTrailPointCount; i++) {
                float t = i / (float)(wandTrailPointCount - 1);
                wandTrailPositions[i] = origin + new Vector2(-100f * t, 70f * MathF.Sin(t * 1.6f)).RotatedBy(angle);
            }
            BeginDraw();
            DrawAutoWandTrail(progress);
            var info = new PlayerDrawSet { DrawDataCache = new() };
            DrawStar(ref info, progress);
            EndBeginDraw(ss: SamplerState.LinearClamp);
            foreach (DrawData data in info.DrawDataCache) data.Draw(Main.spriteBatch);
            Main.spriteBatch.End();
        }
    }
    public static class WandScene {
        public static void Draw(Vector2 origin, float angle, float progress, int tick, bool moving, bool water) {
            Main.timeForVisualEffects = tick;
            WandReplayBase action = water ? new WaterWandReplay() : new WandReplayBase();
            action.Replay(origin, angle, progress, moving);
        }
        public static void Check() {
            Main.timeForVisualEffects = 15;
            var water = new WaterWandReplay();
            Color hold = water.ReadStar(0);
            if (hold.B <= hold.G || hold.G <= hold.R || hold.R == 0)
                throw new Exception("Water action frame 0 must show a blue star.");
            int last = hold.B;
            foreach (float progress in new[] { .05f, .25f, .5f, .75f, 1f }) {
                Color current = water.ReadStar(progress);
                if (current.B >= last) throw new Exception("Recovery star must fade monotonically.");
                last = current.B;
            }
            if (last != 0) throw new Exception("Recovery star must finish transparent.");
            var normal = new WandReplayBase();
            Color start = normal.ReadStar(0), middle = normal.ReadStar(.5f);
            if (start.R != 0 || start.B != 0 || middle.R <= middle.G || middle.B <= middle.G)
                throw new Exception("Default action's purple star or fade-in changed.");
            Console.WriteLine("PASS wand: blue star visible at held frame 0, fades across recovery, other actions retain purple/fade-in.");
        }
    }
}

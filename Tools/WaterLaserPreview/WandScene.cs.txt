// Only PlayerDrawSet/DrawData storage is adapted; the action overrides, base dispatch,
// color/fade calculations, trail shader and star textures are extracted from game sources.
namespace Terraria.DataStructures {
    public struct PlayerDrawSet { public List<DrawData> DrawDataCache; public float shadow; }
    public struct DrawData(Texture2D texture, Vector2 position, Rectangle? sourceRect, Color color,
        float rotation, Vector2 origin, Vector2 scale, SpriteEffects effects, float layerDepth) {
        public bool ignorePlayerRotation;
        public Color Color = color;
        public static readonly List<Color> DrawnColors = new();
        public void Draw(SpriteBatch batch) {
            DrawnColors.Add(Color);
            batch.Draw(texture, position, sourceRect, Color, rotation, origin, scale, effects, layerDepth);
        }
    }
}
namespace 伊蕾娜.ElainaActions {
    public partial class WandReplayBase {
        protected Vector2 StarCenter;
        private bool DrawWandTrail => true;
        private Vector2[] wandTrailPositions;
        private int wandTrailPointCount;

        public Color ReadStar(float progress, float shadow = 0f) {
            LayerDrawRequestSystem.ClearFrame();
            DrawData.DrawnColors.Clear();
            var info = new PlayerDrawSet { DrawDataCache = new(), shadow = shadow };
            DrawStar(ref info, progress);
            if (info.DrawDataCache.Count != 0) throw new Exception("Action star still enters the player layer.");
            bool particlesPassed = false;
            LayerDrawRequestSystem.Request("Preview:Particles", LayerDrawRequestSystem.DrawTargetLayer.Dust,
                LayerDrawRequestSystem.DrawTiming.After, () => {
                    if (DrawData.DrawnColors.Count != 0) throw new Exception("Action star drew before skill particles.");
                    particlesPassed = true;
                }, order: LayerDrawRequestSystem.DustOrder);
            LayerDrawRequestSystem.Flush(LayerDrawRequestSystem.DrawTargetLayer.Dust, LayerDrawRequestSystem.DrawTiming.After);
            if (!particlesPassed || DrawData.DrawnColors.Count != (shadow == 0f ? 3 : 0))
                throw new Exception("Action star was missing, duplicated, or queued for a player shadow.");
            // The deferred request must return with the batch closed.
            BeginDraw();
            Main.spriteBatch.End();
            return DrawData.DrawnColors.Count == 0 ? Color.Transparent : DrawData.DrawnColors[0];
        }

        public void Replay(Vector2 origin, float angle, float progress, bool moving) {
            StarCenter = origin;
            wandTrailPointCount = moving ? 30 : 0;
            wandTrailPositions = new Vector2[wandTrailPointCount];
            for (int i = 0; i < wandTrailPointCount; i++) {
                float t = i / (float)(wandTrailPointCount - 1);
                wandTrailPositions[i] = origin + new Vector2(-100f * t, 70f * MathF.Sin(t * 1.6f)).RotatedBy(angle);
            }
            BeginDraw();
            DrawAutoWandTrail(progress);
            var info = new PlayerDrawSet { DrawDataCache = new() };
            DrawStar(ref info, progress);
            EndBeginDraw(ss: SamplerState.LinearClamp);
            foreach (DrawData data in info.DrawDataCache) data.Draw(Main.spriteBatch);
            Main.spriteBatch.End();
        }
    }
    public static class WandScene {
        public static void DrawIce(Vector2 origin, float angle, int frame, int tick, bool legacy) {
            Main.timeForVisualEffects = tick;
            WandReplayBase action = legacy ? new WandReplayBase() : new IceWandReplay();
            action.Replay(origin, angle, IceWandReplay.Progress(frame), true);
        }
        public static void CheckIce() {
            Main.timeForVisualEffects = 15;
            var ice = new IceWandReplay();
            var normal = new WandReplayBase();
            Color full = normal.ReadStar(.5f);
            Color hold = ice.ReadStar(IceWandReplay.Progress(1));
            if (hold != full || ice.ReadStar(0) != full)
                throw new Exception("Ice Cone star must reach normal brightness at start and while held.");
            Color legacy = normal.ReadStar(IceWandReplay.Progress(1));
            if (legacy.R >= hold.R / 3) throw new Exception("Legacy brightness regression no longer reproduced.");
            int last = hold.R;
            for (int frame = 2; frame <= 21; frame++) {
                Color current = ice.ReadStar(IceWandReplay.Progress(frame));
                if (current.R >= last) throw new Exception("Ice Cone star must fade throughout recovery.");
                last = current.R;
            }
            if (last != 0) throw new Exception("Ice Cone star must disappear at recovery end.");
            ice.ReadStar(IceWandReplay.Progress(1), shadow: .5f);
            Console.WriteLine("PASS Ice Cone star: full held brightness, original purple/blink, 20-frame fade and no shadow duplicate.");
        }
        public static void Draw(Vector2 origin, float angle, float progress, int tick, bool moving, bool water) {
            Main.timeForVisualEffects = tick;
            WandReplayBase action = water ? new WaterWandReplay() : new WandReplayBase();
            action.Replay(origin, angle, progress, moving);
        }
        public static void Check() {
            Main.timeForVisualEffects = 15;
            var water = new WaterWandReplay();
            Color hold = water.ReadStar(0);
            if (hold.B <= hold.G || hold.G <= hold.R || hold.R == 0)
                throw new Exception("Water action frame 0 must show a blue star.");
            int last = hold.B;
            foreach (float progress in new[] { .05f, .25f, .5f, .75f, 1f }) {
                Color current = water.ReadStar(progress);
                if (current.B >= last) throw new Exception("Recovery star must fade monotonically.");
                last = current.B;
            }
            if (last != 0) throw new Exception("Recovery star must finish transparent.");
            var normal = new WandReplayBase();
            Color start = normal.ReadStar(0), middle = normal.ReadStar(.5f);
            if (start.R != 0 || start.B != 0 || middle.R <= middle.G || middle.B <= middle.G)
                throw new Exception("Default action's purple star or fade-in changed.");
            normal.ReadStar(.5f, shadow: .5f);
            Console.WriteLine("PASS wand: stars draw after skill particles, shadows do not queue stars, batch closes; blue/purple colors and recovery fade preserved.");
        }
    }

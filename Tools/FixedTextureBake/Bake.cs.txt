var entries = new List<object>();
void Export(string path, int width, int height, Color[] pixels)
{
    byte[] rgba = new byte[pixels.Length * 4];
    for (int i = 0; i < pixels.Length; i++)
    {
        Color c = pixels[i];
        rgba[i * 4] = c.R; rgba[i * 4 + 1] = c.G; rgba[i * 4 + 2] = c.B; rgba[i * 4 + 3] = c.A;
    }
    string file = path.Replace('/', '_') + ".rgba";
    File.WriteAllBytes(Path.Combine(output, file), rgba);
    entries.Add(new { Path = path + ".png", Width = width, Height = height, Raw = file,
        Sha256 = Convert.ToHexString(SHA256.HashData(rgba)).ToLowerInvariant(),
        Encoding = pixels.All(c => c.A == 255) ? "opaque-data" : "premultiplied-RGBA" });
}

// Hairline ring for the level-up rune circle (drawn squashed, so SoftRing's band is too heavy).
// Radius 120 of 128, Gaussian cross-section with sigma 2 px; premultiplied (a, a, a, a) like the other UI masks.
static Color[] ThinRing()
{
    const int size = 256;
    var pixels = new Color[size * size];
    for (int y = 0; y < size; y++)
        for (int x = 0; x < size; x++)
        {
            float dx = x + .5f - size / 2f, dy = y + .5f - size / 2f;
            float d = (MathF.Sqrt(dx * dx + dy * dy) - 120f) / 2f;
            int a = (int)MathF.Round(255 * MathF.Exp(-d * d));
            pixels[y * size + x] = new Color(a, a, a, a);
        }
    return pixels;
}

// White four-point flare for the level-up head star, shaped after KL T_StartFlare001 but without its
// rainbow-dispersed horizontal arm (at small draw sizes that sampled as a red line once tinted lavender).
// Arms thicken toward the cusp core; a faint haze matches the original's soft surround. Premultiplied (a, a, a, a).
// Haze and cusp are windowed to reach exactly zero inside the texture, otherwise the border reads as a box once lifted to HDR.
static Color[] StarFlare()
{
    const int size = 128;
    var pixels = new Color[size * size];
    static float Arm(float along, float across)
    {
        float width = .9f + 3.4f / (1 + along / 5f), t = along / (size / 2f);
        return MathF.Exp(-(across / width) * (across / width)) * MathF.Pow(MathF.Max(0, 1 - t * t), 1.2f);
    }
    for (int y = 0; y < size; y++)
        for (int x = 0; x < size; x++)
        {
            float dx = MathF.Abs(x + .5f - size / 2f), dy = MathF.Abs(y + .5f - size / 2f);
            float r = MathF.Sqrt(dx * dx + dy * dy);
            float core = MathF.Exp(-(r / 7f) * (r / 7f));
            float window = MathF.Pow(MathF.Max(0, 1 - r * r / (60f * 60f)), 2);
            float cusp = MathF.Min(1, 1 / (1 + dx * dy / 18f)) * MathF.Exp(-r / 26f) * window;
            float haze = .18f * MathF.Exp(-(r / 45f) * (r / 45f)) * window;
            float coverage = 1 - (1 - Arm(dx, dy)) * (1 - Arm(dy, dx)) * (1 - core) * (1 - cusp) * (1 - haze);
            int a = (int)MathF.Round(255 * Math.Clamp(coverage, 0, 1));
            pixels[y * size + x] = new Color(a, a, a, a);
        }
    return pixels;
}

// Charge-point glow: the resource-bar star4 (quadratic arms, control at 0.16 R) with R = 12 of 64,
// Gaussian-blurred twice (tight sigma 1.6 keeps the star outline, wide sigma 4.5 spreads it) plus a faint round halo.
// Windowed to zero before the border; premultiplied (a, a, a, a). Drawn additively (vertex alpha 0) by ChargePointHud.
static Color[] StarGlow()
{
    const int size = 64, samples = 8;
    const float radius = 12f, control = .16f * radius;
    var outline = new List<Vector2>();
    for (int q = 0; q < 4; q++)
        for (int k = 0; k < 24; k++)
        {
            float t = k / 24f, u = 1 - t;
            // First quadrant arm from (0, -R) to (R, 0) through (c, -c), then rotated by quarter turns.
            var p = new Vector2(2 * t * u * control + t * t * radius, -(u * u * radius + 2 * t * u * control));
            outline.Add(q switch { 0 => p, 1 => new(-p.Y, p.X), 2 => -p, _ => new(p.Y, -p.X) });
        }
    bool Inside(float x, float y)
    {
        bool inside = false;
        for (int i = 0, j = outline.Count - 1; i < outline.Count; j = i++)
        {
            Vector2 a = outline[i], b = outline[j];
            if ((a.Y > y) != (b.Y > y) && x < (b.X - a.X) * (y - a.Y) / (b.Y - a.Y) + a.X) inside = !inside;
        }
        return inside;
    }
    var mask = new float[size * size];
    for (int y = 0; y < size; y++)
        for (int x = 0; x < size; x++)
        {
            int hits = 0;
            for (int sy = 0; sy < samples; sy++)
                for (int sx = 0; sx < samples; sx++)
                    if (Inside(x + (sx + .5f) / samples - size / 2f, y + (sy + .5f) / samples - size / 2f)) hits++;
            mask[y * size + x] = hits / (float)(samples * samples);
        }
    float[] Blur(float[] source, float sigma)
    {
        int reach = (int)MathF.Ceiling(sigma * 3);
        var kernel = new float[reach * 2 + 1];
        float sum = 0;
        for (int i = -reach; i <= reach; i++) sum += kernel[i + reach] = MathF.Exp(-i * i / (2 * sigma * sigma));
        for (int i = 0; i < kernel.Length; i++) kernel[i] /= sum;
        var pass = new float[size * size];
        var result = new float[size * size];
        for (int y = 0; y < size; y++)
            for (int x = 0; x < size; x++)
            {
                float v = 0;
                for (int i = -reach; i <= reach; i++)
                    if (x + i >= 0 && x + i < size) v += source[y * size + x + i] * kernel[i + reach];
                pass[y * size + x] = v;
            }
        for (int y = 0; y < size; y++)
            for (int x = 0; x < size; x++)
            {
                float v = 0;
                for (int i = -reach; i <= reach; i++)
                    if (y + i >= 0 && y + i < size) v += pass[(y + i) * size + x] * kernel[i + reach];
                result[y * size + x] = v;
            }
        return result;
    }
    float[] tight = Blur(mask, 1.6f), wide = Blur(mask, 4.5f);
    float tightPeak = tight.Max(), widePeak = wide.Max();
    var pixels = new Color[size * size];
    for (int y = 0; y < size; y++)
        for (int x = 0; x < size; x++)
        {
            float dx = x + .5f - size / 2f, dy = y + .5f - size / 2f, r2 = dx * dx + dy * dy;
            float window = MathF.Pow(MathF.Max(0, 1 - r2 / (30f * 30f)), 2);
            float halo = .32f * MathF.Exp(-r2 / (11f * 11f));
            float coverage = 1 - (1 - .85f * tight[y * size + x] / tightPeak) * (1 - .7f * wide[y * size + x] / widePeak) * (1 - halo);
            int a = (int)MathF.Round(255 * Math.Clamp(coverage * window, 0, 1));
            pixels[y * size + x] = new Color(a, a, a, a);
        }
    return pixels;
}

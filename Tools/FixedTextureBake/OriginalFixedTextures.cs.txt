// Original CPU formulas moved out of production on 2026-10-25.
// Uses the same .NET/FNA Color and float arithmetic, not an approximate redraw.
using Microsoft.Xna.Framework;
internal static class OriginalFixedTextures
{
    private const int NoiseSize = 512;
    private const float NoiseRange = 9f;
    internal static Color[] CloudNoise()
    {
            var random = new Random(19781024);
            var pixels = new Color[256 * 256];
            for (int i = 0; i < pixels.Length; i++)
            {
                byte value = (byte)random.Next(256);
                pixels[i] = new Color(value, value, value, (byte)255);
            }
            // 消融场只在初始化时烘入同一纹理的 G 通道；不增加纹理槽或逐帧 CPU 噪声计算。
            float SmoothNoise(int x, int y, int cellSize)
            {
                int gx = x / cellSize, gy = y / cellSize;
                float fx = x % cellSize / (float)cellSize, fy = y % cellSize / (float)cellSize;
                fx = fx * fx * (3 - 2 * fx); fy = fy * fy * (3 - 2 * fy);
                float Value(int a, int b) => pixels[((b + cellSize * 3) & 255) * 256 + ((a + cellSize * 7) & 255)].R / 255f;
                return MathHelper.Lerp(MathHelper.Lerp(Value(gx, gy), Value(gx + 1, gy), fx),
                    MathHelper.Lerp(Value(gx, gy + 1), Value(gx + 1, gy + 1), fx), fy);
            }
            for (int y = 0; y < 256; y++)
            for (int x = 0; x < 256; x++)
                pixels[y * 256 + x].G = (byte)((SmoothNoise(x, y, 32) * .65f
                    + SmoothNoise(x, y, 16) * .25f + SmoothNoise(x, y, 8) * .10f) * 255);
        return pixels;
    }
    internal static float[] BuildRevealThresholds(Color[] pixels)
    {
        // 只在创建噪声时统计一次。采样云体覆盖区的 G 通道，匹配 Shader 的 LinearWrap 与中心偏置。
        var values = new global::System.Collections.Generic.List<float>(128 * 64);
        for (int y = 0; y < 64; y++)
        for (int x = 0; x < 128; x++)
        {
            float u = .06f + (x + .5f) / 128 * .88f;
            float v = .12f + (y + .5f) / 64 * .66f;
            float ellipseX = (u - .5f) / .46f, ellipseY = (v - .45f) / .33f;
            if (ellipseX * ellipseX + ellipseY * ellipseY > 1) continue;
            float sx = u * 256 - .5f, sy = v * .45f * 256 - .5f;
            int ix = (int)MathF.Floor(sx), iy = (int)MathF.Floor(sy);
            float Pixel(int px, int py) => pixels[(py & 255) * 256 + (px & 255)].G / 255f;
            float sampled = MathHelper.Lerp(MathHelper.Lerp(Pixel(ix, iy), Pixel(ix + 1, iy), sx - ix),
                MathHelper.Lerp(Pixel(ix, iy + 1), Pixel(ix + 1, iy + 1), sx - ix), sy - iy);
            values.Add(sampled * .82f + MathF.Abs((u - .5f) * 6.6f) * .058f);
        }
        values.Sort();
        var thresholds = new float[129];
        thresholds[0] = -.12f;
        thresholds[^1] = 1.12f;
        for (int i = 1; i < thresholds.Length - 1; i++)
            thresholds[i] = values[(int)(i / 128f * (values.Count - 1))];
        return thresholds;
    }
    internal static Dictionary<string, Color[]> UiMasks()
    {
        var textures = new Dictionary<string, Color[]>();
        textures["GlowDot"] = BuildTexture(128, (x, y) =>
        {
            float d = Dist01(x, y, 128);
            float v = MathF.Pow(Math.Clamp(1f - d, 0f, 1f), 2.4f);
            return v;
        });

        textures["SoftRing"] = BuildTexture(128, (x, y) =>
        {
            float d = Dist01(x, y, 128);
            float sigma = 0.09f;
            float delta = (d - 0.78f) / sigma;
            return MathF.Exp(-delta * delta);
        });

        textures["Star4"] = BuildTexture(128, (x, y) =>
        {
            float nx = Math.Abs(x - 63.5f) / 64f;
            float ny = Math.Abs(y - 63.5f) / 64f;
            float armH = MathF.Pow(1f - Math.Clamp(ny, 0f, 1f), 9f) * MathF.Pow(1f - Math.Clamp(nx, 0f, 1f), 2.2f);
            float armV = MathF.Pow(1f - Math.Clamp(nx, 0f, 1f), 9f) * MathF.Pow(1f - Math.Clamp(ny, 0f, 1f), 2.2f);
            float core = MathF.Pow(Math.Clamp(1f - MathF.Sqrt(nx * nx + ny * ny) * 1.6f, 0f, 1f), 3f);
            return Math.Clamp(MathF.Max(MathF.Max(armH, armV), core), 0f, 1f);
        });

        textures["DiamondFill"] = BuildTexture(64, (x, y) =>
        {
            float nx = Math.Abs(x - 31.5f) / 32f;
            float ny = Math.Abs(y - 31.5f) / 32f;
            float d = nx + ny;
            return Math.Clamp((0.98f - d) / 0.06f, 0f, 1f);
        });
            return textures;
    }
    private static float Dist01(int x, int y, int size)
    {
        float half = (size - 1) * 0.5f;
        float dx = (x - half) / half;
        float dy = (y - half) / half;
        return MathF.Sqrt(dx * dx + dy * dy);
    }
    private static Color[] BuildTexture(int size, Func<int, int, float> intensity)
    {
        var data = new Color[size * size];
        for (int y = 0; y < size; y++)
            for (int x = 0; x < size; x++)
            {
                float v = Math.Clamp(intensity(x, y), 0f, 1f);
                data[y * size + x] = Color.White * v;
            }
        return data;
    }
    internal static Color[] PortalLine()
    {
        var data = new Color[32];
        for (int y = 0; y < data.Length; y++)
        {
            float d = Math.Abs(y + .5f - 16) / 16; // 0 在中心，1 在边缘
            float t = Math.Clamp((1 - d) / .45f, 0, 1);
            float v = t * t * (3 - 2 * t);
            data[y] = new Color(v, v, v, v);
        }
        return data;
    }
    internal static Color[] IceNoise()
    {
        Color[] data = new Color[NoiseSize * NoiseSize];
        for (int y = 0; y < NoiseSize; y++)
        {
            float qy = ((y + 0.5f) / NoiseSize * 2f - 1f) * NoiseRange;
            for (int x = 0; x < NoiseSize; x++)
            {
                float qx = ((x + 0.5f) / NoiseSize * 2f - 1f) * NoiseRange;
                byte value = (byte)Math.Clamp((int)MathF.Round(Fbm(qx, qy) * 255f), 0, 255);
                data[y * NoiseSize + x] = new Color(value, value, value, 255);
            }
        }

        return data;
    }
    private static float Fract(float x) => x - MathF.Floor(x);
    private static float Hash(float x, float y)
    {
        x = Fract(x * 123.34f);
        y = Fract(y * 456.21f);
        float d = x * (x + 45.32f) + y * (y + 45.32f);
        return Fract((x + d) * (y + d));
    }
    private static float Noise(float x, float y)
    {
        float ix = MathF.Floor(x), iy = MathF.Floor(y);
        float fx = x - ix, fy = y - iy;
        fx = fx * fx * (3f - 2f * fx);
        fy = fy * fy * (3f - 2f * fy);
        float bottom = MathHelper.Lerp(Hash(ix, iy), Hash(ix + 1f, iy), fx);
        float top = MathHelper.Lerp(Hash(ix, iy + 1f), Hash(ix + 1f, iy + 1f), fx);
        return MathHelper.Lerp(bottom, top, fy);
    }
    private static float Fbm(float x, float y)
    {
        float sum = 0f, amplitude = 0.5f;
        for (int i = 0; i < 4; i++)
        {
            sum += amplitude * Noise(x, y);
            x = x * 2.03f + 1.7f;
            y = y * 2.03f + 1.7f;
            amplitude *= 0.5f;
        }

        return sum;
    }
}

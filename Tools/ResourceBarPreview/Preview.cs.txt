using System;
using System.IO;
using System.Collections.Generic;
using Microsoft.Xna.Framework;
using Microsoft.Xna.Framework.Content;
using Microsoft.Xna.Framework.Graphics;
using ReLogic.Graphics;
using SDL2;
using 伊蕾娜.ElainaAttribute;

internal sealed class ResourcePreview : Game
{
    private readonly string root, output;
    private readonly Dictionary<string, Texture2D> textures = new();
    private ResourceBarDrawing drawing;
    private PortraitHudDrawing portrait;
    private ContentManager fonts;
    private bool done;
    // Same as PORTRAIT_STATES in bake.cjs: level, experience, cap, displayed life, flash/pulse, time.
    private static readonly (string Name, int Level, int Experience, int Cap, float Life, float Burst, float Time)[] PortraitStates =
    [("normal",24,65,30,.84f,0,4), ("low",24,12,30,.84f,0,4), ("capped",30,100,30,.84f,0,4),
     ("levelup",25,5,30,.84f,.6f,4), ("lowhp",7,40,30,.2f,0,.26f), ("wide",120,50,120,.84f,0,4)];
    private ResourcePreview(string root, string output)
    {
        this.root = root; this.output = output;
        new GraphicsDeviceManager(this) { PreferredBackBufferWidth = 400, PreferredBackBufferHeight = 200,
            GraphicsProfile = GraphicsProfile.HiDef, SynchronizeWithVerticalRetrace = false };
        IsFixedTimeStep = false;
        SDL.SDL_HideWindow(Window.Handle);
    }
    protected override void LoadContent()
    {
        Content.RootDirectory = root;
        drawing = new ResourceBarDrawing(Texture, Content.Load<Effect>("Effects/Content/ResourceBarClip"));
        // The real KL Gelasio XNB, as loaded by FontManager in game.
        fonts = new ContentManager(Services, Path.GetFullPath(Path.Combine(root, "../KL/Fonts")));
        var gelasio = fonts.Load<DynamicSpriteFont>("Gelasio");
        foreach (char c in "0123456789") Check(gelasio.IsCharacterSupported(c), $"Missing Gelasio digit {c}");
        portrait = new PortraitHudDrawing(Texture, gelasio);
        // Compare with the installed FNA's real SpriteEffect instead of assuming a half-pixel convention.
        using var probe = new SpriteBatch(GraphicsDevice);
        int matched = 0;
        foreach (float scale in new[] { 1f, 1.5f, 2.25f, 3f })
        {
            var ui = Matrix.CreateScale(scale);
            probe.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.LinearClamp,
                DepthStencilState.None, RasterizerState.CullNone, null, ui);
            probe.Draw(Texture("Pixel"), Vector2.Zero, Color.White); probe.End();
            foreach (var field in typeof(SpriteBatch).GetFields(System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic))
                if (field.GetValue(probe) is Effect effect)
                    foreach (EffectParameter parameter in effect.Parameters)
                        if (parameter.Name == "MatrixTransform")
                        {
                            var viewport = GraphicsDevice.Viewport;
                            Check(parameter.GetValueMatrix() == ResourceBarDrawing.ClipTransform(ui, viewport.Width, viewport.Height),
                                $"Fill/frame transform mismatch at UI scale {scale}");
                            matched++;
                        }
        }
        Check(matched == 4, "FNA SpriteEffect transform checks did not execute");
        Console.WriteLine("PASS: fill/frame projection matches installed FNA at four UI scales.");
    }
    private Texture2D Texture(string name)
    {
        if (textures.TryGetValue(name, out var texture)) return texture;
        using var stream = File.OpenRead(Path.Combine(root, "ElainaAttribute/ResourceBarAssets", name + ".png"));
        texture = Texture2D.FromStream(GraphicsDevice, stream);
        // Same premultiplied bytes as tML's rawimg loader (only zero-alpha pixels are zeroed).
        var pixels = new Color[texture.Width * texture.Height]; texture.GetData(pixels);
        for (int i = 0; i < pixels.Length; i++) if (pixels[i].A == 0) pixels[i] = Color.Transparent;
        texture.SetData(pixels); textures[name] = texture; return texture;
    }
    protected override void Update(GameTime gameTime)
    {
        if (done) return; done = true; SDL.SDL_HideWindow(Window.Handle);
        Directory.CreateDirectory(output); Checks();
        using var batch = new SpriteBatch(GraphicsDevice);
        foreach (float scale in new[] { 1f, 1.5f, 3f }) foreach (bool day in new[] { false, true })
            foreach (var entry in new[] { ("normal",.84f,.84f,4f), ("damage",.18f,.84f,4f), ("empty",0f,0f,4f),
                ("full",1f,1f,4f), ("tiny",.003f,.012f,4f), ("motion",.84f,.84f,6f) })
                Render(batch, scale, day, entry.Item1, entry.Item2, entry.Item3, entry.Item4);
        foreach (float scale in new[] { 1f, 1.5f, 3f }) foreach (bool day in new[] { false, true })
            foreach (var entry in PortraitStates) RenderPortrait(batch, scale, day, entry);
        RenderGame(batch, 1);
        RenderGame(batch, 1.5f);
        RenderGame(batch, 1.5f, true);
        for (int i = 0; i <= 5; i++) RenderAnimation(batch, i * .2f);
        RenderLevelUp(batch);
        Console.WriteLine("PASS: state checks, 72 shared-renderer comparisons, layout, six damage frames and a level-up strip.");
        Exit();
    }
    private void Save(RenderTarget2D target, string name)
    {
        GraphicsDevice.SetRenderTarget(null);
        using var stream = File.Create(Path.Combine(output, name + ".png"));
        target.SaveAsPng(stream, target.Width, target.Height);
    }
    private static ResourceBarState State(bool mana, float value, float ghost)
    {
        var state = new ResourceBarState(mana);
        state.Reset(ghost * 1000, 1000); state.Update(value * 1000, 1000, 0, 0, false);
        return state;
    }
    private void Render(SpriteBatch batch, float scale, bool day, string name, float value, float ghost, float time)
    {
        using var target = new RenderTarget2D(GraphicsDevice, (int)(322 * scale), (int)(76 * scale));
        GraphicsDevice.SetRenderTarget(target); GraphicsDevice.Clear(day ? new Color(120,167,245) : new Color(20,18,29));
        var matrix = Matrix.CreateScale(scale);
        drawing.Draw(batch, matrix, new Vector2(34,20), false, time, State(false,value,ghost));
        drawing.Draw(batch, matrix, new Vector2(34,48), true, time+.7f, State(true,value==.84f?.68f:value,ghost==.84f?.68f:ghost));
        Save(target,$"{name}-{(day?"day":"dark")}-{scale:0.0}");
    }
    private static PortraitHudState Portrait(int level, int experience, int cap, float burst)
    {
        var state = new PortraitHudState(); state.Reset(level, experience, 100, cap); state.ShowBurst(burst);
        return state;
    }
    private void RenderPortrait(SpriteBatch batch, float scale, bool day,
        (string Name, int Level, int Experience, int Cap, float Life, float Burst, float Time) entry)
    {
        using var target = new RenderTarget2D(GraphicsDevice, (int)(160 * scale), (int)(140 * scale));
        GraphicsDevice.SetRenderTarget(target); GraphicsDevice.Clear(day ? new Color(120,167,245) : new Color(20,18,29));
        portrait.Draw(batch, Matrix.CreateScale(scale), new Vector2(60, 56),
            Portrait(entry.Level, entry.Experience, entry.Cap, entry.Burst), entry.Life, entry.Time);
        Save(target, $"portrait-{entry.Name}-{(day?"day":"dark")}-{scale:0.0}");
    }
    private void RenderLevelUp(SpriteBatch batch)
    {
        // Level 24 at 95/100 gains 15 experience: the band reaches the apex, levels up, then restarts at 10.
        var state = Portrait(24, 95, 30, 0);
        float[] moments = [0, .1f, .2f, .35f, .6f];
        using var target = new RenderTarget2D(GraphicsDevice, moments.Length * 480, 420);
        GraphicsDevice.SetRenderTarget(target); GraphicsDevice.Clear(new Color(20,18,29));
        int frame = 0;
        for (int i = 0; i < moments.Length; i++)
        {
            while (frame < MathF.Round(moments[i] * 60)) { state.Update(25, 10, 100, 30, 1f / 60); frame++; }
            portrait.Draw(batch, Matrix.CreateScale(3) * Matrix.CreateTranslation(i * 480, 0, 0), new Vector2(60, 56), state, .84f, 4 + moments[i]);
        }
        Save(target, "portrait-levelup-strip");
    }
    private void RenderGame(SpriteBatch batch, float scale, bool wide = false)
    {
        using var target=new RenderTarget2D(GraphicsDevice,(int)(520*scale),(int)(150*scale));
        GraphicsDevice.SetRenderTarget(target);GraphicsDevice.Clear(new Color(120,167,245));
        var matrix=Matrix.CreateScale(scale);
        using var stream=File.OpenRead(Path.Combine(root,"Tools/ResourceBarPreview/MinimapFrame.png"));
        using var map=Texture2D.FromStream(GraphicsDevice,stream);
        batch.Begin(SpriteSortMode.Deferred,BlendState.AlphaBlend,SamplerState.PointClamp,DepthStencilState.None,RasterizerState.CullNone,null,matrix);
        batch.Draw(map,new Vector2(224,79),Color.White);batch.End();
        drawing.Draw(batch,matrix,ResourceBarDrawing.LifePosition(520),false,4,State(false,.84f,.84f));
        drawing.Draw(batch,matrix,ResourceBarDrawing.ManaPosition(520),true,4.7f,State(true,.68f,.68f));
        portrait.Draw(batch,matrix,PortraitHudDrawing.Position(520),wide ? Portrait(120,50,120,1) : Portrait(24,65,30,0),.84f,4);
        Save(target,$"game-layout-{scale:0.0}{(wide ? "-wide" : "")}");
    }
    private void RenderAnimation(SpriteBatch batch,float elapsed)
    {
        var hp=State(false,.84f,.84f);var mp=State(true,.68f,.68f);
        hp.Update(90,1000,0,4);mp.Update(200,1000,0,4.7f);
        for(int i=0;i<(int)MathF.Round(elapsed*60);i++){hp.Update(90,1000,1f/60,4+i/60f);mp.Update(200,1000,1f/60,4.7f+i/60f);}
        using var target=new RenderTarget2D(GraphicsDevice,966,228);
        GraphicsDevice.SetRenderTarget(target);GraphicsDevice.Clear(new Color(20,18,29));
        drawing.Draw(batch,Matrix.CreateScale(3),new Vector2(34,20),false,4+elapsed,hp);
        drawing.Draw(batch,Matrix.CreateScale(3),new Vector2(34,48),true,4.7f+elapsed,mp);
        Save(target,$"animation-{elapsed:0.0}");
    }
    private static void Check(bool pass,string label){if(!pass)throw new Exception(label);}
    private static void Checks()
    {
        var s=new ResourceBarState(false);s.Reset(420,500);s.Update(90,500,0,0,false);
        Check(Math.Abs(s.Value-.18f)<.00001f&&Math.Abs(s.Ghost-.84f)<.00001f,"damage must keep previous ghost");
        for(int i=0;i<12;i++)s.Update(90,500,1f/60,i/60f,false);
        Check(Math.Abs(s.Ghost-.84f)<.00001f,"ghost must wait 0.4 seconds");
        for(int i=0;i<240;i++)s.Update(90,500,1f/60,i/60f,false);
        Check(Math.Abs(s.Ghost-s.Value)<.00001f,"ghost must converge");
        s.Update(190,500,1f/60,5,false);Check(s.Value>.18f&&s.Value<.38f,"healing easing");
        s.Update(190,1000,1f/60,5,false);Check(s.Value==s.Ghost&&s.Particles.Count==0,"max value change resets visual state");
        s.Reset(200,200);s.Update(0,200,1f/60,6);Check(s.Particles.Count>0,"loss particles");
        for(int i=0;i<300;i++)s.Update(0,200,1f/60,6+i/60f);Check(s.Particles.Count==0,"particles expire at zero");
        Check(ResourceBarState.Fraction(1,0)==0&&ResourceBarState.Fraction(float.NaN,200)==0,"invalid resource denominators");
        Check(ResourceBarDrawing.ManaPosition(520).Y-ResourceBarDrawing.LifePosition(520).Y==28,"shared HUD spacing");
        var p=new PortraitHudState();p.Reset(1,0,100,30);int ups=0,shown=1;
        for(int i=0;i<600;i++){p.Update(4,50,100,30,1f/60);if(p.Shown>shown){ups++;shown=p.Shown;}}
        Check(ups==3&&p.Shown==4&&Math.Abs(p.Fraction-.5f)<.001f,"multi-level gain levels up once per level and keeps the remainder");
        Check(p.Sparks.Count==0,"level-up sparks expire");
        p.Update(3,0,100,30,1f/60);Check(p.Shown==3&&p.Fraction==0,"lost progress snaps without a level-up");
        p.Reset(30,100,100,30);Check(p.Capped&&p.Fraction==1,"cap stores one full level");
        p.Reset(5,40,0,30);Check(float.IsFinite(p.Fraction)&&p.Fraction==0,"zero experience requirement");
        p.Reset(1,0,100,90);for(int i=0;i<20;i++)p.Update(90,0,100,90,1f/60);Check(p.Sparks.Count<=66,"sparks stay bounded");
        var centre=PortraitHudDrawing.Position(520);
        float radius=PortraitHudDrawing.DisplayOuter;
        Check(PortraitHudDrawing.Contains(centre,centre+new Vector2(radius-1,0))&&!PortraitHudDrawing.Contains(centre,centre+new Vector2(radius*.7f,-radius*.7f)),"scaled diamond hover test");
        var label=centre+(PortraitHudDrawing.LevelAnchor+new Vector2(0,10))*PortraitHudDrawing.DisplayScale;
        Check(PortraitHudDrawing.Contains(centre,label)&&!PortraitHudDrawing.Contains(centre,centre+new Vector2(0,radius+20))&&!PortraitHudDrawing.Contains(centre,PortraitHudDrawing.TooltipPosition(centre)),"bottom level hover and tooltip clearance");
    }
    protected override void UnloadContent(){foreach(var texture in textures.Values)texture.Dispose();fonts?.Dispose();base.UnloadContent();}
    public static void Main(string[] args){using var game=new ResourcePreview(args[0],args[1]);game.Run();}
}

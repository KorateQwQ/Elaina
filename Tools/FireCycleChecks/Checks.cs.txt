    {

        typeof(NPC).GetField("_globals", All)!.SetValue(npc, new GlobalNPC[] { burn, new WaterExposureNPC(), new ElementalGlobalNPC() });
        GameMain.npc[index] = npc;
        return npc;
    }
    static Player Player(int index)
    {
        var player = new Player { whoAmI = index, active = true, statLife = 100, statLifeMax2 = 100 };
        var parts = new ModPlayer[] { new FireCyclePlayer(), new ElainaSkillModPlayer(), new ElainaAttributeModPlayer(), new ActionModPlayer() };
        for (ushort i = 0; i < parts.Length; i++) {
            SetEntity(parts[i], player); typeof(ModPlayer).GetProperty("Index", All)!.SetValue(parts[i], i);
            if (index == 0) ContentInstance.Register(parts[i]);
        }
        typeof(Player).GetField("modPlayers", All)!.SetValue(player, parts);
        GameMain.player[index] = player;
        return player;
    }
    static Projectile Proj(int index, ModProjectile type, int owner = 0)
    {
        var p = new Projectile { whoAmI = index, type = ProjectileID.WoodenArrowFriendly, active = true, identity = index + 100, owner = owner };

        typeof(Projectile).GetProperty("ModProjectile", All)!.SetValue(p, type);
        typeof(Projectile).GetField("_globals", All)!.SetValue(p, new GlobalProjectile[] { new FireWindState() });
        GameMain.projectile[index] = p;
        return p;
    }

        Check(!burst.CanUseSkill(),"empty charges cannot cast");
        File.WriteAllText(Path.Combine(args[0],".vissandbox/fire-cycle-checks/progress-snapshots.json"),
            System.Text.Json.JsonSerializer.Serialize(progressFrames));
        Console.WriteLine($"PASS {passed} fire-cycle assertions (real compiled skills, NPC state, packets and strike boundary).");
    }

    static void CheckDamageIgnition(FireCyclePlayer cycle, Projectile beam)
    {
        int events=0;
        void Reward(NPC _,int __)=>events++;
        FireBurnSystem.Ignited+=Reward;
        try {
            void Hit(FireBurnHit hit,int damage) => cycle.ApplyBeamHit(hit.TargetIndex,hit.TargetGeneration,
                hit.AnchorIndex,hit.AnchorGeneration,hit.WasBurning,beam.identity,beam.type,18,5,damage);
            float Progress(NPC npc)=>npc.GetGlobalNPC<ElementalGlobalNPC>().ElementAccumulation[ElementType.Fire].Current;
            var low=Npc(30); low.life=low.lifeMax=1000;
            for(int i=1;i<=9;i++) {Clock((uint)(10000+i*18));Hit(FireBurnNPC.CaptureHit(low)!.Value,1);}
            Check(Progress(low)==162 && !FireBurnNPC.IsBurning(low),"low damage keeps the original nine-pulse progress");
            Clock(10306);Hit(FireBurnNPC.CaptureHit(low)!.Value,1);
            Check(FireBurnNPC.IsBurning(low) && events==1,"low damage still ignites on the tenth pulse");

            var high=Npc(31);high.life=high.lifeMax=1000;
            Clock(11000);Hit(FireBurnNPC.CaptureHit(high)!.Value,250);
            Check(Progress(high)==90 && !FireBurnNPC.IsBurning(high),"twenty-five percent damage gives fifty percent buildup, without adding time on top");
            CaptureProgress("damage-25percent",high);
            Clock(11018);Hit(FireBurnNPC.CaptureHit(high)!.Value,250);
            Check(FireBurnNPC.IsBurning(high) && events==2,"fifty percent total damage ignites after two hits");

            var fraction=Npc(32);fraction.life=fraction.lifeMax=1001;
            for(int i=1;i<=8;i++){Clock((uint)(12000+i*18));Hit(FireBurnNPC.CaptureHit(fraction)!.Value,60);}
            Check(Math.Abs(Progress(fraction)-8*60*360d/1001)<.0001 && !FireBurnNPC.IsBurning(fraction),
                "fractional percentage buildup accumulates without per-hit integer truncation");

            var dead=Npc(33);dead.life=dead.lifeMax=100;
            FireBurnHit lethal=FireBurnNPC.CaptureHit(dead)!.Value;
            dead.life=0;dead.active=false;
            int before=events;
            Clock(13000);Hit(lethal,100);Hit(lethal,100);
            Check(events==before+1 && !dead.active && dead.life==0 && !dead.HasBuff<ElementalAffliction_Fire>(),
                "confirmed one-shot kill rewards once without reviving or applying buffs to the corpse");
            var reused=Npc(33);reused.life=reused.lifeMax=100;
            Hit(lethal,100);
            Check(events==before+1 && !reused.GetGlobalNPC<ElementalGlobalNPC>().ElementAccumulation.ContainsKey(ElementType.Fire),
                "late lethal receipt cannot ignite a reused NPC slot");

            var root=Npc(34);root.life=root.lifeMax=1000;
            var a=Npc(35,34);var b=Npc(36,34);
            a.lifeMax=20;b.lifeMax=50;
            var ha=FireBurnNPC.CaptureHit(a)!.Value;var hb=FireBurnNPC.CaptureHit(b)!.Value;
            Clock(14000);Hit(ha,20);
            Check(Progress(root)==18,"small body damage uses root maximum health and retains the time minimum");
            Clock(14001);Hit(hb,20);Hit(hb,20);
            Check(Progress(root)==18,"later body hits do not add another attachment or damage compensation");
            Clock(14018);Hit(FireBurnNPC.CaptureHit(a)!.Value,230);
            var last=FireBurnNPC.CaptureHit(b)!.Value;
            root.life=0;root.active=false;b.life=0;b.active=false;
            before=events;
            Clock(14019);Hit(last,230);
            Check(events==before && !root.active,"same-pulse lethal body hit cannot add a second attachment or ignition reward");

            var immune=Npc(37);immune.dontTakeDamage=true;
            Check(FireBurnNPC.CaptureHit(immune)==null,"immune targets cannot produce a valid hit snapshot");
            var zero=Npc(38);var hz=FireBurnNPC.CaptureHit(zero)!.Value;
            Hit(hz,0);Hit(hz,-1);
            Check(!zero.GetGlobalNPC<ElementalGlobalNPC>().ElementAccumulation.ContainsKey(ElementType.Fire),"zero or negative actual damage cannot add time or percentage progress");
            GameMain.netMode=NetmodeID.MultiplayerClient;
            Hit(hz,zero.lifeMax);
            Check(!zero.HasBuff<ElementalAffliction_Fire>(),"clients cannot settle damage ignition locally");
            GameMain.netMode=NetmodeID.SinglePlayer;

            // 检查真实弹幕入口：ModifyHitNPC 在存活时取身份，OnHitNPC 在死亡后提交实际 damageDone。
            var actual=(FireLaser)beam.ModProjectile;
            var target=Npc(39);target.life=target.lifeMax=100;
            int requests=0;FireBurnHit submitted=default;int submittedDamage=0;int submittedTicks=0;
            using var requestHook=new Hook(typeof(FireCyclePlayer).GetMethod("RequestBeamHit")!,
                (Action<FireCyclePlayer,FireBurnHit,Projectile,int,int>)((self,snapshot,proj,ticks,damage)=>{
                    requests++;submitted=snapshot;submittedDamage=damage;submittedTicks=ticks;Hit(snapshot,damage);
                }));
            var modifiers=new NPC.HitModifiers();actual.ModifyHitNPC(target,ref modifiers);
            target.life=0;target.active=false;before=events;
            actual.OnHitNPC(target,new NPC.HitInfo {Damage=99999},100);
            actual.OnHitNPC(target,new NPC.HitInfo {Damage=99999},100);
            Check(requests==1 && submitted.AnchorIndex==target.whoAmI && submittedDamage==100 && submittedTicks==10 && events==before+1,
                "real lethal callback forwards actual damage and consumes its pre-hit snapshot once");

            // 回归：附着去重只针对“同一脉冲 × 同一元素主体”，在发起端完成，与弹幕 identity/脉冲序号无关。
            // 旧实现由服务器按 identity + 序号判断，重新施法复用旧槽位时会把整段新脉冲当成重复包。
            var ignited=(HashSet<(int Index,int Generation)>)typeof(FireLaser).GetField("ignitedAnchors",All)!.GetValue(actual)!;
            var pulseRoot=Npc(43);var pulseBody=Npc(44,43);var pulseOther=Npc(45);
            foreach(var n in new[]{pulseRoot,pulseBody,pulseOther}) n.life=n.lifeMax=100000;
            void BeamHit(NPC n) {var m=new NPC.HitModifiers();actual.ModifyHitNPC(n,ref m);actual.OnHitNPC(n,new NPC.HitInfo{Damage=10},10);}
            ignited.Clear();requests=0;
            Clock(15000);BeamHit(pulseBody);BeamHit(pulseOther);
            Clock(15003);BeamHit(pulseRoot);
            Check(requests==2 && Progress(pulseRoot)==18 && Progress(pulseOther)==18,
                "one pulse submits one attachment per element anchor; separate enemies each get their own");
            for(int i=1;i<=3;i++){ignited.Clear();Clock((uint)(15000+i*10));BeamHit(i%2==0?pulseRoot:pulseBody);}
            Check(requests==5 && Progress(pulseRoot)==72,"every new paid pulse attaches again under the same projectile identity");

            var weakFinish=Npc(40);weakFinish.lifeMax=1000;weakFinish.life=10;
            var finish=FireBurnNPC.CaptureHit(weakFinish)!.Value;
            weakFinish.life=0;weakFinish.active=false;before=events;
            Hit(finish,10);
            Check(events==before,"a lethal hit below both buildup thresholds does not receive an unconditional kill bonus");

            var serverKill=Npc(41);serverKill.life=serverKill.lifeMax=100;
            var serverReceipt=FireBurnNPC.CaptureHit(serverKill)!.Value;
            serverKill.life=0;serverKill.active=false;before=events;
            GameMain.netMode=NetmodeID.Server;GameMain.myPlayer=255;
            Hit(serverReceipt,100);Hit(serverReceipt,100);
            Check(events==before+1 && !serverKill.active,"server accepts the matching defeated instance once after the strike packet");
            GameMain.netMode=NetmodeID.SinglePlayer;GameMain.myPlayer=0;

            var skill=(FireBurstSkill)GameMain.player[0].GetModPlayer<ElainaSkillModPlayer>().UnlockedSkill[nameof(FireBurstSkill)].ModSkill;
            skill.Stack=2;skill.CurrentCD=4;
            var small=Npc(42);small.life=small.lifeMax=100;
            var receipt=FireBurnNPC.CaptureHit(small)!.Value;small.life=0;small.active=false;
            Hit(receipt,100);Hit(receipt,100);
            Check(skill.Stack==2 && skill.CurrentCD==4,"lethal ignition does not grant a Flame Burst charge or change recovery");
        }
        finally {FireBurnSystem.Ignited-=Reward;GameMain.netMode=NetmodeID.SinglePlayer;GameMain.myPlayer=0;}
    }
    static void CheckIgnitionWithoutBurstRewards(Player player, FireBurstSkill burst)
    {
        // Load the old subscriber when testing an old assembly, so this regression fails before the removal.
        var oldType=typeof(FireCyclePlayer).Assembly.GetType("伊蕾娜.ElainaModSkills.Skills.Fire.FireCycleSystem");
        var oldBridge=oldType == null ? null : (ModSystem)Activator.CreateInstance(oldType)!;
        oldBridge?.Load();
        int spawned=0;
        using var spawnHook=new Hook(typeof(Projectile).GetMethod("NewProjectile",All,null,
            [typeof(IEntitySource),typeof(Vector2),typeof(Vector2),typeof(int),typeof(int),typeof(float),typeof(int),typeof(float),typeof(float),typeof(float)],null)!,
            (Func<IEntitySource,Vector2,Vector2,int,int,float,int,float,float,float,int>)((_,_,_,_,_,_,_,_,_,_)=>{spawned++;return 0;}));
        var channel=new DummyAction();
        var actions=player.GetModPlayer<ActionModPlayer>();
        typeof(ActionModPlayer).GetProperty("CurrentAnimAction",All)!.SetValue(actions,channel);
        float mana=player.GetModPlayer<ElainaAttributeModPlayer>().MagicPoint;
        try {
            foreach(int charges in new[]{0,1,3}) {
                burst.Stack=charges; burst.CurrentCD=charges==3 ? 0 : 4;
                float cooldown=burst.CurrentCD;
                var target=Npc(50+charges); var body=Npc(60+charges,50+charges);
                Clock((uint)(20000+charges*100));
                var hit=FireBurnNPC.CaptureHit(body)!.Value;
                Check(FireBurnNPC.ApplyBeamHit(hit,player.whoAmI,18,1,5000),"actual beam ignition still succeeds");
                Check(FireBurnNPC.IsBurning(target),"removing rewards preserves burning");
                Check(burst.Stack==charges && burst.CurrentCD==cooldown,"ignition at "+charges+" charges does not grant charges or alter cooldown");
                Check(spawned==0,"ignition never spawns an automatic Flame Burst, including at full charges");
                Clock((uint)(20002+charges*100));
                target.GetGlobalNPC<FireBurnNPC>().ApplyExposure(target,player.whoAmI,18,1);
                Check(burst.Stack==charges && burst.CurrentCD==cooldown && spawned==0,"refreshing burning has no Flame Burst reward");
            }
            Check(ReferenceEquals(actions.CurrentAnimAction,channel)
                && player.GetModPlayer<ElainaAttributeModPlayer>().MagicPoint==mana,"ignition preserves channel and mana");
            Check(!burst.CanUseSkill(),"manual burst cannot interrupt active channel");
            Check(typeof(FireCyclePlayer).GetMethod("ReceiveReward")==null && typeof(FireBurstSkill).GetMethod("GrantIgnitionCharge")==null,
                "old reward RPC and charge-grant entry points are removed");
        }
        finally {
            oldBridge?.Unload(); burst.Stack=3; burst.CurrentCD=0;
            typeof(ActionModPlayer).GetProperty("CurrentAnimAction",All)!.SetValue(actions,null);
        }
    }
    static ElementalGlobalNPC.BuildUpProgressContext ProgressRoundTrip(NPC npc, bool wasVisible=false)
    {
        using var stream=new MemoryStream();
        npc.GetGlobalNPC<ElementalGlobalNPC>().SendExtraAI(npc,new Terraria.ModLoader.IO.BitWriter(),new BinaryWriter(stream));
        stream.Position=0;
        var received=new ElementalGlobalNPC();
        if(wasVisible)received.ElementAccumulation[ElementType.Fire]=new() { Current=200,Max=300,VisualTime=120 };
        received.ReceiveExtraAI(npc,null!,new BinaryReader(stream));
        return received.ElementAccumulation[ElementType.Fire];
    }

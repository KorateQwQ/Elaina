    public static class Main

        public const int maxPlayers = 255, maxProjectiles = 1000;
        public static int netMode, myPlayer;
        public static bool dedServ => netMode == 2;
        public static Player[] player = [];
        public static Projectile[] projectile = [];
        public static Player LocalPlayer => player[myPlayer];
        public static IEnumerable<Player> ActivePlayers => player.Where(p => p.active);
        public static SpriteBatch spriteBatch;
        public static Vector2 screenPosition;
    }
    public sealed class Player
    {
        public int whoAmI;
        public bool active, dead, ghost;
        public Vector2 Center;
        readonly Dictionary<Type, ModPlayer> components = new();
        public T GetModPlayer<T>() where T : ModPlayer, new()
        {
            if (!components.TryGetValue(typeof(T), out var value))
                components[typeof(T)] = value = new T { Player = this };
            return (T)value;
        }
    }
    public sealed class NPC
    {
        public int type = 1, whoAmI = 10, lifeMax = 100, realLife = -1;
        public bool friendly, townNPC, SpawnedFromStatue;
        public Vector2 Center;
        public string FullName => "Test enemy";
        public object GetSource_Loot() => this;
    }
    public sealed class Projectile
    {
        public int owner, width, height, aiStyle, penetrate, timeLeft;
        public bool active, friendly, hostile, tileCollide, ignoreWater, netImportant, netUpdate;
        public float[] ai = new float[3];
        public Vector2 Center, velocity;
        public ModProjectile ModProjectile;
        public void Kill() => active = false;
        public static int NewProjectile(object source, Vector2 position, Vector2 velocity, int type, int damage,
            float knockback, int owner, float ai0, float ai1)
        {
            int index = Array.FindIndex(Main.projectile, p => !p.active);
            if (index < 0) return Main.maxProjectiles;
            var p = new Projectile { active = true, owner = owner, Center = position, velocity = velocity };
            p.ModProjectile = new ExperienceOrb { Projectile = p }; p.ModProjectile.SetDefaults();
            // Actual engine's owner guard: creating client-owned projectiles on the server loses initial AI.
            if (owner == Main.myPlayer) { p.ai[0] = ai0; p.ai[1] = ai1; }
            Main.projectile[index] = p;
            return index;
        }
    }

    public static class VectorExtensions
    {
        public static Vector2 RotatedBy(this Vector2 p, double angle) => new(
            p.X * (float)Math.Cos(angle) - p.Y * (float)Math.Sin(angle),
            p.X * (float)Math.Sin(angle) + p.Y * (float)Math.Cos(angle));
    }

    public class TagCompound : Dictionary<string, object>
    {

int passed = 0;
void Check(bool value, string description)
{ if (!value) throw new Exception(description); Console.WriteLine("PASS " + description); passed++; }
void Reset(int mode = NetmodeID.Server)
{
    TMain.netMode = mode; TMain.myPlayer = mode == NetmodeID.Server ? 255 : 0;
    TMain.player = Enumerable.Range(0, TMain.maxPlayers).Select(i => new Player { whoAmI = i }).ToArray();
    TMain.projectile = Enumerable.Range(0, TMain.maxProjectiles).Select(_ => new Projectile()).ToArray();
    ModPacket.Sent.Clear();
}
Player Activate(int index, Vector2 center, bool elaina = true)
{
    Player p = TMain.player[index]; p.active = true; p.Center = center;
    p.GetModPlayer<ElainaModplayer>().Elaina = elaina; return p;
}
Projectile[] Orbs() => TMain.projectile.Where(p => p.active).ToArray();
void Tick(int count)
{
    for (int i = 0; i < count; i++) foreach (Projectile p in Orbs())
    {
        p.ModProjectile.AI();
        if (!p.active) continue;
        p.Center += p.velocity;
        if (--p.timeLeft <= 0) p.Kill();
    }
}
void Receive((int Target, byte[] Data) packet)
{
    using var reader = new BinaryReader(new MemoryStream(packet.Data)); reader.ReadByte();
    ExperienceNetwork.HandlePacket(reader, reader.ReadInt32());
}

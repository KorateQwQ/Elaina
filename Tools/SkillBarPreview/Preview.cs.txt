using 伊蕾娜.ElainaModSkills.UICore;
using System;
using System.IO;
using System.Collections.Generic;
using Microsoft.Xna.Framework;
using Microsoft.Xna.Framework.Graphics;
using ReLogic.Graphics;
using SDL2;
using 伊蕾娜.ElainaModSkills.ElainaSkillUI;

internal sealed class PreviewGame : Game
{
    private readonly string root;
    private readonly string output;
    private readonly Dictionary<string, Texture2D> textures = new();
    private bool rendered;
    private UIRasterShapes shapes;
    private static bool ScaleOnly;
    private Effect cooldown;
    private UIRasterFont numberFont;
    private DynamicSpriteFont captionFont;
    private Effect sealEffect;
    private Texture2D pixel, glow, diamond;

    private PreviewGame(string root, string output)
    {
        this.root = root;
        this.output = output;
        new GraphicsDeviceManager(this) { PreferredBackBufferWidth = 320, PreferredBackBufferHeight = 180,
            GraphicsProfile = GraphicsProfile.HiDef, SynchronizeWithVerticalRetrace = false };
        IsFixedTimeStep = false;
        SDL.SDL_HideWindow(Window.Handle);
    }

    private Texture2D Texture(string relative)
    {
        if (textures.TryGetValue(relative, out var texture)) return texture;
        using var stream = File.OpenRead(Path.Combine(root, relative + ".png"));
        texture = Texture2D.FromStream(GraphicsDevice, stream);
        // tML's rawimg conversion preserves RGB except at zero alpha.
        var data = new Color[texture.Width * texture.Height];
        texture.GetData(data);
        for (int i = 0; i < data.Length; i++) if (data[i].A == 0) data[i] = Color.Transparent;
        texture.SetData(data);
        textures.Add(relative, texture);
        return texture;
    }

    private readonly Dictionary<string, UIRasterTexture> artwork = new();
    private UIRasterTexture ArtTexture(string name)
    {
        const string root = "ElainaModSkills/ElainaSkillUI/BattleBar/";
        if (!artwork.TryGetValue(name, out var art)) artwork[name] = art = new(Texture(root + name), level => Texture(root + name + "Lod" + level));
        return art;
    }

    protected override void LoadContent()
    {
        Content.RootDirectory = Path.GetFullPath(Path.Combine(root, "../KL"));
        cooldown = Content.Load<Effect>("Effects/Content/CDEffect");
        captionFont = Content.Load<DynamicSpriteFont>("Fonts/NotoSerifSC");
        numberFont = new("NotoSerifSC", captionFont, 9, 48, path => File.ReadAllBytes(Path.Combine(root, path)), Texture);
        Content.RootDirectory = root;
        sealEffect = Content.Load<Effect>("Effects/Content/SkillSeal");
        pixel = Texture("Effects/Tex/Baked/WhitePixel");
        shapes = new(pixel, Content.Load<Effect>("Effects/Content/UIShape"));
        glow = Texture("ElainaModSkills/UICore/Assets/GlowDot");
        diamond = Texture("ElainaModSkills/UICore/Assets/DiamondFill");
    }

    protected override void Update(GameTime time)
    {
        if (rendered) return;
        rendered = true;
        SDL.SDL_HideWindow(Window.Handle);
        Directory.CreateDirectory(output);
        using var batch = new SpriteBatch(GraphicsDevice);
        if (ScaleOnly)
        {
            foreach (float uiScale in new[] { 1f, 1.125f, 1.25f, 2f })
            foreach (float barScale in new[] { 1f, .75f, .5f })
                RenderSeals(batch, uiScale, false, barScale);
            Exit();
            return;
        }
        foreach (float scale in new[] { 1f, 1.5f })
        {
            RenderSeals(batch, scale, false);
            RenderSeals(batch, scale, true);
            Render(batch, scale, .7f, 2, -1, "selected");
            Render(batch, scale, 1.4f, 2, -1, "orbit");
            Render(batch, scale, .16f, 0, 6, "empty-hover");
            Render(batch, scale, .7f, 2, -1, "cooldown");
            foreach (float seconds in new[] { 0f, .055f, .12f, .24f, .45f, .7f })
                Render(batch, scale, seconds, 2, -1, $"arrival-{seconds:0.000}");
        }
        Render(batch, 1f, 0, -1, -1, "no-selection");
        Render(batch, 1f, .7f, 7, -1, "last-slot");
        for (int frame = 0; frame < 60; frame++)
        {
            float seconds = frame / 30f;
            Render(batch, 1.5f, seconds % 1, seconds < 1 ? 1 : 2, -1, $"animation-{frame:000}");
        }
        Exit();
    }

    private void RenderSeals(SpriteBatch batch, float scale, bool day, float barScale = 1)
    {
        using var target = new RenderTarget2D(GraphicsDevice, (int)(710 * scale), (int)(150 * scale));
        GraphicsDevice.SetRenderTarget(target);
        GraphicsDevice.Clear(day ? new Color(130, 174, 215) : new Color(19, 20, 28));
        var matrix = Matrix.CreateScale(scale);
        void Begin(Effect effect = null) => batch.Begin(effect == null ? SpriteSortMode.Deferred : SpriteSortMode.Immediate,
            BlendState.AlphaBlend, SamplerState.LinearClamp, DepthStencilState.None, RasterizerState.CullNone, effect, matrix);
        var position = new Vector2(32, 20);
        var stateArt = new SkillBarDrawing.StateArt(pixel, glow, diamond, shapes);
        var icon = Texture("ElainaModSkills/Skills/Ice/ShardbindSkill");
        var atlas = ArtTexture("Ornaments");
        Begin(); SkillBarDrawing.DrawRack(batch, ArtTexture("Rack"), position, barScale);
        string[] labels = ["sealed", "sealed+CD", "denied", "0.06s", "0.25s", "0.50s", "open+CD", "ready"];
        for (int i = 0; i < 8; i++)
        {
            Vector2 center = position + new Vector2(57 + 78 * i, 55) * barScale;
            bool sealedState = i < 3;
            float releaseAge = i switch { 3 => .06f, 4 => .25f, 5 => .5f, _ => -1 };
            float remaining = i is 1 or 4 or 6 ? .45f : 0;
            bool bound = SkillBarDrawing.HasSealOverlay(sealedState, releaseAge);
            SkillBarDrawing.DrawSlotBackground(batch, atlas, center, i == 2, true, 0, .7f, scale: barScale);
            batch.End();
            sealEffect.Parameters["time"].SetValue(1 - remaining);
            sealEffect.Parameters["flashTime"].SetValue(0f);
            sealEffect.Parameters["ready"].SetValue(remaining > 0 ? 0f : 1f);
            sealEffect.Parameters["seal"].SetValue(SkillBarDrawing.SealTone(sealedState, releaseAge));
            Begin(sealEffect);
            batch.Draw(icon, center, null, Color.White, 0, new Vector2(icon.Width, icon.Height) / 2,
                50f * barScale / Math.Max(icon.Width, icon.Height), SpriteEffects.None, 0);
            batch.End(); Begin();
            float flash = SkillBarDrawing.SealReleasePulse(releaseAge);
            if (flash > 0) batch.Draw(icon, center, null, new Color(245, 222, 255, 0) * flash, 0,
                new Vector2(icon.Width, icon.Height) / 2, 50f * barScale / Math.Max(icon.Width, icon.Height), SpriteEffects.None, 0);
            SkillBarDrawing.DrawSlotFrame(batch, atlas, center, i == 2, true, 0, .7f, scale: barScale);
            SkillBarDrawing.DrawSkillState(batch, stateArt, numberFont, center, 60 * remaining, remaining, remaining > 0 ? 0 : 1, ScaleOnly && i == 6 ? 3 : ScaleOnly && i == 7 ? 8 : 1, !bound, barScale);
            if (bound) SkillBarDrawing.DrawSealState(batch, stateArt, center, .7f, i == 2 ? .12f : 9, releaseAge, scale: barScale);
            batch.DrawString(captionFont, labels[i], center + new Vector2(0, 59 * barScale), day ? Color.Black : Color.White,
                0, captionFont.MeasureString(labels[i]) * new Vector2(.5f, 0), 10f / 48 * barScale, SpriteEffects.None, 0);
        }
        batch.End(); GraphicsDevice.SetRenderTarget(null);
        string file = Path.Combine(output, ScaleOnly
            ? $"seal-states-{(day ? "day" : "dark")}-ui{scale:0.00}-bar{barScale:0.00}.png"
            : $"seal-states-{(day ? "day" : "dark")}-{scale:0.0}.png");
        using var stream = File.Create(file); target.SaveAsPng(stream, target.Width, target.Height);
        Console.WriteLine(file);
    }

    private void Render(SpriteBatch batch, float scale, float seconds, int selected, int hover, string state)
    {
        using var target = new RenderTarget2D(GraphicsDevice, (int)(710 * scale), (int)(150 * scale));
        GraphicsDevice.SetRenderTarget(target);
        GraphicsDevice.Clear(new Color(19, 20, 28));
        var position = new Vector2(32, 20);
        var atlas = ArtTexture("Ornaments");
        batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.LinearClamp,
            DepthStencilState.None, RasterizerState.CullNone, null, Matrix.CreateScale(scale));
        SkillBarDrawing.DrawRack(batch, ArtTexture("Rack"), position);
        for (int i = 0; i < 8; i++)
        {
            Vector2 center = position + new Vector2(57 + 78 * i, 55);
            bool occupied = i == 1 || i == 2;
            SkillBarDrawing.DrawSlotBackground(batch, atlas, center, selected == i, occupied, hover == i ? 1 : 0, seconds);
            if (occupied)
            {
                var icon = Texture(i == 1 ? "ElainaModSkills/Skills/Fire/FireLaserSkill" : "ElainaModSkills/Skills/Fire/FireTornadoSkill");
                if (state == "cooldown" && i == 2)
                {
                    batch.End();
                    cooldown.Parameters["time"].SetValue(.4f);
                    cooldown.Parameters["flashTime"].SetValue(0f);
                    batch.Begin(SpriteSortMode.Immediate, BlendState.AlphaBlend, SamplerState.LinearClamp,
                        DepthStencilState.None, RasterizerState.CullNone, cooldown, Matrix.CreateScale(scale));
                }
                batch.Draw(icon, center, null, Color.White, 0, new Vector2(icon.Width, icon.Height) / 2,
                    50f / Math.Max(icon.Width, icon.Height), SpriteEffects.None, 0);
                if (state == "cooldown" && i == 2)
                {
                    batch.End();
                    batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.LinearClamp,
                        DepthStencilState.None, RasterizerState.CullNone, null, Matrix.CreateScale(scale));
                    UIRasterText.Draw(batch, numberFont, "2.4", center, Color.White,
                        UIRasterText.Pixels(batch, 22, 11), new Vector2(.5f));
                }
                if (state == "cooldown" && i == 1)
                    UIRasterText.Draw(batch, numberFont, "3", center + new Vector2(18, 16), Color.White,
                        UIRasterText.Pixels(batch, 14), new Vector2(.5f));
            }
            SkillBarDrawing.DrawSlotFrame(batch, atlas, center, selected == i, occupied, hover == i ? 1 : 0, seconds);
        }
        batch.End();
        GraphicsDevice.SetRenderTarget(null);
        string file = Path.Combine(output, $"{state}-{scale:0.0}.png");
        using var stream = File.Create(file);
        target.SaveAsPng(stream, target.Width, target.Height);
        Console.WriteLine(file);
    }

    protected override void UnloadContent()
    {
        foreach (var texture in textures.Values) texture.Dispose();
        base.UnloadContent();
    }

    [STAThread]
    private static void Main(string[] args)
    {
        ScaleOnly = Array.IndexOf(args, "--scale-only") >= 0;
        using var game = new PreviewGame(args[0], args[1]);
        game.Run();
    }
}

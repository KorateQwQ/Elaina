static class Checks
{
    static int passed;
    static bool registeredPlayers;
    static void Check(bool value, string reason)
    {
        if (!value) throw new Exception(reason);
        passed++;
    }

    static (ElainaStatePlayer State, ElainaSkillModPlayer Skills) CharacterFixture(int index = 0)
    {
        const BindingFlags flags = BindingFlags.Instance | BindingFlags.NonPublic | BindingFlags.Public;
        var entity = new Player { active = true, whoAmI = index };
        var state = new ElainaStatePlayer(); var skills = new ElainaSkillModPlayer();
        ModPlayer[] components = [state, skills];
        for (ushort i = 0; i < components.Length; i++) {
            typeof(ModPlayer).GetProperty("Entity", flags)!.SetValue(components[i], entity);
            typeof(ModPlayer).GetProperty("Index", flags)!.SetValue(components[i], i);
            if (!registeredPlayers) ContentInstance.Register(components[i]);
        }
        registeredPlayers = true;
        typeof(Player).GetField("modPlayers", flags)!.SetValue(entity, components);
        return (state, skills);
    }

    static void Main()
    {
        Terraria.Program.SavePath = Path.GetTempPath();
        GameMain.netMode = NetmodeID.SinglePlayer; GameMain.myPlayer = 0;
        var manager = typeof(KLGameStateManager);
        const BindingFlags flags = BindingFlags.Static | BindingFlags.NonPublic;
        var originals = (IDictionary)manager.GetField("origBossInfos", flags)!.GetValue(null)!;
        var originalType = manager.GetNestedType("OrigBossChecklistBossInfo", BindingFlags.NonPublic)!;
        var bosses = KLGameStateManager.GetBossInfos();
        // World predicates must never be evaluated by character progression.
        Func<bool> forbiddenWorldRead = () => throw new Exception("Read world boss state!");
        void Original(int id, float progress)
        {
            originals.Add(id, Activator.CreateInstance(originalType, progress, forbiddenWorldRead));
            ((LocalizedText[])typeof(Lang).GetField("_npcNameCache", flags)!.GetValue(null)!)[id] = Language.GetOrRegister("ProgressChecks.NPC" + id, () => "NPC" + id);
        }
        KLGameStateManager.BossChecklistBossInfo Boss(string key, float progress, bool isBoss = true)
        {
            var info = new KLGameStateManager.BossChecklistBossInfo {
                progression = progress, isBoss = isBoss,
                displayName = Language.GetOrRegister("ProgressChecks." + key, () => key)
            };
            typeof(KLGameStateManager.BossChecklistBossInfo).GetField("key", BindingFlags.Instance | BindingFlags.NonPublic)!.SetValue(info, key);
            typeof(KLGameStateManager.BossChecklistBossInfo).GetField("downed", BindingFlags.Instance | BindingFlags.NonPublic)!.SetValue(info, forbiddenWorldRead);
            bosses[key] = info;
            return info;
        }

        var (player, skills) = CharacterFixture();
        player.DebugResetProgress();
        Check(!player.DebugGetNextBoss(out _, out _), "Uninitialized lists have no next stage.");
        Original(NPCID.KingSlime, 1);
        Original(NPCID.EyeofCthulhu, 2);
        Original(NPCID.EaterofWorldsHead, 3);
        Original(NPCID.EaterofWorldsBody, 3);
        Original(NPCID.BrainofCthulhu, 3);
        Original(NPCID.Retinazer, 9);
        Original(NPCID.Spazmatism, 9);
        Original(NPCID.MoonLordCore, 18);
        Check(player.GetMaxLevel() == 5, "Reset cap is 5.");
        Check(player.DebugGetNextBoss(out var name, out var state) && state == 1 && name == "NPC50", "Empty external list uses original King Slime.");
        Check(player.DebugGetNextBossTooltip().Contains("5 → Lv. 10"), "Tooltip predicts the new cap.");
        Check(player.Progress == 0 && player.ProgressSet.Count == 0, "Preview never mutates character.");
        player.Experience = 65;
        Check(player.DebugMaxLevel() && player.CurrentLevel == 5 && player.Experience == 0,
            "Max level reaches the initial cap and clears XP.");
        Check(skills.SkillPoint == 100, "Debug max level grants the initial four levels and Lv5 bonus.");
        Check(player.Progress == 0 && player.ProgressSet.Count == 0 && !player.DebugMaxLevel(),
            "Max level is repeatable without advancing progress or recording bosses.");
        player.CurrentLevel = 5;
        player.Experience = 42;
        Check(player.DebugAdvanceMaxLevel(out _, out state) && state == 1, "Advance to King Slime.");
        Check(player.Progress == 1 && player.GetMaxLevel() == 10 && player.ProgressSet.SequenceEqual(new[] { 1f }), "Advance progress, record and cap together.");
        Check(player.CurrentLevel == 5 && player.Experience == 42, "Raising cap does not grant levels or erase XP.");
        Check(skills.SkillPoint == 220, "Cap 5 to 10 grants 120 study points in advance.");
        Check(player.DebugMaxLevel() && player.CurrentLevel == 10 && player.Experience == 0
            && player.Progress == 1 && player.ProgressSet.SequenceEqual(new[] { 1f }),
            "Max level reevaluates the raised cap and preserves boss history.");
        Check(skills.SkillPoint == 220, "Max level does not repeat prepaid study points.");
        Check(player.DebugGetNextBossTooltip().Contains("NPC4"), "Tooltip immediately moves on to Eye.");
        player.DebugAdvanceMaxLevel(out _, out _);
        Check(player.DebugAdvanceMaxLevel(out name, out state) && state == 3 && name.Contains(" / "), "Evil bosses share one stage.");
        Check(player.DebugAdvanceMaxLevel(out name, out state) && state == 9 && name.Contains(" / "), "Segments and equal states are skipped; twins share a stage.");
        player.DebugAdvanceMaxLevel(out _, out _);
        int finalPoints = skills.SkillPoint;
        Check(!player.DebugAdvanceMaxLevel(out _, out _) && player.Progress == 18 && skills.SkillPoint == finalPoints,
            "Final boss is a non-mutating no-op without repeated points.");
        player.DebugResetProgress();
        Check(player.DebugGetNextBoss(out _, out state) && state == 1 && player.CurrentLevel == 1 && player.Experience == 0, "Reset returns to first boss.");

        Boss("Later", 2);
        Boss("Event", .25f, false);
        Boss("First", .5f);
        Boss("SameStage", .5f);
        Check(player.DebugAdvanceMaxLevel(out name, out state) && state == .5f && name == "First", "External bosses use sorted character progress and stable ties, not events or world state.");
        Check(player.GetMaxLevel() == KLGameStateManager.GetLevelCap(.5f), "Fractional progression obeys KL cap mapping even when unchanged.");
        Check(skills.SkillPoint == 0, "Reset clears points and equal-cap bosses do not grant points.");
        Check(player.DebugGetNextBoss(out name, out state) && name == "Later" && state == 2, "Equal progression is not advanced twice.");
        player.DebugAdvanceMaxLevel(out _, out _);
        Check(!player.DebugGetNextBoss(out _, out _), "Do not mix vanilla fallback states into an active external list.");
        Check(player.DebugMaxLevel() && player.CurrentLevel == player.GetMaxLevel(), "Max level follows the external progression cap.");
        var save = new TagCompound();
        player.SaveData(save);
        var (loaded, _) = CharacterFixture();
        loaded.LoadData(save);
        Check(loaded.Progress == player.Progress && loaded.GetMaxLevel() == player.GetMaxLevel() && loaded.ProgressSet.SequenceEqual(player.ProgressSet), "Save/load retains advanced progress and cap.");
        Check(loaded.CurrentLevel == loaded.GetMaxLevel() && loaded.Experience == 0, "Save/load retains max level and XP.");
        loaded.DebugResetProgress();
        Check(loaded.DebugGetNextBoss(out _, out state) && state == .5f, "Reset on external list starts at character zero.");
        Check(player.Progress == 2, "Resetting another character does not change this one.");
        var legacyRemainder = new ElainaStatePlayer();
        legacyRemainder.LoadData(new TagCompound { ["progress"] = 2f, ["currentLevel"] = 10,
            ["experience"] = 440, ["progressSet"] = new List<float>() });
        Check(legacyRemainder.CurrentLevel == 12 && legacyRemainder.Experience == 50,
            "Reduced XP requirements consume the old saved remainder before clamping.");
        var converted = new TagCompound(); legacyRemainder.SaveData(converted);
        legacyRemainder.LoadData(converted);
        Check(legacyRemainder.CurrentLevel == 12 && legacyRemainder.Experience == 50,
            "Converted remainder survives actual TagCompound save/load without repeated level grants.");
        bosses.Clear();
        CheckStudyPoints();
        CheckStudyCosts();
        CheckBalanceMigration();
        Console.WriteLine($"PASS: {passed} character progress checks against production assemblies.");
    }

    static void CheckStudyPoints()
    {
        var (state, skills) = CharacterFixture();
        Check(ExperienceRules.StudyPointsForLevels(1, 1) == 0
            && ExperienceRules.StudyPointsForLevels(1, 35) == 820
            && ExperienceRules.StudyPointsForLevels(4, 5) == 40
            && ExperienceRules.StudyPointsForLevels(10, 5) == 0
            && ExperienceRules.StudyPointsForLevels(1, int.MaxValue) > int.MaxValue,
            "Study-point totals cover milestones, backwards bounds and large values.");
        state.AddExperience(100);
        Check(state.CurrentLevel == 2 && skills.SkillPoint == 20, "Normal experience grants twenty study points per level.");
        state.AddExperience(int.MaxValue);
        Check(state.CurrentLevel == 5 && skills.SkillPoint == 100, "Multi-level normal awards include the Lv5 bonus.");
        state.AddExperience(1000); state.GetLevel(); state.GetLevel();
        Check(skills.SkillPoint == 100, "Banked experience and repeated level reads do not pay at the cap.");
        // Simulate a real Boss unlocking a new cap, without the debug prepayment.
        state.Progress = 1;
        Check(state.GetLevel() == 6 && skills.SkillPoint == 120, "A banked level consumed after a Boss pays once.");
        state.AddExperience(int.MaxValue);
        Check(state.CurrentLevel == 10 && skills.SkillPoint == 220, "Normal Lv10 total is 220 study points.");
        var (prepaid, prepaidSkills) = CharacterFixture();
        prepaid.DebugMaxLevel();
        Check(prepaid.DebugGetNextBossTooltip().Contains("补发 120 研习点"), "Cap tooltip predicts the unpaid reward.");
        prepaid.DebugAdvanceMaxLevel(out _, out _);
        Check(prepaid.CurrentLevel == 5 && prepaidSkills.SkillPoint == 220, "Debug cap advance prepays without increasing current level.");
        var save = new TagCompound(); prepaid.SaveData(save);
        var pointsSave = new TagCompound(); prepaidSkills.SaveData(pointsSave);
        var (loaded, loadedSkills) = CharacterFixture();
        loaded.LoadData(save); loadedSkills.LoadData(pointsSave);
        loaded.AddExperience(int.MaxValue);
        Check(loaded.CurrentLevel == 10 && loadedSkills.SkillPoint == 220, "Prepaid claim history and balance survive save/load without double payment.");
        loadedSkills.SkillPoint = 85;
        loaded.DebugResetProgress();
        Check(loadedSkills.SkillPoint == 0 && loaded.CurrentLevel == 1 && loaded.Experience == 0,
            "Reset clears study points together with level and experience.");
        loaded.DebugResetProgress();
        var resetSave = new TagCompound(); loaded.SaveData(resetSave);
        var resetPointsSave = new TagCompound(); loadedSkills.SaveData(resetPointsSave);
        loaded.LoadData(resetSave); loadedSkills.LoadData(resetPointsSave);
        Check(loadedSkills.SkillPoint == 0 && resetSave.GetInt("studyPointRewardLevel") == 1,
            "Repeated reset and save/load preserve zero balance and reset reward history.");
        loaded.AddExperience(100);
        Check(loaded.CurrentLevel == 2 && loadedSkills.SkillPoint == 20,
            "Leveling after reset grants study points again.");
        loaded.DebugMaxLevel(); loaded.DebugAdvanceMaxLevel(out _, out _);
        Check(loadedSkills.SkillPoint == 220, "Debug leveling after reset pays each level once.");
        loaded.DebugMaxLevel();
        Check(loadedSkills.SkillPoint == 220, "Reset does not disable prepaid reward deduplication.");
        var (legacy, legacySkills) = CharacterFixture();
        legacySkills.SkillPoint = 37;
        legacy.LoadData(new TagCompound { ["progress"] = 1f, ["currentLevel"] = 8, ["experience"] = 0 });
        Check(legacySkills.SkillPoint == 37, "Old saved levels are not retroactively paid.");
        legacy.AddExperience(ExperienceRules.RequiredForLevel(8));
        Check(legacy.CurrentLevel == 9 && legacySkills.SkillPoint == 57, "Old saves receive study points for their next actual level.");
        legacySkills.SkillPoint = int.MaxValue - 5;
        legacy.AddExperience(ExperienceRules.RequiredForLevel(9));
        Check(legacySkills.SkillPoint == int.MaxValue, "Study-point awards saturate instead of overflowing.");
        var (remote, remoteSkills) = CharacterFixture(1);
        Check(remote.AddExperience(100) == 0 && remoteSkills.SkillPoint == 0, "Remote characters cannot receive local rewards.");
        GameMain.netMode = NetmodeID.Server;
        var (server, serverSkills) = CharacterFixture();
        Check(server.AddExperience(100) == 0 && serverSkills.SkillPoint == 0, "Server does not write client-owned study points.");
        GameMain.netMode = NetmodeID.SinglePlayer;
        var (fresh, freshSkills) = CharacterFixture();
        fresh.DebugAdvanceMaxLevel(out _, out _);
        Check(fresh.CurrentLevel == 1 && freshSkills.SkillPoint == 220, "First debug cap grant includes unpaid initial levels.");
    }

    static void CheckBalanceMigration()
    {
        var oldSave = new TagCompound { ["progress"] = 6f, ["currentLevel"] = 35,
            ["studyPointRewardLevel"] = 35, ["experience"] = 0 };
        var (state, skills) = CharacterFixture();
        state.LoadData(oldSave);
        // Simulate the skill component loading later, with previously spent points.
        skills.SkillPoint = 13;
        var ownedDoll = new Skill();
        typeof(Skill).GetProperty("ModSkill", BindingFlags.Instance | BindingFlags.Public | BindingFlags.NonPublic)!
            .SetValue(ownedDoll, new DollControlSkill { Level = 7 });
        skills.UnlockedSkill[nameof(DollControlSkill)] = ownedDoll;
        Check(skills.SkillPoint == 13, "Loading state cannot pay before the skill balance has loaded");
        state.OnEnterWorld();
        Check(skills.SkillPoint == 483, "Lv35 migration pays 410 income difference and 60 doll upgrade refund");
        state.OnEnterWorld();
        Check(skills.SkillPoint == 483, "Entering another world cannot repeat migration");
        var save = new TagCompound(); state.SaveData(save);
        var (loaded, loadedSkills) = CharacterFixture();
        loaded.LoadData(save); loadedSkills.SkillPoint = 483; loaded.OnEnterWorld();
        Check(loadedSkills.SkillPoint == 483, "Migration version survives save/load without paying again");

        var (prepaid, prepaidSkills) = CharacterFixture();
        prepaid.LoadData(new TagCompound { ["progress"] = 1f, ["currentLevel"] = 5, ["studyPointRewardLevel"] = 10 });
        prepaidSkills.SkillPoint = 7; prepaid.OnEnterWorld();
        Check(prepaidSkills.SkillPoint == 117, "Migration includes previously prepaid levels");
        prepaid.AddExperience(int.MaxValue);
        Check(prepaidSkills.SkillPoint == 117 && prepaid.CurrentLevel == 10, "Normal leveling cannot repeat migrated prepaid rewards");

        var (legacy, legacySkills) = CharacterFixture();
        legacy.LoadData(new TagCompound { ["progress"] = 1f, ["currentLevel"] = 8 });
        legacySkills.SkillPoint = 37; legacy.OnEnterWorld();
        Check(legacySkills.SkillPoint == 117, "Old saves without reward history receive only the balance increase");
        legacy.AddExperience(ExperienceRules.RequiredForLevel(8));
        Check(legacySkills.SkillPoint == 137, "First new level after migration earns the new reward once");

        var (pending, pendingSkills) = CharacterFixture(); pending.LoadData(oldSave);
        var pendingSave = new TagCompound(); pending.SaveData(pendingSave);
        pending.LoadData(pendingSave); pending.OnEnterWorld();
        Check(pendingSkills.SkillPoint == 410, "Saving before world entry preserves the pending migration");
        var (saturated, saturatedSkills) = CharacterFixture(); saturated.LoadData(oldSave);
        saturatedSkills.SkillPoint = int.MaxValue - 1; saturated.OnEnterWorld();
        Check(saturatedSkills.SkillPoint == int.MaxValue, "Migration saturates without integer overflow");
        var (remote, remoteSkills) = CharacterFixture(1); remote.LoadData(oldSave); remote.OnEnterWorld();
        Check(remoteSkills.SkillPoint == 0, "Migration never writes another client's balance");
        GameMain.netMode = NetmodeID.Server;
        var (server, serverSkills) = CharacterFixture(); server.LoadData(oldSave); server.OnEnterWorld();
        Check(serverSkills.SkillPoint == 0, "Server cannot migrate client-owned points");
        GameMain.netMode = NetmodeID.SinglePlayer;
        var (fresh, freshSkills) = CharacterFixture(); fresh.OnEnterWorld();
        Check(freshSkills.SkillPoint == 0, "New characters have no legacy bonus");
        fresh.LoadData(oldSave); fresh.DebugResetProgress(); fresh.OnEnterWorld();
        Check(freshSkills.SkillPoint == 0, "Debug reset cancels a pending legacy refund");
    }

    static IEnumerable<SkillUnlockCondition> Leaves(SkillUnlockCondition condition) =>
        condition is CompositeSkillUnlockCondition group ? group.Conditions.SelectMany(Leaves) : [condition];

    static void CheckStudyCosts()
    {
        var (_, skills) = CharacterFixture();
        Check(SkillBalanceConfig.TryParse(File.ReadAllText(SkillBalanceConfig.ResourcePath), out var config, out var error), error);
        int total = 0;
        var plan = new (ElainaSkill Skill, int Unlock, int Upgrade, int Level)[] {
            (new AshenWitchSkill(), 0, 0, 1), (new MagicMissileSkill(), 0, 5, 15),
            (new MagicBarrierSkill(), 10, 0, 1), (new MultiMissileSkill(), 20, 15, 7),
            (new AlchemySkill(), 10, 15, 7), (new DollControlSkill(), 20, 5, 7),
            (new FlightSkill(), 15, 10, 3), (new WindSlashSkill(), 20, 5, 9),
            (new HealSkill(), 20, 15, 2), (new WaterLaserSKill(), 25, 5, 5),
            (new WaterBallSkill(), 25, 15, 3), (new RecallSkill(), 20, 0, 1) };
        foreach (var (skill, unlock, upgrade, level) in plan) {
            string id = skill.GetType().Name;
            var leaves = Leaves(skill.UnlockCondition).ToArray();
            Check(leaves.OfType<SkillPointUnlockCondition>().Sum(c => c.SkillPointCost) == unlock, id + " planned unlock cost");
            var items = leaves.OfType<ItemSkillUnlockCondition>().SelectMany(c => c.Items).ToArray();
            Check(skill is RecallSkill ? items.Length == 1 && items[0].ItemType == ItemID.PotionOfReturn && items[0].Stack == 1
                : items.Length == 0, id + " no placeholder materials; only Recall consumes a return potion");
            if (level > 1) {
                Check(config.GetSkillLevelCap(config.Skills[id], 35) == level, id + " meat-pre progression nodes match budget");
                var owned = new Skill();
                typeof(Skill).GetProperty("ModSkill", BindingFlags.Instance | BindingFlags.Public | BindingFlags.NonPublic)!.SetValue(owned, skill);
                skill.Skill = owned;
                for (int next = 2; next <= level; next++) {
                    var cost = skill.GetConstellationUpgradeCondition(next);
                    Check(cost is SkillPointUnlockCondition p && p.SkillPointCost == upgrade, id + " point-only upgrade at " + next);
                    skills.SkillPoint = upgrade - 1;
                    Check(!cost.TryUnlock(skills, skill) && skills.SkillPoint == upgrade - 1, id + " shortage cannot spend");
                    skills.SkillPoint = upgrade;
                    Check(cost.TryUnlock(skills, skill) && skills.SkillPoint == 0, id + " exact upgrade points consumed");
                }
            }
            total += unlock + (level - 1) * upgrade;
        }
        Check(total == 590, "All planned pre-Hardmode skills at Lv35 cost 590 points including Recall");
        foreach (int characterLevel in Enumerable.Range(5, 31)) {
            int budget = 0;
            foreach (var (skill, unlock, upgrade, _) in plan) {
                if (config.Skills.TryGetValue(skill.GetType().Name, out var entry)) {
                    if (characterLevel < entry.UnlockLevel) continue;
                    budget += unlock + (config.GetSkillLevelCap(entry, characterLevel) - 1) * upgrade;
                } else budget += unlock;
            }
            long earned = ExperienceRules.StudyPointsForLevels(1, characterLevel);
            Check(earned >= budget, $"Lv{characterLevel}: {earned} points cover all available skills ({budget}), including Recall");
            Console.WriteLine($"Study budget Lv{characterLevel}: earned {earned}, all available skills {budget}, spare {earned - budget}.");
        }
        Check(!Enum.GetNames<DollCommand>().Contains("Attack") && !Enum.GetNames<DollCommand>().Contains("PlacePlanters")
            && (int)DollCommand.Cancel == 5 && (int)DollCommand.Dismiss == 6,
            "Attack and Place Planters removed without changing existing command IDs");
    }

    public class Projectile
    {
        public Vector2 Center, velocity;
        public int width, height, aiStyle, penetrate, timeLeft, owner, frame, frameCounter, direction, spriteDirection;
        public bool active, friendly, hostile, tileCollide, ignoreWater, netImportant, netUpdate;
        public float rotation, scale = 1f;
        public float[] ai = new float[3];
        public ModProjectile ModProjectile;
        public void Kill() => active = false;
        public Color GetAlpha(Color color) => color;
        public DataStructures.IEntitySource GetSource_FromThis() => null;
        public static int NewProjectile(DataStructures.IEntitySource source, Vector2 center, Vector2 velocity,
            int type, int damage, float knockback, int owner)
        {
            if (owner != Main.myPlayer || Main.netMode == 2) throw new Exception("Non-owner spawn");
            int index = Array.FindIndex(Main.projectile, p => !p.active);
            if (index < 0) throw new Exception("Projectile pool exhausted");
            var mod = new KumaPetProjectile();
            mod.SetDefaults();
            var p = mod.Projectile;
            p.Center = center; p.velocity = velocity; p.owner = owner; p.active = true;
            Main.projectile[index] = p;
            return index;
        }
    }

    public class ElainaAttributeModPlayer
    {
        public float MagicPoint = 100f;
        public bool ConsumeMagicPoint(float cost, bool consume = true)
        {
            if (MagicPoint < cost) return false;

        }

namespace 伊蕾娜.ElainaModSkills.Skills.DollControl
{
    /// <summary>
    /// 指令/标记/收集物宿主的替身：正式版依赖 WorldGen 与网络，这里只按队列给出目标格，
    /// 每格敲 <see cref="HitsPerTile"/> 下算挖掉并记一件收集物，用来检查玩偶 AI 如何使用这些接口。
    /// </summary>
    public class DollCommandPlayer
    {
        public readonly List<Point> Targets = [];
        public bool TargetsWalls;
        public int HitsPerTile = 3, Interval = 6, Hits, Loot, Delivered;
        public Vector2 DeliveredAt;
        public bool HasTask => Targets.Count > 0;
        public bool HasLoot => Loot > 0;
        internal bool TrySelectTarget(Vector2 from, Point current, bool currentWall, out Point target, out bool wall)
        {
            wall = TargetsWalls;
            target = Targets.Count > 0 ? Targets[0] : new Point(-1, -1);
            return Targets.Count > 0;
        }
        internal int HitInterval(bool walls) => Interval;
        internal void Hit(Point point, bool walls)
        {
            if (++Hits % HitsPerTile != 0) return;
            Targets.Remove(point);
            Loot++;
        }
        public void DeliverLoot(Vector2 from, Terraria.DataStructures.IEntitySource source)
        {
            if (Loot == 0) return;
            Delivered += Loot;
            Loot = 0;
            DeliveredAt = from;
        }
    }

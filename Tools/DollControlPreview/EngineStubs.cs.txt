        public DataStructures.IEntitySource GetSource_FromThis() => null;
        public static int NewProjectile(DataStructures.IEntitySource source, Vector2 center, Vector2 velocity,
            int type, int damage, float knockback, int owner)
        {
            if (owner != Main.myPlayer || Main.netMode == 2) throw new Exception("Non-owner spawn");
            int index = Array.FindIndex(Main.projectile, p => !p.active);
            if (index < 0) throw new Exception("Projectile pool exhausted");
            var mod = (ModProjectile)Activator.CreateInstance(ModContent.ProjectileTypes[type]);
            mod.SetDefaults();
            var p = mod.Projectile;
            p.Center = center; p.velocity = velocity; p.owner = owner; p.active = true;
            Main.projectile[index] = p;
            return index;
        }
    }

    /// <summary>
    /// 指令/标记/收集物宿主的替身：正式版依赖 WorldGen 与网络，这里只按队列给出目标格，
    /// 每格敲 <see cref="HitsPerTile"/> 下算挖掉并记一件收集物，用来检查玩偶 AI 如何使用这些接口。
    /// </summary>
    public sealed partial class DollCommandPlayer
    {
        public Player Player;
        public sealed class MarkSet : HashSet<Point>
        {
            public void AddRange(IEnumerable<Point> points) => UnionWith(points);
            public Point this[int index] => this.ElementAt(index);
        }
        public sealed class PlanterSet : Dictionary<Point, int>
        {
            public void Add(Point p) => this[p] = 1;
            public void AddRange(IEnumerable<Point> points) { foreach (Point p in points) Add(p); }
        }
        public readonly MarkSet TileMarks = [], WallMarks = [];
        public MarkSet Targets => TargetsWalls ? WallMarks : TileMarks;
        public readonly PlanterSet PlanterMarks = [];
        public PlanterSet PlanterTargets => PlanterMarks;
        public readonly HashSet<Point> BlockedTiles = [], BlockedWalls = [], InvalidTiles = [], InvalidWalls = [], BlockedPlanters = [];
        public readonly Dictionary<Point, Point> UnlockWallsAfterRemoval = [];
        private int validateTimer;
        private enum Work { Ready, Later, Never }
        private Work Check(Point p, bool wall) => (wall ? InvalidWalls : InvalidTiles).Contains(p) ? Work.Never
            : (wall ? BlockedWalls : BlockedTiles).Contains(p) ? Work.Later : Work.Ready;
        private bool IsMarkable(int x, int y, bool wall) => !(wall ? InvalidWalls : InvalidTiles).Contains(new(x, y));
        private bool InRange(Point p) => Vector2.DistanceSquared(Player.Center, p.ToWorldCoordinates()) <= 2400f * 2400f;
        private bool CanBuild(Point p) => !BlockedPlanters.Contains(p);
        public void ClearTasks() { TileMarks.Clear(); WallMarks.Clear(); PlanterMarks.Clear(); workClaims.Clear(); }
        public void ResumeGarden() { }
        public Point? PreviewChestTarget;
        public DollPetProjectile PreviewChestHelper;
        public bool RemoteChestActionFinished;
        public int PreviewChestOpened;
        internal bool IsRemoteChestHelper(DollPetProjectile worker) => worker == PreviewChestHelper && PreviewChestTarget != null;
        internal bool TrySelectRemoteChest(DollPetProjectile worker, out Point target)
        { target = PreviewChestTarget ?? new Point(-1, -1); return IsRemoteChestHelper(worker); }
        internal void CompleteRemoteChestGesture(DollPetProjectile worker)
        { if (IsRemoteChestHelper(worker) && !RemoteChestActionFinished) { RemoteChestActionFinished = true; PreviewChestOpened++; } }
        public Point? PreviewGardenTarget;
        public int GardenActions;
        internal bool TrySelectGardenTarget(Vector2 from, Point current, out Point target, DollPetProjectile worker = null)
        { target = PreviewGardenTarget ?? new(-1, -1); return PreviewGardenTarget != null; }
        internal void TendGarden(Point point) { GardenActions++; }
        public int PlacedPlanters;
        public bool TargetsWalls;
        public int HitsPerTile = 3, Interval = 6, Hits, Loot, Delivered;
        public Vector2 DeliveredAt;
        public bool HasTask => TileMarks.Count > 0 || WallMarks.Count > 0 || PlanterTargets.Count > 0;
        public bool HasLoot => Loot > 0;
        internal void PlacePlanter(Point point) { if (PlanterTargets.Remove(point)) PlacedPlanters++; }
        internal int HitInterval(bool walls) => Interval;
        readonly Dictionary<(Point, bool), int> damage = [];
        internal void Hit(Point point, bool walls)
        {
            var marks = walls ? WallMarks : TileMarks;
            if (!marks.Contains(point)) throw new Exception("Duplicate or stale hit");
            Hits++;
            var key = (point, walls);
            damage.TryGetValue(key, out int prior);
            damage[key] = prior + 1;
            if (damage[key] < HitsPerTile) return;
            damage.Remove(key);
            marks.Remove(point);
            if (walls && UnlockWallsAfterRemoval.TryGetValue(point, out Point exposed)) BlockedWalls.Remove(exposed);
            Loot++;
        }
        public void DeliverLoot(Vector2 from, Terraria.DataStructures.IEntitySource source)
        {
            if (Loot == 0) return;
            Delivered += Loot;
            Loot = 0;
            DeliveredAt = from;
        }
    }

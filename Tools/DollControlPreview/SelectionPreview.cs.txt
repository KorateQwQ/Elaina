using System;
using System.IO;
using System.Linq;
using Terraria.Localization;
using Microsoft.Xna.Framework.Content;
using ReLogic.Graphics;
using SDL2;
using 伊蕾娜.ElainaModSkills.Skills.DollControl;

internal sealed class SelectionPreview : Game
{
    private readonly string root, output;
    private Texture2D pixel, mine, wall, stars;
    private DynamicSpriteFont serif, sans;
    private ContentManager fonts;
    private bool done;
    private SelectionPreview(string root, string output)
    {
        this.root = root; this.output = output;
        new GraphicsDeviceManager(this) { PreferredBackBufferWidth = 320, PreferredBackBufferHeight = 180,
            GraphicsProfile = GraphicsProfile.HiDef, SynchronizeWithVerticalRetrace = false };
        IsFixedTimeStep = false;
        SDL.SDL_HideWindow(Window.Handle);
    }
    protected override void LoadContent()
    {
        fonts = new ContentManager(Services, Path.GetFullPath(Path.Combine(root, "../KL/Fonts")));
        serif = fonts.Load<DynamicSpriteFont>("NotoSerifSC");
        sans = fonts.Load<DynamicSpriteFont>("HarmonyOS_Sans_SC");
        // 与游戏的 TextureAssets.MagicPixel 同为 1×1000，绘制必须只取 (0,0,1,1)，否则预览也会出现满屏拉伸线。
        pixel = new Texture2D(GraphicsDevice, 1, 1000); pixel.SetData(Enumerable.Repeat(Color.White, 1000).ToArray());
        mine = Icon("Mine"); wall = Icon("BreakWall");
        // tML 加载 PNG 时预乘 alpha；FromStream 不会，这里补上以匹配游戏内绘制。
        using (var stream = File.OpenRead(Path.Combine(root, "ElainaModAlchemy/UI/Assets/NotebookStars.png")))
            stars = Texture2D.FromStream(GraphicsDevice, stream);
        var texels = new Color[stars.Width * stars.Height]; stars.GetData(texels);
        for (int i = 0; i < texels.Length; i++) texels[i] = Color.FromNonPremultiplied(texels[i].ToVector4());
        stars.SetData(texels);
    }
    private Texture2D Icon(string name)
    {
        using var stream = File.OpenRead(Path.Combine(root, "ElainaModSkills/Skills/DollControl/Commands/DollCommand_" + name + ".png"));
        return Texture2D.FromStream(GraphicsDevice, stream);
    }
    protected override void Update(GameTime gameTime)
    {
        if (done) return; done = true;
        SDL.SDL_HideWindow(Window.Handle);
        Directory.CreateDirectory(output);
        using var batch = new SpriteBatch(GraphicsDevice);
        DrawFurniture(batch);
        var cases = new[] {
            ("ready", "zh-Hans", 1f, 1280, 800, false, 126, true),
            ("empty", "zh-Hans", 1f, 1280, 800, false, 0, false),
            ("walls-150", "zh-Hans", 1.5f, 1280, 800, true, 268, false),
            ("english-150", "en-US", 1.5f, 1280, 800, true, 123456, true),
            ("compact-150", "zh-Hans", 1.5f, 800, 600, false, 53, false),
            ("compact-en", "en-US", 1.5f, 800, 600, true, 32, true)
        };
        var crops = new List<Texture2D>();
        foreach (var (name, language, scale, width, height, walls, count, hover) in cases)
        {
            // Match tML's real HJSON parser, including quoted values and nested localization groups.
            Language.Load(root, language);
            string Text(string key) => Language.GetTextValue("Mods.伊蕾娜.UI." + key);
            var layout = DollSelectionBarDrawing.GetLayout(new Vector2(width, height) / scale);
            using var target = new RenderTarget2D(GraphicsDevice, width, height);
            GraphicsDevice.SetRenderTarget(target); GraphicsDevice.Clear(new Color(140, 98, 66));
            batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.LinearClamp,
                DepthStencilState.None, RasterizerState.CullNone, null, Matrix.CreateScale(scale));
            DollSelectionBarDrawing.Draw(batch, new(pixel, stars, serif, sans), walls ? wall : mine, layout,
                hover ? new Vector2(layout.Start.Center.X, layout.Start.Center.Y) : new(-1), scale,
                Text(walls ? "DollCommandBreakWall" : "DollCommandMine"), Text("DollModeSettings"),
                Text("DollMarkKeyHint").Replace("{0}", name == "english-150" ? "Mouse4 / RightControl" : "F"), Text("DollSettingsDone"),
                Text("DollBringBackItems"), name != "walls-150" && name != "compact-150",
                Text(walls ? "DollChainWalls" : "DollChainMining"), walls,
                Text("DollToggleOn"), Text("DollToggleOff"));
            batch.End(); GraphicsDevice.SetRenderTarget(null);
            using (var stream = File.Create(Path.Combine(output, name + ".png"))) target.SaveAsPng(stream, width, height);
            var cropRect = new Rectangle((int)(layout.Panel.X * scale) - 6, (int)(layout.Panel.Y * scale) - 6,
                (int)(layout.Panel.Width * scale) + 14, (int)(layout.Panel.Height * scale) + 16);
            var data = new Color[cropRect.Width * cropRect.Height]; target.GetData(0, cropRect, data, 0, data.Length);
            var crop = new Texture2D(GraphicsDevice, cropRect.Width, cropRect.Height); crop.SetData(data); crops.Add(crop);
        }
        using (var sheet = new RenderTarget2D(GraphicsDevice, 850, crops.Sum(t => t.Height + 18) + 18))
        {
            GraphicsDevice.SetRenderTarget(sheet); GraphicsDevice.Clear(new Color(25, 29, 38));
            batch.Begin(); int y = 18;
            foreach (Texture2D crop in crops) { batch.Draw(crop, new Vector2((850 - crop.Width) / 2, y), Color.White); y += crop.Height + 18; }
            batch.End(); GraphicsDevice.SetRenderTarget(null);
            using var stream = File.Create(Path.Combine(output, "selection-states.png")); sheet.SaveAsPng(stream, sheet.Width, sheet.Height);
        }
        foreach (var crop in crops) crop.Dispose();
        Console.WriteLine("PASS: shared FNA renderer, real Noto Serif SC / HarmonyOS Sans SC fonts, NotebookStars and localized labels; settings/hover/walls/English/compact at 100% and 150% UI scale.");
        Console.WriteLine(Path.Combine(output, "selection-states.png"));
        Exit();
    }
    protected override void UnloadContent() { pixel?.Dispose(); mine?.Dispose(); wall?.Dispose(); stars?.Dispose(); fonts?.Dispose(); }
    private void DrawFurniture(SpriteBatch batch)
    {
        Terraria.GameContent.TextureAssets.MagicPixel.Value = pixel;
        Terraria.Main.screenPosition = Vector2.Zero;
        foreach (float zoom in new[] { 2f, 3f })
        {
            using var target = new RenderTarget2D(GraphicsDevice, 640, (int)(230 * zoom));
            GraphicsDevice.SetRenderTarget(target); GraphicsDevice.Clear(new Color(30, 41, 48));
            batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.PointClamp,
                DepthStencilState.None, RasterizerState.CullNone, null, Matrix.CreateScale(zoom));
            foreach (var (type, width, height, y, label) in new[] {
                (18, 2, 1, 3, "工作台"), (79, 4, 2, 7, "床"), (14, 3, 2, 11, "桌子") })
            {
                Terraria.ObjectData.TileObjectData.Fixtures[type] = new() { Width = width, Height = height };
                Terraria.Main.tileFrameImportant[type] = true;
                // Two touching identical objects; clicking the rightmost cell of the first must select just that object.
                for (int objectIndex = 0; objectIndex < 2; objectIndex++)
                for (int x = 0; x < width; x++) for (int row = 0; row < height; row++)
                {
                    int column = 2 + objectIndex * width + x;
                    Terraria.Main.tile[column, y + row] = new Tile { HasTile = true, TileType = (ushort)type,
                        TileFrameX = (short)(x * 18), TileFrameY = (short)(row * 18) };
                    batch.Draw(pixel, new Rectangle(column * 16 + 2, (y + row) * 16 + 3, 12, 11), new Color(112, 85, 65));
                }
                Point clicked = new(2 + width - 1, y + height - 1);
                var cells = DollTileSelection.RectangleCells(clicked, clicked, false);
                FurnitureMarkPreview.DrawMarks(batch, cells, new Color(150, 235, 160), new Rectangle(0, 0, 50, 40), .5f);
                batch.Draw(pixel, new Rectangle(clicked.X * 16 + 6, clicked.Y * 16 + 6, 4, 4), Color.White);
                batch.DrawString(sans, label, new Vector2(32, y * 16 - 24), Color.White, 0, Vector2.Zero, .25f, SpriteEffects.None, 0);
            }
            batch.End(); GraphicsDevice.SetRenderTarget(null);
            using var stream = File.Create(Path.Combine(output, $"furniture-selection-{zoom:0}.png"));
            target.SaveAsPng(stream, target.Width, target.Height);
        }
    }
    [STAThread]
    private static void Main(string[] args)
    {
        SelectionChecks.Run();
        using var game = new SelectionPreview(args[0], args[1]); game.Run();
    }
}

using System;
using System.IO;
using System.Linq;
using Terraria.Localization;
using Microsoft.Xna.Framework.Content;
using ReLogic.Graphics;
using SDL2;
using 伊蕾娜.ElainaModSkills.Skills.DollControl;

internal sealed class SelectionPreview : Game
{
    private readonly string root, output;
    private Texture2D pixel, mine, wall, planters, stars;
    private DynamicSpriteFont serif, sans;
    private ContentManager fonts;
    private bool done;
    private SelectionPreview(string root, string output)
    {
        this.root = root; this.output = output;
        new GraphicsDeviceManager(this) { PreferredBackBufferWidth = 320, PreferredBackBufferHeight = 180,
            GraphicsProfile = GraphicsProfile.HiDef, SynchronizeWithVerticalRetrace = false };
        IsFixedTimeStep = false;
        SDL.SDL_HideWindow(Window.Handle);
    }
    protected override void LoadContent()
    {
        fonts = new ContentManager(Services, Path.GetFullPath(Path.Combine(root, "../KL/Fonts")));
        serif = fonts.Load<DynamicSpriteFont>("NotoSerifSC");
        sans = fonts.Load<DynamicSpriteFont>("HarmonyOS_Sans_SC");
        // 与游戏的 TextureAssets.MagicPixel 同为 1×1000，绘制必须只取 (0,0,1,1)，否则预览也会出现满屏拉伸线。
        pixel = new Texture2D(GraphicsDevice, 1, 1000); pixel.SetData(Enumerable.Repeat(Color.White, 1000).ToArray());
        mine = Icon("Mine"); wall = Icon("BreakWall"); planters = Icon("PlacePlanters");
        // tML 加载 PNG 时预乘 alpha；FromStream 不会，这里补上以匹配游戏内绘制。
        using (var stream = File.OpenRead(Path.Combine(root, "ElainaModAlchemy/UI/Assets/NotebookStars.png")))
            stars = Texture2D.FromStream(GraphicsDevice, stream);
        var texels = new Color[stars.Width * stars.Height]; stars.GetData(texels);
        for (int i = 0; i < texels.Length; i++) texels[i] = Color.FromNonPremultiplied(texels[i].ToVector4());
        stars.SetData(texels);
    }
    private Texture2D Icon(string name)
    {
        using var stream = File.OpenRead(Path.Combine(root, "ElainaModSkills/Skills/DollControl/Commands/DollCommand_" + name + ".png"));
        return Texture2D.FromStream(GraphicsDevice, stream);
    }
    protected override void Update(GameTime gameTime)
    {
        if (done) return; done = true;
        SDL.SDL_HideWindow(Window.Handle);
        Directory.CreateDirectory(output);
        using var batch = new SpriteBatch(GraphicsDevice);
        var cases = new[] {
            ("ready", "zh-Hans", 1f, 1280, 800, false, 126, true),
            ("empty", "zh-Hans", 1f, 1280, 800, false, 0, false),
            ("walls-150", "zh-Hans", 1.5f, 1280, 800, true, 268, false),
            ("english-150", "en-US", 1.5f, 1280, 800, true, 123456, true),
            ("compact-150", "zh-Hans", 1.5f, 800, 600, false, 53, false),
            ("weak-pick", "zh-Hans", 1.5f, 1280, 800, false, 32, true),
            ("locked-wall", "zh-Hans", 1.5f, 1280, 800, true, 32, true),
            ("locked-wall-en", "en-US", 1.5f, 800, 600, true, 32, true)
        };
        var crops = new List<Texture2D>();
        foreach (var (name, language, scale, width, height, walls, count, hover) in cases)
        {
            // Match tML's real HJSON parser, including quoted values and nested localization groups.
            Language.Load(root, language);
            string Text(string key) => Language.GetTextValue("Mods.伊蕾娜.UI." + key);
            var layout = DollSelectionBarDrawing.GetLayout(new Vector2(width, height) / scale);
            using var target = new RenderTarget2D(GraphicsDevice, width, height);
            GraphicsDevice.SetRenderTarget(target); GraphicsDevice.Clear(new Color(140, 98, 66));
            batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.LinearClamp,
                DepthStencilState.None, RasterizerState.CullNone, null, Matrix.CreateScale(scale));
            DollSelectionBarDrawing.Draw(batch, new(pixel, stars, serif, sans), walls ? wall : mine, layout,
                hover ? new Vector2(layout.Start.Center.X, layout.Start.Center.Y) : new(-1), scale,
                Text(walls ? "DollCommandBreakWall" : "DollCommandMine"), Text(walls ? "DollSelectionWalls" : "DollSelectionTiles"), count,
                Text(walls ? "DollSelectionEmptyWalls" : "DollSelectionEmptyTiles"), Text(walls ? "DollMarkHintChain" : "DollMarkHintRect"),
                Text(walls ? "DollSelectionStartWalls" : "DollSelectionStart"), Text("DollSelectionExit"),
                bringBackLabel: Text("DollBringBackItems"), bringBackItems: name != "walls-150" && name != "compact-150",
                chainLabel: Text(walls ? "DollChainWalls" : "DollChainMining"), chain: walls,
                onLabel: Text("DollToggleOn"), offLabel: Text("DollToggleOff"),
                warning: name == "weak-pick" ? Text("DollPickPowerInsufficient")
                    : name.StartsWith("locked-wall") ? Text("DollWallRequiresGolem") : null);
            batch.End(); GraphicsDevice.SetRenderTarget(null);
            using (var stream = File.Create(Path.Combine(output, name + ".png"))) target.SaveAsPng(stream, width, height);
            var cropRect = new Rectangle((int)(layout.Panel.X * scale) - 6, (int)(layout.Panel.Y * scale) - 6,
                (int)(layout.Panel.Width * scale) + 14, (int)(layout.Panel.Height * scale) + 16);
            var data = new Color[cropRect.Width * cropRect.Height]; target.GetData(0, cropRect, data, 0, data.Length);
            var crop = new Texture2D(GraphicsDevice, cropRect.Width, cropRect.Height); crop.SetData(data); crops.Add(crop);
        }
        using (var sheet = new RenderTarget2D(GraphicsDevice, 850, crops.Sum(t => t.Height + 18) + 18))
        {
            GraphicsDevice.SetRenderTarget(sheet); GraphicsDevice.Clear(new Color(25, 29, 38));
            batch.Begin(); int y = 18;
            foreach (Texture2D crop in crops) { batch.Draw(crop, new Vector2((850 - crop.Width) / 2, y), Color.White); y += crop.Height + 18; }
            batch.End(); GraphicsDevice.SetRenderTarget(null);
            using var stream = File.Create(Path.Combine(output, "selection-states.png")); sheet.SaveAsPng(stream, sheet.Width, sheet.Height);
        }
        foreach (var crop in crops) crop.Dispose();
        DrawPlanters(batch);
        DrawMaterialMenus(batch);
        Console.WriteLine("PASS: shared FNA renderer, real Noto Serif SC / HarmonyOS Sans SC fonts, NotebookStars and localized labels; ready/disabled/hover/walls/English/compact at 100% and 150% UI scale.");
        Console.WriteLine(Path.Combine(output, "selection-states.png"));
        Exit();
    }
    protected override void UnloadContent() { pixel?.Dispose(); mine?.Dispose(); wall?.Dispose(); planters?.Dispose(); stars?.Dispose(); fonts?.Dispose(); }
    private void DrawPlanters(SpriteBatch batch)
    {
        foreach (var (name, language, scale, width, height, count) in new[] {
            ("planters", "zh-Hans", 1f, 1280, 800, 120),
            ("planters-150", "zh-Hans", 1.5f, 1280, 800, 120),
            ("planters-en-compact", "en-US", 1.5f, 800, 600, 120),
            ("planters-empty", "zh-Hans", 1f, 1280, 800, 0) })
        {
            Language.Load(root, language);
            string Text(string key) => Language.GetTextValue("Mods.伊蕾娜.UI." + key);
            var layout = DollSelectionBarDrawing.GetLayout(new Vector2(width, height) / scale, true);
            using var target = new RenderTarget2D(GraphicsDevice, width, height);
            GraphicsDevice.SetRenderTarget(target); GraphicsDevice.Clear(new Color(42, 70, 75));
            batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.PointClamp,
                DepthStencilState.None, RasterizerState.CullNone, null, Matrix.CreateScale(scale));
            for (int y = 0; y < (count > 0 ? 3 : 0); y++) for (int x = 0; x < 12; x++)
                DollPlanterDrawing.Draw(batch, pixel, new Vector2(70 + x * 16, 60 + y * 32), x == 10, .9f);
            batch.End();
            batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.LinearClamp,
                DepthStencilState.None, RasterizerState.CullNone, null, Matrix.CreateScale(scale));
            string Option(string label, string item, int owned) => string.Format(Text("DollMaterialOption"), Text(label), item, owned, count);
            bool en = language == "en-US";
            DollSelectionBarDrawing.Draw(batch, new(pixel, stars, serif, sans), planters, layout,
                new Vector2(layout.Planter.Center.X, layout.Planter.Center.Y), scale,
                Text("DollCommandPlacePlanters"), Text("DollSelectionPlanters"), count, Text("DollSelectionEmptyPlanters"),
                Text("DollPlanterHint"), Text("DollSelectionStartPlanters"), Text("DollSelectionExit"),
                new(planters, planters.Bounds, Option("DollPlanterLabel", en ? "Deathweed Planter Box" : "死亡草种植盆", 80), 80 >= count));
            batch.End(); GraphicsDevice.SetRenderTarget(null);
            using var stream = File.Create(Path.Combine(output, name + ".png")); target.SaveAsPng(stream, width, height);
        }
        Console.WriteLine("PASS: planter bar + shared 2-layer world markers, Chinese/English, 100%/150%, empty/short supply/compact.");
    }
    private void DrawMaterialMenus(SpriteBatch batch)
    {
        foreach (var (name, language, scale, width, height, total, offset) in new[] {
            ("material-menu-planters", "zh-Hans", 1f, 1280, 800, 3, 0),
            ("material-menu-planters-150", "zh-Hans", 1.5f, 1280, 800, 8, 0),
            ("material-menu-en-compact", "en-US", 1.5f, 800, 600, 8, 3),
            ("material-menu-empty", "zh-Hans", 1f, 1280, 800, 0, 0) })
        {
            Language.Load(root, language);
            string Text(string key) => Language.GetTextValue("Mods.伊蕾娜.UI." + key);
            bool english = language == "en-US";
            var bar = DollSelectionBarDrawing.GetLayout(new Vector2(width, height) / scale, true);
            var popup = DollSelectionBarDrawing.GetMaterialMenuLayout(bar, total, offset);
            string[] zh = ["太阳花种植盆", "月光草种植盆", "闪耀根种植盆", "死亡草种植盆", "血腥死亡草种植盆", "寒颤棘种植盆", "火焰花种植盆", "水叶草种植盆"];
            string[] en = ["Daybloom Planter Box", "Moonglow Planter Box", "Blinkroot Planter Box", "Deathweed Planter Box", "Crimson Planter Box", "Shiverthorn Planter Box", "Fireblossom Planter Box", "Waterleaf Planter Box"];
            var entries = new List<DollSelectionBarDrawing.MaterialMenuEntry>();
            for (int i = 0; i < total; i++)
            {
                var texture = planters;
                string label = string.Format(Text("DollMaterialListOption"), english ? en[i % en.Length] : zh[i % zh.Length], 999 - i * 73);
                entries.Add(new(new(texture, texture.Bounds, label, true), i == (offset > 0 ? offset + 1 : 1)));
            }
            using var target = new RenderTarget2D(GraphicsDevice, width, height);
            GraphicsDevice.SetRenderTarget(target); GraphicsDevice.Clear(new Color(42, 70, 75));
            batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.LinearClamp,
                DepthStencilState.None, RasterizerState.CullNone, null, Matrix.CreateScale(scale));
            string Option(string key, string item, int owned) => string.Format(Text("DollMaterialOption"), Text(key), item, owned, 120);
            DollSelectionBarDrawing.Draw(batch, new(pixel, stars, serif, sans), planters, bar,
                new Vector2(popup.Row(0).Center.X, popup.Row(0).Center.Y), scale,
                Text("DollCommandPlacePlanters"), Text("DollSelectionPlanters"), 120, Text("DollSelectionEmptyPlanters"),
                Text("DollPlanterHint"), Text("DollSelectionStartPlanters"), Text("DollSelectionExit"),
                new(planters, planters.Bounds, Option("DollPlanterLabel", total == 0 ? Text("DollMaterialNone") : english ? "Moonglow Planter Box" : "月光草种植盆", total == 0 ? 0 : 926), total > 0),
                total > 0, new(popup, entries, Text("DollChoosePlanter"), Text("DollMaterialNone")), true);
            batch.End(); GraphicsDevice.SetRenderTarget(null);
            using var stream = File.Create(Path.Combine(output, name + ".png")); target.SaveAsPng(stream, width, height);
        }
        Console.WriteLine("PASS: production material popup renderer, choices/selected/hover/empty/paged, Chinese/English, compact and 150% UI.");
    }
    [STAThread]
    private static void Main(string[] args)
    {
        SelectionChecks.Run();
        PlanterSelectionChecks.Run();
        using var game = new SelectionPreview(args[0], args[1]); game.Run();
    }
}

using System;
using Microsoft.Xna.Framework.Input;
using Terraria.GameInput;
using Terraria.ModLoader.IO;
using 伊蕾娜.Config;
using 伊蕾娜.ElainaModSkills.Skills.DollControl;
using Host = Terraria.Main;

internal static class SelectionChecks
{
    private static int assertions;
    internal static void Run()
    {
        for (int x = 0; x < Host.maxTilesX; x++) for (int y = 0; y < Host.maxTilesY; y++) Host.tile[x, y] = new Tile();
        Point a = new(20, 20), b = new(21, 20), other = new(22, 20), old = new(30, 20);
        Host.tile[a.X, a.Y] = new Tile { HasTile = true, TileType = 1, WallType = 1 };
        Host.tile[b.X, b.Y] = new Tile { HasTile = true, TileType = 1, WallType = 1 };
        Host.tile[other.X, other.Y] = new Tile { HasTile = true, TileType = 2, WallType = 2 };
        foreach (float scale in new[] { 1f, 1.5f, 2f })
        foreach (bool walls in new[] { false, true })
        foreach (bool chain in new[] { false, true })
        {
            Host.UIScale = scale;
            var c = new DollCommandPlayer();
            c.OpenCommandWheel(); DollCommandWheel.Choose(walls ? DollCommand.BreakWall : DollCommand.Mine);
            Check(c.SettingsOpen && c.Tool == null && !c.HasTask, "command opens settings without starting selection or task");
            var layout = DollSelectionBarDrawing.GetLayout(PlayerInput.OriginalScreenSize / scale);
            if (chain) Button(c, layout.Chain);
            Button(c, layout.BringBack);
            Check(!c.BringBackItems && DollCommands.IsChain(c.MarkMode) == chain, "settings toggle once per click at every UI scale");
            Button(c, layout.Start);
            Check(!c.SettingsOpen && !c.HasTask, "Done only closes settings");
            c.Player.HeldItem.ModItem = new object();
            伊蕾娜.ElainaModSkills.ElainaSkillModPlayer.GetActiveSkill[0].ModSkill = new object();
            var queue = walls ? c.WallMarks : c.TileMarks;
            queue.Add(old);
            Tick(c, true, a, left: true);
            Check(c.Tool != null && queue.SetEquals([old]) && c.SelectionMarks.Count == (chain ? 2 : 1), "F hold previews without submitting under another spell or item");
            Check(c.Player.controlUseItem && !c.Player.mouseInterface && Host.mouseLeft, "marking leaves attack input untouched");
            Tick(c, false, a, left: true);
            Check(queue.SetEquals(chain ? new[] {old, a, b} : new[] {old, a}) && c.Tool == null && c.SelectionMarks.Count == 0,
                "F release immediately appends selected object/chain and clears transient state");
            Tick(c, false, other, left: true);
            Check(!queue.Contains(other), "left attack does not mark targets");
            Tick(c, true, a); Tick(c, true, other);
            Check(c.SelectionMarks.SetEquals([a, b, other]), "drag selects rectangle even when chain is enabled");
            Tick(c, false, other);
            Check(queue.SetEquals([old, a, b, other]), "rectangle submits once without a Start button");
            Tick(c, true, a); Tick(c, true, other); Tick(c, false, a);
            Check(c.SelectionMarks.Count == 0 && queue.Count == 4, "repeated assignment is idempotent and never removes queued tasks");
            var save = new TagCompound(); c.SaveData(save);
            var loaded = new DollCommandPlayer(); loaded.LoadData(save);
            Check(loaded.MarkMode == c.MarkMode && !loaded.BringBackItems && !loaded.SettingsOpen && loaded.Tool == null,
                "mode and collection preference persist without transient UI or selections");
            c.ClearTasks();
            Check(!c.HasTask && c.Tool == null && !c.SettingsOpen, "Cancel Tasks clears tasks and gesture");
        }
        Host.UIScale = 1;
        foreach (string interrupt in new[] { "focus", "chat", "map", "options", "ui", "chest", "dead", "absent", "escape", "wheel" })
        {
            var c = new DollCommandPlayer();
            Tick(c, true, a);
            switch (interrupt)
            {
                case "focus": Host.hasFocus = false; break;
                case "chat": Host.drawingPlayerChat = true; break;
                case "map": Host.mapFullscreen = true; break;
                case "options": Host.ingameOptionsWindow = true; break;
                case "chest": c.Player.chest = 0; break;
                case "dead": c.Player.dead = true; break;
                case "absent": DollPetProjectile.Present = false; break;
                case "escape": Host.keyState = new KeyboardState(Keys.Escape); break;
                case "wheel": DollCommandWheel.OwnsPointer = true; break;
            }
            Tick(c, false, b, ui: interrupt == "ui");
            Check(!c.HasTask && c.Tool == null && c.SelectionMarks.Count == 0, interrupt + " cancels transient selection on release");
            Host.hasFocus = true; Host.drawingPlayerChat = Host.mapFullscreen = Host.ingameOptionsWindow = false;
            Host.keyState = default; DollPetProjectile.Present = true; DollCommandWheel.OwnsPointer = false;
        }
        var restricted = new DollCommandPlayer();
        Host.tile[a.X, a.Y].RequiredPick = 200;
        Tick(restricted, true, a); Tick(restricted, false, a);
        Check(!restricted.HasTask && restricted.Tool == null && CombatText.Last == "DollPickPowerInsufficient", "blocked targets reject and clear gesture instead of leaving a draft");
        Host.tile[a.X, a.Y].RequiredPick = 0;
        Tick(restricted, true, a); Host.tile[a.X, a.Y].HasTile = false; Tick(restricted, false, a);
        Check(!restricted.HasTask, "release revalidates tiles destroyed while holding");
        CheckFurniture();
        Console.WriteLine($"PASS: {assertions} production hotkey input, settings, persistence, restriction and furniture checks.");
    }
    private static void CheckFurniture()
    {
        foreach (var (type, width, height) in new[] { (79, 4, 2), (18, 2, 1), (14, 3, 2), (13, 1, 1), (4, 1, 1), (15, 1, 2), (19, 1, 1), (380, 1, 1) })
        foreach (bool chain in new[] { false, true })
        {
            Host.tileFrameImportant[type] = true;
            Terraria.ObjectData.TileObjectData.Fixtures[type] = new() { Width = width, Height = height };
            for (int x = 90; x < 100; x++) for (int y = 30; y < 40; y++) Host.tile[x, y] = new Tile();
            List<Point> cells = [];
            for (int x = 0; x < width; x++) for (int y = 0; y < height; y++)
            {
                Point p = new(90 + x, 30 + y); cells.Add(p);
                Host.tile[p.X, p.Y] = new Tile { HasTile = true, TileType = (ushort)type, TileFrameX = (short)(x * 18), TileFrameY = (short)(y * 18) };
            }
            var c = new DollCommandPlayer(); c.OpenCommandWheel(); DollCommandWheel.Choose(DollCommand.Mine);
            if (chain) Button(c, DollSelectionBarDrawing.GetLayout(PlayerInput.OriginalScreenSize).Chain);
            Tick(c, true, cells[^1]);
            Check(c.SelectionMarks.SetEquals(cells), $"{type}: one cell previews complete furniture");
            Tick(c, false, cells[^1]);
            Check(c.TileMarks.SetEquals(cells) && c.Tool == null, $"{type}: complete furniture submitted on release");
        }
        Host.tileAxe[5] = true; Host.tileHammer[26] = true;
        foreach (ushort type in new ushort[] { 5, 26 })
        {
            Host.tile[100, 40] = new Tile { HasTile = true, TileType = type };
            Check(!DollCommandPlayer.IsMarkable(100, 40, false), "pickaxe selection excludes axe/hammer targets");
        }
        Check(!DollCommandPlayer.IsMarkable(0, 0, false), "world boundary excluded");
    }
    private static void Tick(DollCommandPlayer c, bool held, Point tile, bool left = false, bool ui = false)
    {
        KeyBind.DollMark.JustPressed = held && !KeyBind.DollMark.Current;
        KeyBind.DollMark.Current = held;
        c.Player.mouseInterface = ui; c.Player.controlUseItem = left; Host.mouseLeft = left;
        Host.MouseWorld = new(tile.X * 16, tile.Y * 16);
        PlayerInput.MouseX = PlayerInput.MouseY = 100;
        Host.GameUpdateCount++;
        c.ProcessTriggers(new());
    }
    private static void Button(DollCommandPlayer c, Rectangle button)
    {
        KeyBind.DollMark.Current = KeyBind.DollMark.JustPressed = false;
        foreach (bool left in new[] { false, true, true, false })
        {
            c.Player.mouseInterface = false; Host.mouseLeft = left;
            PlayerInput.MouseX = (int)(button.Center.X * Host.UIScale); PlayerInput.MouseY = (int)(button.Center.Y * Host.UIScale);
            c.ProcessTriggers(new());
        }
    }
    private static void Check(bool condition, string message) { assertions++; if (!condition) throw new Exception(message); }
}

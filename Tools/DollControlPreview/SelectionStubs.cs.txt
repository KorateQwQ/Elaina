namespace Terraria.ModLoader.IO
{
    public class TagCompound : Dictionary<string, object>
    {
        public bool GetBool(string key) => TryGetValue(key, out var value) && (bool)value;
        public int GetInt(string key) => TryGetValue(key, out var value) ? (int)value : 0;
        public IList<T> GetList<T>(string key) => TryGetValue(key, out var value) ? (IList<T>)value : new List<T>();
        public TagCompound GetCompound(string key) => (TagCompound)this[key];
    }
    public static class ItemIO
    {
        public static TagCompound Save(Item item) => new() { ["type"] = item.type };
        public static Item Load(TagCompound tag) => new(tag.GetInt("type"));
    }

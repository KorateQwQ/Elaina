using System;
using System.IO;
using System.Linq;
using 伊蕾娜.ElainaAttribute;
using 伊蕾娜.ElainaModSkills.Skills.DollControl;
using World = Terraria.Main;

static partial class Checks
{
    static int assertions;
    static void Check(bool result, string message)
    {
        if (!result) throw new Exception(message);
        assertions++;
    }
    public static KumaPetProjectile Reset(int level = 7)
    {
        World.myPlayer = 0; World.netMode = 0;
        World.player = Enumerable.Range(0, World.maxPlayers).Select(i => new Player { whoAmI = i }).ToArray();
        World.projectile = Enumerable.Range(0, World.maxProjectiles).Select(_ => new Projectile()).ToArray();
        var skill = new DollControlSkill { Level = level }; skill.Initialize(); skill.PreUseSkill();
        return (KumaPetProjectile)World.projectile[0].ModProjectile;
    }
    public static void Tick(DollPetProjectile doll)
    {
        var owner = World.player[doll.Projectile.owner];
        owner.Center += owner.velocity;
        doll.AI();
        if (doll.Projectile.active) doll.Projectile.Center += doll.Projectile.velocity;
    }
    public static void Run(string repo)
    {
        Check(伊蕾娜.ElainaModSkills.SkillBalanceConfig.TryParse(File.ReadAllText(Path.Combine(repo,
            "ElainaModSkills/Data/SkillBalance.json")), out var config, out var error), error);
        伊蕾娜.ElainaModSkills.ElainaSkill.Configuration = config;
        var doll = Reset();
        var skill = new DollControlSkill { Level = 7 }; skill.Initialize();
        var owner = World.player[0];
        var mana = owner.GetModPlayer<ElainaAttributeModPlayer>();
        Check(mana.MagicPoint == 80f, "First cast must cost exactly 20");
        Check(doll.Projectile.owner == 0 && doll.CanDamage() == false && doll.CanCutTiles() == false,
            "Doll must be owned, harmless and unable to cut grass");
        Check(doll.Projectile.netImportant && !doll.Projectile.tileCollide, "Doll must sync on join and pass through terrain");
        var idleFrames = new System.Collections.Generic.HashSet<int>();
        for (int i = 0; i < 600; i++) { Tick(doll); idleFrames.Add(doll.Projectile.frame); }
        Check(!doll.IsMoving && !doll.IsMining && doll.Projectile.active, "Idle bob must not start move animation or expire");
        Check(idleFrames.SetEquals(new[] { 0, 1, 2 }), "Idle must loop three frames");
        owner.HeldItem.pick = 35; owner.itemAnimation = 20;
        for (int i = 0; i < 60; i++) Tick(doll);
        Check(!doll.IsMining, "Owner's own pickaxe swing no longer triggers mining");
        owner.HeldItem.pick = 0; owner.itemAnimation = 0;

        // 挖掘标记：飞到目标格朝向主人的一侧，面向格子循环四帧挖矿；主人端按间隔敲击，挖完回到身边交还收集物。
        var commands = owner.GetModPlayer<DollCommandPlayer>();
        Point first = new((int)(owner.Center.X / 16f) + 14, (int)(owner.Center.Y / 16f) + 3);
        Point second = new(first.X + 1, first.Y + 2);
        commands.Interval = 10;
        commands.Targets.AddRange(new[] { first, second });
        var mineFrames = new System.Collections.Generic.HashSet<int>();
        bool placed = false;
        for (int i = 0; i < 900 && commands.HasTask; i++)
        {
            Tick(doll);
            if (!doll.IsMining) continue;
            mineFrames.Add(doll.Projectile.frame);
            Vector2 tileCenter = (commands.HasTask ? commands.Targets[0] : second).ToWorldCoordinates();
            placed |= Vector2.Distance(doll.Projectile.Center, tileCenter) < 40f && doll.Projectile.Center.X < tileCenter.X
                && doll.Projectile.spriteDirection == 1;
        }
        Check(!commands.HasTask && commands.Hits == 6 && commands.Loot == 2, "Doll must dig every marked tile at the hit interval");
        Check(mineFrames.SetEquals(new[] { 0, 1, 2, 3 }), "Digging must loop four mining frames");
        Check(placed, "Doll must hover beside the target on the owner's side and face it");
        for (int i = 0; i < 300 && commands.Delivered == 0; i++) Tick(doll);
        Check(commands.Delivered == 2 && commands.Loot == 0 && Vector2.Distance(commands.DeliveredAt, owner.Center) < 120f,
            "Collected items are handed over once the doll is back beside its owner");
        for (int i = 0; i < 180; i++) Tick(doll);
        Check(!doll.IsMining && !doll.IsMoving, "Finished tasks restore idle follow");

        // 其他端只收到同步的目标格：同样飞过去播放挖矿动画，但不敲击。
        commands.Targets.Add(first); commands.Hits = 0;
        doll.Projectile.netUpdate = false; Tick(doll);
        Check(doll.HasWork && doll.Projectile.netUpdate, "New target must be synced");
        var stream = new MemoryStream();
        doll.SendExtraAI(new BinaryWriter(stream));
        var remote = new KumaPetProjectile(); remote.SetDefaults();
        remote.Projectile.owner = 0; remote.Projectile.active = true; remote.Projectile.Center = doll.Projectile.Center;
        stream.Position = 0; remote.ReceiveExtraAI(new BinaryReader(stream));
        World.myPlayer = 1;
        for (int i = 0; i < 240; i++) Tick(remote);
        World.myPlayer = 0;
        Check(remote.IsMining && commands.Hits == 0 && commands.HasTask, "Remote copies animate at the synced target without hitting");
        commands.Targets.Clear();
        for (int i = 0; i < 300; i++) Tick(doll);
        Check(!doll.HasWork && !doll.IsMining, "Cleared tasks send the doll back to following");
        commands.PlacementTargets.AddRange(new[] { first, second });
        int priorHits = commands.Hits;
        doll.AI();
        using (var plantStream = new MemoryStream())
        {
            doll.SendExtraAI(new BinaryWriter(plantStream));
            plantStream.Position = 0;
            remote.ReceiveExtraAI(new BinaryReader(plantStream));
            World.myPlayer = 1;
            for (int i = 0; i < 240; i++) Tick(remote);
            World.myPlayer = 0;
            Check(remote.IsUsing && commands.PlacedBlocks == 0 && commands.PlacementTargets.Count == 2,
                "Synced placement targets animate remotely without consuming or placing");
        }
        for (int i = 0; i < 900 && commands.HasTask; i++) Tick(doll);
        Check(commands.PlacedBlocks == 2 && commands.Hits == priorHits && !commands.HasTask,
            "Owner flies to placement targets and calls placement, never mining");
        commands.PlacementTargets.Add(first); Tick(doll); commands.PlacementTargets.Clear(); Tick(doll);
        Check(!doll.HasWork, "Cancel a placement task immediately restores following");
        foreach (float speed in new[] { 3f, 12f, 30f })
        {
            owner.velocity = new Vector2(speed, 0f);
            var frames = new System.Collections.Generic.HashSet<int>();
            float min = float.MaxValue, max = float.MinValue;
            for (int i = 0; i < 600; i++)
            {
                Tick(doll); frames.Add(doll.Projectile.frame);
                if (i > 300) { min = Math.Min(min, doll.Projectile.Center.Y); max = Math.Max(max, doll.Projectile.Center.Y); }
            }
            Check(doll.IsMoving && frames.SetEquals(new[] { 0, 1, 2, 3 }), $"Four movement frames at speed {speed}");
            Check(Vector2.Distance(owner.Center, doll.Projectile.Center) < 110f, $"Must keep up at speed {speed}");
            Check(max - min > 5f && max - min < 30f, $"Flight must gently bob and drift at speed {speed}: {max - min}");
            owner.velocity = Vector2.Zero;
            for (int i = 0; i < 180; i++) Tick(doll);
            Check(!doll.IsMoving && !doll.IsMining, "Stopping must restore idle");
        }
        owner.direction = -1; owner.velocity = new Vector2(-5f, 0f);
        for (int i = 0; i < 180; i++) Tick(doll);
        Check(doll.Projectile.spriteDirection == -1 && doll.Projectile.Center.X > owner.Center.X, "Leftward follow and facing");
        owner.Center += new Vector2(5000f, 3000f);
        doll.Projectile.netUpdate = false; doll.AI();
        Check(Vector2.Distance(doll.Projectile.Center, owner.Center) < 100f && doll.Projectile.velocity == Vector2.Zero && doll.Projectile.netUpdate,
            "Owner teleport must recall and sync immediately");
        doll.Projectile.Center += new Vector2(300f, 0f);
        Vector2 before = doll.Projectile.Center;
        Check(!skill.BeginWandUse() && !skill.PreUseSkill() && mana.MagicPoint == 80f
            && World.projectile.Count(p => p.active) == 4 && before == doll.Projectile.Center,
            "While the doll is out, left-click belongs to commands: no swing, cost, duplicate or recall");
        mana.MagicPoint = 19f;
        Check(!skill.PreUseSkill() && mana.MagicPoint == 19f && before == doll.Projectile.Center, "Insufficient mana must have no effect");
        World.myPlayer = 1; skill.Initialize();
        Check(skill.PreUseSkill() && World.projectile.Count(p => p.active) == 8 && doll.Projectile.owner == 0,
            "Each player keeps their own doll");
        World.myPlayer = 0;
        owner.gravDir = -1f; doll.RecallToOwner(owner);
        Check(doll.Projectile.Center.Y > owner.Center.Y, "Inverted gravity follows above inverted player");
        World.myPlayer = 255; World.netMode = 2;
        Check(!skill.PreUseSkill(), "Dedicated server must not cast or access LocalPlayer");
        doll.Projectile.Center += new Vector2(5000f, 0f); before = doll.Projectile.Center;
        doll.Projectile.netUpdate = false; doll.AI();
        Check(doll.Projectile.Center == before && !doll.Projectile.netUpdate, "Server must not originate teleport");
        World.myPlayer = 0; World.netMode = 0;
        commands.Loot = 3; DollPetProjectile.DismissAll(owner);
        Check(doll.Projectile.active && doll.IsDismissing && commands.Delivered == 5 && DollPetProjectile.FindOwned(owner) == null,
            "Dismiss hands over collected items at once and starts the vanish animation");
        var team = World.projectile.Where(p => p.active && p.owner == 0).ToArray();
        DollPetProjectile backRow = team.Select(p => (DollPetProjectile)p.ModProjectile).First(d => d.FormationSlot == 3);
        int vanishTicks = 0, backRowGone = -1;
        while (team.Any(p => p.active) && vanishTicks++ < 200)
        {
            foreach (var p in team.Where(p => p.active)) p.ModProjectile.AI();
            if (!backRow.Projectile.active && backRowGone < 0) backRowGone = vanishTicks;
        }
        Check(backRowGone == DollSummonVisuals.VanishTicks, "Back row vanishes first, without waiting for the stagger");
        Check(!doll.Projectile.active && vanishTicks == DollSummonVisuals.VanishTicks + 3 * DollSummonVisuals.StaggerTicks,
            "Every doll is removed once its staggered vanish animation finishes");
        commands.Loot = 4; mana.MagicPoint = 100f; owner.gravDir = 1f;
        Check(skill.BeginWandUse() && skill.PreUseSkill() && commands.Delivered == 9 && mana.MagicPoint == 80f,
            "Summoning again hands over items left by a doll that vanished early");
        doll = (KumaPetProjectile)DollPetProjectile.FindOwned(owner);
        owner.dead = true; doll.AI();
        Check(!doll.Projectile.active, "Death must remove doll");
        doll = Reset(); World.player[0].active = false; doll.AI();
        Check(!doll.Projectile.active, "Disconnect must remove doll");
        doll = Reset(); doll.Projectile.owner = 255; doll.AI();
        Check(!doll.Projectile.active, "Invalid owner must not index player array");
        Reset(); World.projectile[0].Kill();
        foreach (var p in World.projectile) { p.active = true; p.ModProjectile = null; }
        mana = World.player[0].GetModPlayer<ElainaAttributeModPlayer>();
        Check(!skill.PreUseSkill() && mana.MagicPoint == 80f, "Full projectile pool must not consume mana");
        skill.DeserializeData(new Terraria.ModLoader.IO.TagCompound());
        Check(skill.MaxCD == 1f && skill.MagicPointCost == 20f, "Old skill saves must restore current cost/cooldown");
        RunTeamChecks();
        doll = Reset(); owner = World.player[0]; commands = owner.GetModPlayer<DollCommandPlayer>();
        doll.Projectile.Center = owner.Center + new Vector2(500, 0);
        commands.Loot = 1; commands.MiningCarrier = doll;
        commands.PreviewGardenTarget = new Point(40, 20);
        commands.TileMarks.Add(new Point(300, 300));
        Tick(doll);
        Check(Target(doll).Point.X < 0 && commands.Delivered == 0,
            "Miner carrying confirmed loot returns instead of taking a garden job or delivering remotely");
        for (int i = 0; i < 300 && commands.Delivered == 0; i++) Tick(doll);
        Check(commands.Delivered == 1 && Vector2.Distance(commands.DeliveredAt, owner.Center) < 120f,
            "Actual carrier AI flies back before releasing confirmed cargo");
        Check(commands.TileMarks.Contains(new Point(300, 300)), "Returning distant-task cargo preserves its queued work");
        Tick(doll);
        Check(Target(doll).Point == commands.PreviewGardenTarget, "After returning cargo the worker can tend a nearby planter");

        Reset();
        owner = World.player[0];
        commands = owner.GetModPlayer<DollCommandPlayer>();
        foreach (var p in World.projectile) p.active = false;
        var commandSkill = new DollControlSkill(); commandSkill.Initialize();
        伊蕾娜.ElainaModSkills.ElainaSkillModPlayer.GetActiveSkill = [new() { ModSkill = commandSkill }];
        mana = owner.GetModPlayer<ElainaAttributeModPlayer>(); mana.MagicPoint = 100;
        commandSkill.OpenCommands();
        Check(Team().Length == 4 && mana.MagicPoint == 80 && commands.CommandOpens == 1,
            "Production command entry summons missing team, pays once, then opens commands");
        mana.MagicPoint = 0;
        commandSkill.OpenCommands();
        Check(Team().Length == 4 && mana.MagicPoint == 0 && commands.CommandOpens == 2,
            "Present dolls open commands without mana or duplicate summon");
        foreach (var p in World.projectile) p.active = false;
        commandSkill.OpenCommands();
        Check(Team().Length == 0 && commands.CommandOpens == 2,
            "Failed summon cannot open commands or spawn a partial team");
        RunProgressionChecks();
        Console.WriteLine($"PASS: {assertions} production skill/AI/target assignment assertions (world/resource host is stubbed).");
    }
    public static void Main(string[] args) => Run(args[0]);
}

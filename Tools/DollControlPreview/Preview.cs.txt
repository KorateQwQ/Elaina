sealed class Preview : Game
{
    readonly string repo, output;
    readonly Dictionary<string, Texture2D> textures = new();
    SpriteBatch batch;
    Texture2D pixel;
    bool done;
    Preview(string repo, string output)
    {
        this.repo = repo; this.output = output;
        new GraphicsDeviceManager(this) { PreferredBackBufferWidth = 640, PreferredBackBufferHeight = 300,
            GraphicsProfile = GraphicsProfile.HiDef, SynchronizeWithVerticalRetrace = false };
        IsFixedTimeStep = false;
        SDL.SDL_HideWindow(Window.Handle);
    }
    protected override void LoadContent()
    {
        batch = new SpriteBatch(GraphicsDevice); World.spriteBatch = batch;
        pixel = new Texture2D(GraphicsDevice, 1, 1); pixel.SetData(new[] { Color.White });
        ModContent.LoadTexture = LoadTexture;
        TextureAssets.Projectile[0] = new Asset<Texture2D>(LoadTexture(new KumaPetProjectile().Texture));
    }
    Texture2D LoadTexture(string path)
    {
        if (textures.TryGetValue(path, out var texture)) return texture;
        string local = path[(path.IndexOf('/') + 1)..].Replace('/', Path.DirectorySeparatorChar) + ".png";
        using var stream = File.OpenRead(Path.Combine(repo, local));
        texture = Texture2D.FromStream(GraphicsDevice, stream);
        var pixels = new Color[texture.Width * texture.Height]; texture.GetData(pixels);
        foreach (Color color in pixels)
            if (color.A != 0 && color.A != 255) throw new Exception("Semi-transparent asset: recheck tML/FNA alpha conversion.");
        textures.Add(path, texture);
        return texture;
    }
    protected override void Draw(GameTime gameTime)
    {
        if (done) return;
        done = true;
        DrawSheet();
        DrawFollow();
        DrawTeamFormations();        DrawProgression();
        DrawChestApproach();
        Console.WriteLine($"FNA preview: {textures.Count} production PNGs, real PreDraw, 150 trajectory frames.");
        Exit();
    }
    void Begin(float scale = 1f) => batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend,
        SamplerState.PointClamp, DepthStencilState.None, RasterizerState.CullNone, null, Matrix.CreateScale(scale));
    void DrawChestApproach()
    {
        var doll = Checks.Reset();
        var owner = World.player[0]; owner.Center = new Vector2(150, 128); owner.velocity = Vector2.Zero;
        owner.direction = 1;
        foreach (var member in Checks.Team()) member.RecallToOwner(owner);
        var commands = owner.GetModPlayer<DollCommandPlayer>();
        commands.PreviewChestTarget = new Point(25, 7);
        commands.PreviewChestHelper = doll;
        commands.RemoteChestActionFinished = false;
        World.screenPosition = Vector2.Zero;
        string folder = Path.Combine(output, "chest-approach"); Directory.CreateDirectory(folder);
        using var target = new RenderTarget2D(GraphicsDevice, 1000, 400);
        var states = new List<string>();
        for (int tick = 0; tick < 160; tick++)
        {
            if (tick == 95) { commands.PreviewChestTarget = null; commands.PreviewChestHelper = null; }
            Checks.TickTeam();
            if (tick % 4 != 0) continue;
            GraphicsDevice.SetRenderTarget(target); GraphicsDevice.Clear(new Color(23, 30, 43));
            Begin(2);
            batch.Draw(pixel, new Rectangle(0, 150, 500, 50), new Color(38, 48, 46));
            // Neutral player and chest fixtures; the doll's travel, hand gesture and PreDraw are production code.
            batch.Draw(pixel, new Rectangle(144, 108, 12, 12), new Color(194, 201, 215));
            batch.Draw(pixel, new Rectangle(141, 121, 18, 29), new Color(105, 119, 151));
            bool open = commands.PreviewChestOpened > 0 && tick < 95;
            batch.Draw(pixel, new Rectangle(400, 128, 32, 22), new Color(116, 68, 38));
            batch.Draw(pixel, new Rectangle(400, open ? 112 : 122, 32, 7), new Color(194, 143, 65));
            batch.Draw(pixel, new Rectangle(405, 129, 3, 20), new Color(225, 182, 76));
            batch.Draw(pixel, new Rectangle(424, 129, 3, 20), new Color(225, 182, 76));
            if (open)
                batch.Draw(pixel, new Rectangle(400, 123, 32, 4), new Color(32, 25, 26));
            foreach (var member in Checks.Team()) { Color light = Color.White; member.Shown().PreDraw(ref light); }
            batch.End();
            Save(target, $"chest-approach/frame-{tick / 4:00}.png");
            states.Add(tick >= 95 ? "return" : open ? "open" : doll.IsUsing ? "gesture" : "travel");
        }
        File.WriteAllLines(Path.Combine(folder, "states.txt"), states);
    }
    void DrawSheet()
    {
        Checks.Reset();
        foreach (var doll in Checks.Team())
        {
            World.screenPosition = Vector2.Zero;
            // 列：Idle 1~3 / Move 1~4 / Mine 1~4 / Use 1~4 / Hammer 1~4。
            var cells = new (float State, int Frame)[] { (0, 0), (0, 1), (0, 2), (1, 0), (1, 1), (1, 2), (1, 3), (2, 0), (2, 1), (2, 2), (2, 3), (3, 0), (3, 1), (3, 2), (3, 3), (4, 0), (4, 1), (4, 2), (4, 3) };
            using var target = new RenderTarget2D(GraphicsDevice, cells.Length * 160, 600);
            GraphicsDevice.SetRenderTarget(target); GraphicsDevice.Clear(new Color(18, 23, 34));
            Begin();
            for (int row = 0; row < 3; row++)
            for (int col = 0; col < cells.Length; col++)
            {
                batch.Draw(pixel, new Rectangle(col * 160 + 2, row * 200 + 2, 156, 196), new Color(29, 35, 47));
                doll.Projectile.spriteDirection = row == 0 ? -1 : 1;
                World.player[0].gravDir = row == 2 ? -1f : 1f;
                (doll.Projectile.ai[1], doll.Projectile.frame) = cells[col];
                doll.Projectile.Center = new Vector2(col * 160 + 80, row * 200 + 76);
                doll.Projectile.scale = 3f;
                Color light = Color.White; doll.Shown().PreDraw(ref light);
                doll.Projectile.Center.Y = row * 200 + 172;
                doll.Projectile.scale = 1f; doll.Shown().PreDraw(ref light);
            }
            batch.End(); Save(target, $"doll-{doll.FormationSlot}-frames.png");
            if (doll.FormationSlot == 0) Save(target, "kuma-frames.png");
        }
    }
    void DrawFollow()
    {
        Directory.CreateDirectory(Path.Combine(output, "frames"));
        var doll = Checks.Reset();
        var team = Checks.Team();
        var owner = World.player[0];
        var commands = owner.GetModPlayer<DollCommandPlayer>();
        commands.HitsPerTile = 1000;
        Point marked = default;
        using var target = new RenderTarget2D(GraphicsDevice, 640, 400);
        for (int tick = 0; tick < 600; tick++)
        {
            if (tick == 480)
            {
                // 地面里的四格标记，四只玩偶分别认领并挖到预览结束。
                marked = new Point((int)(owner.Center.X / 16f) + 2, (int)((owner.Center.Y + 23f) / 16f) + 1);
                for (int i = 0; i < 4; i++) commands.Targets.Add(new Point(marked.X + i * 2, marked.Y));
            }
            if (tick == 60) owner.velocity = new Vector2(5f, 0f);
            if (tick == 210 || tick == 420) owner.velocity = Vector2.Zero;
            if (tick == 270) { owner.direction = -1; owner.velocity = new Vector2(-5f, 0f); }
            Checks.TickTeam();
            if (tick % 4 != 0) continue;
            GraphicsDevice.SetRenderTarget(target); GraphicsDevice.Clear(new Color(23, 30, 43));
            Begin(2f);
            World.screenPosition = owner.Center - new Vector2(160f, 104f);
            batch.Draw(pixel, new Rectangle(0, 127, 320, 73), new Color(38, 48, 46));
            for (int x = -32; x < 352; x += 32)
                batch.Draw(pixel, new Rectangle(x - (int)(owner.Center.X % 32), 128, 1, 72), new Color(48, 60, 53));
            // Neutral player marker; the doll uses production AI, texture selection and PreDraw.
            batch.Draw(pixel, new Rectangle(155, 85, 10, 10), new Color(194, 201, 215));
            batch.Draw(pixel, new Rectangle(153, 96, 14, 22), new Color(105, 119, 151));
            batch.Draw(pixel, new Rectangle(153, 118, 5, 9), new Color(82, 91, 113));
            batch.Draw(pixel, new Rectangle(162, 118, 5, 9), new Color(82, 91, 113));
            foreach (Point mark in commands.Targets)
            {
                Vector2 cell = new Vector2(mark.X * 16, mark.Y * 16) - World.screenPosition;
                var area = new Rectangle((int)cell.X, (int)cell.Y, 16, 16);
                batch.Draw(pixel, area, new Color(255, 196, 92) * .3f);
                batch.Draw(pixel, new Rectangle(area.X, area.Y, 16, 1), new Color(255, 196, 92));
                batch.Draw(pixel, new Rectangle(area.X, area.Bottom - 1, 16, 1), new Color(255, 196, 92));
                batch.Draw(pixel, new Rectangle(area.X, area.Y, 1, 16), new Color(255, 196, 92));
                batch.Draw(pixel, new Rectangle(area.Right - 1, area.Y, 1, 16), new Color(255, 196, 92));
            }
            foreach (var member in team) { Color light = Color.White; member.Shown().PreDraw(ref light); }
            batch.End(); Save(target, $"frames/{tick / 4:D3}.png");
            if (tick == 596) Save(target, "team-working.png");
        }
    }

        Begin(2f);
        for (int panel = 0; panel < 3; panel++)
        {
            Checks.Reset();
            var owner = World.player[0];
            owner.direction = panel == 1 ? -1 : 1;
            owner.gravDir = panel == 2 ? -1f : 1f;
            owner.Center = new Vector2(panel * 200 + (panel == 1 ? 40 : 140), panel == 2 ? 60 : 130);
            foreach (var member in Checks.Team()) member.RecallToOwner(owner);
            for (int i = 0; i < 180; i++) Checks.TickTeam();
            World.screenPosition = Vector2.Zero;
            batch.Draw(pixel, new Rectangle(panel * 200 + 2, 2, 196, 196), new Color(29, 35, 47));
            batch.Draw(pixel, new Rectangle((int)owner.Center.X - 7, (int)owner.Center.Y - 20, 14, 40), new Color(105, 119, 151));
            foreach (var member in Checks.Team()) { Color light = Color.White; member.Shown().PreDraw(ref light); }
        }
        batch.End(); Save(target, "team-formations.png");
    }

    protected override void UnloadContent()
    {

    }

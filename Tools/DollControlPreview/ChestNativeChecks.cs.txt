    private sealed class RemoteChestFixture : ModTile
    {
        internal static bool CheckOwnRange;
        internal static int Calls, Opens;
        public override bool RightClick(int i, int j)
        {
            Calls++;
            var player = Host.LocalPlayer;
            if (CheckOwnRange && !player.IsInTileInteractionRange(i, j, TileReachCheckSettings.Simple)) return false;
            if (player.inventory[0].type != ItemID.Ruby || player.inventory[0].stack <= 0) return false;
            DollChestReachSystem.TryChestOrigin(i, j, out var p);
            int chest = Chest.FindChest(p.X, p.Y);
            if (chest < 0) return false;
            player.inventory[0].stack--; Opens++;
            player.OpenChest(p.X, p.Y, chest);
            return true;
        }
    }

        void CheckModChest()
        {
            ushort type = (ushort)remoteFixture.Type;
            TileID.Sets.BasicChest[type] = true;
            TileObjectData.newTile.CopyFrom(TileObjectData.Style2x2); TileObjectData.addTile(type);

            RemoteChestFixture.CheckOwnRange = true; player.inventory[0] = new(ItemID.Ruby, 2);
            ClickAndWait();
            Check(RemoteChestFixture.Calls == 1 && RemoteChestFixture.Opens == 0 && player.chest == -1, "Mod RightClick runs but its own range check still rejects");

                "Standard mod chest opens through its own callback and custom key rule");
            CheckLifetime(); Check(player.chest == index, "Mod chest using vanilla UI retains vanilla-lifetime extension"); Close();
        }

    {

        protected virtual void OnScreenSizeChanged(Layout.Size current, Layout.Size previous) { }
        protected virtual void OnExitTree() { }
        protected virtual void UpdateStatus(GameTime gameTime) { }
        protected virtual void Update(GameTime gameTime) { }
        protected virtual void Draw(GameTime gameTime, SpriteBatch batch) => throw new NotSupportedException();

        protected void SetTop(float value, float percent, float alignment) { }

        public void Tick(GameTime gameTime) { if (Enabled) Update(gameTime); }

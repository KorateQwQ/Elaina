internal static class FireBurstPreview
{
    static void Main(string[] args)
    {
        if (OperatingSystem.IsWindows())
            System.Runtime.InteropServices.NativeLibrary.Load(Path.Combine(Environment.SystemDirectory, "d3d11.dll"));
        using var game = new BurstProbe(args[0], args[1], args[2], args.Contains("--glow-comparison"), args.Contains("--domain"), args.Contains("--cast"), args.Contains("--ring-bloom"), args.Contains("--impact-only"));
        game.RunOneFrame();
        if (!game.Exported) throw new InvalidOperationException("Fire burst preview did not complete.");
    }
}

    void AssertHidden()
    {
        var flags = (SDL.SDL_WindowFlags)SDL.SDL_GetWindowFlags(Window.Handle);
        if ((flags & SDL.SDL_WindowFlags.SDL_WINDOW_HIDDEN) == 0
            || (flags & SDL.SDL_WindowFlags.SDL_WINDOW_SHOWN) != 0)
            throw new InvalidOperationException($"Offline preview window unexpectedly visible: {flags}");
    }

    Color[] Render(string name, int frame, bool enabled = true, bool day = false, int count = 1, bool active = true, bool save = true,
        float zoom = 1f, bool onlyRings = false, int ringSide = 0, bool layerProbe = false, int bloomIterations = 0,
        bool legacyImpactGlow = false, float bodyPeakGain = 3f, bool keepParticles = false, bool domainPrepass = false)
    {
        AssertHidden();
        using var target = new RenderTarget2D(GraphicsDevice, Width, Height, false, SurfaceFormat.Color, DepthFormat.None);
        GraphicsDevice.SetRenderTarget(target);
        GraphicsDevice.Clear(day ? new Color(132, 170, 204) : new Color(14, 18, 30));
        queue.GetMethod("ClearFrame").Invoke(null, null);
        Set(renderer.GetType(), "PreviewBloomIterations", bloomIterations);
        Set(visual, "Calls", 0);
        Set(visual, "CircleCalls", 0);
        Set(visual, "RearCalls", 0); Set(visual, "FrontCalls", 0);
        Set(visual, "OnlyRings", onlyRings);
        Set(visual, "RingSide", ringSide);
        Set(visual, "LayerProbe", layerProbe);
        Set(visual, "LegacyImpactGlow", legacyImpactGlow);
        Set(visual, "BodyPeakGain", bodyPeakGain);
        object view = main.GetField("GameViewMatrix").GetValue(null);
        view.GetType().GetField("TransformationMatrix").SetValue(view,
            Matrix.CreateTranslation(-550, -870, 0) * Matrix.CreateScale(zoom, zoom, 1) * Matrix.CreateTranslation(550, 870, 0));
        // 游戏中其他绘制会在各采样槽留下 PointClamp；绘制前按游戏状态设置，绘制方法须自行指定需要的采样方式
        for (int slot = 1; slot < 4; slot++) GraphicsDevice.SamplerStates[slot] = SamplerState.PointClamp;
        var centers = new List<Vector2>();
        for (int i = 0; i < count; i++) {
            float x = count == 1 ? 550f : 400f + 300f * i;
            visual.GetMethod("Submit").Invoke(Activator.CreateInstance(visual), new object[] { frame, x, active });
            centers.Add(ScreenPosition + new Vector2(x, 870));
        }
        // 失效弹幕的检查不含粒子（粒子生命周期独立于弹幕）
        if ((!active && !keepParticles) || onlyRings) centers.Clear();
        pipeline.GetType("KL.Drawing.ParticleSim").GetMethod("Run").Invoke(null, new object[] { centers, frame });
        pipeline.GetType("KL.Drawing.ParticleSim").GetMethod("Submit").Invoke(null, null);
        var requests = (System.Collections.IDictionary)queue.GetField("BloomRequests", All).GetValue(null);
        string[] layers = requests.Keys.Cast<object>().Select(k => k.ToString()).ToArray();
        if (!layers.SequenceEqual(new[] { "(OverPlayers, Before)" })) throw new Exception("Incorrect circle/body stage layers");
        if (frame > 30) {
            // 后圈和主体必须在同一个 Bloom 捕获组内，仅用 order 排序。
            var groups = (System.Collections.IDictionary)requests.Values.Cast<object>().Single();
            var keys = groups.Keys.Cast<object>().Select(k => ((string Id, int Order))k).ToArray();
            if (keys.Single(k => k.Id == "Elaina:FireBurst:WindRear").Order >= keys.Single(k => k.Id == "Elaina:FireBurst").Order)
                throw new Exception("Orange rear must draw before the body within their shared Bloom layer");
        }
        var plainRequests = (System.Collections.IDictionary)queue.GetField("Requests", All).GetValue(null);
        if (plainRequests.Count != (frame > 30 ? 1 : 0)
            || (frame > 30 && plainRequests.Keys.Cast<object>().Single().ToString() != "(Dust, After)"))
            throw new Exception("Black foreground must be a normal draw after the completed projectile/particle Bloom");
        // Ground tiles are drawn before the projectile layer; the burst is composited in front of them.
        batch.Begin(); batch.Draw(pixel, new Rectangle(0, 900, Width, 100), new Color(45, 50, 40)); batch.End();
        if (domainPrepass)
        {
            var request = queue.GetMethods().Single(m => m.Name == "RequestBloom" && m.GetParameters().Length == 5
                && m.GetParameters()[3].ParameterType == typeof(Action));
            Type layer = pipeline.GetType("KL.Drawing.LayerDrawRequestSystem+DrawTargetLayer");
            Type timing = pipeline.GetType("KL.Drawing.LayerDrawRequestSystem+DrawTiming");
            Action drawDomain = () => pipeline.GetType("KL.Drawing.DomainRingPreview").GetMethod("Draw").Invoke(null,
                new object[] { ScreenPosition + new Vector2(550, 870), 10000f, 100f, 500f });
            request.Invoke(null, new object[] { "Probe:Domain", Enum.Parse(layer, "OverPlayers"), Enum.Parse(timing, "Before"), drawDomain, -101 });
        }
        renderer.GetType().GetMethod("RunForeground").Invoke(renderer, new object[] { enabled });
        // 原版 Main.DrawDust 自管 Begin/End；Dust.After 在它返回之后执行，此时没有活动批次。
        batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.PointClamp, DepthStencilState.None, RasterizerState.CullNone, null, Matrix.Identity);
        LastParticles = (int)pipeline.GetType("KL.Drawing.ParticleSim").GetMethod("Draw").Invoke(null, null);
        batch.End();
        renderer.GetType().GetMethod("RunPlainLayer").Invoke(renderer, new object[] { 6 });
        if ((bool)typeof(SpriteBatch).GetField("beginCalled", All).GetValue(batch))
            throw new Exception("Dust.After callback leaked an active SpriteBatch");
        // 后续绘制必须仍能独立开启/关闭自己的批次。
        batch.Begin(); batch.End();
        if ((int)visual.GetField("Calls").GetValue(null) != (active ? count : 0)) throw new Exception("Incorrect draw callback count");
        if ((int)visual.GetField("CircleCalls").GetValue(null) != (active ? count : 0)) throw new Exception("Incorrect circle callback count");
        if ((int)visual.GetField("RearCalls").GetValue(null) != (active && frame > 30 ? count : 0)
            || (int)visual.GetField("FrontCalls").GetValue(null) != (active && frame > 30 ? count : 0))
            throw new Exception("Incorrect split ring callback count");
        var pixels = new Color[Width * Height]; target.GetData(pixels);
        GraphicsDevice.SetRenderTarget(null);
        if (save) {
            using var image = new Texture2D(GraphicsDevice, Width, Height); image.SetData(pixels);
            using var stream = File.Create(Path.Combine(output, name + ".png")); image.SaveAsPng(stream, Width, Height);
        }
        return pixels;
    }

    protected override void Draw(GameTime gameTime)
    {
        AssertHidden();
        if (done) return; done = true;
        if (Environment.GetCommandLineArgs().Contains("--wind-state"))
        {
            Render("wind-isolated", 40, day: true, zoom: 2f, onlyRings: true);
            Render("wind-domain-first", 40, day: true, zoom: 2f, onlyRings: true, domainPrepass: true);
            Render("wind-full-domain-first", 40, day: true, zoom: 2f, domainPrepass: true);
            return;
        }
        if (glowComparison) { SaveGlowComparison(); Exit(); return; }
        if (ringBloomMode) { SaveRingBloomPreview(); Exit(); return; }
        if (domainMode) { SaveDomainPreview(); Exit(); return; }
        if (impactOnly) {
            foreach (bool day in new[] { false, true })
                foreach (int frame in new[] { 44, 46, 50, 56, 64 })
                    RenderCast(frame, day, 1.25f, Vector2.Zero, $"cast/impact-only-{(day ? "day" : "night")}-f{frame}");
            Console.WriteLine("Rendered impact-only frames with formal DrawImpact and KL Bloom.");
            Exit(); return;
        }
        if (castMode && Environment.GetCommandLineArgs().Contains("--wand-trail")) { SaveWandTrailPreview(); Exit(); return; }
        if (castMode && Environment.GetCommandLineArgs().Contains("--launch-ticks")) { SaveLaunchTicks(); Exit(); return; }
        if (castMode) { SaveCastPreview(); Exit(); return; }
        CheckBodyGlow();
        CheckFixedRingHalves();
        // 强 HDR 火柱与较晚的粒子层必须被前侧黑面遮挡，而不能把黑面重新冲亮。
        foreach (bool bloomEnabled in new[] { false, true }) {
            var probe = Render($"foreground-hdr-probe-{bloomEnabled}", 36, enabled: bloomEnabled,
                day: true, onlyRings: true, layerProbe: true);
            int occluded = 0;
            for (int y = 815; y < 855; y++) for (int x = 542; x < 558; x++) {
                Color c = probe[y * Width + x];
                if (c.R < 180 && c.G < 180 && c.B < 180) occluded++;
            }
            if (occluded < 20) throw new Exception($"Foreground failed to cover late HDR content: Bloom={bloomEnabled}, pixels={occluded}");
        }
        var on = Render("night-bloom-f36", 36);
        var wideBloom = Render("comparison-bloom5-f36", 36, bloomIterations: 5);
        SaveBloomComparison(on, wideBloom);
        var off = Render("night-no-bloom-f36", 36, enabled: false);
        int brighter = on.Zip(off, (a, b) => a.R > b.R + 2 || a.G > b.G + 2 || a.B > b.B + 2 ? 1 : 0).Sum();
        if (brighter < 100) throw new Exception("Bloom produced no visible halo");
        var starOn = Render("star-bloom-f06", 6);
        var starOff = Render("star-no-bloom-f06", 6, enabled: false);
        if (starOn.Zip(starOff, (a, b) => a.R > b.R + 2 || a.G > b.G + 2 ? 1 : 0).Sum() < 100) throw new Exception("Star Bloom produced no halo");
        for (int frame = 1; frame <= 30; frame++) {
            var glow = Render("star-frame-on", frame, save: false);
            var plain = Render("star-frame-off", frame, enabled: false, save: false);
            int halo = glow.Zip(plain, (a, b) => a.R > b.R + 2 || a.G > b.G + 2 || a.B > b.B + 2 ? 1 : 0).Sum();
            if (halo < 100) throw new Exception($"Star lost Bloom at frame {frame}");
        }
        Render("day-bloom-f36", 36, day: true);
        Render("day-star-f06", 6, day: true);
        Render("day-star-f30", 30, day: true);
        Render("day-zoom125-f40", 40, day: true, zoom: 1.25f);
        var rings = Render("rings-day-no-bloom-f36", 36, enabled: false, day: true, onlyRings: true);
        int black = 0, orange = 0;
        for (int y = 600; y < 880; y++) for (int x = 250; x < 850; x++) {
            Color c = rings[y * Width + x];
            if (c.R < 95 && c.G < 125 && c.B < 155) black++;
            // 内面允许调成橙色或偏黄的高亮，不把绿色上限写死为旧调色值。
            if (c.R > 200 && c.B < 125) orange++;
        }
        if (black < 100 || orange < 100) throw new Exception($"Ring layers missing: black={black}, orange={orange}");
        Render("rings-day-bloom-f36", 36, day: true, onlyRings: true);
        Render("rings-day-no-bloom-f42", 42, enabled: false, day: true, onlyRings: true);
        // 分别渲染可见的两面：旧的径向双色条带会在每个半圈同时出现黑、橙，必须失败。
        for (int side = -1; side <= 1; side += 2) {
            var face = Render(side < 0 ? "ring-inner-face" : "ring-outer-face", 36,
                enabled: false, day: true, onlyRings: true, ringSide: side);
            int darkPixels = 0, hotPixels = 0;
            for (int y = 600; y < 880; y++) for (int x = 250; x < 850; x++) {
                Color c = face[y * Width + x];
                if (c.R < 95 && c.G < 125 && c.B < 155) darkPixels++;
                if (c.R > 200 && c.B < 125) hotPixels++;
            }
            if (side < 0 ? darkPixels != 0 || hotPixels < 100 : hotPixels != 0 || darkPixels < 100)
                throw new Exception($"Ribbon face color regression: side={side}, dark={darkPixels}, hot={hotPixels}");
        }
        var expiredRings = Render("rings-expired-f60", 60, day: true, onlyRings: true, save: false);
        if (expiredRings[800 * Width + 550] != new Color(132, 170, 204)) throw new Exception("Expired rings still visible");
        foreach (int frame in new[] { 1, 15, 29, 30, 31, 42, 50, 59 }) Render($"night-bloom-f{frame:00}", frame);
        SaveEdgeCrop("day-edge-x3-f33", Render("day-bloom-f33", 33, day: true, save: false));
        SaveParticleSheet();
        SaveBaseSheet();
        // 爆发冲击粒子局部：第 3、6 帧落点附近，夜晚
        foreach (int age in new[] { 33, 36 }) {
            var image = Render("impact", age, save: false);
            var area = new Rectangle(330, 560, 440, 340);
            var crop = new Color[area.Width * area.Height];
            for (int y = 0; y < area.Height; y++) for (int x = 0; x < area.Width; x++) crop[y * area.Width + x] = image[(area.Y + y) * Width + area.X + x];
            using var t = new Texture2D(GraphicsDevice, area.Width, area.Height); t.SetData(crop);
            using var st = File.Create(Path.Combine(output, $"impact-f{age}.png")); t.SaveAsPng(st, area.Width, area.Height);
        }
        Render("multiple", 36, count: 2);
        Render("multiple-stars", 6, count: 2);
        Render("invalidated-star", 6, active: false);
        var invalidated = Render("invalidated", 36, active: false);
        if (invalidated[Width * 500 + 550] != new Color(14, 18, 30)) throw new Exception("Inactive projectile still draws");
        Console.WriteLine($"PASS: actual FireBurst + circle + two-sided ribbons + KL Bloom; {brighter} brighter burst pixels; black={black}, orange={orange}; rear/body share one ordered Bloom layer; Dust.After enters/exits with no active SpriteBatch; fixed halves through 360 degrees; foreground covers late HDR content with Bloom on/off; harmless windup, server guard, synced age, invalidation, 125% zoom and multiple instances checked.");
        Exit();
    }

    // 读取正式绘制实际写入 Shader 的色带值，验证峰值、淡出及 alpha 不变。
    void CheckBodyGlow()
    {
        float[] expected = [3f, 2.9140625f, 2.6875f, 2.3671875f, 2f, 1.6328125f, 1.3125f, 1.0859375f, 1f, 1f];
        var shaders = (System.Collections.IDictionary)pipeline.GetType("KL.Drawing.ModContent").GetField("Effects").GetValue(null);
        var flame = (Effect)shaders["伊蕾娜/Effects/Content/FireBurstFlame"];
        (string name, Vector3 rgb)[] bands = [
            ("BandOuter", new(2f, .46f, .06f)), ("BandMid", new(2f, .82f, .09f)), ("BandInner", new(3.6f, 2.23f, .65f))];
        foreach (bool day in new[] { false, true })
        for (int tick = 1; tick <= 10; tick++)
        {
            Render($"body-glow-{(day ? "day" : "night")}-t{tick:00}", 30 + tick, day: day);
            foreach (var band in bands) {
                Vector4 actual = flame.Parameters[band.name].GetValueVector4();
                Vector4 wanted = new(band.rgb * expected[tick - 1], 1f);
                if (Vector4.Distance(actual, wanted) > .0001f)
                    throw new Exception($"Body glow/coverage regression: tick={tick}, band={band.name}, actual={actual}, expected={wanted}");
            }
            Console.WriteLine($"Body glow: day={day}, tick={tick}, gain={expected[tick - 1]:F4}, all three bands preserve alpha=1");
        }
    }

    // 同一帧、同一尺寸的参数对照：左侧游戏配置，右侧以前预览误用的 5 层采样。

    Color[] RenderCast(float t, bool day, float zoom, Vector2 camera, string name = null, bool bloom = true, float aim = CastAim, bool particlesOnly = false)
    {
        Vector2 center = DomainCenter;
        Vector2 screen = center + camera - new Vector2(Width / 2f, Height / 2f);
        main.GetField("screenPosition").SetValue(null, screen);
        using var target = new RenderTarget2D(GraphicsDevice, Width, Height, false, SurfaceFormat.Color, DepthFormat.None);
        GraphicsDevice.SetRenderTarget(target);
        GraphicsDevice.Clear(day ? new Color(132, 170, 204) : new Color(14, 18, 30));
        queue.GetMethod("ClearFrame").Invoke(null, null);
        Set(renderer.GetType(), "PreviewBloomIterations", 0);
        Matrix view = Matrix.CreateTranslation(-Width / 2f, -Height / 2f, 0) * Matrix.CreateScale(zoom, zoom, 1)
            * Matrix.CreateTranslation(Width / 2f, Height / 2f, 0);
        object viewContext = main.GetField("GameViewMatrix").GetValue(null);
        viewContext.GetType().GetField("TransformationMatrix").SetValue(viewContext, view);
        for (int slot = 1; slot < 4; slot++) GraphicsDevice.SamplerStates[slot] = SamplerState.PointClamp;

        var cast = pipeline.GetType("KL.Drawing.CastPreview");
        cast.GetField("ParticlesOnly").SetValue(null, particlesOnly);
        cast.GetMethod("Start").Invoke(null, new object[] { center, center + new Vector2(0, 21), aim, 7 });
        AdvanceLikeGame(cast, t);
        cast.GetMethod("Submit").Invoke(null, new object[] { t });

        // 地形、敌人剪影、玩家与魔杖：在弹幕层之前绘制（游戏中玩家在 Projectiles 之后、OverPlayers 之前）
        float groundY = center.Y + 21f;
        batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.PointClamp, DepthStencilState.None, RasterizerState.CullNone, null, view);
        int groundScreen = (int)(groundY - screen.Y);
        batch.Draw(pixel, new Rectangle(-Width, groundScreen, Width * 3, Height * 2), day ? new Color(74, 62, 46) : new Color(30, 27, 24));
        batch.Draw(pixel, new Rectangle(-Width, groundScreen, Width * 3, 6), day ? new Color(70, 120, 50) : new Color(32, 52, 30));
        foreach (Vector2 e in GroundEnemies) {
            Vector2 p = center + new Vector2(e.X, 21) - screen;
            batch.Draw(pixel, new Rectangle((int)p.X - 16, (int)p.Y - 44, 32, 44), new Color(40, 30, 52));
        }
        Vector2 player = center - screen;
        batch.Draw(pixel, new Rectangle((int)player.X - 10, (int)player.Y - 21, 20, 42), new Color(96, 80, 150));
        if (t < CastTotal()) {
            float arm = (float)cast.GetMethod("WorldArm").Invoke(null, new object[] { t });
            var pose = cast.GetField("Pose").GetValue(null);
            Vector2 shoulder = (Vector2)pose.GetType().GetField("ArmCenter").GetValue(pose) - screen;
            Vector2 dir = new(MathF.Cos(arm), MathF.Sin(arm));
            batch.Draw(pixel, shoulder, null, new Color(60, 48, 100), arm, new Vector2(0, 0.5f), new Vector2(12, 4), SpriteEffects.None, 0);

            var wand = (Texture2D)textures["伊蕾娜/Items/魔杖"];
            batch.Draw(wand, shoulder + dir * 10, null, Color.White, arm + MathHelper.PiOver4, new Vector2(0, wand.Height), 0.5f, SpriteEffects.None, 0);
        }
        batch.End();

        renderer.GetType().GetMethod("RunForeground").Invoke(renderer, new object[] { bloom });
        if ((bool)typeof(SpriteBatch).GetField("beginCalled", All).GetValue(batch))
            throw new Exception("Cast visuals leaked an active SpriteBatch");
        var pixels = new Color[Width * Height]; target.GetData(pixels);
        GraphicsDevice.SetRenderTarget(null);
        if (name != null) SavePixels(name, pixels, Width, Height);
        return pixels;
    }

    void CheckSeedHandover()
    {
        var visuals = pipeline.GetType("伊蕾娜.ElainaModSkills.Skills.Fire.FireBurstCastVisuals");
        object timeline = visuals.GetProperty("Timeline").GetValue(null);
        float Tl(string name) => (float)timeline.GetType().GetProperty(name).GetValue(timeline);
        float Core(float t) => (float)visuals.GetMethod("WandCoreSize").Invoke(null, new object[] { t });
        float Seed(float t) => (float)visuals.GetMethod("SeedSize").Invoke(null, new object[] { t });
        float launch = Tl("Launch"), fall = Tl("Fall"), step = 0.5f;   // 动画速度 0.5 时每游戏帧推进半个设计帧
        float worst = 0, worstAt = 0, previous = Core(launch - step);
        for (float t = launch; t < fall; t += step) {
            float size = Seed(t), change = MathF.Abs(size - previous) / previous;
            if (change > worst) { worst = change; worstAt = t; }
            previous = size;
        }
        Console.WriteLine($"seed handover: core {Core(launch - step):0.0} -> seed {Seed(launch):0.0}, worst step {worst:P1} at t{worstAt}");
        if (worst > 0.1f) throw new Exception($"Seed size jumps {worst:P1} at t{worstAt} (wand core -> seed handover not continuous)");
    }

    // 游戏中的顺序：KL ActionModPlayer 先 Update 第 N 帧再把帧号加一，绘制取到的是 N+1。
    // 所以绘制设计帧 t 时，Update 只处理到 t - 步长（动画速度 0.5 时步长为半个设计帧）。
    const float GameTickStep = 0.5f;
    static void AdvanceLikeGame(Type cast, float t)
    {
        for (float x = 0; x <= t - GameTickStep + 1e-4f; x += GameTickStep)
            cast.GetMethod("Advance").Invoke(null, new object[] { x });
    }

    void CheckWandTrailStartup()
    {
        var cast = pipeline.GetType("KL.Drawing.CastPreview");
        foreach (float aim in new[] { CastAim, 1.4f, MathHelper.Pi - 1.4f }) {
            cast.GetMethod("Start").Invoke(null, new object[] { DomainCenter, DomainCenter + new Vector2(0, 21), aim, 7 });
            for (float t = 0.5f; t <= 13.5f; t += 0.5f) {
                var points = (Vector2[])cast.GetMethod("WandTrail").Invoke(null, new object[] { t });
                if (points == null) throw new Exception($"Startup trail missing at {t}");
                // 尾部最后 20% 必须沿实际弧线展开，不能靠 0.001 像素错位把所有宽度挤在起点。
                float tailLength = 0;
                for (int i = points.Length * 4 / 5; i < points.Length - 1; i++)
                    tailLength += Vector2.Distance(points[i], points[i + 1]);
                if (tailLength < 0.5f)
                    throw new Exception($"Startup taper collapsed at aim={aim}, t={t}: {tailLength:0.000}px");
            }
        }
        Console.WriteLine("PASS: startup tail taper occupies visible arc length for right/left and downward aim");
    }

    void CheckWandTrailTip()
    {
        var cast = pipeline.GetType("KL.Drawing.CastPreview");
        foreach (float aim in new[] { 1.4f, MathHelper.Pi - 1.4f }) {
            cast.GetMethod("Start").Invoke(null, new object[] { DomainCenter, DomainCenter + new Vector2(0, 21), aim, 7 });
            for (int t = 0; t <= 60; t++) {
                cast.GetMethod("Advance").Invoke(null, new object[] { (float)t });
                var points = (Vector2[])cast.GetMethod("WandTrail").Invoke(null, new object[] { (float)t });
                if (points == null) continue;
                var pose = cast.GetField("Pose").GetValue(null);
                var widths = (float[])pose.GetType().GetMethod("Widths").Invoke(pose, new object[] { points });
                if (widths[^1] != 0 || widths.Any(w => !float.IsFinite(w) || w < 0))
                    throw new Exception($"Invalid wand tip widths at {t}");
                // 尾端四分之一像素内不得仍有可见的粗截面（特别覆盖悬停后的下挥）。
                float distance = 0;
                for (int i = points.Length - 2; i >= 0; i--) {
                    distance += Vector2.Distance(points[i], points[i + 1]);
                    if (distance > 0.25f) break;
                    if (widths[i] > 0.1f)
                        throw new Exception($"Blunt stationary wand tail at t={t}, half-width={widths[i]:0.00}px");
                }
            }
        }
        Console.WriteLine("PASS: wand tails close to zero width, including the strike after hovering");
    }

    // ---------------- 炎爆领域外观候选 ----------------
    static readonly Vector2 DomainCenter = new(6000, 3000);
    // 敌人相对领域中心的位置：地面行走（脚底在地面）与飞行。
    static readonly Vector2[] GroundEnemies = [new(-820, 0), new(-560, 0), new(-300, 0), new(280, 0), new(520, 0), new(760, 0), new(900, 0)];
    static readonly Vector2[] FlyingEnemies = [new(-650, -260), new(-200, -380), new(420, -300), new(700, -180)];
    static readonly string[] VariantNames = ["Ring"];

    Color[] RenderDomain(int variant, int domainAge, int remaining, bool day, Vector2 camera, string name = null)
    {
        float groundY = DomainCenter.Y + 21f; // 玩家中心 + 半身高
        Vector2 screen = DomainCenter + camera - new Vector2(Width / 2f, Height / 2f);
        main.GetField("screenPosition").SetValue(null, screen);
        using var target = new RenderTarget2D(GraphicsDevice, Width, Height, false, SurfaceFormat.Color, DepthFormat.None);
        GraphicsDevice.SetRenderTarget(target);
        GraphicsDevice.Clear(day ? new Color(132, 170, 204) : new Color(14, 18, 30));
        queue.GetMethod("ClearFrame").Invoke(null, null);
        Set(renderer.GetType(), "PreviewBloomIterations", 0);
        Set(visual, "OnlyRings", false); Set(visual, "LayerProbe", false); Set(visual, "LegacyImpactGlow", false); Set(visual, "BodyPeakGain", 3f);
        object view = main.GetField("GameViewMatrix").GetValue(null);
        view.GetType().GetField("TransformationMatrix").SetValue(view, Matrix.Identity);
        for (int slot = 1; slot < 4; slot++) GraphicsDevice.SamplerStates[slot] = SamplerState.PointClamp;

        // 领域外观：与炎爆同在 OverPlayers.Before 的 Bloom 组内，排在火柱之前。
        Type layerType = pipeline.GetType("KL.Drawing.LayerDrawRequestSystem+DrawTargetLayer");
        Type timingType = pipeline.GetType("KL.Drawing.LayerDrawRequestSystem+DrawTiming");
        var request = queue.GetMethods().Single(m => m.Name == "RequestBloom" && m.GetParameters().Length == 5
            && m.GetParameters()[3].ParameterType == typeof(Action));
        // 正式范围圈：半径从 FireBurstDomain.Radius 抽取
        var drawDomain = pipeline.GetType("KL.Drawing.DomainRingPreview").GetMethod("Draw");
        Action domainDraw = () => drawDomain.Invoke(null, new object[] { DomainCenter, DomainRadius, (float)domainAge, (float)remaining });
        request.Invoke(null, new object[] { "Elaina:FireBurstDomain:Ring", Enum.Parse(layerType, "OverPlayers"), Enum.Parse(timingType, "Before"), domainDraw, -101 });

        // 炎爆：展开时中心不再降下（由施法爆燃标示位置）；每 4 帧锁定一名敌人，同一敌人每 75 帧重新锁定。
        var centers = new List<Vector2>(); var ages = new List<int>();
        void Strike(Vector2 world, int age) {
            if (age < 1) return;
            if (age <= 60) visual.GetMethod("SubmitAt").Invoke(Activator.CreateInstance(visual), new object[] { age, world });
            if (age <= 75) { centers.Add(world); ages.Add(age); }
        }
        var enemies = GroundEnemies.Select(e => DomainCenter + new Vector2(e.X, 21)).Concat(FlyingEnemies.Select(e => DomainCenter + e + new Vector2(0, 13))).ToArray();
        for (int j = 0; j < enemies.Length; j++) {
            int t = domainAge - 2 - j * 4;
            if (t < 0 || remaining <= 0) continue;
            Strike(enemies[j], t % 75 + 1);
        }
        var sim = pipeline.GetType("KL.Drawing.ParticleSim");
        sim.GetMethod("RunMixed").Invoke(null, new object[] { centers, ages });
        sim.GetMethod("Submit").Invoke(null, null);

        // 地形、敌人剪影、玩家位置：均在弹幕层之前绘制。
        batch.Begin();
        int groundScreen = (int)(groundY - screen.Y);
        batch.Draw(pixel, new Rectangle(0, groundScreen, Width, Height - groundScreen), day ? new Color(74, 62, 46) : new Color(30, 27, 24));
        batch.Draw(pixel, new Rectangle(0, groundScreen, Width, 6), day ? new Color(70, 120, 50) : new Color(32, 52, 30));
        foreach (Vector2 e in GroundEnemies) {
            Vector2 p = DomainCenter + new Vector2(e.X, 21) - screen;
            batch.Draw(pixel, new Rectangle((int)p.X - 16, (int)p.Y - 44, 32, 44), new Color(40, 30, 52));
        }
        foreach (Vector2 e in FlyingEnemies) {
            Vector2 p = DomainCenter + e - screen;
            batch.Draw(pixel, new Rectangle((int)p.X - 20, (int)p.Y - 13, 40, 26), new Color(40, 30, 52));
        }
        Vector2 player = DomainCenter - screen;
        batch.Draw(pixel, new Rectangle((int)player.X - 10, (int)player.Y - 21, 20, 42), new Color(96, 80, 150));
        batch.End();

        renderer.GetType().GetMethod("RunForeground").Invoke(renderer, new object[] { true });
        batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.PointClamp, DepthStencilState.None, RasterizerState.CullNone, null, Matrix.Identity);
        pipeline.GetType("KL.Drawing.ParticleSim").GetMethod("Draw").Invoke(null, null);
        batch.End();
        renderer.GetType().GetMethod("RunPlainLayer").Invoke(renderer, new object[] { 6 });
        var pixels = new Color[Width * Height]; target.GetData(pixels);
        GraphicsDevice.SetRenderTarget(null);
        if (name != null) SavePixels(name, pixels, Width, Height);
        return pixels;
    }

    void SavePixels(string name, Color[] pixels, int w, int h)
    {
        string path = Path.Combine(output, name + ".png");
        Directory.CreateDirectory(Path.GetDirectoryName(path));
        using var image = new Texture2D(GraphicsDevice, w, h); image.SetData(pixels);
        using var stream = File.Create(path); image.SaveAsPng(stream, w, h);
    }

    static Color[] Half(Color[] src, int w, int h)
    {
        int hw = w / 2, hh = h / 2;
        var dst = new Color[hw * hh];
        for (int y = 0; y < hh; y++) for (int x = 0; x < hw; x++) {
            Vector4 sum = Vector4.Zero;
            for (int dy = 0; dy < 2; dy++) for (int dx = 0; dx < 2; dx++) sum += src[(y * 2 + dy) * w + x * 2 + dx].ToVector4();
            dst[y * hw + x] = new Color(sum / 4);
        }
        return dst;
    }

    void SaveDomainPreview()
    {
        int duration = 600;
        Vector2 offset = new(DomainRadius - 500, 0); // 玩家右移，右侧边界在画面 x≈1460
        int hw = Width / 2, hh = Height / 2;
        for (int v = 0; v < VariantNames.Length; v++) {
            string n = VariantNames[v];

            int[] seq = [6, 14, 24, 40, 80, 130];

                for (int y = 0; y < hh; y++) Array.Copy(half, y * hw, sheet, (k / 3 * hh + y) * hw * 3 + k % 3 * hw, hw);
            }
            SavePixels($"domain/{n}-spread-sequence", sheet, hw * 3, hh * 2);
            foreach (bool day in new[] { false, true }) {
                RenderDomain(v, 300, 300, day, Vector2.Zero, $"domain/{n}-steady-{(day ? "day" : "night")}-center");

                // 飞到高处：可见顶部弧（领域上缘在中心上方 1000 像素）
                RenderDomain(v, 300, 300, day, offset + new Vector2(0, 400 - DomainRadius), $"domain/{n}-steady-{(day ? "day" : "night")}-high");
            }
            RenderDomain(v, 600, 25, false, offset, $"domain/{n}-ending-night-offset");
            // 动画帧：0～240 帧每 3 帧一张（半尺寸，夜晚，偏离中心的镜头），由 compose_domain.py 合成 GIF
            for (int age = 0; age <= 240; age += 3)

            Console.WriteLine($"Domain variant {n} rendered");
        }
        Console.WriteLine("PASS: domain candidates rendered with actual FireBurst strikes and KL Bloom");
    }

    // 单独检查范围圈，排除爆燃闪光、冲击波、星光和后续炎爆对 Bloom 差值的干扰。
    Color[] RenderRingBloom(float age, float remaining, bool day, bool bloom, float zoom, string name, bool atSeam = false)
    {
        float open = Math.Clamp(age / 24f, 0, 1);
        float ring = DomainRadius * (1 - MathF.Pow(1 - open, 3));
        Vector2 centerScreen = new Vector2(Width, Height) / 2 - (atSeam ? new Vector2(-ring, 0) : new Vector2(ring / MathF.Sqrt(2)));
        main.GetField("screenPosition").SetValue(null, DomainCenter - centerScreen);
        object view = main.GetField("GameViewMatrix").GetValue(null);
        view.GetType().GetField("TransformationMatrix").SetValue(view,
            Matrix.CreateTranslation(-Width / 2f, -Height / 2f, 0) * Matrix.CreateScale(zoom, zoom, 1)
            * Matrix.CreateTranslation(Width / 2f, Height / 2f, 0));
        using var target = new RenderTarget2D(GraphicsDevice, Width, Height, false, SurfaceFormat.Color, DepthFormat.None);
        GraphicsDevice.SetRenderTarget(target);
        GraphicsDevice.Clear(day ? new Color(132, 170, 204) : new Color(8, 12, 16));
        queue.GetMethod("ClearFrame").Invoke(null, null);
        Set(renderer.GetType(), "PreviewBloomIterations", 0);
        Type layer = pipeline.GetType("KL.Drawing.LayerDrawRequestSystem+DrawTargetLayer");
        Type timing = pipeline.GetType("KL.Drawing.LayerDrawRequestSystem+DrawTiming");
        var request = queue.GetMethods().Single(m => m.Name == "RequestBloom" && m.GetParameters().Length == 5
            && m.GetParameters()[3].ParameterType == typeof(Action));
        Action draw = () => pipeline.GetType("KL.Drawing.DomainRingPreview").GetMethod("Draw").Invoke(null,
            new object[] { DomainCenter, DomainRadius, age, remaining });
        request.Invoke(null, new object[] { "Elaina:FireBurstDomain:Ring", Enum.Parse(layer, "OverPlayers"), Enum.Parse(timing, "Before"), draw, -101 });
        renderer.GetType().GetMethod("RunForeground").Invoke(renderer, new object[] { bloom });
        var pixels = new Color[Width * Height];
        target.GetData(pixels);
        GraphicsDevice.SetRenderTarget(null);
        SavePixels(name, pixels, Width, Height);
        return pixels;
    }

    void SaveRingBloomPreview()
    {
        foreach (float zoom in new[] { 1f, 1.25f })
        foreach (bool day in new[] { false, true })
        foreach (int age in new[] { 2, 4, 6, 10, 12, 20, 24, 40, 300 })
        {
            string name = $"ring-bloom/{(day ? "day" : "night")}-z{zoom:0.##}-t{age:000}";
            var on = RenderRingBloom(age, 600 - age, day, true, zoom, name + "-on");
            var off = RenderRingBloom(age, 600 - age, day, false, zoom, name + "-off");
            if (age == 300 && day && zoom == 1f) {
                // 初次修改公共 Bloom 公式之前，常驻圈同一景别的 FNA 原始采样。
                foreach (var (x, reference) in new[] { (-20, new Color(255,216,209)), (0, new Color(255,255,149)), (30, new Color(212,196,207)) }) {
                    var actual = on[Height / 2 * Width + Width / 2 + x];
                    if (Math.Max(Math.Abs(actual.R-reference.R), Math.Max(Math.Abs(actual.G-reference.G), Math.Abs(actual.B-reference.B))) > 22)
                        throw new InvalidOperationException($"Domain ring palette changed at {x}: {actual}");
                }
            }
            float open = Math.Clamp(age / 24f, 0, 1);
            float radius = DomainRadius * (1 - MathF.Pow(1 - open, 3));
            Vector2 center = new Vector2(Width, Height) / 2 - new Vector2(radius / MathF.Sqrt(2));
            int halo = 0; float gain = 0;
            for (int i = -20; i <= 20; i++)
            {
                float a = MathHelper.PiOver4 + i * 10f / radius;
                Vector2 at = center + new Vector2(MathF.Cos(a), MathF.Sin(a)) * (radius - 24);
                at = new Vector2(Width, Height) / 2 + (at - new Vector2(Width, Height) / 2) * zoom;
                int index = (int)at.Y * Width + (int)at.X;
                int difference = on[index].R - off[index].R;
                gain += difference;
                if (difference >= 3) halo++;
            }
            Console.WriteLine($"Ring Bloom {(day ? "day" : "night")} z{zoom} t{age}: inner halo {halo}/41, mean red gain {gain / 41:0.00}");
            if (halo < 30 || gain / 41 < 5)
                throw new Exception($"Expanding ring has no continuous Bloom halo at age={age}, zoom={zoom}, day={day}");
        }
        var seam = RenderRingBloom(300, 300, true, true, 1f, "ring-bloom/left-seam", atSeam: true);
        int seamError = 0;
        for (int x = Width / 2 - 100; x < Width / 2 + 20; x++) {
            var upper = seam[(Height / 2 - 1) * Width + x];
            var lower = seam[Height / 2 * Width + x];
            seamError += Math.Abs(upper.G - lower.G);
        }
        if (seamError > 120 * 4) throw new InvalidOperationException($"Domain ring noise seam: {seamError}");
        var expiredOn = RenderRingBloom(600, 0, false, true, 1f, "ring-bloom/expired-on");
        var expiredOff = RenderRingBloom(600, 0, false, false, 1f, "ring-bloom/expired-off");
        if (!expiredOn.SequenceEqual(expiredOff) || expiredOn.Any(c => c != new Color(8, 12, 16)))
            throw new Exception("Expired domain ring left a visible body or Bloom halo");
        Console.WriteLine("PASS: expanding ring has a Bloom halo on day/night backgrounds at 1x/1.25x; expired ring is invisible");
    }

    protected override void UnloadContent()
    {
        if (pipeline != null) {
            var textures = (System.Collections.IDictionary)pipeline.GetType("KL.Drawing.ModContent").GetField("Textures").GetValue(null);
            foreach (Texture2D texture in textures.Values) texture.Dispose();
        }
        batch?.Dispose(); pixel?.Dispose(); effects?.Dispose(); elainaEffects?.Dispose();
    }

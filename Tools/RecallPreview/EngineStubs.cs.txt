        public T GetModPlayer<T>() where T : new()
        {
            if (!mods.TryGetValue(typeof(T), out var value))
            {
                mods[typeof(T)] = value = new T();
                if (value is ModPlayer mp) mp.Player = this;
            }
            return (T)value;
        }

    public sealed class Projectile
    {
        public bool active, tileCollide, ignoreWater, netImportant, netUpdate;
        public int width, height, penetrate, timeLeft, owner, identity;
        public float[] ai = new float[3];
        public Vector2 Center;
        public ModProjectile ModProjectile;
        public void Kill()
        {
            if (!active) return;
            active = false;
            if (Main.netMode != 0 && owner == Main.myPlayer) Wire.SendProjectile(this, kill: true);
        }
        public static int NewProjectile(object source, Vector2 pos, Vector2 velocity, int type, int damage, float knockback,
            int owner, float ai0, float ai1, float ai2)
        {
            int index = Array.FindIndex(Main.projectile, p => !p.active);
            if (index < 0) return Main.maxProjectiles;
            // Match the actual tML NewProjectile owner guard, including dedicated-server myPlayer=255.
            var p = new Projectile { active = true, Center = pos, owner = owner, identity = index,
                ai = owner == Main.myPlayer ? [ai0, ai1, ai2] : new float[3] };
            p.ModProjectile = new RecallPortalProjectile { Projectile = p };
            p.ModProjectile.SetDefaults();
            Main.projectile[index] = p;
            Wire.Spawns.Add((Main.netMode, Main.myPlayer, owner));
            if (Main.netMode != 0 && owner == Main.myPlayer) NetMessage.SendData(Terraria.ID.MessageID.SyncProjectile, number: index);
            return index;
        }
    }
    public static class NetMessage
    {
        public static void SendData(int type, int number)
        {
            if (type == Terraria.ID.MessageID.SyncProjectile) Wire.SendProjectile(Main.projectile[number], false);
        }
    }

    }

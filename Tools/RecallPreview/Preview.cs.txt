/// <summary>
/// 回城法阵离线预览。运行时用 Roslyn 编译正式的 RecallPortalDrawing/Script/Pattern/Visual，
/// 连同 KL 的 LayerDrawRequestSystem 与从 DrawSystem/RenderHelper/DrawHelper 提取的完整 RequestBloom 合成流程，

    public Preview(string root, string output, string toolDirectory, bool overlapOnly)
    {
        this.root = root;
        this.output = output;
        this.toolDirectory = toolDirectory;
        this.overlapOnly = overlapOnly;
        _ = new GraphicsDeviceManager(this) { PreferredBackBufferWidth = overlapOnly ? 1280 : Width, PreferredBackBufferHeight = overlapOnly ? 600 : Height,
            GraphicsProfile = GraphicsProfile.HiDef, SynchronizeWithVerticalRetrace = false };
        IsFixedTimeStep = false;
        SDL.SDL_HideWindow(Window.Handle);
    }

    string BloomSource()
    {
        // 与 Tools/IceDepthPreview 相同的提取方式：只替换 Terraria 设备、相机与少量无关诊断。
        // 方法按 ToString 提取，不带前导 #region 等指令。
        string kl = Path.GetFullPath(Path.Combine(root, "../KL"));
        CompilationUnitSyntax Read(string path) => CSharpSyntaxTree.ParseText(File.ReadAllText(Path.Combine(kl, path))).GetCompilationUnitRoot();
        string Methods(CompilationUnitSyntax tree, params string[] names) => string.Join("\n", tree.DescendantNodes()
            .OfType<MethodDeclarationSyntax>().Where(m => names.Contains(m.Identifier.Text)).Select(m => m.ToString()));
        string Fields(CompilationUnitSyntax tree, Func<string, bool> select) => string.Join("\n", tree.DescendantNodes()
            .OfType<FieldDeclarationSyntax>().Where(f => f.Declaration.Variables.Any(v => select(v.Identifier.Text))).Select(f => f.ToFullString()));
        var draw = Read("Drawing/DrawSystem.cs");
        var render = Read("Drawing/RenderHelper.cs");
        var effects = Read("Drawing/DrawHelper.Effects.cs");
        var drawing = Read("Drawing/DrawHelper.DrawingMethod.cs");
        string source = """
            namespace KL.Drawing {
            public static class TimeStopManager { public static bool IsRenderingPlayerTarget = false; }
            public static class RenderPerformanceDiagnostics { public static void RecordAllocation(long start) {} }
            public static class BloomReplay {
                public static bool Enabled = true;
                private static readonly DrawSystem renderer = new();
                public static void Flush(LayerDrawRequestSystem.DrawTargetLayer layer) => renderer.Run(layer, Enabled);
            }
            public partial class DrawSystem {
                public void Run(LayerDrawRequestSystem.DrawTargetLayer layer, bool bloomEnabled) {
                    // 与 KL BloomConfig 默认值一致：强度 1、迭代 4。
                    SetBloomInfo(1f, 4, shouldBloom: bloomEnabled);
                    RenderRequestedBloom(layer, LayerDrawRequestSystem.DrawTiming.After);
                }
            """;
        string[] fields = ["CanUseRender", "TotalBloomStrength", "TotalBloomBackGroundStrength", "bloomStr", "bloomItr", "ShouldBloom2", "bloomDrawOffset",
            "capturingBloom", "BloomAdditiveBlend", "BloomStraightAlphaBlend"];
        source += Fields(draw, n => fields.Contains(n) || n.StartsWith("SpriteBatch"));
        source += Methods(draw, "RenderRequestedBloom", "RenderBloomRequestsWithoutSceneTarget", "RenderBloomRequestsWithoutPostProcess",
            "EndSpriteBatchIfActive", "EnsureSpriteBatchActive", "CaptureSpriteBatchState", "RestoreSpriteBatchState", "DownSampler", "UpSampler", "SetBloomInfo",
            "CaptureBloomRequests", "GetBlendStateForBloom");
        source += draw.DescendantNodes().OfType<RecordDeclarationSyntax>().Single(r => r.Identifier.Text == "SpriteBatchState").ToFullString();
        source += "}\n public class RenderHelper {" + Fields(render, _ => true) + Methods(render, "CreateRender", "EnsureRenderTargets", "SwitchRender", "DisposeTargets", "DetachRenderTargets", "TargetMatches") + "}\n";
        source += effects.DescendantNodes().OfType<EnumDeclarationSyntax>().Single(e => e.Identifier.Text == "ReColorState").ToFullString();
        source += """
            public partial class DrawHelper {
                public static Effect bloom, reColor;
                public static BlendState ReverseBS = BlendState.AlphaBlend, AlphaBlendNormal = BlendState.AlphaBlend;
                public static void DrawInScreen(Texture2D texture, Vector2? position = null, Color? color = null,
                    Vector2? scale = null, float rotation = 0, SpriteEffects spriteEffects = SpriteEffects.None) {
                    Main.spriteBatch.Draw(texture, position ?? Vector2.Zero, null, color ?? Color.White, rotation,
                        new Vector2(texture.Width, texture.Height) / 2, scale ?? Vector2.One, spriteEffects, 0);
                }
            """;
        source += Methods(effects, "GaussianBlur", "GaussianBlurTwice", "DrawUnderflowColorEffect", "DrawOverflowColorEffect", "ReColorEffect");
        source += Methods(drawing, "BeginDraw", "GetBlendState");
        source += drawing.DescendantNodes().OfType<MethodDeclarationSyntax>().Single(m => m.Identifier.Text == "EndBeginDraw"
            && m.ParameterList.Parameters[0].Type.ToString() == "int").ToFullString();
        return source + "}}";
    }

    protected override void LoadContent()
    {
        SDL.SDL_HideWindow(Window.Handle);
        Directory.CreateDirectory(output);
        replay = Compile();
        scene = replay.GetType("RecallPreviewScene");
        batch = new SpriteBatch(GraphicsDevice);
        pixel = new Texture2D(GraphicsDevice, 1, 1);
        pixel.SetData(new[] { Color.White });
        frame = new RenderTarget2D(GraphicsDevice, Width, Height);
        var effects = new ContentManager(Services, Path.GetFullPath(Path.Combine(root, "../KL/Effects/Content")));
        using (var stream = File.OpenRead(Path.Combine(root, "ElainaModSkills/Skills/Recall/Assets/PortalLine.png")))
            portalLine = Texture2D.FromStream(GraphicsDevice, stream);
        scene.GetMethod("Setup").Invoke(null, [GraphicsDevice, batch,
            effects.Load<Effect>("VertexDraw"), effects.Load<Effect>("Bloom"), effects.Load<Effect>("ReColor"), portalLine]);
    }

    /// <summary>在场景目标上画背景与人物参考，再按游戏流程提交并合成法阵。</summary>
    Color[] Render(float age, bool destination, bool hovered, bool bloom = true, float zoom = 1, Vector2? center = null, float burstStart = 90)
    {
        GraphicsDevice.SetRenderTarget(frame);
        GraphicsDevice.Clear(Sky);
        batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.PointClamp, DepthStencilState.None, RasterizerState.CullNone);
        int ground = (int)Center.Y + 1;
        batch.Draw(pixel, new Rectangle(0, ground, Width, Height - ground), Ground);
        // 灰色人物只作定位参考；游戏中玩家绘制在 Projectiles 层之后，会盖住法阵。
        batch.Draw(pixel, new Rectangle(194, ground - 42, 12, 12), new Color(197, 201, 210));
        batch.Draw(pixel, new Rectangle(192, ground - 29, 16, 28), new Color(98, 109, 130));
        batch.End();
        scene.GetMethod("Draw").Invoke(null, [age, destination, hovered, bloom, center ?? Center, zoom, burstStart]);
        GraphicsDevice.SetRenderTarget(null);
        var data = new Color[Width * Height];
        frame.GetData(data);
        return data;
    }

    void Save(Color[] data, int width, int height, string name)
    {
        using var texture = new Texture2D(GraphicsDevice, width, height);
        texture.SetData(data);
        using var stream = File.Create(Path.Combine(output, name));
        texture.SaveAsPng(stream, width, height);
    }

    static int Brightness(Color c) => c.R * 30 + c.G * 59 + c.B * 11;

    protected override void Draw(GameTime time)
    {
        if (done) return;
        done = true;
        if (overlapOnly) { RenderOverlap(); Exit(); return; }
        Verify();
        RenderSheet();
        RenderAnimation();
        Exit();
    }

    void RenderOverlap()
    {
        using var target = new RenderTarget2D(GraphicsDevice, 1280, 600);
        using var fonts = new ContentManager(Services, Path.GetFullPath(Path.Combine(root, "../KL")));
        var font = fonts.Load<DynamicSpriteFont>("Fonts/NotoSerifSC");
        foreach (var item in new[] { (1, 1200f, 1f, "zh-Hans"), (1, 1200f, 1.25f, "en-US"),
            (2, 1200f, 1f, "zh-Hans"), (3, 1200f, 1f, "zh-Hans"),
            (3, 96f, 1f, "zh-Hans"), (2, 1200f, 1.25f, "en-US"), (2, 45f, 1f, "zh-Hans") })
        {
            GraphicsDevice.SetRenderTarget(target);
            GraphicsDevice.Clear(Sky);
            batch.Begin();
            batch.Draw(pixel, new Rectangle(0, 421, 1280, 179), Ground);
            batch.Draw(pixel, new Rectangle(630, 378, 20, 42), new Color(133, 148, 168));
            batch.End();
            string tooltip = (string)scene.GetMethod("DrawOverlap").Invoke(null, [item.Item1, item.Item2, item.Item3, root, item.Item4]);
            // 原版 MouseText 的排版由游戏负责；这里用正式 Tooltip 文本及项目 FNA 字体检查长行和缩放。
            batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.LinearClamp, DepthStencilState.None, RasterizerState.CullNone);
            batch.DrawString(font, tooltip, new Vector2(40, 38), Color.White, 0, Vector2.Zero, .55f * item.Item3, SpriteEffects.None, 0);
            batch.End();
            GraphicsDevice.SetRenderTarget(null);
            string path = Path.Combine(output, $"overlap-{item.Item1}-{item.Item2}-{item.Item3}-{item.Item4}.png");
            using var stream = File.Create(path);
            target.SaveAsPng(stream, 1280, 600);
            Console.WriteLine(path);
        }
    }

    void Verify()
    {
        var reference = Render(0, false, false, false);
        // 1. 关闭 bloom 时不应有超出线条的光晕；开启后有光晕且只增亮。
        var plain = Render(45, false, false, false);
        var glowing = Render(45, false, false, true);
        int halo = 0, darker = 0;
        for (int i = 0; i < plain.Length; i++)
        {
            if (plain[i] == reference[i] && glowing[i] != reference[i]) halo++;
            if (Brightness(glowing[i]) + 300 < Brightness(plain[i])) darker++;
        }
        if (halo < 200) throw new Exception($"Bloom produced no halo beyond the lines ({halo} pixels).");
        if (darker > 0) throw new Exception($"Bloom darkened {darker} pixels.");
        Console.WriteLine($"PASS bloom: {halo} halo pixels beyond the unprocessed lines, none darkened.");

        // 2. 光晕局部化：描绘中的笔尖要有强光；写完静止时只有少量像素被明显提亮，不能糊成一片雾。
        (int Strong, int Lit) Gain(float age, bool destination)
        {
            var off = Render(age, destination, false, false);
            var on = Render(age, destination, false, true);
            if (Environment.GetEnvironmentVariable("RECALL_GAIN_MAP") != null)
                Save(off.Zip(on).Select(p => { int g = Math.Clamp((Brightness(p.Second) - Brightness(p.First)) / 25, 0, 255); return new Color(g, g, g); }).ToArray(),
                    Width, Height, $"gain-{age}-{destination}.png");
            int strong = 0, lit = 0;
            for (int i = 0; i < off.Length; i++)
            {
                int gain = Brightness(on[i]) - Brightness(off[i]);
                if (gain >= 60 * 100) strong++;
                if (gain >= 12 * 100) lit++;
            }
            return (strong, lit);
        }
        var drawing = Gain(45, false);
        foreach (var (age, destination, name) in new[] { (1200f, false, "start"), (1200f, true, "exit") })
        {
            var idle = Gain(age, destination);
            float share = idle.Lit / (float)(Width * Height);
            Console.WriteLine($"Idle {name}: {idle.Lit} pixels brightened by >=12/255 ({share:P1}), {idle.Strong} by >=60/255.");
            if (share > .12f) throw new Exception($"Idle {name} bloom spreads over {share:P1} of the frame; overall glow should stay subtle.");
        }
        Console.WriteLine($"Drawing: {drawing.Lit} pixels brightened by >=12/255, {drawing.Strong} by >=60/255.");
        if (drawing.Strong < 50) throw new Exception($"Pen tips lack strong local bloom ({drawing.Strong} pixels >= 60/255).");
        Console.WriteLine("PASS localized bloom: strong glow at pen tips, subtle when idle (set RECALL_GAIN_MAP=1 to save gain-*.png maps).");

        // 3. 范围：可见改动都在法阵附近，防止顶点或矩阵错误把线条拉出屏幕。
        foreach (float age in new[] { 15f, 45, 75, 90, 130, 210, 1800, 1880 })
        foreach (bool bloom in new[] { false, true })
        {
            bool destination = age == 45 || age == 210;
            var data = Render(age, destination, age == 210, bloom);
            int outside = 0, visible = 0;
            float margin = bloom ? 48 : 3;
            // 激活后 50 帧内是向上的完成爆发，允许向上延伸到 1.1 倍半径。
            float above = age >= 90 && age < 140 ? 160 * 1.1f : 50;
            for (int y = 0; y < Height; y++)
            for (int x = 0; x < Width; x++)
            {
                int i = y * Width + x;
                if (Math.Abs(Brightness(data[i]) - Brightness(reference[i])) < 800) continue;
                visible++;
                if (Math.Abs(x - Center.X) > 162 + margin || y - Center.Y > 50 + margin || Center.Y - y > above + margin) outside++;
            }
            if (outside > 0 || visible == 0)
            {
                Save(data, Width, Height, "recall-bounds-failure.png");
                throw new Exception($"Age {age} bloom={bloom}: {outside} pixels outside, {visible} visible.");
            }
        }
        Console.WriteLine("PASS bounds: 8 ages with bloom on/off stay around the circle.");

        // 4. 完成爆发：激活后法阵上方出现明显光束，结束后上方恢复干净。
        int Above(float age, float burstStart = 90)
        {
            var data = Render(age, false, false, burstStart: burstStart);
            int lit = 0;
            for (int y = 0; y < Center.Y - 70; y++)
            for (int x = 0; x < Width; x++)
                if (Brightness(data[y * Width + x]) - Brightness(reference[y * Width + x]) >= 40 * 100) lit++;
            return lit;
        }
        int peak = Above(96), after = Above(150), before = Above(80);
        Console.WriteLine($"Burst: {before} pixels lit above the circle at 80, {peak} at 96, {after} at 150.");
        if (peak < 1500 || after > 0 || before > 0) throw new Exception("Completion burst should only light the area above the circle right after activation.");
        int travel = Above(1206, 1200), still = Above(1206);
        Console.WriteLine($"Travel burst: {travel} pixels lit above the circle 6 frames after a travel, {still} without.");
        if (travel < 1500 || still > 0) throw new Exception("Travel should replay the upward burst on the circle.");
        Console.WriteLine("PASS burst: upward rays right after activation and again after each travel.");

        // 5. 游戏缩放：顶点矩阵与 GameViewMatrix 叠加后，法阵按缩放倍数放大。
        var screenCenter = new Vector2(Width / 2f, Height / 2f);
        var empty = Render(0, false, false, false, 1.2f, screenCenter);
        var zoomed = Render(1200, false, false, false, 1.2f, screenCenter);
        int left = Width, right = 0;
        for (int i = 0; i < zoomed.Length; i++)
            if (Math.Abs(Brightness(zoomed[i]) - Brightness(empty[i])) >= 800) { left = Math.Min(left, i % Width); right = Math.Max(right, i % Width); }
        float halfWidth = (right - left + 1) / 2f;
        if (Math.Abs(halfWidth - 160 * 1.2f) > 6) throw new Exception($"Zoom 1.2: half width {halfWidth}, expected {160 * 1.2f}.");
        Console.WriteLine($"PASS zoom: half width {halfWidth} at 1.2x (expected {160 * 1.2f}).");
    }

    void RenderSheet()
    {
        // 从左到右、从上到下：起点 15/45/75 帧描绘过程、96 帧完成爆发；
        // 出口 45 帧（与起点同步描绘、悬停）/96 帧完成爆发/210 帧/到期前 1880 帧。
        (float Age, bool Destination, bool Hovered)[] cells =
            [(15, false, false), (45, false, false), (75, false, false), (96, false, false),
             (45, true, true), (96, true, false), (210, true, false), (1880, true, false)];
        var sheet = new Color[Width * 4 * Height * 2];
        for (int n = 0; n < cells.Length; n++)
        {
            var data = Render(cells[n].Age, cells[n].Destination, cells[n].Hovered);
            int ox = n % 4 * Width, oy = n / 4 * Height;
            for (int y = 0; y < Height; y++) Array.Copy(data, y * Width, sheet, (oy + y) * Width * 4 + ox, Width);
        }
        Save(sheet, Width * 4, Height * 2, "recall-frames.png");
        Console.WriteLine(Path.Combine(output, "recall-frames.png"));
    }

    /// <summary>
    /// 按游戏帧（60 FPS）逐帧回放，生成带播放控制的单文件 HTML。
    /// 各场景按年龄顺序连续渲染，以后加入有状态的粒子时也能在这里逐帧推进。
    /// </summary>
    void RenderAnimation()
    {
        // BurstStart：最近一次向上爆发的起始年龄，画完时为 90，往返传送场景在第 1200 帧有人传送。
        (string Label, int From, int To, bool Destination, float BurstStart)[] scenes =
        [
            ("起点：施法蓄力", 0, 150, false, 90),
            ("出生地出口（与起点同步）", 0, 150, true, 90),
            ("往返传送光效", 1190, 1260, true, 1200),
            ("到期淡出", 1830, 1895, false, 90),
        ];
        var json = new StringBuilder("[");
        int frames = 0;
        foreach (var item in scenes)
        {
            json.Append($"{{\"label\":\"{item.Label}\",\"from\":{item.From},\"destination\":{(item.Destination ? "true" : "false")},\"frames\":[");
            for (int age = item.From; age <= item.To; age++)
            {
                Render(age, item.Destination, false, burstStart: item.BurstStart);
                using var stream = new MemoryStream();
                frame.SaveAsPng(stream, Width, Height);
                if (age > item.From) json.Append(',');
                json.Append('"').Append(Convert.ToBase64String(stream.ToArray())).Append('"');
                frames++;
            }
            json.Append("]},");
        }
        json.Length--;
        json.Append(']');
        string html = File.ReadAllText(Path.Combine(toolDirectory, "player.html")).Replace("/*SCENES*/[]", json.ToString());
        string path = Path.Combine(output, "recall-player.html");
        File.WriteAllText(path, html, new UTF8Encoding(false));
        Console.WriteLine($"{frames} animation frames: {path}");
    }

    protected override void UnloadContent() { frame?.Dispose(); pixel?.Dispose(); portalLine?.Dispose(); batch?.Dispose(); }

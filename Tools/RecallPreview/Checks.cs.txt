static class Checks
{
    static int passed;
    static readonly RecallPortalSystem System = new();
    static Player P(int i) => TMain.player[i];
    static RecallPortalProjectile Portal(int side = 0) => TMain.ActiveProjectiles.Select(p => (RecallPortalProjectile)p.ModProjectile).First(p => p.Side == side);
    static ElainaAttributeModPlayer Mana => P(0).GetModPlayer<ElainaAttributeModPlayer>();
    static void Check(bool result, string label) { if (!result) throw new Exception(label); passed++; }
    static RecallSkill Setup(int mode = 0)
    {
        System.OnWorldUnload(); TMain.netMode = mode; TMain.myPlayer = mode == NetmodeID.Server ? 255 : 0;
        Wire.Queue.Clear(); Wire.Spawns.Clear(); TMain.Notices.Clear();
        for (int i = 0; i < TMain.maxPlayers; i++) TMain.player[i] = new Player { whoAmI = i, Bottom = new Vector2(800, 900) };
        for (int i = 0; i < TMain.maxProjectiles; i++) TMain.projectile[i] = new Projectile();
        P(0).active = true;
        Collision.Solid = Collision.Lava = _ => false;
        ModPacket.Sent.Clear(); RemoteClient.SectionChecks = 0; ElainaHealingWheel.Reset();
        TMain.mouseRight = false; TMain.mouseRightRelease = true;
        var skill = new RecallSkill { Player = P(0) }; skill.Initialize();
        var book = P(0).GetModPlayer<ElainaSkillModPlayer>(); book.Owner = skill; book.ActiveSkill.Add(skill.Skill);
        return skill;
    }
    static void Tick(int count)
    {
        for (int n = 0; n < count; n++)
        {
            foreach (Player p in TMain.player) if (p.active) p.GetModPlayer<RecallPortalPlayer>().PreUpdate();
            System.PostUpdatePlayers();
            foreach (Projectile p in TMain.ActiveProjectiles.ToArray()) { p.ModProjectile.AI(); if (--p.timeLeft <= 0) p.Kill(); }
        }
    }
    static void Packet(RecallPortalSystem.Operation op, int sender, int claimed, params int[] args)
    {
        using var stream = new MemoryStream(); using var writer = new BinaryWriter(stream);
        writer.Write((byte)op); foreach (int arg in args) writer.Write(arg);
        stream.Position = 0; RecallPortalSystem.HandlePacket(new BinaryReader(stream), sender, claimed);
    }
    static void Main()
    {
        Selection(); TimingAndTravel(); Lifecycle(); Input(); AtSpawn(); Overlap(); TransitionChecks(); NetworkChecks.Run(Check);
        Console.WriteLine($"PASS: {passed} recall behavior assertions (production source, engine/network doubles).");
    }
    static void AtSpawn()
    {
        foreach (float offset in new[] { 0f, 80f })
        {
            Setup(); P(0).Bottom = RecallPortalSystem.HomeFoot(P(0)) + new Vector2(offset, 0);
            RecallPortalSystem.Cast(P(0), 0); Tick(90); RecallPortalPresentation.Refresh();
            var origin = Portal(); var exit = Portal(1);
            Check(origin.DisplayCenter != exit.DisplayCenter && P(0).Bottom == exit.DisplayCenter,
                "Casting at/near spawn arrives at the visible displaced exit");
            foreach (var departure in new[] { exit, origin })
            {
                Tick(45); P(0).mouseInterface = false;
                TMain.MouseWorld = departure.DisplayCenter; TMain.mouseRight = TMain.mouseRightRelease = true;
                Vector2 before = P(0).Bottom;
                P(0).GetModPlayer<RecallPortalPlayer>().SetControls();
                Vector2 expected = departure == exit ? origin.DisplayCenter : exit.DisplayCenter;
                Check(P(0).Bottom == expected && P(0).Bottom != before,
                    "Right-click at spawn moves between the two visible circles, never in place");
            }
        }
    }
    static void Overlap()
    {
        Setup(); RecallPortalSystem.Cast(P(0), 0); Tick(135);
        RecallPortalPresentation.Refresh();
        Check(RecallPortalPresentation.Tooltip(P(0), Portal(1)) == "RecallInteract",
            "A single pair's two endpoints do not enable destination details");
        P(1).active = true;
        Check(RecallPortalPresentation.Tooltip(P(0), Portal(1)) == "RecallInteract",
            "Another player without a recall pair does not enable destination details");
        Setup(); P(1).active = true; P(1).name = "Second caster";
        RecallPortalSystem.Cast(P(0), 0);
        Vector2 secondOrigin = new(2400, 1600);
        P(1).Bottom = secondOrigin; TMain.myPlayer = 1; RecallPortalSystem.Cast(P(1), 1); TMain.myPlayer = 0;
        Tick(135);
        var first = Portal(1);
        var second = TMain.ActiveProjectiles.Select(p => (RecallPortalProjectile)p.ModProjectile).Single(p => p.Projectile.owner == 1 && p.Side == 1);
        RecallPortalPresentation.Refresh();
        Check(first.Projectile.Center == second.Projectile.Center && first.DisplayCenter != second.DisplayCenter,
            "Shared spawn anchors produce distinct visible destinations");
        Check(P(1).Bottom == second.DisplayCenter, "Second caster initially arrives at the displaced home circle");
        Check(!first.Hovered(second.DisplayCenter) && !second.Hovered(first.DisplayCenter), "Separated hit regions do not overlap");
        Color light = Color.White;
        TMain.screenPosition = second.DisplayCenter - new Vector2(640, 360);
        RecallPortalVisual.Requests = 0; second.PreDraw(ref light);
        Check(RecallPortalVisual.Requests == 1 && RecallPortalVisual.LastCenter == new Vector2(640, 360),
            "PreDraw submits the displayed position, matching mouse hit testing");
        TMain.screenPosition = new Vector2(20000, 1000); second.PreDraw(ref light);
        Check(RecallPortalVisual.Requests == 1, "Offscreen display centers submit no bloom geometry");
        TMain.screenPosition = Vector2.Zero;
        TMain.MouseWorld = second.DisplayCenter;
        Check(RecallPortalPlayer.FindHovered(P(0)) == null, "The old shared anchor does not retain an invisible interaction region");
        P(0).Bottom = second.DisplayCenter;
        Check(RecallPortalPlayer.FindHovered(P(0)) == second, "Shared home selects the second caster by displayed center");
        P(0).GetModPlayer<RecallPortalPlayer>().SetControls();
        Check(TMain.HoverText.Contains("Second caster") && TMain.HoverText.Contains("RecallDepthSurface|150|100"),
            "Tooltip names the caster and the other endpoint's original tile coordinates");
        // 站在错位后的图形旁，已超出原始阵心半径，也应能通过服务器校验。
        P(0).Bottom = second.DisplayCenter;
        TMain.mouseRight = TMain.mouseRightRelease = true;
        P(0).GetModPlayer<RecallPortalPlayer>().SetControls();
        Check(P(0).Bottom == secondOrigin, "Clicking displaced second exit travels to second caster's original origin");
        var secondStart = (RecallPortalProjectile)TMain.ActiveProjectiles.Single(p => p.owner == 1 &&
            ((RecallPortalProjectile)p.ModProjectile).Side == 0).ModProjectile;
        Check(RecallPortalPresentation.Tooltip(P(0), secondStart).Contains("RecallDepthSurface|122|200"),
            "Destination tooltip reports displaced exit coordinates instead of its shared anchor");
        Tick(45); RecallPortalSystem.RequestTravel(P(0), secondStart);
        Check(P(0).Bottom == second.DisplayCenter, "Forward travel lands at the same visible displaced home circle");
        P(0).mouseInterface = false; P(0).Bottom = new Vector2(30000, 1000); TMain.mouseRight = false;
        TMain.MouseWorld = second.DisplayCenter;
        Check(RecallPortalPlayer.FindHovered(P(0)) == null && RecallPortalPlayer.FindHovered(P(0), true) == second,
            "Distant players can inspect destination without reserving travel input");
        P(0).GetModPlayer<RecallPortalPlayer>().SetControls();
        Check(TMain.HoverText.Contains("RecallMoveCloser"), "Distant hover explains travel range");
        P(0).Bottom = second.DisplayCenter; P(0).team = 2;
        Check(!RecallPortalPresentation.CanTravel(P(0), second) && RecallPortalPresentation.Tooltip(P(0), second).Contains("RecallAlliesOnly"),
            "Offset cannot bypass ally restriction");
        P(0).team = 0;
        second.Projectile.ai[1] = 20;
        Check(!RecallPortalPresentation.CanTravel(P(0), second) && RecallPortalPresentation.Tooltip(P(0), second).Contains("RecallCharging"),
            "Charging circles expose their destination without allowing travel");
        var counterpart = TMain.ActiveProjectiles.Single(p => p.owner == 1 && ((RecallPortalProjectile)p.ModProjectile).Side == 0);
        counterpart.Kill();
        Check(!RecallPortalPresentation.CanTravel(P(0), second) && RecallPortalPresentation.Tooltip(P(0), second).Contains("RecallLinkPending"),
            "Missing counterpart never invents a destination or consumes travel input");
        first.Projectile.Kill(); RecallPortalPresentation.Refresh();
        Check(second.DisplayCenter == second.Projectile.Center, "Remaining circle returns to its original center");
        TMain.ActiveProjectiles.Single(p => p.owner == 0).Kill();
        Check(RecallPortalPresentation.Tooltip(P(0), second) == "RecallLinkPending",
            "Details disappear immediately when only one pair remains, including a missing counterpart");

        Check(RecallPortalPresentation.DepthKey(50) == "RecallDepthSky" &&
            RecallPortalPresentation.DepthKey(250) == "RecallDepthSurface" &&
            RecallPortalPresentation.DepthKey(400) == "RecallDepthUnderground" &&
            RecallPortalPresentation.DepthKey(600) == "RecallDepthCaverns" &&
            RecallPortalPresentation.DepthKey(601) == "RecallDepthUnderworld", "Depth boundary labels use world metadata only");
        // 相邻但并非完全相同的阵心、世界边界、不同到包顺序。
        foreach (float x in new[] { 170f, 16000f, 31830f })
        {
            var entries = Enumerable.Range(0, 10).Select(i => new RecallPortalLayout.Entry(i, 1, 1, new Vector2(x + i % 2 * 8, 3000))).ToList();
            RecallPortalLayout.Arrange(entries, 32000);
            var expected = entries.ToDictionary(e => e.Owner, e => e.Center);
            entries.Reverse(); RecallPortalLayout.Arrange(entries, 32000);
            Check(entries.All(e => e.Center == expected[e.Owner]), "Layout is independent of projectile enumeration order");
            Check(entries.All(e => e.Center.X >= 160 && e.Center.X <= 31840), "Displaced circles stay within horizontal world bounds");
            for (int i = 0; i < entries.Count; i++)
            for (int j = i + 1; j < entries.Count; j++)
                Check(Math.Abs(entries[i].Center.X - entries[j].Center.X) >= RecallPortalLayout.HorizontalSpacing ||
                    Math.Abs(entries[i].Center.Y - entries[j].Center.Y) >= RecallPortalLayout.VerticalSpacing, "Nearby circles remain visibly separated");
        }
    }
    static void TransitionChecks()
    {
        var fade = new RecallTransition();
        fade.Begin(200);
        Check(fade.Opacity == 0, "Nearby travel does not black out");
        fade.Begin(1120);
        Check(Math.Abs(fade.Opacity - .5f) < .001f, "Medium travel uses a softer veil");
        fade.Clear(); fade.Begin(4000);
        Check(fade.Opacity == 1, "Long travel hides the scene cut on arrival");
        fade.Advance(.03f);
        Check(fade.Opacity == 1, "Black veil holds briefly across the first destination frame");
        foreach (int fps in new[] { 30, 60, 144 })
        {
            fade.Clear(); fade.Begin(4000);
            float previous = fade.Opacity;
            for (int frame = 0; frame < fps / 2; frame++)
            {
                fade.Advance(1f / fps);
                Check(fade.Opacity >= 0 && fade.Opacity <= previous, "Fade restores brightness monotonically at " + fps + " FPS");
                previous = fade.Opacity;
            }
            Check(fade.Opacity == 0, "Transition finishes within half a second at " + fps + " FPS");
        }
        fade.Begin(4000); fade.Advance(.2f); float alpha = fade.Opacity; fade.Begin(1120);
        Check(fade.Opacity >= alpha, "Rapid travel cannot flash brighter before a new fade");
        fade.Advance(10);
        Check(fade.Opacity == 0, "Long frame cannot leave a stuck black overlay");
        fade.Begin(float.NaN); fade.Begin(float.PositiveInfinity);
        Check(fade.Opacity == 0, "Invalid distance cannot corrupt opacity");
        Setup(); P(1).active = true; RecallPortalSystem.Cast(P(0), 0); Tick(90);
        Check(P(0).GetModPlayer<RecallPortalPlayer>().Transition.Opacity == 1, "Initial group arrival starts the local transition");
        Check(P(1).Teleports == 1 && P(1).GetModPlayer<RecallPortalPlayer>().Transition.Opacity == 0, "Other players teleport without darkening this client's screen");
        P(0).GetModPlayer<RecallPortalPlayer>().Transition.Advance(1); Tick(45);
        RecallPortalSystem.RequestTravel(P(0), Portal(1));
        Check(P(0).GetModPlayer<RecallPortalPlayer>().Transition.Opacity == 1, "Right-click return also starts the transition");
        P(0).GetModPlayer<RecallPortalPlayer>().OnEnterWorld();
        Check(P(0).GetModPlayer<RecallPortalPlayer>().Transition.Opacity == 0, "Entering a world clears the previous transition");
        Setup(); RecallPortalSystem.Cast(P(0), 0); Collision.Solid = _ => true; Tick(90);
        Check(P(0).GetModPlayer<RecallPortalPlayer>().Transition.Opacity == 0, "Failed landing never darkens the screen");
        Setup(NetmodeID.Server);
        Packet(RecallPortalSystem.Operation.Teleport, 0, 0);
        Check(P(0).GetModPlayer<RecallPortalPlayer>().Transition.Opacity == 0, "Server rejects client teleport and produces no overlay");
    }

    static void Selection()
    {
        var s = Setup();
        Check(s.BeginWandUse() && ElainaHealingWheel.Opens == 0, "No allies: wand enters direct self-cast without a wheel");
        P(0).GetModPlayer<ElainaSkillModPlayer>().UseSkill(0);
        Check(ElainaHealingWheel.Opens == 0 && TMain.ActiveProjectiles.Count() == 2 && Mana.Spent == s.MagicPointCost && s.Stack == 0,
            "Solo direct cast pays exactly once and opens one pair");
        s = Setup(); s.PreUseSkill();
        Check(ElainaHealingWheel.Opens == 0 && TMain.ActiveProjectiles.Count() == 2 && Mana.Spent == s.MagicPointCost,
            "Solo non-wand entry also casts directly on self");
        foreach (string excluded in new[] { "dead", "offline", "team", "target-pvp", "caster-pvp" })
        {
            s = Setup(); P(1).active = true;
            switch (excluded)
            {
                case "dead": P(1).dead = true; break;
                case "offline": P(1).active = false; break;
                case "team": P(1).team = 2; break;
                case "target-pvp": P(1).hostile = true; break;
                case "caster-pvp": P(0).hostile = true; break;
            }
            Check(s.BeginWandUse() && ElainaHealingWheel.Opens == 0, "No eligible ally opens no wheel: " + excluded);
        }
        s = Setup(); P(1).active = true;
        Check(!s.BeginWandUse() && ElainaHealingWheel.OwnsPointer && ElainaHealingWheel.Recall, "Nearby ally opens recall wheel");
        for (int i = 0; i < 100; i++) s.BeginWandUse();
        Check(ElainaHealingWheel.Opens == 1 && Mana.Spent == 0 && !TMain.ActiveProjectiles.Any(), "Holding neither reopens nor spends");
        ElainaHealingWheel.Cancel();
        Check(Mana.Spent == 0 && s.Stack == 1, "Cancel consumes nothing");
        s.BeginWandUse(); ElainaHealingWheel.Confirm(0);
        Check(Mana.Spent == s.MagicPointCost && s.Stack == 0 && TMain.ActiveProjectiles.Count() == 2, "Confirmation consumes once and creates both charging circles");
        ElainaHealingWheel.Confirmed(0);
        Check(TMain.ActiveProjectiles.Count() == 2 && Mana.Spent == s.MagicPointCost, "Stale callback cannot duplicate a cast");
        s = Setup(); P(1).active = true; P(1).Center = P(0).Center + new Vector2(3001, 0); s.BeginWandUse();
        Check(ElainaHealingWheel.ValidTarget(0) && !ElainaHealingWheel.ValidTarget(1), "Far ally yields a self-only wheel");
        P(1).Center = P(0).Center + new Vector2(3000, 0);
        Check(ElainaHealingWheel.ValidTarget(1), "Exact selection boundary is allowed");
        s = Setup(); P(1).active = true; P(1).Center = P(0).Center + new Vector2(3001, 0); s.PreUseSkill();
        Check(ElainaHealingWheel.OwnsPointer && ElainaHealingWheel.ValidTarget(0) && !ElainaHealingWheel.ValidTarget(1) && Mana.Spent == 0,
            "Far ally also requires self confirmation through non-wand entry");
        foreach (string invalid in new[] { "far", "dead", "offline", "team", "pvp", "mana", "unequipped", "disabled", "charge" })
        {
            s = Setup(); P(1).active = true; s.BeginWandUse();
            switch (invalid)
            {
                case "far": P(1).Center += new Vector2(3100, 0); break;
                case "dead": P(1).dead = true; break;
                case "offline": P(1).active = false; break;
                case "team": P(1).team = 2; break;
                case "pvp": P(1).hostile = true; break;
                case "mana": Mana.Available = 0; break;
                case "unequipped": P(0).GetModPlayer<ElainaSkillModPlayer>().ActiveSkill.Clear(); break;
                case "disabled": Mana.UniqueMagicEnabled = false; break;
                case "charge": s.Stack = 0; break;
            }
            ElainaHealingWheel.Confirm(1);
            Check(!TMain.ActiveProjectiles.Any() && Mana.Spent == 0, "Revalidate on confirmation: " + invalid);
        }
    }
    static void TimingAndTravel()
    {
        var s = Setup(); P(1).active = P(2).active = P(3).active = true;
        P(1).SpawnX = 400; P(1).SpawnY = 300; P(2).team = 2;
        P(3).Bottom = P(1).Bottom + new Vector2(161, 0);
        Vector2 origin = P(1).Bottom;
        s.BeginWandUse(); ElainaHealingWheel.Confirm(1);
        P(1).Bottom += new Vector2(25, 0);
        Check(Portal().Projectile.Center == origin, "Circle stays where target stood on cast");
        Check(TMain.ActiveProjectiles.Count() == 2 && Portal(1).Projectile.ai[1] == 0 && Portal().Projectile.ai[1] == 0,
            "Destination opens together with origin and starts charging in sync");
        Tick(89);
        Check(P(0).Teleports == 0 && Portal().Projectile.ai[1] == 89 && Portal(1).Projectile.ai[1] == 89 && !Portal(1).Ready,
            "No teleport before delay; both endpoints charge frame by frame together");
        Tick(1);
        Check(P(0).Teleports == 1 && P(1).Teleports == 1 && P(2).Teleports == 0 && P(3).Teleports == 0, "Initial pulse affects only allies inside the area");
        Check(P(0).Bottom == new Vector2(6408, 4800) && P(1).Bottom == P(0).Bottom, "Group uses selected ally's bed, not each individual's spawn");
        Check(TMain.ActiveProjectiles.Count() == 2 && Portal().Ready && Portal(1).Ready &&
            Portal().Projectile.ai[1] == Portal(1).Projectile.ai[1], "Both endpoints finish drawing on the same frame");
        RecallPortalSystem.RequestTravel(P(0), Portal(1));
        Check(P(0).Teleports == 1, "Immediate repeat blocked by travel cooldown");
        Tick(45); P(0).velocity = new Vector2(12, 80);
        RecallPortalSystem.RequestTravel(P(0), Portal(1));
        Check(P(0).Bottom == origin && P(0).Teleports == 2 && P(0).velocity == Vector2.Zero, "Free return resets motion and lands at original circle");
        Check(Portal().BurstStart == Portal().Projectile.ai[1] && Portal(1).BurstStart == Portal(1).Projectile.ai[1] && Portal().BurstStart > RecallPortalSystem.DelayTicks,
            "Travel replays the upward burst on both circles");
        Check(Mana.Spent == s.MagicPointCost && s.Stack == 0, "Travel does not spend mana or charges");
        Tick(45); RecallPortalSystem.RequestTravel(P(0), Portal());
        Check(P(0).Bottom == P(1).Bottom && P(0).Teleports == 3, "Origin can send player back to destination");
        Tick(1709); Check(TMain.ActiveProjectiles.Count() == 2, "Both portals last through the 30-second window");
        Tick(1); Check(!TMain.ActiveProjectiles.Any(), "Both portals expire together at 30 seconds after activation");
    }
    static void Lifecycle()
    {
        Setup(); RecallPortalSystem.Cast(P(0), 0);
        Check(Portal(1).Projectile.Center == new Vector2(1608, 3200), "Without a bed the exit opens at world spawn");
        Tick(90); Check(P(0).Bottom == new Vector2(1608, 3200), "Group lands at the exit circle");
        Setup(); P(0).SpawnX = 400; P(0).SpawnY = 300; RecallPortalSystem.Cast(P(0), 0);
        P(0).SpawnX = 500; Tick(90);
        Check(P(0).Bottom == new Vector2(6408, 4800), "Exit position is fixed when the circles open");
        Setup(); P(1).active = true; RecallPortalSystem.Cast(P(0), 1); P(1).dead = true; Tick(90);
        Check(P(0).Teleports == 1 && P(1).Teleports == 0, "Drawing completes and teleports the group even if the selected ally dies");
        Setup(); RecallPortalSystem.Cast(P(0), 0);
        Collision.Solid = _ => true; Tick(90);
        Check(TMain.ActiveProjectiles.Count() == 2 && P(0).Teleports == 0, "Completely blocked exit never places player inside blocks");
        Setup(); RecallPortalSystem.Cast(P(0), 0); Tick(90); Tick(45);
        Collision.Lava = _ => true; RecallPortalSystem.RequestTravel(P(0), Portal(1));
        Check(P(0).Teleports == 1 && Portal().BurstStart == RecallPortalSystem.DelayTicks, "Blocked return never places player in lava and plays no travel burst");
        Setup(); RecallPortalSystem.Cast(P(0), 0); Tick(90);
        Portal(1).Projectile.Kill(); Tick(1);
        Check(!TMain.ActiveProjectiles.Any(), "Losing one endpoint cleans up the other");
        Setup(); RecallPortalSystem.Cast(P(0), 0); Tick(90); P(0).active = false; Tick(1);
        Check(!TMain.ActiveProjectiles.Any(), "Caster disconnect cleans up both endpoints");
        Setup(); RecallPortalSystem.Cast(P(0), 0); Tick(30); int oldId = Portal().PairId;
        RecallPortalSystem.Cast(P(0), 0);
        Check(TMain.ActiveProjectiles.Count() == 2 && Portal().PairId != oldId && Portal().Projectile.ai[1] == 0,
            "Recasting closes the caster's previous pair, even while it is still drawing");
        Setup(); TMain.netMode = NetmodeID.Server; TMain.myPlayer = 255;
        Check(!RecallPortalSystem.Cast(P(0), 0) && !TMain.ActiveProjectiles.Any(), "Server never spawns client-owned circles");
        Setup();
        var late = new RecallPortalProjectile { Projectile = new Projectile { active = true, owner = 0, ai = [8, 1800, 1] } };
        late.SetDefaults(); late.AI();
        Check(late.Projectile.timeLeft <= 90 && late.CanDamage() == false && !late.ShouldUpdatePosition(), "Late projectile sync retains remaining lifetime and cannot damage/move");
    }

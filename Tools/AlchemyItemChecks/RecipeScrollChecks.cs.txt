using System.Reflection;
using Terraria;
using Terraria.GameContent.ItemDropRules;
using Terraria.ID;
using Terraria.ModLoader;
using Terraria.ModLoader.IO;
using 伊蕾娜.ElainaModAlchemy.Gameplay;
using 伊蕾娜.ElainaModAlchemy.item;
using 伊蕾娜.ElainaModAlchemy.item.Potions;
using 伊蕾娜.ElainaModAlchemy.item.Recipes;
using 伊蕾娜.ElainaModAlchemy.UI;

// Real tML inventory, progression, shop entries and drop rules; no world or saved character.
internal static class RecipeScrollChecks
{
    private static int passed;
    private const BindingFlags Hidden = BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Static;
    private static void Check(bool condition, string message)
    { if (!condition) throw new Exception(message); passed++; }

    public static void Run(Mod mod)
    {
        var scrolls = new AlchemyRecipeScroll[] { new BloodthirstPotionRecipe(), new StarPowerPotionRecipe(),
            new ConcentrationPotionRecipe(), new ResonancePotionRecipe(), new FeatherlightPotionRecipe() };
        foreach (var scroll in scrolls)
        {
            ((ILoadable)scroll).Load(mod); ContentInstance.Register(scroll); scroll.SetDefaults();
        }
        var progressType = new AlchemyProgressPlayer();
        ((ILoadable)progressType).Load(mod); ContentInstance.Register(progressType);
        typeof(PlayerLoader).GetMethod("ResizeArrays", Hidden)!.Invoke(null, null);
        SetFactory.ResizeArrays(false);
        typeof(ItemLoader).GetMethod("ResizeArrays", Hidden)!.Invoke(null, [false]);
        Main.dedServ = false; Main.netMode = NetmodeID.SinglePlayer; Main.myPlayer = 0;
        Main.player[0] = new Player { active = true, whoAmI = 0 };
        var player = Main.LocalPlayer;
        var progress = player.GetModPlayer<AlchemyProgressPlayer>();
        var oldResolver = AlchemyCatalog.ItemTypeResolver;
        var potions = new AlchemyItem[] { ModContent.GetInstance<BloodthirstPotion>(), ModContent.GetInstance<StarPowerPotion>(),
            ModContent.GetInstance<ConcentrationPotion>(), ModContent.GetInstance<ResonancePotion>(), ModContent.GetInstance<FeatherlightPotion>() };
        AlchemyCatalog.ItemTypeResolver = id => potions.FirstOrDefault(p => p.EntryId == id)?.Type ?? 0;
        void Inventory()
        {
            for (int i = 0; i < player.inventory.Length; i++) player.inventory[i] = new Item();
            for (int i = 0; i < player.bank4.item.Length; i++) player.bank4.item[i] = new Item();
        }
        void Level(int level, params string[] unlocked) => progress.LoadData(new TagCompound {
            ["Level"] = level, ["Experience"] = 0, ["Unlocked"] = unlocked.ToList() });

        Check(scrolls.Select(s => s.Type).Distinct().Count() == 5, "Scrolls have distinct inventory identities");
        foreach (var scroll in scrolls)
        {
            string id = scroll.EntryId;
            int requiredLevel = id == "featherlight" ? 2 : 10;
            var requirement = AlchemyProgressPlayer.RequirementFor(id);
            Check(requirement.RequiredLevel == requiredLevel && requirement.RequiresRecipe && requirement.RecipeItemType == scroll.Type &&
                requirement.RecipeStack == 1 && requirement.ConsumeRecipe, id + " requires its own scroll and configured level");
            Check(scroll.Texture == AlchemyCatalog.RecipeScrollPixelPath && scroll.Item.maxStack == Item.CommonMaxStack &&
                !scroll.Item.consumable && scroll.Item.useStyle == ItemUseStyleID.None, id + " uses shared art and notebook-only consumption");
            Inventory(); Level(requiredLevel - 1); player.inventory[0] = new Item(scroll.Type, 2);
            Check(!progress.TryResearch(id, out _) && player.inventory[0].stack == 2, id + " insufficient level preserves scrolls");
            Inventory(); Level(requiredLevel);
            Check(!progress.TryResearch(id, out _), id + " meeting the level still needs a scroll");
            player.inventory[0] = new Item(scrolls.First(s => s != scroll).Type);
            Check(!progress.TryResearch(id, out _) && player.inventory[0].stack == 1, id + " cannot use another formula");
            Inventory(); player.inventory[58] = new Item(scroll.Type); player.bank4.item[0] = new Item(scroll.Type);
            Check(!progress.TryResearch(id, out _), id + " trash and closed void storage do not count as carried recipes");
            Inventory(); player.inventory[0] = new Item(scroll.Type, 2);
            var status = progress.ResearchStatus(id);
            Check(status.CanResearch && status.RecipeOwned == 2 && status.RecipeTexture == scroll.Texture, id + " UI has live count and icon");
            Check(progress.TryResearch(id, out _) && player.inventory[0].stack == 1 && progress.IsUnlocked(id), id + " learns and consumes exactly one");
            Check(!progress.TryResearch(id, out _) && player.inventory[0].stack == 1, id + " repeated research consumes nothing");
            var tips = new List<TooltipLine>(); scroll.ModifyTooltips(tips);
            Check(tips.Any(t => t.Name == "AlreadyLearned"), id + " spare copy reports already learned");
            var saved = new TagCompound(); progress.SaveData(saved); progress.ResetProgress(); progress.LoadData(saved);
            Check(progress.IsUnlocked(id) && progress.Level == requiredLevel, id + " learned formula survives save and load");
            Level(requiredLevel - 1, id); Inventory();
            Check(progress.IsUnlocked(id), id + " legacy lower-level unlock survives migration");
        }
        Level(10); Inventory();
        player.inventory[0] = new Item(scrolls[0].Type);
        player.dead = true;
        Check(!progress.TryResearch(scrolls[0].EntryId, out _) && player.inventory[0].stack == 1, "Dead players cannot consume a scroll");
        player.dead = false; Main.dedServ = true;
        Check(!progress.TryResearch(scrolls[0].EntryId, out _), "Dedicated server cannot execute the local research transaction");
        Main.dedServ = false;

        var sources = new AlchemyRecipeSources();
        foreach (var (npc, scroll, night) in new[] {
            (NPCID.Wizard, scrolls[1], false), (NPCID.ArmsDealer, scrolls[2], true), (NPCID.WitchDoctor, scrolls[3], true) })
        {
            var shop = new NPCShop(npc); sources.ModifyShop(shop);
            Check(shop.Entries.Count == 1 && shop.Entries[0].Item.type == scroll.Type && shop.Entries[0].Item.value == Item.buyPrice(gold: 5),
                scroll.Name + " appears in its shop at the base price");
            Main.dayTime = true;
            Check(shop.Entries[0].Conditions.All(c => c.IsMet()) == !night, scroll.Name + " daytime availability");
            Main.dayTime = false;
            Check(shop.Entries[0].Conditions.All(c => c.IsMet()), scroll.Name + " nighttime availability");
        }
        var otherShop = new NPCShop(NPCID.Merchant); sources.ModifyShop(otherShop);
        Check(otherShop.Entries.Count == 0, "Unrelated shops receive no scrolls");
        var enemy = new NPC { lifeMax = 100, damage = 20, value = 100, realLife = -1, whoAmI = 2 };
        var loot = new NPCLoot(NPCID.Zombie, new ItemDropDatabase()); sources.ModifyNPCLoot(enemy, loot);
        var rule = loot.Get(false).Single() as ItemDropWithConditionRule;
        Check(rule != null && rule.itemId == scrolls[0].Type && rule.chanceDenominator == 100, "Hostile enemy receives the one-percent bloodthirst rule");
        var attempt = new DropAttemptInfo { npc = enemy, player = player };
        Main.hardMode = Main.bloodMoon = true; Main.dayTime = false;
        Check(rule.CanDrop(attempt), "Hardmode Blood Moon enemy can drop the recipe");
        Main.hardMode = false; Check(!rule.CanDrop(attempt), "Pre-hardmode blocks recipe drop"); Main.hardMode = true;
        Main.bloodMoon = false; Check(!rule.CanDrop(attempt), "Ordinary nights block recipe drop"); Main.bloodMoon = true;
        Main.dayTime = true; Check(!rule.CanDrop(attempt), "Daytime blocks recipe drop"); Main.dayTime = false;
        enemy.SpawnedFromStatue = true; Check(!rule.CanDrop(attempt), "Statues cannot farm recipe drops"); enemy.SpawnedFromStatue = false;
        enemy.realLife = 1; Check(!rule.CanDrop(attempt), "Shared-life segments cannot duplicate recipe drops"); enemy.realLife = -1;
        enemy.friendly = true;
        Check(!rule.CanDrop(attempt), "Friendly entities cannot drop combat recipes");
        var friendlyLoot = new NPCLoot(NPCID.Bunny, new ItemDropDatabase()); sources.ModifyNPCLoot(enemy, friendlyLoot);
        Check(friendlyLoot.Get(false).Count == 0, "Friendly entities do not advertise combat recipe drops");
        CheckChestLoot(scrolls[4].Type);
        AlchemyCatalog.ItemTypeResolver = oldResolver;
        Console.WriteLine($"Recipe scrolls: {passed} registration, research, persistence, shop and loot checks passed.");
    }

    private static void CheckChestLoot(int scrollType)
    {
        Main.maxTilesX = 256; Main.maxTilesY = 600; Main.worldSurface = 50;
        Main.tile = (Tilemap)Activator.CreateInstance(typeof(Tilemap), BindingFlags.Instance | BindingFlags.NonPublic,
            null, new object[] { (ushort)Main.maxTilesX, (ushort)Main.maxTilesY }, null)!;
        Chest Treasure(int x, int y, int style = 1, ushort tileType = TileID.Containers)
        {
            var chest = new Chest { x = x, y = y };
            for (int i = 0; i < chest.item.Length; i++) chest.item[i] = new Item();
            chest.item[0] = new Item(ItemID.CloudinaBottle);
            chest.item[1] = new Item(ItemID.Torch, 17);
            Tile tile = Main.tile[x, y];
            tile.HasTile = true; tile.TileType = tileType; tile.TileFrameX = (short)(style * 36);
            return chest;
        }
        Check(!AlchemyRecipeChestLoot.IsUndergroundGoldChest(null), "Missing chest is ignored");
        var surface = Treasure(5, 49);
        var hell = Treasure(5, Main.UnderworldLayer);
        var wood = Treasure(6, 60, 0);
        var locked = Treasure(7, 60, 2);
        var frozen = Treasure(8, 60, 11);
        var trap = Treasure(9, 60, 52);
        var fake = Treasure(10, 60, 1, TileID.Containers2);
        var empty = Treasure(11, 60); empty.item[0].TurnToAir(); empty.item[1].TurnToAir();
        var removed = Treasure(12, 60); Tile removedTile = Main.tile[12, 60]; removedTile.HasTile = false;
        var full = Treasure(13, 60);
        for (int i = 2; i < full.item.Length; i++) full.item[i] = new Item(ItemID.Torch, 99);
        Chest[] excluded = [surface, hell, wood, locked, frozen, trap, fake, empty, removed];
        Check(excluded.All(c => !AlchemyRecipeChestLoot.IsUndergroundGoldChest(c)),
            "Surface, hell, non-gold, locked, biome, trap, empty and removed chests are excluded");
        Check(AlchemyRecipeChestLoot.IsUndergroundGoldChest(Treasure(14, 50)), "Underground boundary includes ordinary gold chests");

        // A fixed seed exercises both outcomes through the actual world-generation hook.
        var ordinary = Enumerable.Range(0, 1000).Select(i => Treasure(20 + i % 100, 100 + i / 100)).ToArray();
        Main.chest = ordinary.Concat(excluded).Append(full).Append(null).ToArray();
        WorldGen.genRand.SetSeed(20261021);
        new AlchemyRecipeChestLoot().PostWorldGen();
        int found = ordinary.Count(c => c.item.Any(i => i.type == scrollType));
        Check(found >= 70 && found <= 130, "Gold chest population has an approximately ten-percent scroll rate");
        Check(ordinary.All(c => c.item[0].type == ItemID.CloudinaBottle && c.item[1].type == ItemID.Torch &&
            c.item[1].stack == 17), "All existing treasure is preserved");
        Check(ordinary.All(c => c.item.Count(i => i.type == scrollType) <= 1 &&
            c.item.Where(i => i.type == scrollType).All(i => i.stack == 1)), "Successful gold chests receive exactly one scroll");
        Check(excluded.Append(full).All(c => c.item.All(i => i.type != scrollType)), "Excluded and full chests receive no scroll");
        Check(full.item.Skip(2).All(i => i.type == ItemID.Torch && i.stack == 99), "Full chest contents are never overwritten");
        Console.WriteLine($"Featherlight chest sample: {found}/1000 ordinary underground Gold Chests received one scroll.");
    }
}

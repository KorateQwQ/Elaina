using static 伊蕾娜.Localization.UIText;
using SDL2;
using 伊蕾娜.ElainaModAlchemy.UI;
using P = 伊蕾娜.ElainaModAlchemy.UI.AlchemyNotebookPresentation;

// The actual production drawing and measured rows are linked by run.ps1. The only
// fixtures here are private notebook state and screen placement; no Terraria saves.
internal sealed class AlchemyPreviewGame : Game
{
    internal static string Repo;
    private static string Tml;
    private static Microsoft.Xna.Framework.Content.ContentManager VanillaContent, TmlContent;
    private static GraphicsDevice Device;
    private static readonly Dictionary<string, Texture2D> Textures = new(StringComparer.Ordinal);
    private static readonly RasterizerState ClipRasterizer = new() { CullMode = CullMode.None, ScissorTestEnable = true };
    private static bool BookOnly;
    private static bool QuillOnly;
    private static bool ScrollOnly;
    private static bool StarPowerOnly;
    private static bool ExportMaterials;
    private static string FocusScene;
    private bool done;

    private AlchemyPreviewGame()
    {
        new GraphicsDeviceManager(this) { PreferredBackBufferWidth = 320, PreferredBackBufferHeight = 180,
            GraphicsProfile = GraphicsProfile.HiDef, SynchronizeWithVerticalRetrace = false };
        SDL.SDL_HideWindow(Window.Handle);
        IsFixedTimeStep = false;
    }

    protected override void LoadContent()
    {
        Device = GraphicsDevice;
        Content.RootDirectory = Path.GetFullPath(Path.Combine(Repo, "../KL/Fonts"));
        KL.Drawing.FontManager.HarmonyOS_Sans_SC = new(Content.Load<DynamicSpriteFont>("HarmonyOS_Sans_SC"));
        KL.Drawing.FontManager.NotoSerifSC = new(Content.Load<DynamicSpriteFont>("NotoSerifSC"));
        KL.Drawing.FontManager.Gelasio = new(Content.Load<DynamicSpriteFont>("Gelasio"));
        VanillaContent = new(Services, Path.GetFullPath(Path.Combine(Tml, "../Terraria/Content")));
        TmlContent = new(Services, Path.Combine(Tml, "Content"));
        var vanillaIds = System.Text.Json.JsonSerializer.Deserialize<Dictionary<string, int>>(
            File.ReadAllText(Path.Combine(Repo, ".vissandbox/alchemy-preview/vanilla-item-ids.json")));
        AlchemyCatalog.ItemTypeResolver = id => vanillaIds.GetValueOrDefault(id);
        // Check the production catalog, not only previously exported source PNGs.
        // A new vanilla ingredient may have neither an export nor an outline yet.
        if (!ExportMaterials)
        {
            var missing = AlchemyCatalog.Materials.Select(m => m.IconPath)
                .Concat(AlchemyCatalog.Recipes.Select(r => r.PixelPath))
                .Select(path => Path.Combine(Repo, "ElainaModAlchemy/UI/Assets/Outlines", path[(path.LastIndexOf('/') + 1)..] + ".png"))
                .Distinct().Where(path => !File.Exists(path)).ToArray();
            if (missing.Length > 0)
                throw new InvalidOperationException("Missing catalog outlines. Run run.ps1 -ExportMaterials, then bake_redesign.py:\n" + string.Join("\n", missing));
        }
        foreach (var material in AlchemyCatalog.Materials.Where(m => m.Art is null))
        {
            if (material.ItemType <= 0) throw new InvalidOperationException("Missing vanilla item ID: " + material.Id);
            _ = VanillaItemTexture(material.ItemType);
        }
        Console.WriteLine($"Loaded {vanillaIds.Count} vanilla material textures from game XNB assets.");
        var pixel = new Texture2D(Device, 1, 1);
        pixel.SetData(new[] { Color.White });
        Terraria.GameContent.TextureAssets.MagicPixel = new(pixel);
    }

    protected override void Update(GameTime gameTime)
    {
        if (done) return;
        done = true;
        string output = Path.Combine(Repo, ".vissandbox/alchemy-preview/output", Terraria.Localization.Language.Culture);
        Directory.CreateDirectory(output);
        if (ExportMaterials)
        {
            string export = Path.Combine(Repo, ".vissandbox/alchemy-preview/material-sources");
            Directory.CreateDirectory(export);
            foreach (var material in AlchemyCatalog.Materials.Where(m => m.Art is null))
            {
                var texture = VanillaItemTexture(material.ItemType);
                using var stream = File.Create(Path.Combine(export, $"Item_{material.ItemType}.png"));
                texture.SaveAsPng(stream, texture.Width, texture.Height);
            }
            Console.WriteLine(export);
            Exit();
            return;
        }
        using var batch = new SpriteBatch(Device);
        if (ScrollOnly)
        {
            AlchemyScrollPreview.Run(Device, batch, output);
            Exit();
            return;
        }
        if (QuillOnly)
        {
            AlchemyQuillPreview.Run(Device, batch, output);
            Exit();
            return;
        }
        if (BookOnly)
        {
            AlchemyBookPreview.Run(Device, batch, output);
            Exit();
            return;
        }
        string[] scenes = ["new", "all", "potion", "bloodlust", "curio", "curio-locked", "food", "material", "acquire", "short", "locked", "research-ready", "recipe-missing", "recipe-ready", "level-max", "header-reference", "close-hover", "brew-050", "detail-bottom", "catalog-bottom", "catalog-middle", "redesign-reference", "starpower"];
        scenes = scenes.Concat(new[] { "curio-dust", "curio-rain", "vanilla-materials", "focus", "resonance", "bread", "stew", "brulee" }).ToArray();
        if (StarPowerOnly) scenes = ["starpower"];
        if (FocusScene != null) scenes = [FocusScene];
        foreach (var (width, height, uiScale) in new[] { (1600, 1000, 1f), (1280, 720, 1f), (1600, 1000, 1.25f) })
        foreach (string scene in scenes)
        {
            using var target = new RenderTarget2D(Device, width, height);
            Device.SetRenderTarget(target);
            Device.Clear(new Color(20, 21, 28));
            Render(batch, width, height, uiScale, scene);
            Device.SetRenderTarget(null);
            using var stream = File.Create(Path.Combine(output, $"{scene}-{width}x{height}-{uiScale:0.00}.png"));
            target.SaveAsPng(stream, target.Width, target.Height);
        }
        Console.WriteLine(output);
        Exit();
    }

    internal static void Render(SpriteBatch batch, int width, int height, float uiScale, string scene,
        AlchemySelectionQuillMotion quill = null, string selectedId = null, float? scrollOffset = null,
        float? detailScrollOffset = null)
    {
        var state = new AlchemyNotebookState(new NotebookFixture(scene));
        switch (scene)
        {
            case "curio": case "curio-locked": state.SetCategory("curio"); break;
            case "curio-dust": state.SetCategory("curio"); state.Select("mimic"); break;
            case "curio-rain": state.SetCategory("curio"); state.Select("rain"); break;
            case "food": state.SetCategory("food"); break;
            case "bread": case "stew": case "brulee": state.SetCategory("food"); state.Select(scene); break;
            case "material": state.SetCategory("material"); break;
            case "potion": state.SetCategory("potion"); break;
            case "vanilla-materials": state.SetCategory("potion"); state.Select("isolation"); break;
            case "catalog-bottom": state.Select(AlchemyCatalog.Recipes.Last().Id); break;
            case "acquire": state.Select("flour"); break;
            case "bloodlust": state.Select("bloodlust"); break;
            case "starpower": state.SetCategory("potion"); state.Select("starpower"); break;
            case "focus": case "resonance": state.SetCategory("potion"); state.Select(scene); break;
            case "detail-bottom": case "redesign-reference": state.Select("painkiller"); break;
            case "locked": state.Select("focus"); break;
            case "research-ready": state.Select("painkiller"); break;
            case "recipe-missing": case "recipe-ready": state.Select("focus"); break;
        }
        if (selectedId != null) state.Select(selectedId);
        float fit = Math.Min(1, Math.Min((width / uiScale - 40) / P.DesignWidth, (height / uiScale - 40) / P.DesignHeight));
        float scale = uiScale * fit;
        var origin = new Vector2((width - P.DesignWidth * scale) / 2, (height - P.DesignHeight * scale) / 2);
        var d = new AlchemyDrawing { Batch = batch, Scale = scale, Origin = origin };
        void Begin(Rectangle? clip = null)
        {
            Device.ScissorRectangle = clip ?? new Rectangle(0, 0, width, height);
            batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.LinearClamp,
                DepthStencilState.None, ClipRasterizer);
        }
        Rectangle Clip(int x, int y, int w, int h) => new((int)(origin.X + x * scale), (int)(origin.Y + y * scale),
            (int)Math.Ceiling(w * scale), (int)Math.Ceiling(h * scale));
        void At(float x, float y) => d.Origin = origin + new Vector2(x, y) * scale;
        float progress = scene switch { "brew-015" => .15f, "brew-050" => .5f, "brew-085" => .85f, _ => 0 };
        Begin();
        P.Chrome(d, state);
        P.Hero(d, state, progress);
        for (int i = 0; i < P.Navigation.Count; i++)
        { At(26 + i * P.TabWidth, 84); P.Category(d, state, P.Navigation[i], false); }
        At(0, 0);
        P.TabMarker(d, Math.Max(0, P.Navigation.ToList().FindIndex(c => c.Id == state.Category)));
        At(P.CraftBounds.X, P.CraftBounds.Y); P.CraftButton(d, state, false, progress);
        At(0, 0);
        P.Footer(d,state,progress>0);
        if (state.IsUnlocked(state.Current) && state.Current.Craftable)
        {
            d.Text("−",840,620,14,P.Muted*.3f,.5f);
            d.Text("+",896,620,14,P.Lavender,.5f);
            d.Text(LocalizeUI("Max"),940,621,10,P.Muted,.5f);
        }
#if DEBUG
        At(P.DebugLevelUpBounds.X,P.DebugLevelUpBounds.Y);
        P.DebugLevelUpButton(d,state,false,progress<=0&&!state.IsBusy);
        At(608,754); P.DebugButton(d,LocalizeUI("AlchemyDebugRestock"),false);
        At(680,754); P.DebugButton(d,LocalizeUI("AlchemyDebugReset"),false);
#endif
        At(0,0);
        At(P.CloseButtonBounds.X, P.CloseButtonBounds.Y);
        P.CloseButton(d, scene == "close-hover");
        batch.End();

        float catalogOffset = AlchemyCatalogLayout.Reveal(state.Category, state.SelectedId, 0, P.CatalogHeight);
        if (scene == "catalog-middle") catalogOffset = (AlchemyCatalogLayout.HeightFor(state.Category) - P.CatalogHeight) / 2f;
        if (scrollOffset.HasValue) catalogOffset = scrollOffset.Value;
        Begin(Clip(P.CatalogX, P.CatalogY, P.CatalogWidth, P.CatalogHeight));
        foreach (var slot in AlchemyCatalogLayout.For(state.Category))
        {
            float y = P.CatalogY + slot.Y - catalogOffset;
            if (y + P.CardHeight < P.CatalogY || y > P.CatalogY + P.CatalogHeight) continue;
            if (slot.Recipe.Id == state.SelectedId) continue;
            At(P.CatalogX + slot.X, y); P.Card(d, state, slot.Recipe, false);
        }
        // Like the SUI overlay, the picked-up card is drawn after its neighbours.
        var selectedSlot = AlchemyCatalogLayout.For(state.Category).First(s => s.Recipe.Id == state.SelectedId);
        At(P.CatalogX + selectedSlot.X, P.CatalogY + selectedSlot.Y - catalogOffset);
        P.Card(d, state, selectedSlot.Recipe, false);
        d.Origin += P.CardOffset(selectedSlot.Recipe) * scale;
        if (quill != null)
            d.Origin = quill.ResolvePosition(state.SelectedId, d.Origin,
                origin + new Vector2(P.CatalogX, P.CatalogY) * scale, new Vector2(0, -catalogOffset * scale), scale);
        P.SelectionQuill(d);
        batch.End();
        Begin(); At(0, 0);
        d.CatalogFade();
        batch.End();

        Begin(Clip(P.BagBounds.X,P.BagBounds.Y,P.BagBounds.Width,P.BagBounds.Height));
        var bag=P.BagMaterials(state);
        for(int i=0;i<bag.Length;i++)
        {
            At(P.BagBounds.X+i*48,P.BagBounds.Y); P.MaterialSlot(d,state,bag[i],false);
        }
        At(0,0);
        if(bag.Length*48>P.BagBounds.Width)
            d.RoundedBar(P.BagBounds.X,P.BagBounds.Y+60,P.BagBounds.Width*P.BagBounds.Width/(float)(bag.Length*48),3,P.Lavender*.6f);
        batch.End();

        var rows = P.DetailRows(d, state);
        float total = rows.Sum(row => row.Height), offset = scene == "detail-bottom" ? Math.Max(0, total - P.DetailHeight) : 0;
        if (detailScrollOffset.HasValue) offset = detailScrollOffset.Value;
        Begin(Clip(P.DetailX, P.DetailY, P.DetailWidth, P.DetailHeight));
        float rowY = P.DetailY - offset;
        foreach (var row in rows) { At(P.DetailX, rowY); row.Paint(d, null); rowY += row.Height; }
        batch.End();
        if (offset + P.DetailHeight < total - 1)
        {
            Begin(); At(0,0); d.DetailFade(); batch.End();
        }

    }

    internal static Texture2D VanillaItemTexture(int itemType)
    {
        string asset = "Images/Item_" + itemType;
        var content = File.Exists(Path.Combine(TmlContent.RootDirectory, asset + ".xnb"))
            ? TmlContent : VanillaContent;
        return content.Load<Texture2D>(asset);
    }

    internal static Texture2D Texture(string asset)
    {
        if (Textures.TryGetValue(asset, out var texture)) return texture;
        string path = Path.Combine(Repo, asset[(asset.IndexOf('/') + 1)..] + ".png");
        using var stream = File.OpenRead(path);
        texture = Texture2D.FromStream(Device, stream);
        // tML ImageIO.ToRaw preserves partial-alpha RGB and zeros fully clear pixels.
        var pixels = new Color[texture.Width * texture.Height];
        texture.GetData(pixels);
        for (int i = 0; i < pixels.Length; i++) if (pixels[i].A == 0) pixels[i] = Color.Transparent;
        texture.SetData(pixels);
        Textures.Add(asset, texture);
        return texture;
    }

    protected override void UnloadContent()
    {
        foreach (var texture in Textures.Values) texture.Dispose();
        Textures.Clear();
        VanillaContent.Dispose();
        TmlContent.Dispose();
        AlchemyCatalog.ItemTypeResolver = null;
        Terraria.GameContent.TextureAssets.MagicPixel.Value.Dispose();
        ClipRasterizer.Dispose();
        base.UnloadContent();
    }

    [STAThread]
    private static void Main(string[] args)
    {
        Repo = Path.GetFullPath(args[0]);
        Tml = Path.GetFullPath(args[1]);
        Terraria.Localization.Language.Load(Repo, args[2]);
        BookOnly = args.Contains("--book-only");
        QuillOnly = args.Contains("--quill-only");
        StarPowerOnly = args.Contains("--star-power-only");
        ExportMaterials = args.Contains("--export-materials");
        FocusScene = args.FirstOrDefault(a => a.StartsWith("--scene="))?[8..];
        ScrollOnly = args.Contains("--scroll-only");
        using var game = new AlchemyPreviewGame();
        game.Run();
    }
}

using static 伊蕾娜.Localization.UIText;
using 伊蕾娜.ElainaModAlchemy.UI;
using 伊蕾娜.ElainaModAlchemy.Gameplay;

// Offline-only fixtures. No demonstration inventory remains in the production state.
internal sealed class NotebookFixture(string scene) : IAlchemyNotebookSource
{
    public AlchemyNotebookResult LastResult => null;
    public bool Request(AlchemyNotebookAction action, string id, int batches = 1) => false;
    public AlchemyNotebookSnapshot Capture()
    {
        bool scrollScene = scene.StartsWith("recipe-");
        bool learned = scrollScene && scene.EndsWith("-learned");
        bool blank = scene is "new" or "locked" or "research-ready" or "curio-locked" or "material-initial" or "food-initial" || scrollScene && !learned;
        int level = scene is "level-max" or "curio-dust" or "curio-rain" or "vanilla-materials" or "starpower" or "focus" or "resonance" ? 20 : scene is "new" or "locked" or "material-initial" or "food-initial" ? 1 : scene is "header-reference" or "redesign-reference" ? 3 : 5;
        if(scrollScene)
        {
            string[] parts = scene.Split('-');
            int requiredLevel = AlchemyProgressionRules.RequiredLevel(parts.Length >= 3 ? parts[1] : "focus");
            level = scene.EndsWith("-level-low") ? Math.Max(1, requiredLevel - 1) : requiredLevel;
        }
        var materials = AlchemyCatalog.Materials.ToDictionary(m => m.Id, m => scene == "short" ? 0 : m.Initial);
        if(scene == "redesign-reference") { materials["moondew"]=2; materials["daybloom"]=10; materials["honey"]=6; }
        var products = AlchemyCatalog.Recipes.ToDictionary(r => r.Id, r => r.Id == "mana" ? 3 : 0);
        var unlocked = AlchemyCatalog.Recipes.Where(r => !blank && (AlchemyProgressionRules.RequiredLevel(r.Id) <= level || r.Id == "bloodlust"))
            .Select(r => r.Id).ToHashSet();
        if(scene == "redesign-reference")
        {
            unlocked = AlchemyCatalog.Recipes.Where(r=>r.InitialUnlocked || r.Id is "mana" or "painkiller" or "rain" or "bread" or "brulee" or "flour" or "sugar" or "egg").Select(r=>r.Id).ToHashSet();
            products["painkiller"]=1;
        }
        unlocked.UnionWith(AlchemyCatalog.InitiallyUnlockedIds);
        var maximum = AlchemyCatalog.Recipes.ToDictionary(r => r.Id, r => !unlocked.Contains(r.Id) || !r.Craftable ? 0
            : r.Ingredients.Min(i => i.Choices.Sum(id => materials[id]) / i.Count));
        var research = new Dictionary<string, AlchemyResearchStatus>();
        foreach (var entry in AlchemyCatalog.Recipes)
        {
            bool scroll = entry.Id is "bloodlust" or "starpower" or "focus" or "resonance" or "featherlight" or "painkiller" or "isolation";
            int neededLevel = AlchemyProgressionRules.RequiredLevel(entry.Id);
            int owned = scroll && (scene.EndsWith("-ready") || scene.EndsWith("-level-low")) ? 1 : 0;
            bool ready = !unlocked.Contains(entry.Id) && level >= neededLevel && (!scroll || owned > 0);
            string name = scroll ? Terraria.Localization.Language.GetTextValue($"Mods.伊蕾娜.Alchemy.Items.{entry.Art}Recipe.DisplayName") : "";
            research[entry.Id] = new(neededLevel, scroll, true, name, owned, 1, true, ready,
                unlocked.Contains(entry.Id) ? LocalizeUI("Researched") : level < neededLevel ? LocalizeUI("NeedAlchemyLevel",neededLevel)
                    : scroll && owned == 0 ? LocalizeUI("NeedRecipe",name,1) : scroll ? LocalizeUI("ResearchConsumesRecipe") : LocalizeUI("ResearchLevelMet"),
                scroll ? AlchemyCatalog.RecipeScrollPixelPath : null);
        }
        return new AlchemyNotebookSnapshot
        {
            Materials = materials, Products = products, Unlocked = unlocked, Research = research,
            MaximumBatches = maximum, Level = level, MaximumLevel = scrollScene ? Math.Max(8, level) : 20,
            Experience = scene == "redesign-reference" ? 40 : scene == "header-reference" ? 35 : level == 20 || scene is "new" or "material-initial" or "food-initial" ? 0 : 45,
            ExperienceToNextLevel = scrollScene && level >= 8 ? 0 : scene == "header-reference" ? 40 : AlchemyProgressionRules.ExperienceToNextLevel(level)
        };
    }
}

using System;
using System.IO;
using Microsoft.Xna.Framework;
using Microsoft.Xna.Framework.Graphics;
using SDL2;

internal static class Program
{
    static void Main(string[] args)
    {
        using var game = new Preview(args[0]);
        game.Run();
    }
}

sealed class Preview : Game
{
    readonly string output;
    IceShardScene scene;
    RenderTarget2D target;
    bool exported;

    public Preview(string output)
    {
        this.output = output;
        _ = new GraphicsDeviceManager(this) {
            PreferredBackBufferWidth = 960, PreferredBackBufferHeight = 480,
            GraphicsProfile = GraphicsProfile.HiDef, SynchronizeWithVerticalRetrace = false
        };
        IsFixedTimeStep = false;
        SDL.SDL_HideWindow(Window.Handle);
    }

    protected override void LoadContent()
    {
        SDL.SDL_HideWindow(Window.Handle);
        Directory.CreateDirectory(output);
        scene = new IceShardScene(GraphicsDevice);
        target = new RenderTarget2D(GraphicsDevice, 960, 480, false, SurfaceFormat.Color, DepthFormat.None);
    }

    protected override void Draw(GameTime gameTime)
    {
        if (exported) return;
        foreach (string mode in new[] { "single", "continuous", "dark" })
        {
            string frames = Path.Combine(output, mode);
            Directory.CreateDirectory(frames);
            int count = mode == "continuous" ? 96 : 60;
            for (int frame = 0; frame < count; frame++)
            {
                GraphicsDevice.SetRenderTarget(target);
                scene.Draw(frame / 30f, mode);
                GraphicsDevice.SetRenderTarget(null);
                using var stream = File.Create(Path.Combine(frames, $"{frame:D3}.png"));
                target.SaveAsPng(stream, 960, 480);
            }
        }
        exported = true;
        Console.WriteLine("FNA export complete: single / continuous / dark; 216 frames at 30 fps.");
        Exit();
    }

    protected override void UnloadContent()
    {
        scene.Dispose();
        target.Dispose();
    }
}

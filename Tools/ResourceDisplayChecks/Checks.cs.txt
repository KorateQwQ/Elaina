using System.Reflection;
using System.Runtime.CompilerServices;
using Microsoft.Xna.Framework;
using Terraria;
using Terraria.GameContent.UI.ResourceSets;
using Terraria.ModLoader;
using Terraria.ModLoader.IO;
using 伊蕾娜;
using 伊蕾娜.ElainaAttribute;
using 伊蕾娜.ElainaModSkills;
using 伊蕾娜.ElainaModSkills.ElainaSkillUI;
using GameMain = Terraria.Main;

static class Checks
{
    const BindingFlags Instance = BindingFlags.Instance | BindingFlags.Public | BindingFlags.NonPublic;
    static int passed;
    static bool registered;
    static readonly ElainaSkillManager System = new();
    static string ElainaKey;

    static void Check(bool value, string reason)
    {
        if (!value) throw new Exception(reason);
        passed++;
    }

    static ElainaModplayer Character(bool elaina, int index = 0)
    {
        var entity = new Player { active = true, whoAmI = index };
        var player = new ElainaModplayer { Elaina = elaina };
        typeof(ModPlayer).GetProperty("Entity", Instance)!.SetValue(player, entity);
        typeof(ModPlayer).GetProperty("Index", Instance)!.SetValue(player, (ushort)0);
        typeof(Player).GetField("modPlayers", Instance)!.SetValue(entity, new ModPlayer[] { player });
        if (!registered) { ContentInstance.Register(player); registered = true; }
        return player;
    }

    static void Enter(ElainaModplayer player)
    {
        GameMain.player[0] = player.Player;
        GameMain.gameMenu = false;
        Restore(player);
    }

    static void Restore(ElainaModplayer player) => typeof(ElainaSkillManager)
        .GetMethod("RestoreResourceDisplay", BindingFlags.Static | BindingFlags.NonPublic)!
        .Invoke(null, [player]);

    static void Select(string key) => typeof(PlayerResourceSetsManager)
        .GetMethod("SetActive", Instance)!.Invoke(GameMain.ResourceSetsManager, [key]);

    static string Selected => GameMain.ResourceSetsManager.ActiveSet.ConfigKey;
    static TagCompound Save(ElainaModplayer player)
    {
        var tag = new TagCompound();
        player.SaveData(tag);
        return tag;
    }

    static void Main()
    {
        Terraria.Program.SavePath = Path.GetTempPath();
        ((ILoadable)new BoolTagSerializer()).Load(null!);
        GameMain.myPlayer = 0;
        GameMain.netMode = 0;
        var display = new ElainaPlayerResourceDisplaySet();
        ContentInstance.Register(display);
        ElainaKey = display.ConfigKey;
        // Use the real tML cycling API, with texture-free stand-ins for display rendering.
        var manager = GameMain.ResourceSetsManager = new PlayerResourceSetsManager();
        var keys = (List<string>)typeof(PlayerResourceSetsManager).GetField("accessKeys", Instance)!.GetValue(manager)!;
        var sets = (Dictionary<string, IPlayerResourcesDisplaySet>)typeof(PlayerResourceSetsManager)
            .GetField("_sets", Instance)!.GetValue(manager)!;
        keys.Add(ElainaKey);
        foreach (string key in keys) sets[key] = new Display(key);
        Select("Default");

        var a = Character(true);
        Enter(a);
        Check(Selected == ElainaKey, "First entry selects Elaina resources.");
        Select("HorizontalBars");
        System.UpdateUI(new GameTime());
        Check(Selected == "HorizontalBars" && !a.UseElainaResourceDisplay, "Manual changes stay selected and update the character bool.");
        var savedA = Save(a);
        System.OnWorldUnload();
        a = Character(true);
        a.LoadData(savedA);
        Enter(a);
        Check(!a.UseElainaResourceDisplay && Selected == "HorizontalBars", "Opt-out survives save/load and re-entry.");

        Select(ElainaKey);
        // Save immediately, before another UI update, exactly as SaveAndQuit sets gameMenu first.
        GameMain.gameMenu = true;
        savedA = Save(a);
        Check(a.UseElainaResourceDisplay && savedA.GetBool(nameof(a.UseElainaResourceDisplay)), "Exit save captures a last-moment switch back even in menu state.");
        System.OnWorldUnload();

        var ordinary = Character(false);
        Enter(ordinary);
        Check(Selected != ElainaKey, "An ordinary character leaves the inherited Elaina resource style.");
        Select("Default");
        Save(ordinary);
        System.OnWorldUnload();
        a = Character(true);
        a.LoadData(savedA);
        Enter(a);
        Check(Selected == ElainaKey, "Elaina preference restores after another character changes global style.");

        var optedOut = Character(true);
        optedOut.LoadData(new TagCompound { ["Elaina"] = true, [nameof(a.UseElainaResourceDisplay)] = false });
        System.OnWorldUnload();
        Enter(optedOut);
        Check(Selected != ElainaKey && !optedOut.UseElainaResourceDisplay, "A second Elaina character's opt-out does not inherit the first one's style.");
        Select(ElainaKey);
        System.UpdateUI(new GameTime());
        Check(optedOut.UseElainaResourceDisplay, "Manually selecting Elaina again re-enables restoration.");
        Select("NewWithText");
        GameMain.gameMenu = true;
        var savedOptOut = Save(optedOut);
        Check(!savedOptOut.GetBool(nameof(a.UseElainaResourceDisplay)), "Exit save also captures a last-moment opt-out.");
        System.OnWorldUnload();
        Select(ElainaKey);
        Save(optedOut);
        Check(!optedOut.UseElainaResourceDisplay, "Saving outside the world cannot overwrite the recorded choice.");

        var legacy = Character(true);
        legacy.LoadData(new TagCompound { ["Elaina"] = true });
        Check(legacy.UseElainaResourceDisplay, "Old saves retain the previous Elaina default.");
        Enter(legacy);
        var remote = Character(true, 1);
        remote.UseElainaResourceDisplay = false;
        Restore(remote);
        Save(remote);
        Check(Selected == ElainaKey && !remote.UseElainaResourceDisplay, "Remote players cannot change local selection or preferences.");
        Select("Default");
        Save(legacy);
        Check(!legacy.UseElainaResourceDisplay, "Remote entry does not replace the local preference owner.");
        var menuCharacter = Character(true);
        Save(menuCharacter);
        Check(menuCharacter.UseElainaResourceDisplay, "Creating or serializing another character does not read local preferences.");

        // No graphics device is needed to check the actual SUI visibility/input gates.
        var bar = (ElainaSkillBar)RuntimeHelpers.GetUninitializedObject(typeof(ElainaSkillBar));
        Check(!typeof(KL.SkillSystem.SilkyUI.IDraggableUI).IsAssignableFrom(bar.GetType()),
            "The skill bar no longer inherits the legacy mouse drag controller.");
        bar.SetLeft(25, 0, .5f);
        bar.SetTop(35, 0, .5f);
        var barPosition = (bar.Left, bar.Top);
        bar.OnLeftMouseDown(new SilkyUIFramework.UIMouseEvent(bar, Vector2.Zero));
        bar.OnLeftMouseUp(new SilkyUIFramework.UIMouseEvent(bar, new Vector2(500, 300)));
        Check((bar.Left, bar.Top) == barPosition, "Mouse drag events cannot change the shortcut-controlled bar position.");
        bar.Enabled = true;
        Check(bar.Enabled, "The skill bar is enabled for an active Elaina character.");
        GameMain.player[0] = ordinary.Player;
        Check(!bar.Enabled && !bar.IsInteractable, "Ordinary characters cannot draw or interact with the skill bar.");
        Select(ElainaKey);
        System.UpdateUI(new GameTime());
        Check(Selected != ElainaKey, "Ordinary characters still skip Elaina resources when cycling settings.");
        GameMain.player[0] = legacy.Player;
        GameMain.gameMenu = true;
        Check(!bar.Enabled && !bar.IsInteractable, "Menus cannot draw or interact with the skill bar.");
        GameMain.gameMenu = false;
        legacy.Player.active = false;
        Check(!bar.Enabled, "Inactive players cannot show the skill bar.");
        legacy.Player.active = true;
        Check(bar.Enabled, "Returning to Elaina re-enables the existing UI instance.");
        bar.Enabled = false;
        Check(!bar.Enabled, "The base UI enable switch remains respected.");
        SlotAnimationChecks();
        SkillBarScaleChecks();
        Console.WriteLine($"PASS: {passed} resource preference and skill-bar checks (real tModLoader APIs and production code).");
    }

    static void SlotAnimationChecks()
    {
        var slot = new ElainaSkillSlot(null);
        var time = new GameTime();
        var elapsed = typeof(ElainaSkillSlot).GetField("_selectionTime", Instance)!;
        var selected = typeof(ElainaSkillSlot).GetField("_selected", Instance)!;
        float Age() => (float)elapsed.GetValue(slot)!;
        ElainaSkillModPlayer.CurrentSkillIndex = 0;
        GameMain.hasFocus = true;
        GameMain.hideUI = false;
        slot.HandleUpdateStatus(time);
        Thread.Sleep(65);
        slot.HandleUpdateStatus(time);
        Check(Age() > .04f, "Skill slot selection advances with zero cached game time.");
        float before = Age();
        var wall = global::System.Diagnostics.Stopwatch.StartNew();
        var stale = new GameTime(TimeSpan.Zero, TimeSpan.FromSeconds(20));
        for (int i = 0; i < 100; i++) slot.HandleUpdateStatus(stale);
        Check(Age() - before <= wall.Elapsed.TotalSeconds + .02, "Repeated cached time cannot accelerate slot selection.");
        GameMain.hasFocus = false;
        slot.HandleUpdateStatus(time);
        before = Age();
        Thread.Sleep(90);
        GameMain.hasFocus = true;
        slot.HandleUpdateStatus(time);
        Check(Age() == before, "Slot focus recovery excludes the unfocused interval.");
        ElainaSkillModPlayer.CurrentSkillIndex = 1;
        slot.HandleUpdateStatus(time);
        Check(Age() == 0 && !(bool)selected.GetValue(slot)!, "Switching skills resets the old selection animation.");
    }

    static void SkillBarScaleChecks()
    {
        var config = new global::伊蕾娜.Config.BasicConfig();
        Check(config.SkillBarScale == 1f, "New and legacy settings keep the original bar size.");
        var restored = Newtonsoft.Json.JsonConvert.DeserializeObject<global::伊蕾娜.Config.BasicConfig>("{\"ShowCD\":true}")!;
        Check(restored.SkillBarScale == 1f, "Existing config JSON without the new setting defaults to 100 percent.");
        config.SkillBarScale = .65f;
        restored = Newtonsoft.Json.JsonConvert.DeserializeObject<global::伊蕾娜.Config.BasicConfig>(
            Newtonsoft.Json.JsonConvert.SerializeObject(config))!;
        restored.OnChanged();
        Check(ElainaSkillManager.BarScale == .65f, "Saved bar scale applies immediately through OnChanged.");
        foreach (var (input, expected) in new[] { (2f, 1f), (-1f, .5f), (0f, .5f), (float.NaN, 1f), (float.PositiveInfinity, 1f) })
        {
            config.SkillBarScale = input;
            config.OnChanged();
            Check(ElainaSkillManager.BarScale == expected, "Invalid/out-of-range config cannot enlarge or hide the bar.");
        }

        var layout = typeof(ElainaSkillBar).Assembly.GetType("伊蕾娜.ElainaModSkills.ElainaSkillUI.SkillBarLayout")!;
        var anchorField = layout.GetField("_anchor", BindingFlags.Static | BindingFlags.NonPublic)!;
        object oldAnchor = anchorField.GetValue(null);
        var bar = (ElainaSkillBar)RuntimeHelpers.GetUninitializedObject(typeof(ElainaSkillBar));
        // Only BaseBody's constructor needs a graphics device. Exercise the real size
        // application on real SUI children without initializing textures or writing saves.
        var children = new List<SilkyUIFramework.Elements.UIView>();
        for (int i = 0; i < 8; i++) children.Add(new ElainaSkillSlot(null));
        typeof(SilkyUIFramework.Elements.UIElementGroup).GetField("<Elements>k__BackingField", Instance)!.SetValue(bar, children);
        var apply = typeof(ElainaSkillBar).GetMethod("ApplyLayout", Instance)!;
        var move = layout.GetMethod("MoveTo")!;
        var dock = typeof(ElainaSkillBar).Assembly.GetType("伊蕾娜.ElainaModSkills.ElainaSkillUI.ConstellationSkillPanel.ConstellationBookLayout")!
            .GetMethod("DockBounds", BindingFlags.Static | BindingFlags.NonPublic)!;
        try
        {
            foreach (float scale in new[] { 1f, .75f, .5f, 1f })
            {
                ElainaSkillManager.BarScale = scale;
                anchorField.SetValue(null, new Vector2(.95f));
                apply.Invoke(bar, null);
                Check(Math.Abs(bar.Width.Pixels - 646 * scale) < .001f && Math.Abs(bar.Height.Pixels - 94 * scale) < .001f,
                    "Live bar bounds resize, including restoration to 100 percent.");
                Check(children.All(c => c.Width.Pixels == 60 * scale && c.Height.Pixels == 60 * scale),
                    "All eight existing SUI slots resize with the bar.");
                Check(bar.Padding.Left == 27 * scale && bar.Padding.Top == 25 * scale && !bar.IsInteractable,
                    "Padding follows scale and shrinking never enables mouse capture.");
                foreach (float uiScale in new[] { 1f, 1.25f, 2f })
                {
                    Vector2 screen = new Vector2(1920, 1080) / uiScale;
                    Vector2 size = new(646 * scale, 94 * scale);
                    Vector2 mouse = screen / 2;
                    move.Invoke(null, [mouse - size / 2, screen, size]);
                    var anchor = (Vector2)layout.GetProperty("Anchor")!.GetValue(null)!;
                    Check(Vector2.Distance((screen - size) * anchor + size / 2, mouse) < .01f,
                        "Shortcut centering uses the scaled bounds at each Terraria UI scale.");
                    move.Invoke(null, [screen * 2, screen, size]);
                    anchor = (Vector2)layout.GetProperty("Anchor")!.GetValue(null)!;
                    Check(anchor == Vector2.One, "Dragging beyond the screen clamps the complete shrunken bar.");
                }
                var bounds = (Rectangle)dock.Invoke(null, [new Vector2(1920, 1080), new Vector2(300, 800), 646 * scale])!;
                Check(bounds.Right == (int)(300 + 646 * scale), "Book/toast dock follows the resized bar's right edge.");
            }
        }
        finally
        {
            anchorField.SetValue(null, oldAnchor);
            ElainaSkillManager.BarScale = 1f;
        }
    }

    sealed class Display(string key) : IPlayerResourcesDisplaySet
    {
        public string ConfigKey => key;
        public string NameKey => key;
        public string DisplayedName => key;
        public void Draw() { }
        public void TryToHover() { }
    }
}

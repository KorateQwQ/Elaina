using System;
using System.IO;
using System.Reflection;
using System.Runtime.InteropServices;

// Runs the legacy XNA/GDI+ generator in the working directory with temporary session fonts.
// Usage: GenerateHost.exe <DynamicFontGenerator.exe> <font.ttf>...
internal static class GenerateHost
{
    [DllImport("gdi32.dll", CharSet = CharSet.Unicode)]
    private static extern int AddFontResourceEx(string path, uint flags, IntPtr reserved);
    [DllImport("gdi32.dll", CharSet = CharSet.Unicode)]
    private static extern bool RemoveFontResourceEx(string path, uint flags, IntPtr reserved);

    [STAThread]
    private static int Main(string[] args)
    {
        string generatorDirectory = Path.GetDirectoryName(Path.GetFullPath(args[0]));
        AppDomain.CurrentDomain.AssemblyResolve += delegate(object sender, ResolveEventArgs request)
        {
            string path = Path.Combine(generatorDirectory, new AssemblyName(request.Name).Name + ".dll");
            return File.Exists(path) ? Assembly.LoadFrom(path) : null;
        };
        for (int i = 1; i < args.Length; i++)
        {
            string fontPath = Path.GetFullPath(args[i]);
            if (AddFontResourceEx(fontPath, 0, IntPtr.Zero) == 0)
                throw new InvalidOperationException("Temporary font registration failed: " + fontPath);
            AppDomain.CurrentDomain.ProcessExit += delegate { RemoveFontResourceEx(fontPath, 0, IntPtr.Zero); };
        }
        var assembly = Assembly.LoadFrom(Path.GetFullPath(args[0]));
        // The generator compiles every *.dynamicfont in the working directory, then calls Environment.Exit.
        try { assembly.EntryPoint.Invoke(null, null); }
        catch (Exception ex)
        {
            while (ex != null)
            {
                Console.Error.WriteLine(ex.GetType().FullName);
                Console.Error.WriteLine(ex.Message);
                Console.Error.WriteLine(ex.StackTrace);
                ex = ex.InnerException;
            }
            return 1;
        }
        return 0;
    }
}

using Microsoft.Xna.Framework;
using Microsoft.Xna.Framework.Graphics;
using SDL2;
using 伊蕾娜.System.Experience;

using var game = new Preview(args[0], args[1]);
game.Run();

// Mirrors LevelUpEffect + LevelUpDust: each tick emits the production schedule at the feet, steps particles
// (dust updates after projectile AI in the same tick), then draws the body at age = tick and every particle.
// Premultiplied AlphaBlend into a coverage-recording target, composited as scene * (1 - a) + rgb * Hdr (ReColor);
// KL Bloom blur is not simulated. Player and ground are neutral fixtures; textures are the shipped PNGs.
sealed class Preview : Game
{
    const int Ticks = 120, Width = 240, Height = 160, Zoom = 3;
    const float Hdr = 2.2f; // LevelUpDust.Hdr
    static readonly (string Name, Color Sky, Color Ground, Color Edge)[] Scenes =
        [("night", new(16, 21, 32), new(39, 38, 46), new(67, 63, 72)), ("day", new(122, 206, 205), new(118, 92, 66), new(92, 168, 70))];
    static readonly BlendState Under = new()
    {
        ColorSourceBlend = Blend.One, ColorDestinationBlend = Blend.InverseSourceAlpha,
        AlphaSourceBlend = Blend.Zero, AlphaDestinationBlend = Blend.One,
    };
    static readonly BlendState Lift = new()
    {
        ColorSourceBlend = Blend.One, ColorDestinationBlend = Blend.One,
        AlphaSourceBlend = Blend.Zero, AlphaDestinationBlend = Blend.One,
    };
    readonly string output, repo;
    SpriteBatch batch;
    Texture2D pixel;
    LevelUpTextures tex;
    bool done;

    public Preview(string output, string repo)
    {
        this.output = output; this.repo = repo;
        _ = new GraphicsDeviceManager(this) { PreferredBackBufferWidth = Width * Zoom, PreferredBackBufferHeight = Height * Zoom,
            GraphicsProfile = GraphicsProfile.HiDef, SynchronizeWithVerticalRetrace = false };
        IsFixedTimeStep = false; SDL.SDL_HideWindow(Window.Handle);
    }

    // tModLoader premultiplies PNG alpha on load; mirror it so blending matches the game.
    Texture2D Load(string path)
    {
        using var stream = File.OpenRead(path);
        var texture = Texture2D.FromStream(GraphicsDevice, stream);
        var data = new Color[texture.Width * texture.Height];
        texture.GetData(data);
        for (int i = 0; i < data.Length; i++) data[i] = Color.FromNonPremultiplied(data[i].ToVector4());
        texture.SetData(data);
        return texture;
    }

    protected override void LoadContent()
    {
        batch = new SpriteBatch(GraphicsDevice);
        pixel = new Texture2D(GraphicsDevice, 1, 1); pixel.SetData([Color.White]);
        string ui = Path.Combine(repo, "ElainaModSkills/UICore/Assets"), kl = Path.Combine(repo, "../KL/Effects/Tex/Sparkle");
        // Same order as LevelUpDust.Textures().
        tex = new LevelUpTextures(Load(Path.Combine(ui, "GlowDot.png")), Load(Path.Combine(ui, "Star4.png")),
            Load(Path.Combine(kl, "T_StartFlare001.PNG")), Load(Path.Combine(kl, "ShotLine.png")),
            Load(Path.Combine(ui, "ThinRing.png")), Load(Path.Combine(repo, "ElainaModAlchemy/Revealing/Assets/MarkGlow.png")));
    }

    protected override void Draw(GameTime time)
    {
        if (done) return; done = true;
        foreach (var scene in Scenes) Render(scene.Name, scene.Sky, scene.Ground, scene.Edge);
        Console.WriteLine("Exported level-up timeline on night and day backgrounds (production LevelUpVisuals and textures).");
        Exit();
    }

    void Render(string name, Color sky, Color ground, Color edge)
    {
        var random = new Random(7);
        var particles = new List<(Vector2 Origin, LevelUpParticle Particle)>();
        var emitted = new List<LevelUpParticle>();
        Vector2 feet = new(Width / 2, 128);
        using var target = new RenderTarget2D(GraphicsDevice, Width, Height);
        using var hdr = new RenderTarget2D(GraphicsDevice, Width, Height, false, SurfaceFormat.HalfVector4, DepthFormat.None);
        using var big = new RenderTarget2D(GraphicsDevice, Width * Zoom, Height * Zoom);
        for (int tick = 0; tick < Ticks; tick++)
        {
            emitted.Clear();
            if (tick < LevelUpVisuals.BodyTicks) LevelUpVisuals.Emit(random, tick, emitted);
            foreach (var p in emitted) particles.Add((feet, p));
            for (int i = particles.Count - 1; i >= 0; i--)
            {
                var entry = particles[i];
                if (!LevelUpVisuals.Step(ref entry.Particle)) particles.RemoveAt(i); else particles[i] = entry;
            }

            GraphicsDevice.SetRenderTarget(hdr);
            GraphicsDevice.Clear(Color.Transparent);
            batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.LinearClamp, DepthStencilState.None, RasterizerState.CullNone);
            LevelUpVisuals.DrawBody(batch, tex, feet, tick);
            foreach (var (origin, particle) in particles) LevelUpVisuals.Draw(batch, tex, origin, particle);
            batch.End();

            GraphicsDevice.SetRenderTarget(target);
            GraphicsDevice.Clear(sky);
            batch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, SamplerState.PointClamp, DepthStencilState.None, RasterizerState.CullNone);
            batch.Draw(pixel, new Rectangle(0, (int)feet.Y, Width, Height), ground);
            batch.Draw(pixel, new Rectangle(0, (int)feet.Y, Width, 2), edge);
            // 20 x 42 player fixture with a hat brim, feet at the effect origin.
            int px = (int)feet.X, py = (int)feet.Y;
            batch.Draw(pixel, new Rectangle(px - 9, py - 30, 18, 30), new Color(148, 132, 183));
            batch.Draw(pixel, new Rectangle(px - 6, py - 38, 12, 9), new Color(224, 195, 174));
            batch.Draw(pixel, new Rectangle(px - 11, py - 41, 22, 3), new Color(70, 62, 86));
            batch.Draw(pixel, new Rectangle(px - 5, py - 50, 10, 9), new Color(70, 62, 86));
            batch.End();
            // scene * (1 - a) + rgb, then add rgb * (Hdr - 1); the 8-bit target clips like the game's scene target.
            batch.Begin(SpriteSortMode.Deferred, Under, SamplerState.PointClamp, DepthStencilState.None, RasterizerState.CullNone);
            batch.Draw(hdr, Vector2.Zero, Color.White);
            batch.End();
            batch.Begin(SpriteSortMode.Deferred, Lift, SamplerState.PointClamp, DepthStencilState.None, RasterizerState.CullNone);
            for (float lift = Hdr - 1; lift > 0; lift -= 1) batch.Draw(hdr, Vector2.Zero, Color.White * Math.Min(1, lift));
            batch.End();

            GraphicsDevice.SetRenderTarget(big);
            batch.Begin(SpriteSortMode.Deferred, BlendState.Opaque, SamplerState.PointClamp, DepthStencilState.None, RasterizerState.CullNone);
            batch.Draw(target, new Rectangle(0, 0, Width * Zoom, Height * Zoom), Color.White);
            batch.End();
            GraphicsDevice.SetRenderTarget(null);
            if (tick % 2 == 0)
                using (var stream = File.Create(Path.Combine(output, $"frame-{name}-{tick:000}.png"))) big.SaveAsPng(stream, Width * Zoom, Height * Zoom);
        }
    }
}

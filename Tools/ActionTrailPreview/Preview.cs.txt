internal static class ActionTrailPreview
{
    internal static string Fault = "none";
    static void Main(string[] args)
    {
        if (args.Length > 2) Fault = args[2];
        if (OperatingSystem.IsWindows())
            System.Runtime.InteropServices.NativeLibrary.Load(Path.Combine(Environment.SystemDirectory, "d3d11.dll"));
        using var game = new TrailProbe(args[0], args[1]);
        game.RunOneFrame();
        if (!game.Exported) throw new Exception("Action trail export did not complete");
    }
}

internal sealed class TrailProbe : Game
{
    const BindingFlags All = BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Static | BindingFlags.Instance;
    readonly string root, output, kl;
    Assembly pipeline;
    Type main, helper, scene;
    object renderer;
    SpriteBatch batch;
    ContentManager content;
    Texture2D trail, pixel, starCross, starLine, starHalo;
    string HaloPath => Path.Combine(root, "ElainaActions/Assets/WandStarHalo.png");
    bool done;
    public bool Exported => done;
    void AssertHidden() {
        var flags = (SDL2.SDL.SDL_WindowFlags)SDL2.SDL.SDL_GetWindowFlags(Window.Handle);
        if ((flags & SDL2.SDL.SDL_WindowFlags.SDL_WINDOW_HIDDEN) == 0 || (flags & SDL2.SDL.SDL_WindowFlags.SDL_WINDOW_SHOWN) != 0)
            throw new Exception($"Preview window unexpectedly visible: {flags}");
    }

    public TrailProbe(string root, string output)
    {
        this.root = root; this.output = output; kl = Path.GetFullPath(Path.Combine(root, "../KL"));
        new GraphicsDeviceManager(this) { PreferredBackBufferWidth = 640, PreferredBackBufferHeight = 360,
            GraphicsProfile = GraphicsProfile.HiDef, SynchronizeWithVerticalRetrace = false };
        SDL2.SDL.SDL_HideWindow(Window.Handle); IsFixedTimeStep = false; AssertHidden();
    }

    string Extract()
    {
        CompilationUnitSyntax Read(string path) => CSharpSyntaxTree.ParseText(File.ReadAllText(path)).GetCompilationUnitRoot();
        string Methods(CompilationUnitSyntax tree, params string[] names) => string.Join("\n", tree.DescendantNodes()
            .OfType<MethodDeclarationSyntax>().Where(m => names.Contains(m.Identifier.Text)).Select(m => m.WithoutLeadingTrivia().ToFullString()));
        var vertex = Read(Path.Combine(kl, "Drawing/DrawHelper.vertexDraw.cs"));
        var drawing = Read(Path.Combine(kl, "Drawing/DrawHelper.DrawingMethod.cs"));
        var style = Read(Path.Combine(root, "ElainaActions/ElainaTrailVisualStyle.cs"));
        var action = Read(Path.Combine(root, "ElainaActions/ElainaAction.cs"));
        var slash = Read(Path.Combine(root, "ElainaActions/SlashTrailActionNode.cs")).DescendantNodes()
            .OfType<ClassDeclarationSyntax>().First();
        slash = slash.WithMembers(new(slash.Members.Where(m => m is not MethodDeclarationSyntax method
            || method.Identifier.Text is not ("Draw" or "DrawDebugPoint" or "GetStarCenter"))));
        var fields = style.DescendantNodes().OfType<FieldDeclarationSyntax>().Where(f => f.Declaration.Variables
            .Any(v => v.Identifier.Text.StartsWith("Default")));
        string source = """
            using System;
            using System.Collections.Generic;
            using Microsoft.Xna.Framework;
            using Microsoft.Xna.Framework.Graphics;
            using KL.Drawing;
            using static KL.Drawing.DrawHelper;
            using ElainaAction = Fixture.ActionReplay;
            namespace KL.Drawing {
            public partial class DrawSystem {
                public void RunTrail(float strength, int iterations, bool enabled) {
                    SetBloomInfo(strength, iterations, shouldBloom: enabled);
                    RenderRequestedBloom(LayerDrawRequestSystem.DrawTargetLayer.OverPlayers, LayerDrawRequestSystem.DrawTiming.Before);
                }
            }
            public partial class DrawHelper {
            public static Effect vertexDraw;
            """ + Methods(vertex, "TrailEffect") + Methods(drawing, "PingPongWave")
            + drawing.DescendantNodes().OfType<EnumDeclarationSyntax>().Single(e => e.Identifier.Text == "PingPongWaveType").ToFullString()
            + Read(Path.Combine(kl, "Drawing/DrawHelper.Assets.cs")).DescendantNodes().OfType<StructDeclarationSyntax>()
                .Single(s => s.Identifier.Text == "CustomVertexInfo").WithoutLeadingTrivia().ToFullString() + "}}"
            + "namespace Fixture { public static partial class ElainaTrailVisualStyle {"
            + string.Join("\n", fields.Select(f => f.ToFullString()))
            + string.Join("\n", style.DescendantNodes().OfType<PropertyDeclarationSyntax>().Where(p => (p.Identifier.Text.StartsWith("Default") || p.Identifier.Text == "WaterWandColor")).Select(p => p.ToFullString()))
            + Methods(style, "RequestTrail", "RequestVisual", "GetActionAlpha", "SmoothOpacity", "DrawTrailOpacity", "DrawTrail", "DrawWandTrail", "DrawSlashTrail", "RequestStar", "QueueStar")
            + "} public partial class ActionReplay {"
            + Methods(action, "SubmitActionVisuals", "ResetWandTrail", "UpdateWandTrail", "RequestAutoWandTrail", "CreateWandTrailVisual", "SubmitTrail", "OnInterrupt", "OnFinish")
            + "} public class WaterActionReplay : ActionReplay { protected override bool DrawActionStar => true;"
            + Methods(Read(Path.Combine(root, "ElainaActions/Action_WaterBall.cs")), "RequestActionStar", "CreateWandTrailVisual")
            + "}" + slash.ToFullString()
            + string.Join("\n", Read(Path.Combine(root, "ElainaActions/ElainaTrailSnapshot.cs")).Members.OfType<FileScopedNamespaceDeclarationSyntax>().Single().Members.Select(m => m.ToFullString()))
            + "public partial class ElainaActionDrawSystem : BaseSystem {"
            + Read(Path.Combine(root, "ElainaActions/ElainaActionDrawSystem.cs")).DescendantNodes().OfType<FieldDeclarationSyntax>().Single().ToFullString()
            + Methods(Read(Path.Combine(root, "ElainaActions/ElainaActionDrawSystem.cs")), "RetainTrails", "PostUpdatePlayers", "OnWorldUnload") + "}"

            + "public class ActionNode { public int StartFrame, EndFrame; public int DurationFrame => EndFrame - StartFrame;"
            + Methods(Read(Path.Combine(kl, "ActionsSystem/ActionNode.cs")), "IsActive", "GetProgress", "Clamp01")
            + "} public class ElainaNode : ActionNode { public ElainaNode(int start, int end) { StartFrame = start; EndFrame = end; }"
            + Methods(Read(Path.Combine(root, "ElainaActions/ElainaNode.cs")), "GetOwner", "GetLocalDrawOffset")
            + "} public static class KLMathF {"
            + Methods(Read(Path.Combine(kl, "Utils/KLMathF.cs")), "ClampLerp") + "}}";
        source = source.Replace("Main.graphics.GraphicsDevice.DrawUserPrimitives(PrimitiveType.TriangleList, triangleList.ToArray()", "DrawCheckedTriangles(PrimitiveType.TriangleList, triangleList.ToArray()");
        // 故障只注入独立程序集，永远不回写正式源码。
        source = source.Replace("\r\n", "\n");
        return ActionTrailPreview.Fault switch {
            "alpha" => source.Replace("startAlpha: 1f", "startAlpha: alpha"),
            "tangent" => source.Replace("if (normalDir.LengthSquared() <= 0.000001f)", "if (false)"),
            "sampling" => source.Replace("lastTrails.Clear();\n        RequestAutoWandTrail", "UpdateWandTrail(); lastTrails.Clear();\n        RequestAutoWandTrail"),
            "interruption" => source.Replace("ElainaActionDrawSystem.RetainTrails(owner, lastTrails);", ""),
            _ => source
        };
    }

    void Set(Type type, string field, object value) => type.GetField(field, All).SetValue(null, value);
    protected override void LoadContent()
    {
        AssertHidden();
        Directory.CreateDirectory(output);
        pipeline = Program.CompilePipeline(kl, Extract(), File.ReadAllText(Path.Combine(root, "Tools/ActionTrailPreview/Scene.cs.txt")));
        main = pipeline.GetType("KL.Drawing.Main"); helper = pipeline.GetType("KL.Drawing.DrawHelper"); scene = pipeline.GetType("Fixture.Scene");
        Type draw = pipeline.GetType("KL.Drawing.DrawSystem"); Set(draw, "CanUseRender", true); renderer = Activator.CreateInstance(draw);
        var device = Activator.CreateInstance(pipeline.GetType("KL.Drawing.DeviceContext"));
        device.GetType().GetField("GraphicsDevice").SetValue(device, GraphicsDevice);
        batch = new SpriteBatch(GraphicsDevice);
        Set(main, "graphics", device); Set(main, "instance", device); Set(main, "spriteBatch", batch);
        Set(main, "screenWidth", 640); Set(main, "screenHeight", 360);
        content = new ContentManager(Services, Path.Combine(kl, "Effects/Content"));
        foreach (string name in new[] { "bloom", "reColor", "vertexDraw" })
            Set(helper, name, content.Load<Effect>(char.ToUpperInvariant(name[0]) + name[1..]));
        using var stream = File.OpenRead(Path.Combine(kl, "Effects/Tex/Trail/LightTrail.png"));
        trail = Texture2D.FromStream(GraphicsDevice, stream);
        CheckTrailTexture();
        Set(pipeline.GetType("Fixture.ElainaTrailVisualStyle"), "Texture", trail);
        Texture2D Load(string path) {
            using var file = File.OpenRead(Path.Combine(kl, path));
            return Texture2D.FromStream(GraphicsDevice, file);
        }
        starCross = Load("Effects/Tex/Sparkle/T_StartFlare001.png");
        starLine = Load("Effects/Tex/Sparkle/ShotLine.png");
        Set(pipeline.GetType("Fixture.ElainaTrailVisualStyle"), "StarCross", starCross);
        Set(pipeline.GetType("Fixture.ElainaTrailVisualStyle"), "StarLine", starLine);
        if (ActionTrailPreview.Fault != "bake-halo") {
            using var png = File.OpenRead(HaloPath);
            starHalo = Texture2D.FromStream(GraphicsDevice, png);
            Set(pipeline.GetType("Fixture.ElainaTrailVisualStyle"), "StarHalo", starHalo);
        }
        pixel = new Texture2D(GraphicsDevice, 1, 1); pixel.SetData(new[] { Color.White });
        Set(pipeline.GetType("KL.Drawing.TextureAssets"), "MagicPixel", Activator.CreateInstance(pipeline.GetType("KL.Drawing.Asset"), pixel));
    }

    void CheckTrailTexture()
    {
        // 对照 tML ImageIO.ToRaw：仅清零 A=0 的像素，不对半透明 RGB 再次预乘。
        using var stream = File.OpenRead(Path.Combine(kl, "Effects/Tex/Trail/LightTrail.png"));
        IntPtr data = FNA3D.ReadImageStream(stream, out int width, out int height, out int length);
        if (data == IntPtr.Zero) throw new Exception("Could not decode formal LightTrail PNG");
        byte[] raw = new byte[length];
        System.Runtime.InteropServices.Marshal.Copy(data, raw, 0, length);
        FNA3D.FNA3D_Image_Free(data);
        var loaded = new Color[width * height]; trail.GetData(loaded);
        for (int i = 0; i < loaded.Length; i++)
        {
            int p = i * 4;
            Color expected = raw[p + 3] == 0 ? Color.Transparent : new Color(raw[p], raw[p+1], raw[p+2], raw[p+3]);
            if (loaded[i] != expected) throw new Exception("Preview LightTrail pixels differ from tML rawimg loading");
        }
    }

    Color[] Render(float progress, bool bloom)
    {
        AssertHidden();
        using var target = new RenderTarget2D(GraphicsDevice, 640, 360);
        GraphicsDevice.SetRenderTarget(target); GraphicsDevice.Clear(new Color(25, 35, 50));
        // Queue while no SpriteBatch is active, matching CacheProjDraws.
        scene.GetMethod("Queue").Invoke(null, [progress]);
        renderer.GetType().GetMethod("RunTrail").Invoke(renderer, [1f, 5, bloom]);
        scene.GetMethod("CheckConsumed").Invoke(null, null);
        var pixels = new Color[640 * 360]; target.GetData(pixels);
        GraphicsDevice.SetRenderTarget(null);
        using var file = File.Create(Path.Combine(output, $"trail-{progress:0.00}-{(bloom ? "bloom" : "plain")}.png"));
        target.SaveAsPng(file, 640, 360);
        // This opaque sprite is drawn after trails without resetting the batch in its callback.
        if (!bloom && pixels[30 * 640 + 30] != Color.White) throw new Exception("Trail leaked vertex shader/batch state into next request.");
        return pixels;
    }

    void CheckOpacity()
    {
        double fullEnergy = 0, fullAlpha = 0;
        foreach (float opacity in new[] { 1f, .5f, .25f, 0f })
        {
            AssertHidden();
            using var target = new RenderTarget2D(GraphicsDevice, 640, 360, false, SurfaceFormat.Vector4, DepthFormat.None);
            GraphicsDevice.SetRenderTarget(target); GraphicsDevice.Clear(Color.Transparent);
            scene.GetMethod("QueueOpacity").Invoke(null, [opacity, false]);
            renderer.GetType().GetMethod("RunTrail").Invoke(renderer, [1f, 5, false]);
            var pixels = new Vector4[640 * 360]; target.GetData(pixels);
            double energy = pixels.Sum(p => (double)p.X);
            double alpha = pixels.Sum(p => (double)p.W);
            if (opacity == 1) { fullEnergy = energy; fullAlpha = alpha; }
            if (fullEnergy <= 0 || Math.Abs(energy / fullEnergy - opacity) > .002)
                throw new Exception($"Opacity was applied more than once: {opacity}, energy ratio {energy / fullEnergy}");
            if (fullAlpha <= 0 || Math.Abs(alpha / fullAlpha - opacity) > .002)
                throw new Exception($"Alpha channel does not follow opacity: {opacity}, ratio {alpha / fullAlpha}");
            GraphicsDevice.SetRenderTarget(null);
            Console.WriteLine($"Opacity {opacity}: RGB energy ratio {energy / fullEnergy:0.000}");
        }
        foreach (bool repeated in new[] { false, true })
        {
            using var target = new RenderTarget2D(GraphicsDevice, 640, 360);
            GraphicsDevice.SetRenderTarget(target); GraphicsDevice.Clear(new Color(25, 35, 50));
            scene.GetMethod("QueueOpacity").Invoke(null, [1f, repeated]);
            renderer.GetType().GetMethod("RunTrail").Invoke(renderer, [1f, 5, true]);
            GraphicsDevice.SetRenderTarget(null);
            using var file = File.Create(Path.Combine(output, $"geometry-{repeated}.png"));
            target.SaveAsPng(file, 640, 360);
        }
        foreach (int tick in new[] { 0, 2, 4, 6, 8 })
        {
            AssertHidden();
            using var target = new RenderTarget2D(GraphicsDevice, 640, 360);
            GraphicsDevice.SetRenderTarget(target); GraphicsDevice.Clear(new Color(25, 35, 50));
            scene.GetMethod("QueueAfterimage").Invoke(null, [tick]);
            renderer.GetType().GetMethod("RunTrail").Invoke(renderer, [1f, 5, true]);
            GraphicsDevice.SetRenderTarget(null);
            using var file = File.Create(Path.Combine(output, $"afterimage-{tick}.png"));
            target.SaveAsPng(file, 640, 360);
        }
    }

    protected override void Draw(GameTime gameTime)
    {
        if (done) return;
        AssertHidden();
        if (ActionTrailPreview.Fault == "bake-halo") { BakeHalo(); done = true; Exit(); return; }
        scene.GetMethod("CheckLogic").Invoke(null, null);
        CheckOpacity();
        CheckBoundedStars();
        foreach (float progress in new[] { .1f, .5f, .9f }) {
            var plain = Render(progress, false); var glow = Render(progress, true);
            // Exclude the following projectile marker; only the two trail regions count.
            if (!Enumerable.Range(70 * 640, 240 * 640).Any(i => plain[i] != glow[i]))
                throw new Exception("Bloom did not change trail rendering.");
        }
        foreach (float progress in new[] { 0f, .5f, 1f }) {
            var plain = RenderStars(progress, false);
            var glow = RenderStars(progress, true);
            bool changed = Enumerable.Range(70 * 640, 240 * 640).Any(i => plain[i] != glow[i]);
            if (changed != (progress == .5f)) throw new Exception("Star Bloom or fade endpoints incorrect");
        }
        foreach (bool spread in new[] {false,true}) {
            var cores = RenderStars(.5f,false,spread,.7f,1.35f);
            var glows = RenderStars(.5f,true,spread,.7f,1.35f);
            if (cores.SequenceEqual(glows)) throw new Exception("Missing baked wand halo");
        }
        done = true;
        Console.WriteLine("PASS: eight overlapping or spread wands use zero Bloom queues, animated texture glow and fade endpoints.");
        Console.WriteLine("PASS: game-tick history, interruption afterimages, finite geometry, single opacity; early auto/slash submission, snapshots, paused samples, active node bounds, shared queue, batch chaining.");
        Exit();
    }

    void CheckBoundedStars()
    {
        const int width = 640, height = 360;
        var draw = renderer.GetType();
        var queue = pipeline.GetType("KL.Drawing.LayerDrawRequestSystem");
        object layer = Enum.Parse(queue.GetNestedType("DrawTargetLayer"), "OverPlayers");
        object timing = Enum.Parse(queue.GetNestedType("DrawTiming"), "After");
        using var target = new RenderTarget2D(GraphicsDevice, width, height);
        foreach (bool backbuffer in new[] { false, true })
        foreach (var center in new[] { new Vector2(318.25f,141.5f), new Vector2(3, 5), new Vector2(638, 359) })
        foreach (bool overlap in new[] { false, true })
        {
            Color[] actual = Render(false), reference = Render(true);
            if (!actual.SequenceEqual(reference)) throw new Exception($"Bounded star differs at {center}, overlap={overlap}, backbuffer={backbuffer}: "
                + actual.Where((c,i) => c != reference[i]).Count() + " pixels");
            if (!backbuffer && overlap && center.X == 318.25f) {
                using var file = File.Create(Path.Combine(output, "star-local-overlap.png"));
                target.SaveAsPng(file, width, height);
            }
            Color[] Render(bool full) {
                AssertHidden();
                GraphicsDevice.SetRenderTarget(backbuffer ? null : target);
                GraphicsDevice.Clear(new Color(24,30,50));
                batch.Begin(); batch.Draw(pixel, new Rectangle(300,35,40,290), new Color(142,103,91)); batch.End();
                scene.GetMethod("QueueReference").Invoke(null,[center,overlap,full,1f]);
                var bounds = (Rectangle?)queue.GetMethod("GetBloomBounds", All).Invoke(null,[layer,timing]);
                if (bounds.HasValue == full) throw new Exception("Unexpected bounds for star queue");
                draw.GetMethod("SetBloomInfo", All).Invoke(null,[1f,5,Vector2.Zero,true]);
                draw.GetMethod("RenderRequestedBloom", All).Invoke(renderer,[layer,timing]);
                var pixels = new Color[width * height];
                if (backbuffer) GraphicsDevice.GetBackBufferData(pixels); else target.GetData(pixels);
                GraphicsDevice.SetRenderTarget(null);
                return pixels;
            }
        }
        Console.WriteLine("PASS: bounded vs full RGBA identical for single/overlapping stars at center and both screen edges, scene target and backbuffer.");
    }

    void BakeHalo()
    {
        const int size = 128;
        using var target = new RenderTarget2D(GraphicsDevice, 640, 360);
        GraphicsDevice.SetRenderTarget(target);
        GraphicsDevice.Clear(Color.Transparent);
        scene.GetMethod("QueueReference").Invoke(null, [new Vector2(320,180), false, true, 1f]);
        var queue = pipeline.GetType("KL.Drawing.LayerDrawRequestSystem");
        renderer.GetType().GetMethod("RenderRequestedBloom", All).Invoke(renderer, [
            Enum.Parse(queue.GetNestedType("DrawTargetLayer"), "OverPlayers"), Enum.Parse(queue.GetNestedType("DrawTiming"), "After")]);
        var pyramid = (RenderTarget2D[])pipeline.GetType("KL.Drawing.RenderHelper").GetField("BloomUpSample", All).GetValue(null);
        var radiance = new Microsoft.Xna.Framework.Graphics.PackedVector.HalfVector4[640*360];
        pyramid[5].GetData(radiance);
        var pixels = new Color[size*size];
        for (int y = 0; y < 360; y++) for (int x = 0; x < 640; x++) {
            int dx = x - 320 + size/2, dy = y - 180 + size/2;
            byte a = (byte)Math.Clamp((int)MathF.Round(255*radiance[y*640+x].ToVector4().X),0,255);
            if (dx <= 0 || dy <= 0 || dx >= size-1 || dy >= size-1) {
                if (a != 0) throw new Exception("Halo cropped at export boundary");
            } else pixels[dy*size+dx] = new Color(a,a,a,a);
        }
        using var texture = new Texture2D(GraphicsDevice,size,size);
        texture.SetData(pixels);
        Directory.CreateDirectory(Path.GetDirectoryName(HaloPath));
        using var file = File.Create(HaloPath); texture.SaveAsPng(file,size,size);
        GraphicsDevice.SetRenderTarget(null);
    }

    Color[] RenderStars(float progress, bool halo, bool spread = true, float blink = 1f, float zoom = 1f)
    {
        AssertHidden();
        using var target = new RenderTarget2D(GraphicsDevice,640,360);
        GraphicsDevice.SetRenderTarget(target); GraphicsDevice.Clear(new Color(25,35,50));
        batch.Begin(); batch.Draw(pixel,new Rectangle(310,0,330,360),new Color(148,135,127)); batch.End();
        var view = main.GetField("GameViewMatrix").GetValue(null);
        view.GetType().GetField("TransformationMatrix").SetValue(view, Matrix.CreateTranslation(-320,-180,0)*Matrix.CreateScale(zoom)*Matrix.CreateTranslation(320,180,0));
        renderer.GetType().GetMethod("SetBloomInfo",All).Invoke(renderer,[1f,5,Vector2.Zero,halo]);
        scene.GetMethod("QueueStars").Invoke(null,[progress,spread,blink]);
        scene.GetMethod("DrawOrdinaryStars").Invoke(null,null);
        batch.End();
        var pixels = new Color[640*360]; target.GetData(pixels);
        GraphicsDevice.SetRenderTarget(null);
        view.GetType().GetField("TransformationMatrix").SetValue(view,Matrix.Identity);
        using var file = File.Create(Path.Combine(output,$"wand-{progress:0.00}-{halo}-{spread}-{blink}-{zoom}.png"));
        target.SaveAsPng(file,640,360);
        return pixels;
    }

    protected override void UnloadContent()
    {
        if (pipeline != null) {
            Type targets = pipeline.GetType("KL.Drawing.RenderHelper");
            object detached = targets.GetMethod("DetachRenderTargets", All).Invoke(null, null);
            targets.GetMethod("DisposeTargets", All).Invoke(null, [detached]);
        }
        batch?.Dispose(); trail?.Dispose(); pixel?.Dispose(); starCross?.Dispose(); starLine?.Dispose(); starHalo?.Dispose(); content?.Dispose();
    }
}

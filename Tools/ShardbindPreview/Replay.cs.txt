    static Assembly Compile(string name, IEnumerable<string> sources)
    {
        var references = ((string)AppContext.GetData("TRUSTED_PLATFORM_ASSEMBLIES")).Split(Path.PathSeparator)
            .Append(typeof(GraphicsDevice).Assembly.Location).Append(typeof(CSharpSyntaxTree).Assembly.Location)
            .Append(typeof(Compilation).Assembly.Location).Distinct().Select(p => MetadataReference.CreateFromFile(p));
        var compilation = CSharpCompilation.Create(name, sources.Select(s => CSharpSyntaxTree.ParseText(s)), references,
            new CSharpCompilationOptions(OutputKind.DynamicallyLinkedLibrary));
        using var bytes = new MemoryStream();
        var result = compilation.Emit(bytes);
        if (!result.Success) throw new InvalidOperationException(string.Join("\n", result.Diagnostics.Where(d => d.Severity == DiagnosticSeverity.Error)));
        return Assembly.Load(bytes.ToArray());
    }

    public void ExportBloomThresholdProbe(GraphicsDevice gd, string output)
    {
        Vector3[] colors = [new(1, 1, 1), new(.1f, .2f, .8f), new(2, .2f, .1f),
            new(.1f, 1.5f, .2f), new(.1f, .2f, 2), new(1.001f), new(2, 3, 4)];
        var samples = (from rgb in colors from alpha in new[] { 0f, .01f, .25f, .5f, 1f }
            select new Vector4(rgb, alpha)).ToArray();
        int width = samples.Length;
        using var input = new Texture2D(gd, width, 1, false, SurfaceFormat.Vector4);
        input.SetData(samples);
        using var captured = new RenderTarget2D(gd, width, 1, false, SurfaceFormat.Vector4, DepthFormat.None);
        gd.SetRenderTarget(captured);
        gd.Clear(Color.Transparent);
        var straight = (BlendState)replay.GetType("KL.Drawing.DrawSystem").GetField("BloomStraightAlphaBlend",
            BindingFlags.NonPublic | BindingFlags.Static).GetValue(null);
        batch.Begin(SpriteSortMode.Immediate, straight, SamplerState.PointClamp, DepthStencilState.None, RasterizerState.CullNone);
        batch.Draw(input, Vector2.Zero, Color.White);
        batch.End();
        using var extracted = new RenderTarget2D(gd, width, 1, false, SurfaceFormat.Vector4, DepthFormat.None);
        gd.SetRenderTarget(extracted);
        gd.Clear(Color.Transparent);
        batch.Begin(SpriteSortMode.Immediate, BlendState.Opaque, SamplerState.PointClamp,
            DepthStencilState.None, RasterizerState.CullNone);
        klContent.Load<Effect>("Bloom").CurrentTechnique.Passes[2].Apply();
        batch.Draw(captured, Vector2.Zero, Color.White);
        batch.End();
        gd.SetRenderTarget(null);
        var result = new Vector4[width];
        extracted.GetData(result);
        var lines = new List<string> { "r,g,b,alpha,extracted_r,extracted_g,extracted_b" };
        for (int i = 0; i < width; i++)
        {
            Vector4 s = samples[i];
            Vector4 c = result[i];
            var expected = new Vector4(Math.Max(s.X - 1, 0), Math.Max(s.Y - 1, 0), Math.Max(s.Z - 1, 0), 0) * s.W;
            if (!(Vector4.Distance(c, expected) < .00001f))
                throw new InvalidOperationException($"Bloom extraction for {s}: expected {expected}, got {c}.");
            lines.Add(FormattableString.Invariant($"{s.X},{s.Y},{s.Z},{s.W},{c.X:F6},{c.Y:F6},{c.Z:F6}"));
        }
        Console.WriteLine($"PASS {width} straight-alpha HDR color/opacity samples through capture and extraction.");
        File.WriteAllLines(Path.Combine(output, "bloom-threshold.csv"), lines);
    }

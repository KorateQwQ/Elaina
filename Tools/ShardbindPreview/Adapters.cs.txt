namespace KL.Dusts
{
    public static class KLPreviewPhysics
    {
        public static bool Update(Dust dust)
        {
            var info = (KLBasicDust.DustInfo)dust.customData;
            if (info == null) return false;
            info.CurrentLifeTime--;
            info.LifeProgress = 1 - info.CurrentLifeTime / info.LifeTime;
            dust.life = info.LifeTime;
            dust.remaining = info.CurrentLifeTime;
            dust.position += dust.velocity;
            dust.rotation = dust.velocity.ToRotation();
            if (info.CurrentLifeTime < 0) dust.active = false;
            return false;
        }
    }
    public static class ExtraDustExtensions
    {
        public static KLBasicDust.DustInfo DustInfo(this Dust dust) => (KLBasicDust.DustInfo)dust.customData;
        public static Color GetRealColor(this Dust dust, Color color) => color;
    }
